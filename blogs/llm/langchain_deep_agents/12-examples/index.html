    12.1 代码助手¶

    场景描述¶

    构建一个智能编程助手，支持： - 代码生成和补全 - 代码审查和优化 - Bug 调试和修复 - 单元测试生成 - 文档编写

    完整实现¶

    # code_assistant.py
import os
from deepagents import create_deep_agent, create_subagent
from deepagents.backends import CompositeBackend, StateBackend, StoreBackend
from langchain_daytona import DaytonaSandbox
from daytona import Daytona, CreateSandboxFromSnapshotParams
from langgraph.store.postgres import PostgresStore

# MCP 配置
MCP_CONFIG = {
    "mcpServers": {
        "github": {
            "command": "npx",
            "args": ["-y", "@modelcontextprotocol/server-github"],
            "env": {"GITHUB_TOKEN": os.environ["GITHUB_TOKEN"]}
        },
        "filesystem": {
            "command": "npx",
            "args": ["-y", "@modelcontextprotocol/server-filesystem", "/workspace"]
        }
    }
}

async def create_code_assistant(runtime):
    """创建代码助手 Agent"""

    # 创建沙箱（用于代码执行）
    daytona = Daytona(api_key=os.environ["DAYTONA_API_KEY"])
    sandbox = await daytona.create(
        CreateSandboxFromSnapshotParams(
            labels={"assistant_id": runtime.context.assistant_id},
            auto_delete_interval=3600
        )
    )

    # 创建持久化存储
    store = PostgresStore.from_conn_string(os.environ["DATABASE_URL"])
    store.setup()

    # 后端配置
    def make_backend(rt):
        return CompositeBackend(
            default=DaytonaSandbox(sandbox=sandbox),
            routes={
                "/memories/": StoreBackend(
                    rt,
                    namespace=lambda ctx: (
                        "code-assistant",
                        ctx.runtime.context.user_id
                    )
                )
            }
        )

    # 系统提示
    system_prompt = """你是专业编程助手。

## 能力
- 代码生成（Python, JavaScript, Go, Rust）
- 代码审查和优化
- Bug 调试和修复
- 单元测试生成
- 技术文档编写

## 工作流程
1. 理解用户需求和项目上下文
2. 阅读 /memories/preferences.txt 获取用户偏好
3. 根据任务类型选择合适的方法
4. 生成代码后运行测试验证
5. 提供清晰的解释和使用说明

## 代码规范
- 遵循项目现有的代码风格
- 添加必要的注释和文档字符串
- 考虑边界情况和错误处理
- 优先使用简洁、可读的代码

## 安全
- 不执行危险命令
- 不处理敏感数据
- 代码在沙箱中运行"""

    return create_deep_agent(
        model="anthropic:claude-opus-4-6",
        backend=make_backend,
        store=store,
        system_prompt=system_prompt,
        mcp_config=MCP_CONFIG
    )

    子代理协调¶

    async def handle_code_request(main_agent, request: str):
    """处理代码请求，使用子代理分工"""

    # 分析请求类型
    if "review" in request.lower() or "审查" in request:
        return await code_review(main_agent, request)
    elif "debug" in request.lower() or "调试" in request:
        return await debug_code(main_agent, request)
    elif "test" in request.lower() or "测试" in request:
        return await generate_tests(main_agent, request)
    else:
        return await generate_code(main_agent, request)

async def code_review(main_agent, request: str):
    """代码审查工作流"""

    # 创建审查子代理
    reviewer = await create_subagent(
        role="高级代码审查员",
        task=f"""审查代码，关注：
1. 代码风格和一致性
2. 潜在的安全漏洞
3. 性能问题
4. 可维护性
5. 测试覆盖率

{request}""",
        model="anthropic:claude-sonnet-4-6"
    )

    review_result = await reviewer.finish()

    # 创建改进建议子代理
    improver = await create_subagent(
        role="代码优化专家",
        task=f"""基于审查结果，提供具体的改进建议：
{review_result}

为每个问题提供：
1. 问题描述
2. 修复方案
3. 修复后的代码示例"""
    )

    improvements = await improver.finish()

    return f"""## 代码审查报告

### 审查结果
{review_result}

### 改进建议
{improvements}"""

from deepagents import create_deep_agent
from deepagents.backends import CompositeBackend, StateBackend, StoreBackend

async def create_data_analyst(runtime):

    分析工作流¶

    async def run_analysis_pipeline(data_path: str, analysis_type: str):
    """运行完整分析流程"""

    # 阶段 1：数据加载和检查
    loader = await create_subagent(
        role="数据工程师",
        task=f"""加载并检查数据文件：{data_path}

输出：
1. 数据维度（行数、列数）
2. 列类型和数据类型
3. 缺失值统计
4. 基本统计信息
5. 数据质量问题"""
    )
    data_info = await loader.finish()

    # 阶段 2：数据清洗
    cleaner = await create_subagent(
        role="数据清洗专家",
        task=f"""基于以下数据信息，进行数据清洗：
{data_info}

处理：
1. 缺失值（填充或删除）
2. 异常值检测和处理
3. 数据类型转换
4. 重复值处理
5. 特征工程（如需要）"""
    )
    clean_code = await cleaner.finish()

    # 阶段 3：探索性分析
    eda = await create_subagent(
        role="数据分析师",
        task=f"""对清洗后的数据进行 EDA：

分析内容：
1. 单变量分析（分布、集中趋势）
2. 双变量分析（相关性、分组比较）
3. 多变量分析（如需要）
4. 关键发现总结

生成可视化图表。"""
    )
    eda_result = await eda.finish()

    # 阶段 4：建模（如适用）
    if analysis_type in ["predictive", "classification", "clustering"]:
        modeler = await create_subagent(
            role="机器学习工程师",
            task=f"""基于 EDA 结果，构建预测模型：
{eda_result}

要求：
1. 特征选择和工程
2. 模型选择和训练
3. 交叉验证
4. 性能评估
5. 超参数调优（如需要）"""
        )
        model_result = await modeler.finish()
    else:
        model_result = "无需建模（描述性分析）"

    # 阶段 5：报告生成
    reporter = await create_subagent(
        role="数据报告撰写员",
        task=f"""整合所有分析结果，生成完整报告：

数据信息：
{data_info}

清洗过程：
{clean_code}

EDA 结果：
{eda_result}

建模结果：
{model_result}

报告结构：
1. 执行摘要
2. 数据概述
3. 分析方法
4. 关键发现
5. 可视化图表
6. 结论和建议"""
    )

    return await reporter.finish()

    async def literature_review(topic: str, years: int = 5):

    searcher = await create_subagent(

- Semantic Scholar

输出：

    )

{paper}

输出：

        )
        return await summarizer.finish()

    # 并行处理多篇论文
    paper_list = parse_papers(papers)

        summarize_paper(paper) for paper in paper_list[:10]

    # 阶段 3：主题聚类
