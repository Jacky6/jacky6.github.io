    第十章：生产部署¶

    LangSmith Deployments | 多租户 | 异步处理 | 生产级配置完全指南

    10.1 部署选项¶

    Deep Agents 支持三种部署方式：

    部署选项：
├── LangSmith Deployments（推荐）
│   ├── 托管服务
│   ├── 自动扩缩容
│   └── 集成监控和日志
│
├── 自托管（Docker/K8s）
│   ├── 完全控制
│   ├── 数据本地化
│   └── 需要运维
│
└── 本地开发
    ├── InMemoryStore
    ├── 快速迭代
    └── 不持久化

    对比¶

    特性 LangSmith 自托管 本地
    部署难度 低 高 低
    运维成本 低 高 低
    数据控制 中 高 高
    可扩展性 高 中 低
    成本 按使用付费 固定成本 免费

    租户配额管理¶

    class QuotaMiddleware(AgentMiddleware):
    """租户配额管理"""

    def __init__(self, db):
        self.db = db

    async def abefore_agent(self, state, runtime):
        tenant_id = runtime.context.tenant_id

        # 检查配额
        usage = await self.db.get_usage(tenant_id, period="month")
        quota = await self.db.get_quota(tenant_id)

        if usage["tokens"] >= quota["tokens"]:
            raise QuotaExceeded(
                f"配额已用尽：{usage['tokens']}/{quota['tokens']}"
            )

        # 记录使用
        await self.db.record_usage(
            tenant_id,
            tokens=estimate_tokens(state)
        )

    后台任务¶

    from langgraph_sdk import get_client

async def start_background_task(assistant_id: str, input: dict):
    """启动后台任务，不等待完成"""

    client = get_client()

    # 异步启动
    run = await client.runs.create(
        assistant_id=assistant_id,
        input=input,
        multitask_strategy="rollback"  # 失败回滚
    )

    return run["id"]  # 返回 run ID 供后续查询

# 使用

    "research-assistant",
    {"messages": [{"role": "user", "content": "研究 AI 趋势"}]}
)

# 稍后查询状态
status = await client.runs.get(run_id)
print(status["status"])  # pending, running, completed, failed

    Token 限流¶

    class TokenLimitMiddleware(AgentMiddleware):
    """Token 使用限制"""

    def __init__(self, max_tokens: int = 100000):
        self.max_tokens = max_tokens
        self.usage = defaultdict(int)

    async def abefore_agent(self, state, runtime):
        user_id = runtime.context.user_id

        # 估算本次 token 使用
        estimated = estimate_tokens(state)

        if self.usage[user_id] + estimated > self.max_tokens:
            raise TokenLimitExceeded(
                f"Token 限制：{self.max_tokens}"
            )

    async def aafter_agent(self, state, runtime):
        user_id = runtime.context.user_id

        # 记录实际使用
        actual = count_tokens(state["messages"])
        self.usage[user_id] += actual

    命令过滤¶

    class CommandFilterMiddleware(AgentMiddleware):
    """危险命令过滤"""

    DANGEROUS_PATTERNS = [
        "rm -rf",
        "DROP TABLE",
        "DELETE FROM",
        "sudo",
        "chmod 777",
        "curl | bash",
    ]

    async def abefore_agent(self, state, runtime):
        # 检查最后一条消息
        last_msg = state["messages"][-1].content

        for pattern in self.DANGEROUS_PATTERNS:
            if pattern in last_msg:
                raise SecurityViolation(
                    f"检测到危险命令：{pattern}"
                )

    输出过滤¶

    class OutputFilterMiddleware(AgentMiddleware):
    """敏感信息过滤"""

    PATTERNS = {
        "api_key": r"sk-[a-zA-Z0-9]{20,}",
        "password": r"password['\"]?\s*[:=]\s*['\"]?[^\s'\"]+",
        "secret": r"secret['\"]?\s*[:=]\s*['\"]?[^\s'\"]+",
    }

    async def aafter_agent(self, state, runtime):
        last_msg = state["messages"][-1]

        for name, pattern in self.PATTERNS.items():
            if re.search(pattern, last_msg.content):
                # 脱敏
                last_msg.content = re.sub(
                    pattern,
                    f"[{name} REDACTED]",
                    last_msg.content
                )

    结构化日志¶

    import structlog

logger = structlog.get_logger()

class LoggingMiddleware(AgentMiddleware):
    """结构化日志"""

    async def abefore_agent(self, state, runtime):
        logger.info(
            "agent_start",
            thread_id=runtime.context.thread_id,
            user_id=runtime.context.user_id,
            message_count=len(state["messages"])
        )

    async def aafter_agent(self, state, runtime):
        logger.info(
            "agent_complete",
            thread_id=runtime.context.thread_id,
            duration=runtime.elapsed_time,
            tokens_used=count_tokens(state["messages"])
        )

    async def aon_error(self, state, runtime, error):
        logger.error(
            "agent_error",
            thread_id=runtime.context.thread_id,
            error=str(error),
            error_type=type(error).__name__
        )

    指标收集¶

    from prometheus_client import Counter, Histogram, Gauge

# 定义指标
AGENT_RUNS = Counter("agent_runs_total", "Total agent runs", ["status"])
AGENT_DURATION = Histogram("agent_duration_seconds", "Agent duration")
AGENT_TOKENS = Histogram("agent_tokens", "Agent tokens used")
ACTIVE_RUNS = Gauge("agent_runs_active", "Active agent runs")

class MetricsMiddleware(AgentMiddleware):
    """Prometheus 指标"""

    async def abefore_agent(self, state, runtime):
        ACTIVE_RUNS.inc()
        self.start_time = time.time()

    async def aafter_agent(self, state, runtime):
        ACTIVE_RUNS.dec()

        duration = time.time() - self.start_time
        tokens = count_tokens(state["messages"])

        AGENT_RUNS.labels(status="success").inc()
        AGENT_DURATION.observe(duration)
        AGENT_TOKENS.observe(tokens)

    async def aon_error(self, state, runtime, error):
        ACTIVE_RUNS.dec()
        AGENT_RUNS.labels(status="error").inc()

    分布式追踪¶

    from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider

trace.set_tracer_provider(TracerProvider())
tracer = trace.get_tracer("deepagents")

class TracingMiddleware(AgentMiddleware):
    """OpenTelemetry 追踪"""

    async def abefore_agent(self, state, runtime):
        self.span = tracer.start_span("agent_run")
        self.span.set_attribute("thread_id", runtime.context.thread_id)
        self.span.set_attribute("user_id", runtime.context.user_id)

    async def aafter_agent(self, state, runtime):
        self.span.set_attribute("duration", runtime.elapsed_time)
        self.span.end()

    async def aon_error(self, state, runtime, error):
        self.span.record_exception(error)
        self.span.set_status(trace.Status(trace.StatusCode.ERROR))
        self.span.end()

    import signal

class GracefulShutdown:
    """优雅关闭处理"""

    def __init__(self, agent):
        self.agent = agent
        self.running_runs = set()

    def register(self):
        signal.signal(signal.SIGTERM, self.handle_shutdown)
        signal.signal(signal.SIGINT, self.handle_shutdown)

    def handle_shutdown(self, signum, frame):
        logger.info("收到关闭信号，等待运行中任务完成...")

        # 等待所有运行中任务完成
        if self.running_runs:
            logger.info(f"等待 {len(self.running_runs)} 个任务完成")
            # 等待超时后强制关闭

        logger.info("关闭完成")
        sys.exit(0)

class AgentInput(BaseModel):
    message: str
    user_id: str

    @validator("message")
    def validate_message(cls, v):
        if len(v) > 10000:
            raise ValueError("消息过长")
        if contains_injection(v):
            raise ValueError("检测到注入尝试")

    @validator("user_id")

    模型路由¶

    class ModelRouter:
    """智能模型路由：根据任务复杂度选择模型"""

    SIMPLE_MODELS = ["gpt-4o-mini", "claude-haiku"]
    COMPLEX_MODELS = ["gpt-4o", "claude-opus"]

    async def route(self, input: str) -> str:
        # 简单任务用便宜模型
        if len(input) < 100 and is_simple_query(input):
            return random.choice(self.SIMPLE_MODELS)

        # 复杂任务用强大模型
        return random.choice(self.COMPLEX_MODELS)

def is_simple_query(query: str) -> bool:
    simple_patterns = ["what is", "who is", "define", "explain"]
    return any(p in query.lower() for p in simple_patterns)

    响应缓存¶

    from cachetools import TTLCache

cache = TTLCache(maxsize=1000, ttl=3600)  # 1 小时 TTL

async def cached_invoke(agent, input: str):
    # 检查缓存
    cache_key = hash(input)
    if cache_key in cache:
        return cache[cache_key]

    # 调用并缓存
    result = await agent.ainvoke(input)
    cache[cache_key] = result
    return result

    Token 优化¶

    def optimize_prompt(prompt: str) -> str:
    """优化提示词，减少 token"""

    # 移除多余空白
    prompt = re.sub(r"\s+", " ", prompt)

    # 移除注释（如果不需要）
    prompt = re.sub(r"#.*\n", "\n", prompt)

    # 缩写常见词
    abbreviations = {
        "please": "plz",
        "thank you": "thx",
        "information": "info",
    }
    for full, abbr in abbreviations.items():
        prompt = prompt.replace(full, abbr)

    return prompt
