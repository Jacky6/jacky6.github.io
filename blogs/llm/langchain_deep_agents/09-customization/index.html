    实现自定义后端¶

    from deepagents.backends import FileData
from deepagents.backends.utils import create_file_data

class S3Backend:
    """S3 存储后端"""

    def __init__(self, bucket: str, prefix: str = ""):
        self.bucket = bucket
        self.prefix = prefix
        self.s3 = boto3.client("s3")

    def _key(self, path: str) -> str:
        return f"{self.prefix}{path}"

    async def read_file(self, path: str) -> FileData:
        response = self.s3.get_object(
            Bucket=self.bucket,
            Key=self._key(path)
        )
        content = response["Body"].read().decode()
        return create_file_data(content)

    async def write_file(self, path: str, data: FileData) -> None:
        content = "\n".join(data["content"])
        self.s3.put_object(
            Bucket=self.bucket,
            Key=self._key(path),
            Body=content.encode()
        )

    async def delete_file(self, path: str) -> None:
        self.s3.delete_object(
            Bucket=self.bucket,
            Key=self._key(path)
        )

    async def list_files(self, prefix: str) -> list[str]:
        response = self.s3.list_objects_v2(
            Bucket=self.bucket,
            Prefix=self._key(prefix)
        )
        return [
            obj["Key"].replace(self.prefix, "")
            for obj in response.get("Contents", [])
        ]

    async def file_exists(self, path: str) -> bool:
        try:
            self.s3.head_object(
                Bucket=self.bucket,
                Key=self._key(path)
            )
            return True
        except ClientError:
            return False

# 使用
backend = S3Backend(bucket="my-deepagents-bucket", prefix="user-123/")

    组合后端¶

    from deepagents.backends import CompositeBackend

class HybridBackend:
    """混合后端：热数据内存 + 冷数据 S3"""

    def __init__(self, hot_backend, cold_backend, hot_threshold=10):
        self.hot = hot_backend
        self.cold = cold_backend
        self.threshold = hot_threshold
        self.hot_paths = set()

    async def read_file(self, path: str):
        if path in self.hot_paths:
            return await self.hot.read_file(path)
        return await self.cold.read_file(path)

    async def write_file(self, path: str, data):
        # 新文件先写热存储
        await self.hot.write_file(path, data)
        self.hot_paths.add(path)

        # 如果热文件太多，迁移最旧的到冷存储
        if len(self.hot_paths) > self.threshold:
            oldest = next(iter(self.hot_paths))
            data = await self.hot.read_file(oldest)
            await self.cold.write_file(oldest, data)
            await self.hot.delete_file(oldest)
            self.hot_paths.remove(oldest)

    日志中间件¶

    import logging
from datetime import datetime

class LoggingMiddleware(AgentMiddleware):
    """记录 Agent 执行日志"""

    def __init__(self, logger: logging.Logger = None):
        self.logger = logger or logging.getLogger("deepagents")

    async def abefore_agent(self, state, runtime):
        self.logger.info(
            f"[{datetime.now().isoformat()}] "
            f"Agent starting. Thread: {runtime.context.thread_id}"
        )
        self.logger.debug(f"State: {state}")

    async def aafter_agent(self, state, runtime):
        self.logger.info(
            f"[{datetime.now().isoformat()}] "
            f"Agent completed. Messages: {len(state['messages'])}"
        )

# 使用
agent = create_deep_agent(
    middleware=[LoggingMiddleware()]

    限流中间件¶

    import asyncio
from collections import defaultdict

class RateLimitMiddleware(AgentMiddleware):
    """API 调用限流"""

    def __init__(self, calls_per_minute: int = 60):
        self.calls_per_minute = calls_per_minute
        self.timestamps = defaultdict(list)
        self.lock = asyncio.Lock()

    async def abefore_agent(self, state, runtime):
        user_id = runtime.context.user_id
        now = asyncio.get_event_loop().time()

        async with self.lock:
            # 清理旧记录
            self.timestamps[user_id] = [
                t for t in self.timestamps[user_id]
                if now - t < 60
            ]

            # 检查限流
            if len(self.timestamps[user_id]) >= self.calls_per_minute:
                wait_time = 60 - (now - self.timestamps[user_id][0])
                raise RateLimitExceeded(
                    f"Rate limit exceeded. Wait {wait_time:.1f}s"
                )

            self.timestamps[user_id].append(now)

# 使用
agent = create_deep_agent(
    middleware=[RateLimitMiddleware(calls_per_minute=30)]
)

    成本追踪中间件¶

    from dataclasses import dataclass, field
from typing import Dict

@dataclass
class CostTracker:
    """成本追踪器"""

    costs: Dict[str, float] = field(default_factory=dict)
    model_prices = {
        "gpt-4o": 0.000005,      # per input token
        "claude-opus": 0.000015,
        # ...
    }

    async def aafter_agent(self, state, runtime):
        # 从 usage 信息计算成本
        for msg in state["messages"]:
            if hasattr(msg, "usage_metadata"):
                model = msg.usage_metadata.get("model", "unknown")
                input_tokens = msg.usage_metadata.get("input_tokens", 0)
                output_tokens = msg.usage_metadata.get("output_tokens", 0)

                price = self.model_prices.get(model, 0)
                cost = (input_tokens + output_tokens) * price

                user_id = runtime.context.user_id
                self.costs[user_id] = self.costs.get(user_id, 0) + cost

# 使用
tracker = CostTracker()
agent = create_deep_agent(
    middleware=[tracker]
)

# 查询成本
print(f"User 123 cost: ${tracker.costs.get('123', 0):.4f}")

    记忆同步中间件¶

    class MemorySyncMiddleware(AgentMiddleware):
    """同步记忆到外部存储"""

    def __init__(self, external_store):
        self.external_store = external_store

    async def aafter_agent(self, state, runtime):
        # 提取记忆更新
        user_id = runtime.context.user_id

        # 查找所有 /memories/ 路径的写操作
        for msg in state["messages"]:
            if msg.type == "tool" and msg.name == "write_file":
                if msg.content.startswith("/memories/"):
                    path = msg.content.split("\n")[0]
                    content = msg.content.split("\n", 1)[1]

                    # 同步到外部存储
                    await self.external_store.put_item(
                        ("memories", user_id),
                        path.replace("/memories/", ""),
                        {"content": content}
                    )

    9.4 工具扩展¶

    自定义工具¶

    from langchain.tools import tool

@tool
def search_web(query: str) -> str:
    """搜索网络获取信息"""
    import requests

    response = requests.get(
        "https://api.search.com/search",
        params={"q": query}
    )
    results = response.json()

    return "\n".join([
        f"{r['title']}: {r['snippet']}"
        for r in results["results"][:5]
    ])

@tool
def calculate(expression: str) -> str:
    """计算数学表达式"""
    try:
        # 安全计算（不用 eval）
        import ast
        import operator

        ops = {
            ast.Add: operator.add,
            ast.Sub: operator.sub,
            ast.Mult: operator.mul,
            ast.Div: operator.truediv,
        }

        tree = ast.parse(expression, mode="eval")
        result = eval(compile(tree, "", "eval"), {"__builtins__": None}, ops)
        return str(result)
    except Exception as e:
        return f"计算错误：{e}"

# 使用
agent = create_deep_agent(tools=[search_web, calculate])

    工具包装器¶

    from functools import wraps

def with_retry(max_attempts=3):
    """工具重试装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    await asyncio.sleep(2 ** attempt)  # 指数退避
        return wrapper
    return decorator

@with_retry(max_attempts=3)
@tool
def fetch_api(endpoint: str) -> str:
    """调用 API（带重试）"""
    import aiohttp

    async with aiohttp.ClientSession() as session:
        async with session.get(endpoint) as response:
            return await response.text()

    工具组合¶

    from langchain.tools import StructuredTool

class ToolChain:
    """工具链：组合多个工具"""

    def __init__(self, *tools):
        self.tools = tools

    @classmethod
    def from_tools(cls, *tools):
        return cls(*tools)

    def create_chain_tool(self, name: str, description: str):
        """创建组合工具"""

        async def execute(**kwargs):
            results = []
            for tool in self.tools:
                result = await tool.ainvoke(kwargs)
                results.append(result)
            return "\n---\n".join(results)

        return StructuredTool.from_function(
            coroutine=execute,
            name=name,
            description=description
        )

# 使用
data_tool = ToolChain.from_tools(
    fetch_data,
    clean_data,
    analyze_data
).create_chain_tool(
    name="full_analysis",
    description="完整数据分析：获取→清洗→分析"
)

agent = create_deep_agent(tools=[data_tool])

    模式 1：插件系统¶

    class PluginSystem:
    """插件系统：动态加载工具"""

    def __init__(self):
        self.plugins = {}

    def register(self, name: str, plugin):
        """注册插件"""
        self.plugins[name] = plugin

    def get_tools(self, plugin_names: list = None):
        """获取指定插件的工具"""
        if plugin_names is None:
            plugin_names = list(self.plugins.keys())

        tools = []
        for name in plugin_names:
            plugin = self.plugins.get(name)
            if plugin:
                tools.extend(plugin.get_tools())
        return tools

# 定义插件
class WeatherPlugin:
    def get_tools(self):
        return [get_weather, get_forecast]

class NewsPlugin:
    def get_tools(self):
        return [search_news, get_headlines]

# 使用
plugins = PluginSystem()
plugins.register("weather", WeatherPlugin())
plugins.register("news", NewsPlugin())

agent = create_deep_agent(
    tools=plugins.get_tools(["weather"])  # 只加载天气插件
)

    模式 2：上下文感知工具¶

    class ContextAwareTool:
    """根据上下文动态调整的工具"""

    def __init__(self, base_tool, context_provider):
        self.base_tool = base_tool
        self.context_provider = context_provider

    async def __call__(self, *args, **kwargs):
        # 获取上下文
        context = await self.context_provider.get_context()

        # 调整工具行为
        if context["mode"] == "safe":
            kwargs["safe_mode"] = True
        if context["verbose"]:
            kwargs["detail_level"] = "high"

        return await self.base_tool(*args, **kwargs)

# 使用
class UserContextProvider:
    async def get_context(self):
        return {
            "mode": "safe",  # 或 "advanced"
            "verbose": True,
            "preferences": {...}
        }

safe_search = ContextAwareTool(
    search_web,
    UserContextProvider()
)

    模式 3：工具版本控制¶

    class VersionedTool:
    """带版本控制的工具"""

    def __init__(self, name: str):
        self.name = name
        self.versions = {}

    def register(self, version: str, func):
        """注册版本"""
        self.versions[version] = func

    def get(self, version: str = "latest"):
        """获取指定版本"""
        if version == "latest":
            return list(self.versions.values())[-1]
        return self.versions.get(version)

# 使用
search_tool = VersionedTool("search")

@search_tool.register("v1")
async def search_v1(query: str):
    """v1: 简单搜索"""
    return basic_search(query)

@search_tool.register("v2")
async def search_v2(query: str, filters: dict = None):
    """v2: 带过滤搜索"""
    return advanced_search(query, filters)

@search_tool.register("v3")
async def search_v3(query: str, filters: dict = None, rerank: bool = True):
    """v3: 带重排序"""
    return premium_search(query, filters, rerank)

# 使用特定版本
agent = create_deep_agent(
    tools=[search_tool.get("v2")]
)
