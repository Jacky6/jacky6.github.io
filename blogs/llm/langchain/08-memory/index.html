    8.2.2 在 Agent 中使用 State¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage
from typing import Annotated, TypedDict
from langgraph.graph.message import add_messages

class AgentState(TypedDict):
    messages: Annotated[list, add_messages]
    context: dict

class SimpleAgent:
    def __init__(self):
        self.model = ChatOpenAI(model="gpt-4o")
        self.state = AgentState(messages=[], context={})

    def chat(self, user_input: str) -> str:
        # 添加用户消息
        self.state["messages"].append(HumanMessage(content=user_input))

        # 调用模型
        response = self.model.invoke(self.state["messages"])

        # 添加 AI 响应
        self.state["messages"].append(AIMessage(content=response.content))

        # 更新上下文
        self.state["context"]["last_topic"] = user_input[:50]

        return response.content

    def get_history(self):
        return self.state["messages"]

    def reset(self):
        self.state = AgentState(messages=[], context={})

# 使用
agent = SimpleAgent()
print(agent.chat("你好，我叫张三"))
print(agent.chat("记住我的名字"))
print(agent.chat("我叫什么？"))  # 应该能回答"张三"

    8.2.3 状态持久化（内存）¶

    import json
from datetime import datetime

class PersistentState:
    def __init__(self, session_id: str):
        self.session_id = session_id
        self.state = {
            "messages": [],
            "created_at": datetime.now().isoformat(),
            "updated_at": datetime.now().isoformat(),
            "metadata": {}
        }

    def add_message(self, role: str, content: str):
        self.state["messages"].append({
            "role": role,
            "content": content,
            "timestamp": datetime.now().isoformat()
        })
        self.state["updated_at"] = datetime.now().isoformat()

    def get_messages(self):
        return self.state["messages"]

    def set_metadata(self, key: str, value):
        self.state["metadata"][key] = value
        self.state["updated_at"] = datetime.now().isoformat()

    def get_metadata(self, key: str, default=None):
        return self.state["metadata"].get(key, default)

    def export(self) -> str:
        return json.dumps(self.state, ensure_ascii=False, indent=2)

    def load(self, json_str: str):
        self.state = json.loads(json_str)

# 使用示例
session = PersistentState("session_001")
session.add_message("user", "你好")
session.add_message("assistant", "你好！有什么帮助？")
session.set_metadata("user_name", "张三")

print(session.export())

    8.3.3 文件存储¶

    import json
import os
from pathlib import Path
from typing import Optional, List, Tuple

class FileStore:
    """基于文件系统的持久化存储"""

    def __init__(self, base_path: str):
        self.base_path = Path(base_path)
        self.base_path.mkdir(parents=True, exist_ok=True)

    def _get_file_path(self, key: str) -> Path:
        """将键转换为文件路径"""
        # user:001:name → user/001/name.json
        parts = key.split(":")
        return self.base_path / "/".join(parts[:-1]) / f"{parts[-1]}.json"

    def set(self, key: str, value: any):
        """存储数据"""
        file_path = self._get_file_path(key)
        file_path.parent.mkdir(parents=True, exist_ok=True)

        with open(file_path, "w", encoding="utf-8") as f:
            json.dump(value, f, ensure_ascii=False, indent=2)

    def get(self, key: str, default=None):
        """读取数据"""
        file_path = self._get_file_path(key)

        if not file_path.exists():
            return default

        with open(file_path, "r", encoding="utf-8") as f:
            return json.load(f)

    def delete(self, key: str):
        """删除数据"""
        file_path = self._get_file_path(key)
        if file_path.exists():
            file_path.unlink()

    def list_keys(self, prefix: str = "") -> List[str]:
        """列出所有键"""
        keys = []
        base_dir = self.base_path / prefix.replace(":", "/")

        if base_dir.exists():
            for file_path in base_dir.rglob("*.json"):
                rel_path = file_path.relative_to(self.base_path)
                key = ":".join([*rel_path.parts[:-1], rel_path.stem])
                keys.append(key)

        return keys

# 使用示例
store = FileStore("./data_store")

store.set("user:001:profile", {
    "name": "张三",
    "email": "zhangsan@example.com"
})

store.set("user:001:preferences", {
    "theme": "dark",
    "language": "zh-CN"
})

profile = store.get("user:001:profile")
print(f"用户档案：{profile}")

all_keys = store.list_keys()
print(f"所有键：{all_keys}")

    8.4 跨会话记忆¶

    8.4.1 用户级别的记忆¶

    from langchain.storage import InMemoryStore
from langchain_core.messages import HumanMessage, AIMessage
import json

class UserMemory:
    """用户级别的长期记忆"""

    def __init__(self, store: InMemoryStore, user_id: str):
        self.store = store
        self.user_id = user_id
        self.prefix = f"user:{user_id}"

    def save_profile(self, profile: dict):
        """保存用户档案"""
        self.store.set(f"{self.prefix}:profile", profile)

    def get_profile(self) -> dict:
        """获取用户档案"""
        return self.store.get(f"{self.prefix}:profile", {})

    def update_profile(self, updates: dict):
        """更新用户档案"""
        profile = self.get_profile()
        profile.update(updates)
        self.save_profile(profile)

    def save_preference(self, key: str, value: any):
        """保存偏好设置"""
        self.store.set(f"{self.prefix}:preference:{key}", value)

    def get_preference(self, key: str, default=None):
        """获取偏好设置"""
        return self.store.get(f"{self.prefix}:preference:{key}", default)

    def add_to_history(self, item: dict):
        """添加到历史记录"""
        history = self.store.get(f"{self.prefix}:history", [])
        history.append({
            **item,
            "timestamp": json.dumps({"$date": str(__import__("datetime").datetime.now())})
        })
        # 只保留最近 100 条
        self.store.set(f"{self.prefix}:history", history[-100:])

    def get_history(self, limit: int = 10) -> list:
        """获取历史记录"""
        history = self.store.get(f"{self.prefix}:history", [])
        return history[-limit:]

# 使用示例
store = InMemoryStore()
user_memory = UserMemory(store, "user_001")

# 保存档案
user_memory.save_profile({
    "name": "张三",
    "email": "zhangsan@example.com",
    "registered_at": "2026-01-01"
})

# 保存偏好
user_memory.save_preference("theme", "dark")
user_memory.save_preference("notifications", True)

# 添加历史
user_memory.add_to_history({
    "action": "login",
    "ip": "192.168.1.1"
})

# 读取
print(f"档案：{user_memory.get_profile()}")
print(f"主题偏好：{user_memory.get_preference('theme')}")
print(f"最近历史：{user_memory.get_history()}")

    8.4.2 对话历史持久化¶

    from langchain.storage import InMemoryStore
from langchain_core.messages import HumanMessage, AIMessage, messages_from_dict, messages_to_dict
import json

class ConversationStore:
    """对话历史存储"""

    def __init__(self, store: InMemoryStore):
        self.store = store

    def save_conversation(self, session_id: str, messages: list):
        """保存对话历史"""
        serialized = messages_to_dict(messages)
        self.store.set(f"conversation:{session_id}", serialized)

    def load_conversation(self, session_id: str) -> list:
        """加载对话历史"""
        serialized = self.store.get(f"conversation:{session_id}", [])
        return messages_from_dict(serialized)

    def delete_conversation(self, session_id: str):
        """删除对话历史"""
        self.store.delete(f"conversation:{session_id}")

    def list_sessions(self, user_id: str = None) -> list:
        """列出所有会话"""
        prefix = f"conversation:"
        if user_id:
            prefix += f"{user_id}:"

        return list(self.store.yield_keys(prefix=prefix))

# 使用示例
store = InMemoryStore()
conv_store = ConversationStore(store)

# 保存对话
messages = [
    HumanMessage(content="你好"),
    AIMessage(content="你好！有什么帮助？"),
    HumanMessage(content="帮我写个 Python 函数"),
    AIMessage(content="好的，请问需要什么功能的函数？"),
]

conv_store.save_conversation("session_001", messages)

# 加载对话
loaded_messages = conv_store.load_conversation("session_001")
print(f"加载了 {len(loaded_messages)} 条消息")
for msg in loaded_messages:
    print(f"  {msg.type}: {msg.content[:30]}...")

    8.5 实战：带记忆的对话助手¶

    8.5.1 完整实现¶

    from langchain_openai import ChatOpenAI
from langchain.storage import InMemoryStore
from langchain_core.messages import HumanMessage, AIMessage, messages_to_dict, messages_from_dict
from typing import Optional, Dict, Any
import json
from datetime import datetime

class MemoryChatBot:
    """带长期记忆的对话机器人"""

    def __init__(self, user_id: str, store: Optional[InMemoryStore] = None):
        self.user_id = user_id
        self.store = store or InMemoryStore()
        self.model = ChatOpenAI(model="gpt-4o")
        self.session_id = f"{user_id}:{datetime.now().strftime('%Y%m%d%H%M%S')}"
        self.messages = []

        # 加载用户档案
        self.profile = self._load_profile()

        # 加载对话历史（如果有）
        self._load_conversation()

    def _load_profile(self) -> Dict[str, Any]:
        """加载用户档案"""
        profile = self.store.get(f"user:{self.user_id}:profile", {})
        return profile

    def _save_profile(self, profile: Dict[str, Any]):
        """保存用户档案"""
        self.profile = profile
        self.store.set(f"user:{self.user_id}:profile", profile)

    def _load_conversation(self):
        """加载最近的对话历史"""
        # 查找最近的会话
        sessions = list(self.store.yield_keys(
            prefix=f"conversation:{self.user_id}:"
        ))

        if sessions:
            # 加载最近的会话
            last_session = sessions[-1]
            serialized = self.store.get(last_session, [])
            self.messages = messages_from_dict(serialized)
            self.session_id = last_session.split(":")[-1]

    def _save_conversation(self):
        """保存对话历史"""
        serialized = messages_to_dict(self.messages)
        self.store.set(f"conversation:{self.user_id}:{self.session_id}", serialized)

    def chat(self, user_input: str) -> str:
        """发送消息并获取响应"""
        # 添加用户消息
        self.messages.append(HumanMessage(content=user_input))

        # 构建系统提示（包含用户档案）
        system_content = "你是一个友好的助手。"
        if self.profile.get("name"):
            system_content += f"用户的名字是{self.profile['name']}。"
        if self.profile.get("preferences"):
            prefs = self.profile["preferences"]
            if prefs.get("language"):
                system_content += f"用户偏好使用{prefs['language']}交流。"

        from langchain_core.messages import SystemMessage

        # 调用模型
        all_messages = [SystemMessage(content=system_content)] + self.messages
        response = self.model.invoke(all_messages)

        # 添加 AI 响应
        self.messages.append(AIMessage(content=response.content))

        # 保存对话
        self._save_conversation()

        # 智能更新用户档案
        self._update_profile_from_conversation(user_input, response.content)

        return response.content

    def _update_profile_from_conversation(self, user_input: str, ai_response: str):
        """从对话中提取信息更新用户档案"""
        # 使用模型提取用户信息
        extraction_model = self.model.with_structured_output({
            "type": "object",
            "properties": {
                "name": {"type": "string", "description": "用户提到的姓名"},
                "preference": {
                    "type": "object",
                    "properties": {
                        "key": {"type": "string"},
                        "value": {"type": "string"}
                    }
                }
            }
        })

        try:
            extracted = extraction_model.invoke(
                f"从以下对话中提取用户信息（姓名、偏好等）：\n用户：{user_input}\nAI: {ai_response}"
            )

            if extracted.get("name") and not self.profile.get("name"):
                self.profile["name"] = extracted["name"]
                self._save_profile(self.profile)

            if extracted.get("preference"):
                if "preferences" not in self.profile:
                    self.profile["preferences"] = {}
                pref = extracted["preference"]
                self.profile["preferences"][pref["key"]] = pref["value"]
                self._save_profile(self.profile)

        except Exception as e:
            print(f"提取信息失败：{e}")

    def reset(self):
        """重置当前对话"""
        self.messages = []
        self.session_id = f"{self.user_id}:{datetime.now().strftime('%Y%m%d%H%M%S')}"

    def get_history(self) -> list:
        """获取对话历史"""
        return self.messages

    def list_sessions(self) -> list:
        """列出所有会话"""
        return list(self.store.yield_keys(
            prefix=f"conversation:{self.user_id}:"
        ))

# 使用示例
if __name__ == "__main__":
    bot = MemoryChatBot("user_001")

    while True:
        user_input = input("\n你：")
        if user_input.lower() in ["exit", "quit"]:
            break

        response = bot.chat(user_input)
        print(f"AI: {response}")

    print(f"\n当前用户档案：{bot.profile}")
    print(f"会话数量：{len(bot.list_sessions())}")

    8.5.2 记忆总结（Memory Summarization）¶

    from langchain_openai import ChatOpenAI
from langchain.storage import InMemoryStore

class SummarizingMemory:
    """带总结的记忆系统"""

    def __init__(self, user_id: str):
        self.user_id = user_id
        self.store = InMemoryStore()
        self.model = ChatOpenAI(model="gpt-4o")
        self.messages = []
        self.summary = ""

        # 加载已有总结和消息
        self._load()

    def _load(self):
        """加载记忆"""
        self.summary = self.store.get(f"memory:{self.user_id}:summary", "")
        serialized = self.store.get(f"memory:{self.user_id}:messages", [])
        from langchain_core.messages import messages_from_dict
        self.messages = messages_from_dict(serialized)

    def _save(self):
        """保存记忆"""
        self.store.set(f"memory:{self.user_id}:summary", self.summary)
        from langchain_core.messages import messages_to_dict
        self.store.set(f"memory:{self.user_id}:messages", messages_to_dict(self.messages))

    def add_message(self, role: str, content: str):
        """添加消息"""
        from langchain_core.messages import HumanMessage, AIMessage

        msg = HumanMessage(content=content) if role == "user" else AIMessage(content=content)
        self.messages.append(msg)

        # 每 10 条消息总结一次
        if len(self.messages) % 10 == 0:
            self._summarize()

        self._save()

    def _summarize(self):
        """总结对话历史"""
        summarizer = self.model.with_structured_output({
            "type": "object",
            "properties": {
                "summary": {"type": "string", "description": "对话总结，200 字以内"},
                "key_facts": {"type": "array", "items": {"type": "string"}, "description": "关键事实列表"}
            }
        })

        # 将消息转换为文本
        messages_text = "\n".join([f"{m.type}: {m.content}" for m in self.messages[-20:]])

        result = summarizer.invoke(f"总结以下对话的关键信息：\n{messages_text}")

        # 合并旧总结和新总结
        if self.summary:
            new_summary = f"之前的总结：{self.summary}\n\n新的对话：{result['summary']}"
        else:
            new_summary = result["summary"]

        self.summary = new_summary

        # 保留最近 20 条消息，其他的只保留在总结中
        self.messages = self.messages[-20:]

    def get_context(self) -> str:
        """获取当前上下文（总结 + 最近消息）"""
        context_parts = []

        if self.summary:
            context_parts.append(f"对话总结：{self.summary}")

        if self.messages:
            recent = "\n".join([f"{m.type}: {m.content}" for m in self.messages[-5:]])
            context_parts.append(f"最近对话：\n{recent}")

        return "\n\n".join(context_parts)

    import hashlib
from cryptography.fernet import Fernet

class SecureMemory:
    """加密记忆存储"""

    def __init__(self, encryption_key: bytes):
        self.fernet = Fernet(encryption_key)
        self.store = InMemoryStore()

    def set(self, key: str, value: str):
        """加密存储"""
        encrypted = self.fernet.encrypt(value.encode())
        self.store.set(f"encrypted:{key}", encrypted)

    def get(self, key: str, default=None) -> str:
        """解密读取"""
        encrypted = self.store.get(f"encrypted:{key}")
        if encrypted is None:
            return default
        return self.fernet.decrypt(encrypted).decode()

    import re

    8.6.3 性能优化¶

    from functools import lru_cache
import time

class CachedMemory:
    """带缓存的记忆系统"""

    def __init__(self, user_id: str):
        self.user_id = user_id
        self.store = InMemoryStore()
        self._cache = {}
        self._cache_time = {}
        self._cache_ttl = 300  # 5 分钟缓存

    @lru_cache(maxsize=100)
    def get_cached(self, key: str) -> any:
        """带缓存的读取"""
        # 检查缓存是否过期
        if key in self._cache_time:
            if time.time() - self._cache_time[key] < self._cache_ttl:
                return self._cache[key]

        # 从存储加载
        value = self.store.get(f"user:{self.user_id}:{key}")

        # 更新缓存
        self._cache[key] = value
        self._cache_time[key] = time.time()

        return value

    def invalidate_cache(self, key: str = None):
        """使缓存失效"""
        if key:
            self._cache.pop(key, None)
            self._cache_time.pop(key, None)
        else:
            self._cache.clear()
            self._cache_time.clear()
