    08. 实战项目¶

    项目 1：客服助手¶

    场景描述¶

    构建一个电商客服助手，能够： - 回答常见问题 - 查询订单状态 - 处理退货请求 - 转接人工客服

    完整代码¶

    from langchain.agents import create_agent
from typing import Dict
import json

# 模拟数据库
orders_db = {
    "ORD001": {"status": "已发货", "date": "2026-03-20"},
    "ORD002": {"status": "处理中", "date": "2026-03-22"},
    "ORD003": {"status": "已送达", "date": "2026-03-18"},
}

# 工具 1：查询订单状态
def check_order_status(order_id: str) -> str:
    """查询订单状态。

    Args:
        order_id: 订单号（如 ORD001）

    Returns:
        订单状态信息
    """
    order = orders_db.get(order_id.upper())
    if order:
        return f"订单 {order_id}：{order['status']}，下单日期：{order['date']}"
    return f"未找到订单 {order_id}"

# 工具 2：处理退货
def process_return(order_id: str, reason: str) -> str:
    """处理退货请求。

    Args:
        order_id: 订单号
        reason: 退货原因

    Returns:
        退货处理结果
    """
    order = orders_db.get(order_id.upper())
    if not order:
        return f"未找到订单 {order_id}"

    if order["status"] == "已送达":
        return f"退货申请已提交，订单：{order_id}，原因：{reason}。客服将在 24 小时内处理。"
    else:
        return f"订单 {order_id} 尚未送达，暂不能申请退货。"

# 工具 3：查询常见问题
def search_faq(question: str) -> str:
    """搜索常见问题解答。

    Args:
        question: 问题关键词

    Returns:
        相关 FAQ 答案
    """
    faq = {
        "运费": "满 99 元免运费，否则运费 10 元。",
        "发货": "工作日 24 小时内发货，周末和节假日顺延。",
        "退货": "支持 7 天无理由退货，商品需保持完好。",
        "发票": "确认收货后可在订单详情申请电子发票。",
        "客服": "客服工作时间：周一至周五 9:00-18:00。",
    }

    for key, answer in faq.items():
        if key in question:
            return answer

    return "抱歉，我没有找到相关信息。您可以联系人工客服。"

# 创建 Agent
system_prompt = """你是电商客服助手，负责帮助客户解决问题。

你的职责：
1. 回答常见问题（使用 search_faq）
2. 查询订单状态（使用 check_order_status）
3. 处理退货请求（使用 process_return）
4. 无法处理时转接人工客服

服务准则：
- 礼貌、专业、耐心
- 先确认客户身份和订单信息
- 提供清晰、准确的答复
- 遇到投诉或复杂问题，建议联系人工客服

可用工具：
- search_faq: 回答常见问题
- check_order_status: 查询订单
- process_return: 处理退货

如果客户情绪激动或问题复杂，请说："我理解您的情况，让我为您转接人工客服。" """

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[search_faq, check_order_status, process_return],
    system_prompt=system_prompt,
)

# 运行示例
def run_customer_service():
    """运行客服助手"""
    print("🤖 客服助手已就绪（输入 'exit' 退出）\n")

    messages = []

    while True:
        user_input = input("客户：").strip()

        if user_input.lower() in ["exit", "quit", "退出"]:
            print("客服助手：感谢您的咨询，祝您生活愉快！再见！")
            break

        messages.append({"role": "user", "content": user_input})

        result = agent.invoke({"messages": messages})

        assistant_message = result["messages"][-1]
        messages.append(assistant_message)

        print(f"客服助手：{assistant_message.content}\n")

# 测试
if __name__ == "__main__":
    run_customer_service()

    项目 2：数据分析助手¶

    场景描述¶

    构建一个数据分析助手，能够： - 执行 SQL 查询 - 生成数据可视化 - 解释分析结果 - 提供业务建议

    完整代码¶

    from langchain.agents import create_agent
import pandas as pd
from datetime import datetime, timedelta

# 模拟销售数据
def generate_sales_data():
    """生成模拟销售数据"""
    dates = pd.date_range(end=datetime.now(), periods=30, freq='D')
    data = {
        'date': dates,
        'revenue': [10000 + i * 100 + (i % 7) * 500 for i in range(30)],
        'orders': [50 + i + (i % 7) * 10 for i in range(30)],
        'customers': [30 + i // 2 for i in range(30)],
    }
    return pd.DataFrame(data)

sales_df = generate_sales_data()

# 工具 1：查询销售数据
def query_sales_data(
    start_date: str = None,
    end_date: str = None,
    metric: str = "revenue"
) -> str:
    """查询销售数据。

    Args:
        start_date: 开始日期（YYYY-MM-DD）
        end_date: 结束日期（YYYY-MM-DD）
        metric: 指标（revenue/orders/customers）

    Returns:
        查询结果
    """
    df = sales_df.copy()

    if start_date:
        df = df[df['date'] >= pd.to_datetime(start_date)]
    if end_date:
        df = df[df['date'] <= pd.to_datetime(end_date)]

    if metric not in df.columns:
        return f"未知指标：{metric}。可用指标：revenue, orders, customers"

    total = df[metric].sum()
    avg = df[metric].mean()
    count = len(df)

    return f"{metric} 统计（{count}天）：\n" \
           f"  总计：{total:,.0f}\n" \
           f"  平均：{avg:,.0f}/天\n" \
           f"  最高：{df[metric].max():,.0f}\n" \
           f"  最低：{df[metric].min():,.0f}"

# 工具 2：计算增长率
def calculate_growth(metric: str = "revenue", days: int = 7) -> str:
    """计算增长率。

    Args:
        metric: 指标名称
        days: 对比天数

    Returns:
        增长率信息
    """
    df = sales_df.copy()

    if metric not in df.columns:
        return f"未知指标：{metric}"

    recent = df.head(days)[metric].sum()
    previous = df.iloc[days:days*2][metric].sum()

    growth_rate = ((recent - previous) / previous) * 100 if previous > 0 else 0

    trend = "增长" if growth_rate > 0 else "下降"

    return f"{metric} 趋势（对比前{days}天）：\n" \
           f"  近期：{recent:,.0f}\n" \
           f"  前期：{previous:,.0f}\n" \
           f"  变化：{growth_rate:+.1f}% ({trend})"

# 工具 3：生成业务建议
def generate_insights() -> str:
    """生成业务洞察和建议。

    Returns:
        业务建议
    """
    df = sales_df.copy()

    # 计算关键指标
    avg_revenue = df['revenue'].mean()
    avg_orders = df['orders'].mean()
    avg_order_value = avg_revenue / avg_orders if avg_orders > 0 else 0

    # 周末 vs 工作日
    df['day_of_week'] = df['date'].dt.dayofweek
    weekend_avg = df[df['day_of_week'] >= 5]['revenue'].mean()
    weekday_avg = df[df['day_of_week'] < 5]['revenue'].mean()

    insights = []
    insights.append("📊 业务洞察报告")
    insights.append("=" * 40)
    insights.append(f"• 平均日收入：¥{avg_revenue:,.0f}")
    insights.append(f"• 平均日订单：{avg_orders:.0f}单")
    insights.append(f"• 平均客单价：¥{avg_order_value:.0f}")
    insights.append(f"• 周末日均收入：¥{weekend_avg:,.0f}")
    insights.append(f"• 工作日均收入：¥{weekday_avg:,.0f}")

    if weekend_avg > weekday_avg * 1.2:
        insights.append("\n💡 建议：周末表现优异，可增加周末营销投入")
    elif weekday_avg > weekend_avg * 1.2:
        insights.append("\n💡 建议：工作日表现更好，可推出工作日促销")

    return "\n".join(insights)

# 创建 Agent
system_prompt = """你是数据分析助手，帮助业务团队分析销售数据。

你的职责：
1. 查询销售数据（使用 query_sales_data）
2. 计算增长趋势（使用 calculate_growth）
3. 提供业务洞察（使用 generate_insights）
4. 用清晰的语言解释数据

分析原则：
- 数据驱动，客观准确
- 提供具体的数字支持
- 给出可执行的建议
- 指出异常和趋势

可用工具：
- query_sales_data: 查询特定时间段数据
- calculate_growth: 计算增长率
- generate_insights: 生成业务建议

回答时：
1. 先总结关键发现
2. 提供具体数据
3. 给出建议"""

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[query_sales_data, calculate_growth, generate_insights],
    system_prompt=system_prompt,
)

# 运行示例
def run_data_analyst():
    """运行数据分析助手"""
    print("🤖 数据分析助手已就绪（输入 'exit' 退出）\n")

    messages = []

    while True:
        user_input = input("问题：").strip()

        if user_input.lower() in ["exit", "quit", "退出"]:

        messages.append({"role": "user", "content": user_input})

        result = agent.invoke({"messages": messages})

        assistant_message = result["messages"][-1]

        print(f"\n{assistant_message.content}\n")

if __name__ == "__main__":

    项目 3：个人助理¶

    场景描述¶

    构建一个个人助理，能够： - 管理待办事项 - 设置提醒 - 记录笔记 - 回答问题

    完整代码¶

    from langchain.agents import create_agent
from datetime import datetime, timedelta
import json
import os

TODO_FILE = "todos.json"
NOTES_FILE = "notes.json"

# 加载数据
def load_data(filename):
    if os.path.exists(filename):
        with open(filename, 'r', encoding='utf-8') as f:
            return json.load(f)
    return []

# 保存数据
def save_data(filename, data):
    with open(filename, 'w', encoding='utf-8') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

# 工具 1：添加待办
def add_todo(task: str, priority: str = "medium", due_date: str = None) -> str:
    """添加待办事项。

    Args:
        task: 任务描述
        priority: 优先级（high/medium/low）
        due_date: 截止日期（YYYY-MM-DD）

    Returns:
        添加结果
    """
    todos = load_data(TODO_FILE)

    todo = {
        "id": len(todos) + 1,
        "task": task,
        "priority": priority,
        "due_date": due_date,
        "created_at": datetime.now().strftime("%Y-%m-%d %H:%M"),
        "completed": False,
    }

    todos.append(todo)
    save_data(TODO_FILE, todos)

    return f"✅ 已添加待办 #{todo['id']}: {task}"

# 工具 2：查看待办
def list_todos(show_completed: bool = False) -> str:
    """查看待办事项列表。

    Args:
        show_completed: 是否显示已完成的

    Returns:
        待办列表
    """
    todos = load_data(TODO_FILE)

    if not show_completed:
        todos = [t for t in todos if not t["completed"]]

    if not todos:
        return "📝 暂无待办事项"

    lines = ["📋 待办事项："]
    for todo in todos:
        status = "✅" if todo["completed"] else "⬜"
        priority_icon = {"high": "🔴", "medium": "🟡", "low": "🟢"}.get(todo["priority"], "")
        lines.append(f"{status} {priority_icon} #{todo['id']}: {todo['task']}")
        if todo["due_date"]:
            lines.append(f"   截止：{todo['due_date']}")

    return "\n".join(lines)

# 工具 3：完成待办
def complete_todo(todo_id: int) -> str:
    """标记待办为已完成。

    Args:
        todo_id: 待办 ID

    Returns:
        操作结果
    """
    todos = load_data(TODO_FILE)

    for todo in todos:
        if todo["id"] == todo_id:
            todo["completed"] = True
            todo["completed_at"] = datetime.now().strftime("%Y-%m-%d %H:%M")
            save_data(TODO_FILE, todos)
            return f"✅ 已完成待办 #{todo_id}: {todo['task']}"

    return f"❌ 未找到待办 #{todo_id}"

# 工具 4：添加笔记
def add_note(title: str, content: str) -> str:
    """添加笔记。

    Args:
        title: 笔记标题
        content: 笔记内容

    Returns:
        添加结果
    """
    notes = load_data(NOTES_FILE)

    note = {
        "id": len(notes) + 1,
        "title": title,
        "content": content,
        "created_at": datetime.now().strftime("%Y-%m-%d %H:%M"),
    }

    notes.append(note)
    save_data(NOTES_FILE, notes)

    return f"📝 已添加笔记 #{note['id']}: {title}"

# 工具 5：搜索笔记
def search_notes(keyword: str) -> str:
    """搜索笔记。

    Args:
        keyword: 搜索关键词

    Returns:
        搜索结果
    """
    notes = load_data(NOTES_FILE)

    results = [n for n in notes if keyword.lower() in n["title"].lower() 
               or keyword.lower() in n["content"].lower()]

    if not results:
        return f"📝 未找到包含 '{keyword}' 的笔记"

    lines = [f"🔍 找到 {len(results)} 篇笔记："]
    for note in results:
        lines.append(f"\n📌 #{note['id']}: {note['title']}")
        lines.append(f"   {note['content'][:100]}...")

    return "\n".join(lines)

# 工具 6：获取时间信息
def get_time_info() -> str:
    """获取当前时间信息。

    Returns:
        时间信息
    """
    now = datetime.now()
    return f"当前时间：{now.strftime('%Y年%m月%d日 %H:%M')}\n" \
           f"星期：{now.strftime('%A')}"

# 创建 Agent
system_prompt = """你是个人助理，帮助用户管理日常事务。

你的职责：
1. 管理待办事项（添加、查看、完成）
2. 管理笔记（添加、搜索）
3. 提供时间信息
4. 回答一般问题

工作原则：
- 友好、乐于助人
- 主动提醒重要事项
- 帮助用户保持条理
- 记住用户的偏好

可用工具：
- add_todo: 添加待办
- list_todos: 查看待办
- complete_todo: 完成待办
- add_note: 添加笔记
- search_notes: 搜索笔记
- get_time_info: 获取时间

主动提醒：
- 如果有即将到期的待办，主动提醒
- 如果用户说"谢谢"，礼貌回应"""

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[add_todo, list_todos, complete_todo, add_note, search_notes, get_time_info],
    system_prompt=system_prompt,
)

# 运行示例
def run_personal_assistant():
    """运行个人助理"""
    print("🤖 个人助理已就绪（输入 'exit' 退出）\n")

    messages = []

    while True:
        user_input = input("你：").strip()

        if user_input.lower() in ["exit", "quit", "退出"]:
            print("助理：祝您有愉快的一天！再见！")
            break

        messages.append({"role": "user", "content": user_input})

        result = agent.invoke({"messages": messages})

        assistant_message = result["messages"][-1]
        messages.append(assistant_message)

        print(f"\n助理：{assistant_message.content}\n")

if __name__ == "__main__":
    run_personal_assistant()

# 3. 限流保护
from functools import wraps
import time

def rate_limit(max_calls, period):
    def decorator(func):
        calls = []
        @wraps(func)
        def wrapper(*args):
            now = time.time()
            calls[:] = [t for t in calls if now - t < period]
            if len(calls) >= max_calls:
                raise Exception("Rate limit exceeded")
            calls.append(now)
            return func(*args)
        return wrapper
    return decorator

    最佳实践总结¶

    ✅ 推荐¶

    1. 明确定义 Agent 职责
2. 编写清晰的工具文档
3. 添加适当的错误处理
4. 使用环境变量管理密钥
5. 启用 LangSmith 追踪
6. 定期测试和评估
7. 监控成本和性能

    ❌ 避免¶

    1. 硬编码敏感信息
2. 忽略错误处理
3. 不限制上下文长度
4. 不监控使用情况
5. 过度依赖单一工具
6. 忽略用户隐私
