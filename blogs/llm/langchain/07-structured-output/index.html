    7.1 为什么需要结构化输出¶

    7.1.1 非结构化输出的问题¶

    from langchain_openai import ChatOpenAI

model = ChatOpenAI(model="gpt-4o")

response = model.invoke("提取以下信息的姓名、年龄和邮箱：张三，25 岁，zhangsan@example.com")
print(response.content)
# 输出可能是：
# "姓名：张三\n年龄：25 岁\n邮箱：zhangsan@example.com"
# 或者：
# "张三，25 岁，zhangsan@example.com"
# 或者任何其他格式...

    问题：每次输出格式可能不同，难以程序化处理。

    7.1.2 结构化输出的优势¶

    from pydantic import BaseModel, Field

class PersonInfo(BaseModel):
    name: str = Field(description="姓名")
    age: int = Field(description="年龄")
    email: str = Field(description="电子邮箱")

model = ChatOpenAI(model="gpt-4o")
structured_model = model.with_structured_output(PersonInfo)

response = structured_model.invoke(
    "提取以下信息的姓名、年龄和邮箱：张三，25 岁，zhangsan@example.com"
)

print(response)
# PersonInfo(name='张三', age=25, email='zhangsan@example.com')

# 可以直接访问属性
print(f"姓名：{response.name}")
print(f"年龄：{response.age}")
print(f"邮箱：{response.email}")

    优势： - ✅ 格式统一，易于解析 - ✅ 类型安全，IDE 自动补全 - ✅ 可直接用于后续处理 - ✅ 自动验证数据有效性

    7.2 使用 Pydantic 定义输出结构¶

    7.2.1 基础 Pydantic 模型¶

    from pydantic import BaseModel, Field
from langchain_openai import ChatOpenAI

# 定义输出结构
class WeatherResponse(BaseModel):
    city: str = Field(description="城市名称")
    temperature: float = Field(description="温度（摄氏度）")
    condition: str = Field(description="天气状况，如：晴、多云、雨")
    humidity: int = Field(description="湿度百分比")

# 创建结构化模型
model = ChatOpenAI(model="gpt-4o")
structured_model = model.with_structured_output(WeatherResponse)

# 调用
response = structured_model.invoke("北京今天天气如何？")
print(response)
# WeatherResponse(city='北京', temperature=25.0, condition='晴', humidity=45)

    7.2.2 字段描述的重要性¶

    from pydantic import BaseModel, Field

# ❌ 不好的做法：缺少描述
class BadResponse(BaseModel):
    name: str
    value: int

# ✅ 好的做法：详细描述
class GoodResponse(BaseModel):
    product_name: str = Field(
        description="产品全称，包括品牌和型号",
        examples=["iPhone 15 Pro", "MacBook Pro 14 英寸"]
    )
    price: float = Field(
        description="产品价格，单位为人民币元",
        ge=0,  # 必须大于等于 0
        examples=[7999.00, 12999.00]
    )
    category: str = Field(
        description="产品类别",
        enum=["手机", "电脑", "平板", "配件"]  # 限定可选值
    )
    in_stock: bool = Field(
        description="是否有库存",
        default=True
    )

    7.2.3 可选字段和默认值¶

    from typing import Optional, List
from pydantic import BaseModel, Field

class UserProfile(BaseModel):
    username: str = Field(description="用户名")
    email: str = Field(description="邮箱")

    # 可选字段
    phone: Optional[str] = Field(
        default=None,
        description="手机号码（可选）"
    )

    # 带默认值
    age: int = Field(
        default=0,
        description="年龄"
    )

    # 列表字段
    tags: List[str] = Field(
        default_factory=list,
        description="用户标签列表"
    )

    # 嵌套对象
    address: Optional["Address"] = Field(
        default=None,
        description="地址信息"
    )

class Address(BaseModel):
    province: str = Field(description="省份")
    city: str = Field(description="城市")
    district: str = Field(description="区县")
    detail: str = Field(description="详细地址")

# 更新前向引用
UserProfile.model_rebuild()

    7.3 使用 TypedDict 定义输出结构¶

    from typing import TypedDict, List, Optional
from langchain_openai import ChatOpenAI

# 定义 TypedDict
class WeatherData(TypedDict):
    city: str
    temperature: float
    condition: str
    humidity: int
    forecast: List[str]  # 未来几天预报

# 创建结构化模型
model = ChatOpenAI(model="gpt-4o")
structured_model = model.with_structured_output(WeatherData)

# 调用
response = structured_model.invoke("上海天气预报")
print(response)
# {'city': '上海', 'temperature': 28.5, 'condition': '多云', 
#  'humidity': 65, 'forecast': ['晴', '小雨', '多云']}

# 访问数据
print(f"城市：{response['city']}")
print(f"温度：{response['temperature']}")

    7.3.1 TypedDict vs Pydantic¶

    特性 Pydantic TypedDict
    运行时验证 ✅ 支持 ❌ 不支持
    默认值 ✅ 支持 ❌ 不支持
    字段描述 ✅ Field(description=...) ❌ 不支持
    数据验证 ✅ 自动验证 ❌ 需要额外库
    嵌套结构 ✅ 完善支持 ✅ 支持
    推荐场景 复杂结构、需要验证 简单结构、轻量级

    建议：优先使用 Pydantic，除非需要极简结构。

    7.4.3 联合类型和枚举¶

    from typing import Union, Literal
from pydantic import BaseModel, Field

class SuccessResponse(BaseModel):
    status: Literal["success"] = "success"
    data: str = Field(description="成功返回的数据")

class ErrorResponse(BaseModel):
    status: Literal["error"] = "error"
    error_code: int = Field(description="错误代码")
    error_message: str = Field(description="错误信息")

# 联合类型
ApiResponse = Union[SuccessResponse, ErrorResponse]

model = ChatOpenAI(model="gpt-4o")
structured_model = model.with_structured_output(ApiResponse)

# 根据输入，模型会返回合适的类型
response1 = structured_model.invoke("请求成功，返回数据：hello")
response2 = structured_model.invoke("请求失败，错误代码 404，未找到资源")

print(type(response1))  # <class 'SuccessResponse'>
print(type(response2))  # <class 'ErrorResponse'>

    7.5 实战：信息提取器¶

    7.5.1 简历信息提取¶

    from typing import List, Optional
from pydantic import BaseModel, Field
from langchain_openai import ChatOpenAI

class Education(BaseModel):
    school: str = Field(description="学校名称")
    degree: str = Field(description="学位，如：本科、硕士、博士")
    major: str = Field(description="专业")
    start_date: str = Field(description="开始日期，格式：YYYY-MM")
    end_date: Optional[str] = Field(default=None, description="结束日期")

class WorkExperience(BaseModel):
    company: str = Field(description="公司名称")
    position: str = Field(description="职位")
    start_date: str = Field(description="开始日期")
    end_date: Optional[str] = Field(default=None, description="结束日期")
    responsibilities: List[str] = Field(description="工作职责列表")

class Skill(BaseModel):
    name: str = Field(description="技能名称")
    proficiency: str = Field(
        description="熟练程度",
        enum=["入门", "熟悉", "精通", "专家"]
    )

class Resume(BaseModel):
    name: str = Field(description="姓名")
    email: str = Field(description="邮箱")
    phone: str = Field(description="电话")
    summary: str = Field(description="个人简介")
    education: List[Education] = Field(description="教育经历")
    work_experience: List[WorkExperience] = Field(description="工作经历")
    skills: List[Skill] = Field(description="技能列表")

# 创建提取器
model = ChatOpenAI(model="gpt-4o")
resume_extractor = model.with_structured_output(Resume)

# 示例简历文本
resume_text = """
张三，邮箱 zhangsan@email.com，电话 138-0013-8000

个人简介：
5 年 Python 开发经验，擅长后端开发和数据分析。

教育经历：
2014.09 - 2018.06  北京大学  计算机科学与技术  本科

工作经历：
2018.07 - 2021.06  阿里巴巴  后端开发工程师
- 负责电商系统后端开发
- 优化数据库查询性能
- 设计和实现 RESTful API

2021.07 - 至今  字节跳动  高级 Python 工程师
- 领导 5 人团队开发数据平台
- 设计微服务架构
- 推动代码审查和单元测试

技能：
Python（精通）、Java（熟悉）、MySQL（精通）、
Redis（熟悉）、Docker（熟悉）、Kubernetes（入门）
"""

# 提取信息
resume = resume_extractor.invoke(resume_text)

print(f"姓名：{resume.name}")
print(f"邮箱：{resume.email}")
print(f"\n教育经历：")
for edu in resume.education:
    print(f"  {edu.school} - {edu.degree} - {edu.major}")
print(f"\n工作经历：")
for work in resume.work_experience:
    print(f"  {work.company} - {work.position}")
print(f"\n技能：")
for skill in resume.skills:
    print(f"  {skill.name}: {skill.proficiency}")

    7.5.2 新闻文章分析¶

    from typing import List
from pydantic import BaseModel, Field

class NewsAnalysis(BaseModel):
    title: str = Field(description="新闻标题")
    summary: str = Field(description="文章摘要，100 字以内")
    key_points: List[str] = Field(description="关键要点列表")
    entities: List[str] = Field(description="提到的人物、组织、地点")
    sentiment: str = Field(
        description="情感倾向",
        enum=["正面", "中性", "负面"]
    )
    category: str = Field(
        description="新闻类别",
        enum=["政治", "经济", "科技", "体育", "娱乐", "其他"]
    )

model = ChatOpenAI(model="gpt-4o")
news_analyzer = model.with_structured_output(NewsAnalysis)

news_text = """
【科技】OpenAI 发布新一代大模型 GPT-5

昨日，OpenAI 公司在旧金山举行发布会，正式推出了新一代大语言模型 GPT-5。
据 OpenAI CEO 山姆·阿尔特曼介绍，GPT-5 在推理能力、代码生成和多模态理解
方面都有显著提升。新模型将于下月向 Plus 用户开放，预计将推动 AI 应用的
进一步发展。分析师认为，这可能会加剧 AI 领域的竞争。
"""

analysis = news_analyzer.invoke(news_text)

print(f"标题：{analysis.title}")
print(f"摘要：{analysis.summary}")
print(f"关键点：{analysis.key_points}")
print(f"情感：{analysis.sentiment}")
print(f"类别：{analysis.category}")

    7.5.3 产品评论分析¶

    from typing import List, Optional
from pydantic import BaseModel, Field

class ProductReview(BaseModel):
    product_name: str = Field(description="产品名称")
    rating: int = Field(description="评分，1-5 分", ge=1, le=5)
    pros: List[str] = Field(description="优点列表")
    cons: List[str] = Field(description="缺点列表")
    would_recommend: bool = Field(description="是否推荐购买")
    summary: str = Field(description="评论总结")

model = ChatOpenAI(model="gpt-4o")
review_analyzer = model.with_structured_output(ProductReview)

review_text = """
这款 iPhone 15 Pro 用了一个月，整体很满意。

优点：
- A17 Pro 芯片性能强劲
- 钛金属边框手感好，重量轻
- 相机升级明显，5 倍长焦很实用
- USB-C 接口终于来了

缺点：
- 价格还是太贵
- 发热问题依然存在
- 电池续航一般

总体来说是一款优秀的旗舰手机，但性价比不高。
"""

analysis = review_analyzer.invoke(review_text)

print(f"产品：{analysis.product_name}")
print(f"评分：{analysis.rating}/5")
print(f"推荐：{'是' if analysis.would_recommend else '否'}")
print(f"\n优点：")
for pro in analysis.pros:
    print(f"  ✓ {pro}")
print(f"\n缺点：")
for con in analysis.cons:
    print(f"  ✗ {con}")

    7.6 实战：数据转换器¶

    7.6.1 JSON 转 Pydantic¶

    from pydantic import BaseModel, Field, ValidationError
from typing import List

class User(BaseModel):
    id: int = Field(description="用户 ID")
    name: str = Field(description="姓名")
    email: str = Field(description="邮箱")
    is_active: bool = Field(description="是否激活")

# 原始 JSON 数据
json_data = """
[
    {"id": 1, "name": "张三", "email": "zhangsan@example.com", "is_active": true},
    {"id": 2, "name": "李四", "email": "lisi@example.com", "is_active": false},
    {"id": 3, "name": "王五", "email": "wangwu@example.com", "is_active": true}
]
"""

import json
data = json.loads(json_data)

# 转换为 Pydantic 模型
users = [User(**item) for item in data]

for user in users:
    print(f"{user.name} ({user.email}) - {'激活' if user.is_active else '未激活'}")

# 数据验证
try:
    invalid_user = User(id="abc", name="测试", email="test@example.com", is_active=True)
except ValidationError as e:
    print(f"验证失败：{e}")

    7.6.2 自然语言转 SQL¶

    from pydantic import BaseModel, Field
from typing import Optional

class SQLQuery(BaseModel):
    query: str = Field(description="生成的 SQL 查询语句")
    explanation: str = Field(description="查询说明")
    tables_used: list[str] = Field(description="涉及的表名")

model = ChatOpenAI(model="gpt-4o")
sql_generator = model.with_structured_output(SQLQuery)

# 表结构信息
schema_info = """
数据库表结构：
- users (id, name, email, created_at)
- orders (id, user_id, amount, status, created_at)
- products (id, name, price, stock)
"""

query = sql_generator.invoke(f"""
基于以下表结构：
{schema_info}

生成 SQL 查询：找出最近 30 天内消费总额超过 1000 元的用户，
按消费总额降序排列，返回用户姓名、邮箱和消费总额。
""")

print(f"SQL: {query.query}")
print(f"说明：{query.explanation}")
print(f"表：{query.tables_used}")

    7.7.2 重试机制¶

    from tenacity import retry, stop_after_attempt, wait_exponential

class DataExtractor:
    def __init__(self, model_name="gpt-4o"):
        self.model = ChatOpenAI(model=model_name)

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=1, max=10)
    )
    def extract_with_retry(self, text, output_schema):
        """带重试的提取"""
        structured_model = self.model.with_structured_output(output_schema)
        return structured_model.invoke(text)

# 使用
extractor = DataExtractor()
result = extractor.extract_with_retry(resume_text, Resume)

    7.8.1 Schema 设计原则¶

    # ✅ 好的 Schema 设计

# 避免每次都创建
