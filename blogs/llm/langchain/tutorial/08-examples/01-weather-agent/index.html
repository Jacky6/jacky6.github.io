    # config.py
import os
from dataclasses import dataclass

@dataclass
class Config:
    """配置"""
    # OpenAI
    openai_api_key: str = os.getenv("OPENAI_API_KEY", "")

    # 天气 API（使用免费 API）
    weather_api_key: str = os.getenv("WEATHER_API_KEY", "")
    weather_base_url: str = "https://api.openweathermap.org/data/2.5"

    # 模型配置
    model_name: str = "gpt-5.5"
    temperature: float = 0.7

# tools.py
from langchain_core.tools import tool
import requests
from config import Config

@tool
def get_current_weather(city: str) -> str:
    """获取当前天气

    Args:
        city: 城市名称，如"北京"、"Shanghai"

    Returns:
        天气信息 JSON 字符串
    """
    try:
        url = f"{Config.weather_base_url}/weather"
        params = {
            "q": city,
            "appid": Config.weather_api_key,
            "units": "metric",
            "lang": "zh_cn"
        }

        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()
        data = response.json()

        return f"""城市：{data['name']}
天气：{data['weather'][0]['description']}
温度：{data['main']['temp']}°C
体感温度：{data['main']['feels_like']}°C
湿度：{data['main']['humidity']}%
风速：{data['wind']['speed']}m/s
"""
    except Exception as e:
        return f"天气查询失败：{str(e)}"

@tool
def get_weather_forecast(city: str, days: int = 3) -> str:
    """获取天气预报

    Args:
        city: 城市名称
        days: 预报天数（1-5）

    Returns:
        天气预报信息
    """
    try:
        url = f"{Config.weather_base_url}/forecast"
        params = {
            "q": city,
            "appid": Config.weather_api_key,
            "units": "metric",
            "lang": "zh_cn"
        }

        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()
        data = response.json()

        # 解析预报（每 3 小时一个数据）
        forecasts = []
        for item in data['list'][:days*8]:  # 每天 8 个数据点
            date = item['dt_txt'].split()[0]
            temp = item['main']['temp']
            desc = item['weather'][0]['description']

            if date not in [f['date'] for f in forecasts]:
                forecasts.append({
                    'date': date,
                    'temp': temp,
                    'desc': desc
                })

        result = f"{city}未来{days}天预报：\n"
        for f in forecasts:
            result += f"{f['date']}: {f['desc']}, {f['temp']}°C\n"

        return result
    except Exception as e:
        return f"天气预报查询失败：{str(e)}"

@tool
def get_weather_advice(weather_info: str) -> str:
    """根据天气信息提供建议

    Args:
        weather_info: 天气信息

    Returns:
        生活建议
    """
    advice = []

    # 温度建议
    if "温度" in weather_info:
        import re
        temps = re.findall(r'温度[:：]\s*(-?\d+\.?\d*)', weather_info)
        if temps:
            temp = float(temps[0])
            if temp < 0:
                advice.append("❄️ 天气严寒，注意保暖，避免长时间户外活动")
            elif temp < 10:
                advice.append("🧥 天气较冷，建议穿厚外套")
            elif temp < 20:
                advice.append("👕 天气凉爽，适合穿长袖")
            elif temp < 30:
                advice.append("😊 天气舒适，穿着单薄即可")
            else:
                advice.append("☀️ 天气炎热，注意防暑降温")

    # 天气状况建议
    if "雨" in weather_info or "雪" in weather_info:
        advice.append("☔ 有降水，请携带雨具")

    if "雾" in weather_info:
        advice.append("🌫️ 有雾，出行注意交通安全")

    # 湿度建议
    if "湿度" in weather_info:
        import re
        humidity = re.findall(r'湿度[:：]\s*(\d+)', weather_info)
        if humidity and int(humidity[0]) > 80:
            advice.append("💧 湿度较大，注意防潮")
        elif humidity and int(humidity[0]) < 30:
            advice.append("🏜️ 空气干燥，注意补水")

    # 风速建议
    if "风速" in weather_info:
        import re
        wind = re.findall(r'风速[:：]\s*(\d+\.?\d*)', weather_info)
        if wind and float(wind[0]) > 10:
            advice.append("💨 风力较大，注意防风")

    if not advice:
        advice.append("🌤️ 天气良好，适合户外活动")

    return "\n".join(advice)

# agent.py
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langchain_core.tools import tool
from typing import List
from config import Config
from tools import get_current_weather, get_weather_forecast, get_weather_advice

class WeatherAgent:
    """天气 Agent"""

    def __init__(self):
        self.model = ChatOpenAI(
            model=Config.model_name,
            temperature=Config.temperature,
        )

        self.tools = [
            get_current_weather,
            get_weather_forecast,
            get_weather_advice,
        ]

        self.model_with_tools = self.model.bind_tools(self.tools)

        self.system_prompt = """你是一个专业的天气助手。
你可以帮助用户查询天气信息并提供生活建议。

请使用提供的工具来回答用户的问题。
如果工具返回错误，请友好地告知用户。

回答时请：
1. 先给出天气概况
2. 然后提供具体数据
3. 最后给出生活建议
"""

        self.history: List = [
            SystemMessage(content=self.system_prompt)
        ]

    def chat(self, user_input: str) -> str:
        """对话"""
        self.history.append(HumanMessage(content=user_input))

        # 调用模型
        response = self.model_with_tools.invoke(self.history)
        self.history.append(response)

        # 处理工具调用
        while response.tool_calls:
            for tool_call in response.tool_calls:
                # 执行工具
                tool_name = tool_call["name"]
                tool_args = tool_call["args"]

                selected_tool = {
                    "get_current_weather": get_current_weather,
                    "get_weather_forecast": get_weather_forecast,
                    "get_weather_advice": get_weather_advice,
                }.get(tool_name)

                if selected_tool:
                    result = selected_tool.invoke(tool_args)
                else:
                    result = f"未知工具：{tool_name}"

                # 添加工具结果
                from langchain_core.messages import ToolMessage
                self.history.append(
                    ToolMessage(content=result, tool_call_id=tool_call["id"])
                )

            # 继续对话
            response = self.model_with_tools.invoke(self.history)
            self.history.append(response)

        return response.content

    def reset(self):
        """重置对话"""
        self.history = [SystemMessage(content=self.system_prompt)]

# main.py
from agent import WeatherAgent

def main():
    """主函数"""
    print("🌤️  天气助手")
    print("输入 'quit' 退出，'reset' 重置对话\n")

    agent = WeatherAgent()

    while True:
        try:
            user_input = input("你：").strip()

            if not user_input:
                continue

            if user_input.lower() == "quit":
                print("再见！")
                break

            if user_input.lower() == "reset":
                agent.reset()
                print("对话已重置")
                continue

            response = agent.chat(user_input)
            print(f"AI: {response}")

        except KeyboardInterrupt:
            print("\n再见！")
            break
        except Exception as e:
            print(f"错误：{e}")

if __name__ == "__main__":
    main()
