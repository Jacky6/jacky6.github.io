# tools.py
from langchain_core.tools import tool
import pandas as pd
import numpy as np
from typing import Optional
import matplotlib.pyplot as plt
import io
import base64

@tool
def load_csv(file_path: str) -> str:
    """加载 CSV 文件

    Args:
        file_path: CSV 文件路径

    Returns:
        数据概览
    """
    try:
        df = pd.read_csv(file_path)

        info = f"""文件：{file_path}
行数：{len(df):,}
列数：{len(df.columns)}

列名：{list(df.columns)}

前 5 行预览：
{df.head().to_string()}

数据类型：
{df.dtypes.to_string()}

缺失值统计：
{df.isnull().sum().to_string()}
"""
        return info
    except Exception as e:
        return f"加载失败：{str(e)}"

@tool
def describe_data(file_path: str, column: str = None) -> str:
    """描述性统计

    Args:
        file_path: 数据文件路径
        column: 可选，指定列名

    Returns:
        统计信息
    """
    try:
        df = pd.read_csv(file_path)

        if column:
            if column not in df.columns:
                return f"列 '{column}' 不存在。可用列：{list(df.columns)}"

            df = df[column]

        if df.select_dtypes(include=[np.number]).empty:
            return "没有数值列可分析"

        stats = df.describe()
        return f"统计信息：\n{stats.to_string()}"
    except Exception as e:
        return f"统计失败：{str(e)}"

@tool
def filter_data(file_path: str, condition: str) -> str:
    """过滤数据

    Args:
        file_path: 数据文件路径
        condition: 过滤条件，如"age > 25"、"city == '北京'"

    Returns:
        过滤后的数据概览
    """
    try:
        df = pd.read_csv(file_path)
        filtered = df.query(condition)

        return f"""过滤条件：{condition}
原始行数：{len(df):,}
过滤后行数：{len(filtered):,}

前 5 行：
{filtered.head().to_string()}
"""
    except Exception as e:
        return f"过滤失败：{str(e)}"

@tool
def group_and_aggregate(file_path: str, group_by: str, agg_column: str, agg_func: str = "mean") -> str:
    """分组聚合

    Args:
        file_path: 数据文件路径
        group_by: 分组列
        agg_column: 聚合列
        agg_func: 聚合函数 (sum, mean, count, min, max)

    Returns:
        聚合结果
    """
    try:
        df = pd.read_csv(file_path)

        if group_by not in df.columns:
            return f"分组列 '{group_by}' 不存在"
        if agg_column not in df.columns:
            return f"聚合列 '{agg_column}' 不存在"

        grouped = df.groupby(group_by)[agg_column].agg(agg_func).reset_index()
        grouped = grouped.sort_values(agg_column, ascending=False)

        return f"""分组：{group_by}
聚合：{agg_column} ({agg_func})

结果：
{grouped.to_string(index=False)}
"""
    except Exception as e:
        return f"聚合失败：{str(e)}"

@tool
def create_chart(file_path: str, chart_type: str, x_column: str, y_column: str, title: str = None) -> str:
    """创建图表

    Args:
        file_path: 数据文件路径
        chart_type: 图表类型 (bar, line, scatter, pie, histogram)
        x_column: X 轴列
        y_column: Y 轴列
        title: 图表标题

    Returns:
        图表保存路径
    """
    try:
        df = pd.read_csv(file_path)

        plt.figure(figsize=(10, 6))

        if chart_type == "bar":
            plt.bar(df[x_column].astype(str), df[y_column])
        elif chart_type == "line":
            plt.plot(df[x_column], df[y_column], marker='o')
        elif chart_type == "scatter":
            plt.scatter(df[x_column], df[y_column])
        elif chart_type == "histogram":
            plt.hist(df[y_column], bins=20, edgecolor='black')
        elif chart_type == "pie":
            plt.pie(df[y_column], labels=df[x_column], autopct='%1.1f%%')
        else:
            return f"未知图表类型：{chart_type}"

        plt.title(title or f"{chart_type} Chart: {x_column} vs {y_column}")
        plt.xticks(rotation=45)
        plt.tight_layout()

        # 保存图表
        output_path = f"/tmp/chart_{int(pd.Timestamp.now().timestamp())}.png"
        plt.savefig(output_path, dpi=150)
        plt.close()

        return f"图表已保存：{output_path}"
    except Exception as e:
        return f"创建图表失败：{str(e)}"

@tool
def correlation_analysis(file_path: str) -> str:
    """相关性分析

    Args:
        file_path: 数据文件路径

    Returns:
        相关性矩阵
    """
    try:
        df = pd.read_csv(file_path)

        # 只分析数值列
        numeric_df = df.select_dtypes(include=[np.number])

        if numeric_df.empty:
            return "没有数值列可分析"

        corr = numeric_df.corr()

        return f"""相关系数矩阵：
{corr.round(2).to_string()}

强相关 (>0.7 或 <-0.7)：
"""
        # 找出强相关
        strong_corr = []
        for i in range(len(corr.columns)):
            for j in range(i+1, len(corr.columns)):
                if abs(corr.iloc[i, j]) > 0.7:
                    strong_corr.append(
                        f"{corr.columns[i]} ↔ {corr.columns[j]}: {corr.iloc[i, j]:.2f}"
                    )

        return "相关系数矩阵:\n" + corr.round(2).to_string() + "\n\n强相关对:\n" + "\n".join(strong_corr) if strong_corr else "无强相关"
    except Exception as e:
        return f"相关性分析失败：{str(e)}"

@tool
def execute_sql_query(file_path: str, query: str) -> str:
    """执行 SQL 查询

    Args:
        file_path: CSV 文件路径
        query: SQL 查询语句

    Returns:
        查询结果
    """
    try:
        from pandasql import sqldf

        df = pd.read_csv(file_path)
        table_name = file_path.split('/')[-1].replace('.csv', '')

        # 创建局部变量供 SQL 使用
        local_context = {table_name: df, 'pd': pd}

        result = sqldf(query, local_context)

        return f"""SQL: {query}

结果 ({len(result)} 行):
{result.to_string(index=False)}
"""
    except Exception as e:
        return f"SQL 查询失败：{str(e)}"

# agent.py
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from typing import List
from config import Config
from tools import (
    load_csv, describe_data, filter_data,
    group_and_aggregate, create_chart,
    correlation_analysis, execute_sql_query
)

class DataAnalystAgent:
    """数据分析 Agent"""

    def __init__(self):
        self.model = ChatOpenAI(
            model=Config.model_name,
            temperature=Config.temperature,
        )

        self.tools = [
            load_csv,
            describe_data,
            filter_data,
            group_and_aggregate,
            create_chart,
            correlation_analysis,
            execute_sql_query,
        ]

        self.model_with_tools = self.model.bind_tools(self.tools)

        self.system_prompt = """你是一个专业的数据分析助手。
你可以帮助用户分析 CSV 数据文件，提供统计信息和可视化。

可用工具：
- load_csv: 加载 CSV 文件
- describe_data: 描述性统计
- filter_data: 过滤数据
- group_and_aggregate: 分组聚合
- create_chart: 创建图表 (bar/line/scatter/pie/histogram)
- correlation_analysis: 相关性分析
- execute_sql_query: SQL 查询

回答时请：
1. 先加载数据了解结构
2. 根据用户需求选择合适的分析工具
3. 解释分析结果的含义
4. 必要时提供可视化

注意：
- 文件路径必须是完整的绝对路径或相对路径
- 列名需要准确匹配
- 图表会保存到 /tmp/ 目录
"""

        self.history: List = [
            SystemMessage(content=self.system_prompt)
        ]

        self.current_file: str = None

    def chat(self, user_input: str) -> str:
        """对话"""
        # 如果有文件路径，更新上下文
        if ".csv" in user_input and not self.current_file:
            import re
            match = re.search(r'[\w./-]+\.csv', user_input)
            if match:
                self.current_file = match.group()

        self.history.append(HumanMessage(content=user_input))

        # 调用模型
        response = self.model_with_tools.invoke(self.history)
        self.history.append(response)

        # 处理工具调用
        while response.tool_calls:
            for tool_call in response.tool_calls:
                tool_name = tool_call["name"]
                tool_args = tool_call["args"]

                # 自动添加文件路径
                if "file_path" in tool_args and not tool_args["file_path"]:
                    tool_args["file_path"] = self.current_file

                selected_tool = {t.name: t for t in self.tools}.get(tool_name)

                if selected_tool:
                    result = selected_tool.invoke(tool_args)
                else:
                    result = f"未知工具：{tool_name}"

                from langchain_core.messages import ToolMessage
                self.history.append(
                    ToolMessage(content=result, tool_call_id=tool_call["id"])
                )

            response = self.model_with_tools.invoke(self.history)
            self.history.append(response)

        return response.content

    def reset(self):
        """重置"""
        self.history = [SystemMessage(content=self.system_prompt)]
        self.current_file = None

# main.py
from agent import DataAnalystAgent

def main():
    print("📊 数据分析助手")
    print("支持 CSV 文件分析、统计、可视化\n")
    print("输入 'quit' 退出，'reset' 重置\n")

    agent = DataAnalystAgent()

    while True:
        try:
            user_input = input("你：").strip()

            if not user_input:
                continue

            if user_input.lower() == "quit":
                print("再见！")
                break

            if user_input.lower() == "reset":
                agent.reset()
                print("已重置")
                continue

            response = agent.chat(user_input)
            print(f"\nAI: {response}\n")

        except KeyboardInterrupt:
            print("\n再见！")
            break
        except Exception as e:
            print(f"错误：{e}")

if __name__ == "__main__":
    main()
