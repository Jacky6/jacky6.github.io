    11.3 上下文传递¶

    11.3.1 在调用链中传递上下文¶

    from typing import TypedDict, Any
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

class Context(TypedDict, total=False):
    user_id: str
    request_id: str
    custom_data: dict

def process_with_context(model: ChatOpenAI, messages: list, context: Context):
    """带上下文的处理"""

    # 在调用中传递上下文
    config = {
        "configurable": {
            "user_id": context.get("user_id"),
            "request_id": context.get("request_id"),
        }
    }

    response = model.invoke(messages, config=config)
    return response

# 使用
model = ChatOpenAI(model="gpt-5.5")
context: Context = {
    "user_id": "user_123",
    "request_id": "req_001",
    "custom_data": {"key": "value"},
}

response = process_with_context(model, [HumanMessage(content="你好")], context)

    11.3.2 上下文管理器¶

    from contextlib import contextmanager
from typing import TypedDict, Any, Dict

class ContextManager:
    """上下文管理器"""

    def __init__(self):
        self._context_stack = []
        self._global_context: Dict[str, Any] = {}

    def set_global(self, key: str, value: Any):
        """设置全局上下文"""
        self._global_context[key] = value

    def get_global(self, key: str, default=None) -> Any:
        """获取全局上下文"""
        return self._global_context.get(key, default)

    @contextmanager
    def scoped_context(self, **kwargs):
        """作用域上下文"""
        # 压入新上下文
        self._context_stack.append(kwargs)
        try:
            yield
        finally:
            # 弹出上下文
            self._context_stack.pop()

    def get_current_context(self) -> Dict[str, Any]:
        """获取当前上下文（合并所有层级）"""
        context = self._global_context.copy()

        # 合并所有作用域上下文
        for scoped in self._context_stack:
            context.update(scoped)

        return context

# 使用
ctx_manager = ContextManager()

# 设置全局上下文
ctx_manager.set_global("app_id", "my_app")
ctx_manager.set_global("environment", "prod")

# 使用作用域上下文
with ctx_manager.scoped_context(user_id="user_123", request_id="req_001"):
    current = ctx_manager.get_current_context()
    print(f"作用域内：{current}")
    # {'app_id': 'my_app', 'environment': 'prod', 'user_id': 'user_123', 'request_id': 'req_001'}

# 作用域外
current = ctx_manager.get_current_context()
print(f"作用域外：{current}")
# {'app_id': 'my_app', 'environment': 'prod'}

        executor.submit(worker, i)

    11.4 上下文感知的 Agent¶

    11.4.1 基础实现¶

    from typing import TypedDict, Optional, List
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage

class AgentContext(TypedDict, total=False):
    """Agent 上下文"""
    user_id: str
    user_name: str
    session_id: str
    conversation_history: List[dict]
    preferences: dict
    constraints: List[str]

class ContextAwareAgent:
    """上下文感知的 Agent"""

    def __init__(self, context: AgentContext = None):
        self.model = ChatOpenAI(model="gpt-5.5")
        self.context = context or {}
        self.messages: List = []

        # 初始化系统消息
        self._update_system_message()

    def _update_system_message(self):
        """根据上下文更新系统消息"""
        context_parts = []

        # 用户信息
        if self.context.get("user_name"):
            context_parts.append(f"当前用户：{self.context['user_name']}")

        # 用户偏好
        prefs = self.context.get("preferences", {})
        if prefs:
            context_parts.append(f"用户偏好：{prefs}")

        # 约束条件
        constraints = self.context.get("constraints", [])
        if constraints:
            context_parts.append(f"约束：{', '.join(constraints)}")

        system_content = "你是一个智能助手。\n\n"
        if context_parts:
            system_content += "\n".join(context_parts)

        # 更新或添加系统消息
        if self.messages and isinstance(self.messages[0], SystemMessage):
            self.messages[0] = SystemMessage(content=system_content)
        else:
            self.messages.insert(0, SystemMessage(content=system_content))

    def set_context(self, **kwargs):
        """设置上下文"""
        self.context.update(kwargs)
        self._update_system_message()

    def chat(self, user_input: str) -> str:
        """对话"""
        self.messages.append(HumanMessage(content=user_input))

        # 传递上下文配置
        config = {
            "configurable": {
                "user_id": self.context.get("user_id"),
                "session_id": self.context.get("session_id"),
            }
        }

        response = self.model.invoke(self.messages, config=config)

        self.messages.append(AIMessage(content=response.content))
        return response.content

    def get_context(self) -> AgentContext:
        """获取当前上下文"""
        return self.context

# 使用
agent = ContextAwareAgent()

# 设置上下文
agent.set_context(
    user_id="user_123",
    user_name="张三",
    preferences={"language": "zh-CN", "tone": "friendly"},
    constraints=["不要提供医疗建议", "不要提供法律建议"]
)

# 对话
response = agent.chat("你好")
print(f"AI: {response}")

    11.4.2 上下文持久化¶

    import json
from pathlib import Path
from typing import TypedDict

class PersistentContext(TypedDict, total=False):
    """持久化上下文"""
    user_id: str
    settings: dict
    history: list
    bookmarks: list

class ContextStore:
    """上下文存储"""

    def __init__(self, base_path: str = "./contexts"):
        self.base_path = Path(base_path)
        self.base_path.mkdir(parents=True, exist_ok=True)

    def _get_file_path(self, user_id: str) -> Path:
        return self.base_path / f"{user_id}.json"

    def save(self, context: PersistentContext):
        """保存上下文"""
        file_path = self._get_file_path(context["user_id"])
        with open(file_path, "w", encoding="utf-8") as f:
            json.dump(context, f, ensure_ascii=False, indent=2)

    def load(self, user_id: str) -> PersistentContext:
        """加载上下文"""
        file_path = self._get_file_path(user_id)

        if not file_path.exists():
            return {"user_id": user_id}

        with open(file_path, "r", encoding="utf-8") as f:
            return json.load(f)

    def delete(self, user_id: str):
        """删除上下文"""
        file_path = self._get_file_path(user_id)
        if file_path.exists():
            file_path.unlink()

# 使用
store = ContextStore()

# 保存
context: PersistentContext = {
    "user_id": "user_123",
    "settings": {"theme": "dark", "language": "zh-CN"},
    "history": ["对话 1", "对话 2"],
    "bookmarks": ["文章 1", "文章 2"],
}
store.save(context)

# 加载
loaded = store.load("user_123")
print(f"加载的上下文：{loaded}")

    11.5.1 环境配置¶

    import os
from typing import TypedDict, Optional
from pathlib import Path
import json

class AppConfig(TypedDict, total=False):
    """应用配置"""
    # 模型配置
    default_model: str
    max_tokens: int
    temperature: float

    # API 配置
    api_base_url: str
    api_timeout: int

    # 功能开关
    enable_streaming: bool
    enable_caching: bool
    enable_logging: bool

    # 路径配置
    data_dir: str
    log_dir: str
    cache_dir: str

class ConfigManager:
    """配置管理器"""

    def __init__(self, config_path: str = None):
        self.config_path = config_path or "./config.json"
        self.config: AppConfig = {}
        self._load()

    def _load(self):
        """加载配置"""
        # 1. 从文件加载
        if Path(self.config_path).exists():
            with open(self.config_path, "r", encoding="utf-8") as f:
                self.config = json.load(f)

        # 2. 环境变量覆盖
        self._load_from_env()

    def _load_from_env(self):
        """从环境变量加载配置"""
        env_mapping = {
            "DEFAULT_MODEL": "default_model",
            "MAX_TOKENS": "max_tokens",
            "TEMPERATURE": "temperature",
            "API_BASE_URL": "api_base_url",
            "API_TIMEOUT": "api_timeout",
            "ENABLE_STREAMING": "enable_streaming",
            "ENABLE_CACHING": "enable_caching",
            "ENABLE_LOGGING": "enable_logging",
        }

        for env_var, config_key in env_mapping.items():
            value = os.environ.get(f"LANGCHAIN_{env_var}")
            if value:
                # 类型转换
                if config_key in ["max_tokens", "api_timeout"]:
                    value = int(value)
                elif config_key in ["temperature"]:
                    value = float(value)
                elif config_key in ["enable_streaming", "enable_caching", "enable_logging"]:
                    value = value.lower() in ["true", "1", "yes"]

                self.config[config_key] = value

    def get(self, key: str, default=None):
        """获取配置"""
        return self.config.get(key, default)

    def set(self, key: str, value):
        """设置配置"""
        self.config[key] = value
        self._save()

    def _save(self):
        """保存配置到文件"""
        with open(self.config_path, "w", encoding="utf-8") as f:
            json.dump(self.config, f, ensure_ascii=False, indent=2)

# 使用
config = ConfigManager()

# 获取配置
model = config.get("default_model", "gpt-5.5")
max_tokens = config.get("max_tokens", 4096)

print(f"模型：{model}")
print(f"最大 token: {max_tokens}")

    11.5.2 多环境配置¶

    from typing import Literal, Dict
from enum import Enum

Environment = Literal["development", "staging", "production"]

class MultiEnvConfig:
    """多环境配置"""

    def __init__(self):
        self.configs: Dict[Environment, dict] = {
            "development": {
                "model": "gpt-5.4-mini",
                "debug": True,
                "logging_level": "DEBUG",
                "api_base": "https://api.dev.example.com",
            },
            "staging": {
                "model": "gpt-5.5",
                "debug": False,
                "logging_level": "INFO",
                "api_base": "https://api.staging.example.com",
            },
            "production": {
                "model": "gpt-5.5",
                "debug": False,
                "logging_level": "WARNING",
                "api_base": "https://api.example.com",
            },
        }

        # 当前环境
        self.current_env: Environment = os.environ.get(
            "LANGCHAIN_ENV", "development"
        )

    def get_current_config(self) -> dict:
        """获取当前环境配置"""
        return self.configs.get(self.current_env, {})

    def get(self, key: str, default=None):
        """获取当前环境的配置"""
        return self.get_current_config().get(key, default)

    def switch_environment(self, env: Environment):
        """切换环境"""
        if env not in self.configs:
            raise ValueError(f"未知环境：{env}")
        self.current_env = env
        print(f"切换到环境：{env}")

# 使用
config = MultiEnvConfig()

print(f"当前环境：{config.current_env}")
print(f"模型：{config.get('model')}")

# 切换环境
config.switch_environment("production")
print(f"生产环境模型：{config.get('model')}")

    11.5.3 配置验证¶

    from pydantic import BaseModel, Field, validator
from typing import Optional

class ValidatedConfig(BaseModel):
    """带验证的配置"""

    # 必填字段
    model_name: str = Field(..., min_length=1)
    api_key: str = Field(..., min_length=10)

    # 带默认值
    max_tokens: int = Field(default=4096, ge=1, le=128000)
    temperature: float = Field(default=0.7, ge=0.0, le=2.0)

    # 可选字段
    api_base: Optional[str] = Field(default=None)
    timeout: int = Field(default=30, ge=1)

    @validator('model_name')
    def validate_model_name(cls, v):
        allowed_models = ["gpt-5.4-mini", "gpt-4", "gpt-5.5", "gpt-4-turbo"]
        if v not in allowed_models:
            raise ValueError(f"不支持的模型：{v}")
        return v

    @validator('api_key')
    def validate_api_key(cls, v):
        if not v.startswith("sk-"):
            raise ValueError("API key 必须以 sk-开头")
        return v

    11.6 实战：完整的上下文系统¶

    from typing import TypedDict, Optional, List, Any, Dict
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
import json
import uuid
from datetime import datetime
from contextlib import contextmanager

class FullContext(TypedDict, total=False):
    """完整上下文"""
    # 标识
    request_id: str
    session_id: str
    user_id: str

    # 用户信息
    user_name: Optional[str]
    user_role: Optional[str]

    # 模型配置
    model_name: str
    temperature: float
    max_tokens: int

    # 对话历史
    conversation: List[Dict[str, str]]

    # 元数据
    created_at: str
    updated_at: str
    metadata: Dict[str, Any]

class ContextSystem:
    """完整的上下文系统"""

    def __init__(self, config_path: str = None):
        self.model = ChatOpenAI(model="gpt-5.5")
        self.context: FullContext = self._create_default_context()
        self._context_stack = []

        # 加载配置
        if config_path:
            self._load_config(config_path)

    def _create_default_context(self) -> FullContext:
        """创建默认上下文"""
        return {
            "request_id": str(uuid.uuid4()),
            "session_id": str(uuid.uuid4()),
            "user_id": "anonymous",
            "model_name": "gpt-5.5",
            "temperature": 0.7,
            "max_tokens": 4096,
            "conversation": [],
            "created_at": datetime.now().isoformat(),
            "updated_at": datetime.now().isoformat(),
            "metadata": {},
        }

    def _load_config(self, path: str):
        """加载配置"""
        try:
            with open(path, "r", encoding="utf-8") as f:
                config = json.load(f)
                if "model_name" in config:
                    self.context["model_name"] = config["model_name"]
                if "temperature" in config:
                    self.context["temperature"] = config["temperature"]
        except Exception as e:
            print(f"加载配置失败：{e}")

    @contextmanager
    def scoped_context(self, **updates):
        """作用域上下文"""
        # 保存当前上下文
        saved_context = self.context.copy()

        # 应用更新
        self.context.update(updates)
        self.context["updated_at"] = datetime.now().isoformat()

        try:
            yield
        finally:
            # 恢复上下文
            self.context = saved_context

    def set_user(self, user_id: str, user_name: str = None, user_role: str = None):
        """设置用户"""
        self.context["user_id"] = user_id
        if user_name:
            self.context["user_name"] = user_name
        if user_role:
            self.context["user_role"] = user_role

    def set_model_config(self, model_name: str = None, temperature: float = None, max_tokens: int = None):
        """设置模型配置"""
        if model_name:
            self.context["model_name"] = model_name
            self.model = ChatOpenAI(model=model_name)
        if temperature is not None:
            self.context["temperature"] = temperature
        if max_tokens is not None:
            self.context["max_tokens"] = max_tokens

    def chat(self, user_input: str) -> str:
        """对话"""
        # 构建系统消息
        system_parts = ["你是一个智能助手。"]

        if self.context.get("user_name"):
            system_parts.append(f"当前用户：{self.context['user_name']}")

        if self.context.get("user_role"):
            system_parts.append(f"用户角色：{self.context['user_role']}")

        system_message = SystemMessage(content="\n".join(system_parts))

        # 构建消息列表
        messages = [system_message]

        # 添加对话历史（最近 10 条）
        for msg in self.context["conversation"][-10:]:
            if msg["role"] == "user":
                messages.append(HumanMessage(content=msg["content"]))
            elif msg["role"] == "assistant":
                messages.append(AIMessage(content=msg["content"]))

        # 添加当前消息
        messages.append(HumanMessage(content=user_input))

        # 调用模型
        config = {
            "configurable": {
                "user_id": self.context["user_id"],
                "session_id": self.context["session_id"],
            }
        }

        response = self.model.invoke(messages, config=config)

        # 更新对话历史
        self.context["conversation"].append({
            "role": "user",
            "content": user_input,
            "timestamp": datetime.now().isoformat(),
        })
        self.context["conversation"].append({
            "role": "assistant",
            "content": response.content,
            "timestamp": datetime.now().isoformat(),
        })

        self.context["updated_at"] = datetime.now().isoformat()

        return response.content

    def export_context(self) -> str:
        """导出上下文"""
        return json.dumps(self.context, ensure_ascii=False, indent=2)

    def reset(self):
        """重置上下文"""
        self.context = self._create_default_context()

# 使用示例
if __name__ == "__main__":
    system = ContextSystem()

    # 设置用户
    system.set_user("user_123", "张三", "developer")

    # 设置模型配置
    system.set_model_config(temperature=0.5)

    # 对话
    print(system.chat("你好"))
    print(system.chat("我叫什么名字？"))

    # 作用域上下文
    with system.scoped_context(metadata={"special_mode": True}):
        print(system.chat("现在是什么特殊模式？"))

    # 导出上下文
    print(f"\n当前上下文：\n{system.export_context()}")

    11.7 context_schema（官方推荐方式）¶

    LangChain 官方推荐使用 context_schema 作为运行时上下文的标准方式， 配合 create_agent 使用，比手动管理 TypedDict 更简洁：

    from dataclasses import dataclass
from langchain.agents import create_agent
from langchain.agents import AgentRuntime
from langchain_core.utils.uuid import uuid7

# 1. 使用 dataclass 定义上下文结构
@dataclass
class Context:
    user_id: str
    language: str = "zh"
    user_name: str = ""

# 2. 创建 Agent 时指定 context_schema
agent = create_agent(
    model="openai:gpt-5.5",
    tools=[],
    system_prompt="你是 helpful assistant",
    context_schema=Context,
)

# 3. 在工具中访问上下文
from langchain.tools import tool

@tool
def get_user_preferences(runtime: AgentRuntime) -> str:
    """获取当前用户偏好"""
    user_id = runtime.context.user_id
    return f"用户 {user_id} 的偏好设置"

# 4. 调用时传递 context
result = agent.invoke(
    {"messages": [{"role": "user", "content": "你好"}]},
    config={"configurable": {"thread_id": str(uuid7())}},
    context=Context(user_id="user-123", language="zh", user_name="小智"),
)

    context vs config 的区别¶

    维度 config context
    用途 基础设施配置 业务数据
    示例 thread_id, timeout, checkpointer user_id, tenant_id, permissions
    传递方式 config={"configurable": {...}} context=Context(...)
    访问位置 内部使用 runtime.context
    schema 类型 不需要 dataclass

    在中间件中使用 context¶

    from langchain.agents.middleware import before_model
from langchain.agents import AgentRuntime

@before_model
def inject_user_info(state, runtime: AgentRuntime):
    """在每次模型调用前注入用户信息"""
    user_name = runtime.context.user_name
    language = runtime.context.language

    # 将用户信息注入到状态或系统提示中
    return {
        "messages": [
            {"role": "system", "content": f"用户名叫 {user_name}，使用语言：{language}"}
        ]
    }

agent = create_agent(
    model="openai:gpt-5.5",
    tools=[],
    system_prompt="你是 helpful assistant",
    context_schema=Context,
    middleware=[inject_user_info],
)
