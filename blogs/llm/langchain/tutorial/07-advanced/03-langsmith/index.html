    13.1 LangSmith 概述¶

    13.1.1 什么是 LangSmith¶

    LangSmith 是 LangChain 官方提供的开发平台，提供：

    ┌─────────────────────────────────────────────────────┐
│                LangSmith 产品线                       │
├─────────────────────────────────────────────────────┤
│                                                     │
│  🔍 Tracing          记录所有 LLM 调用和中间步骤     │
│                                                     │
│  🔧 Engine           自动检测 trace 中的问题并       │
│                      提议修复方案，可直接创建 PR     │
│                                                     │
│  🏗️ Fleet            无代码 Agent 构建器，提供       │
│                      模板、集成和日常自动化          │
│                                                     │
│  📊 Evaluation       测试和优化 Agent 性能           │
│                                                     │
│  📈 Monitoring       生产环境性能和错误追踪          │
│                                                     │
│  🚀 Deployment       为长期运行有状态工作流设计的     │
│                      可扩展部署平台                  │
│                                                     │
│  🤝 Collaboration    团队共享和反馈                │
│                                                     │
└─────────────────────────────────────────────────────┘

    13.1.2 核心概念¶

    langsmith_concepts = {

    13.3.2 复杂链路的追踪¶

    from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage

# 定义工具
@tool
def search_web(query: str) -> str:
    """搜索网络"""
    return f"搜索结果：关于{query}的信息"

@tool
def calculate(expression: str) -> str:
    """计算"""
    return f"计算结果：{expression} = 42"

# 创建带工具的模型
model = ChatOpenAI(model="gpt-5.5")
model_with_tools = model.bind_tools([search_web, calculate])

# 多步执行
messages = [HumanMessage(content="搜索 Python 教程，然后计算 100 + 200")]

# 第一步：获取工具调用
response1 = model_with_tools.invoke(messages)
messages.append(response1)

# 执行工具
for tool_call in response1.tool_calls:
    if tool_call["name"] == "search_web":
        result = search_web.invoke(tool_call["args"])
    elif tool_call["name"] == "calculate":
        result = calculate.invoke(tool_call["args"])

    messages.append(ToolMessage(content=result, tool_call_id=tool_call["id"]))

# 第二步：获取最终响应
response2 = model_with_tools.invoke(messages)

# 在 LangSmith 中可以查看完整的执行链路

    13.5.2 运行评估¶

    from langsmith import Client
from langsmith.evaluation import evaluate
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

client = Client()

# 定义被评估的函数
def chatbot(inputs: dict) -> dict:
    """聊天机器人函数"""
    model = ChatOpenAI(model="gpt-5.5")
    response = model.invoke([HumanMessage(content=inputs["question"])])
    return {"answer": response.content}

# 定义评估器
def accuracy_evaluator(run, example) -> dict:
    """准确性评估器"""
    predicted = run.outputs.get("answer", "")
    expected = example.outputs.get("answer", "")

    # 简单比较（实际应该用更复杂的评估）
    is_correct = expected.lower() in predicted.lower()

    return {
        "key": "accuracy",
        "score": 1.0 if is_correct else 0.0,
    }

def relevance_evaluator(run, example) -> dict:
    """相关性评估器"""
    predicted = run.outputs.get("answer", "")

    # 检查回答是否相关（简化版）
    if len(predicted) < 10:
        return {"key": "relevance", "score": 0.0}

    return {"key": "relevance", "score": 0.8}

# 运行评估
results = evaluate(
    chatbot,
    data=client.list_examples(dataset_name="chatbot-evaluation"),
    evaluators=[accuracy_evaluator, relevance_evaluator],
    experiment_prefix="chatbot-v1",
)

print(f"评估完成")

    13.5.3 自定义评估器¶

    from langsmith.evaluation import EvaluationResult
from langsmith import Run

def length_evaluator(run: Run, example) -> EvaluationResult:
    """长度评估器"""
    answer = run.outputs.get("answer", "")
    length = len(answer)

    # 理想长度 50-500 字
    if 50 <= length <= 500:
        score = 1.0
    elif length < 50:
        score = 0.5
    else:
        score = 0.3

    return EvaluationResult(
        key="answer_length",
        score=score,
        comment=f"回答长度：{length}字",
    )

def tone_evaluator(run: Run, example) -> EvaluationResult:
    """语气评估器"""
    answer = run.outputs.get("answer", "")

    # 检查是否友好
    friendly_words = ["你好", "请", "谢谢", "帮助", "欢迎"]
    friendly_count = sum(1 for word in friendly_words if word in answer)

    score = min(1.0, friendly_count / 2)

    return EvaluationResult(
        key="tone",
        score=score,
        comment=f"友好词数：{friendly_count}",
    )

def run_custom_evaluation():
    from langsmith import Client
    from langsmith.evaluation import evaluate

    client = Client()

    results = evaluate(
        chatbot,
        data=client.list_examples(dataset_name="chatbot-evaluation"),
        evaluators=[length_evaluator, tone_evaluator],
    )

    return results

    13.5.4 使用 LLM 作为评估器¶

    from langsmith.evaluation import EvaluationResult
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage

llm_judge = ChatOpenAI(model="gpt-5.5", temperature=0)

def llm_evaluator(run, example) -> EvaluationResult:
    """使用 LLM 作为评估器"""

    question = example.inputs.get("question", "")
    expected = example.outputs.get("answer", "")
    predicted = run.outputs.get("answer", "")

    # 构建评估提示
    eval_prompt = f"""
请评估以下回答的质量：

问题：{question}
预期回答：{expected}
实际回答：{predicted}

请从以下维度评分（0-1）：
1. 准确性：回答是否正确
2. 完整性：是否覆盖了关键点
3. 清晰度：表达是否清晰

请以 JSON 格式返回：
{{
    "accuracy": 0.0-1.0,
    "completeness": 0.0-1.0,
    "clarity": 0.0-1.0,
    "comment": "简短评价"
}}
"""

    response = llm_judge.invoke([
        SystemMessage(content="你是一个公正的评估器。"),
        HumanMessage(content=eval_prompt),
    ])

    # 解析评估结果
    import json
    try:
        scores = json.loads(response.content)
        overall = (scores["accuracy"] + scores["completeness"] + scores["clarity"]) / 3

        return EvaluationResult(
            key="llm_evaluation",
            score=overall,
            comment=scores.get("comment", ""),
            metadata=scores,
        )
    except:
        return EvaluationResult(
            key="llm_evaluation",
            score=0.5,
            comment="评估失败",
        )

    13.6 生产监控¶

    13.6.1 设置监控告警¶

    from langsmith import Client
from datetime import datetime, timedelta

client = Client()

def check_error_rate(project_name: str, threshold: float = 0.1) -> bool:
    """检查错误率"""

    # 获取最近的运行
    runs = list(client.list_runs(
        project_name=project_name,
        limit=100,
    ))

    if not runs:
        return True

    # 计算错误率
    error_count = sum(1 for run in runs if run.status == "error")
    error_rate = error_count / len(runs)

    if error_rate > threshold:
        print(f"⚠️  错误率过高：{error_rate:.2%}")
        return False

    return True

def check_latency(project_name: str, threshold: float = 10.0) -> bool:
    """检查延迟"""

    runs = list(client.list_runs(
        project_name=project_name,
        limit=100,
    ))

    if not runs:
        return True

    # 计算平均延迟
    latencies = [
        (run.end_time - run.start_time).total_seconds()
        for run in runs
        if run.end_time and run.start_time
    ]

    avg_latency = sum(latencies) / len(latencies) if latencies else 0

    if avg_latency > threshold:
        print(f"⚠️  平均延迟过高：{avg_latency:.2f}s")
        return False

    return True

# 定期检查
def monitor_project():
    project_name = "production-chatbot"

    if not check_error_rate(project_name):
        send_alert("错误率过高")

    if not check_latency(project_name):
        send_alert("延迟过高")

    """发送告警"""

    13.6.2 性能仪表板¶

    from langsmith import Client
import json

client = Client()

def generate_dashboard(project_name: str) -> dict:
    """生成性能仪表板"""

    # 获取运行数据
    runs = list(client.list_runs(project_name=project_name, limit=1000))

    # 计算指标
    total_runs = len(runs)
    successful_runs = sum(1 for run in runs if run.status == "success")
    failed_runs = sum(1 for run in runs if run.status == "error")

    latencies = [
        (run.end_time - run.start_time).total_seconds()
        for run in runs
        if run.end_time and run.start_time
    ]

    avg_latency = sum(latencies) / len(latencies) if latencies else 0
    p95_latency = sorted(latencies)[int(len(latencies) * 0.95)] if latencies else 0

    # Token 统计
    total_tokens = sum(
        run.metadata.get("token_usage", {}).get("total_tokens", 0)
        for run in runs
        if run.metadata
    )

    dashboard = {
        "project": project_name,
        "generated_at": datetime.now().isoformat(),
        "metrics": {
            "total_runs": total_runs,
            "success_rate": successful_runs / total_runs if total_runs > 0 else 0,
            "error_rate": failed_runs / total_runs if total_runs > 0 else 0,
            "avg_latency_seconds": avg_latency,
            "p95_latency_seconds": p95_latency,
            "total_tokens": total_tokens,
        },
    }

    return dashboard

# 使用
dashboard = generate_dashboard("production-chatbot")
print(json.dumps(dashboard, indent=2))

    13.6.3 反馈收集¶

    from langsmith import Client

client = Client()

def collect_feedback(run_id: str, score: float, comment: str = None):
    """收集用户反馈"""

    client.create_feedback(
        run_id,
        key="user_feedback",
        score=score,
        comment=comment,
    )

    print(f"反馈已记录：{score}/1.0")

def get_feedback_stats(project_name: str) -> dict:
    """获取反馈统计"""

    runs = list(client.list_runs(project_name=project_name, limit=100))

    feedback_scores = []

    for run in runs:
        # 获取反馈
        feedbacks = client.list_feedback(run_id=run.id)
        for feedback in feedbacks:
            if feedback.key == "user_feedback" and feedback.score is not None:
                feedback_scores.append(feedback.score)

    if not feedback_scores:
        return {"count": 0, "avg_score": 0}

    return {
        "count": len(feedback_scores),
        "avg_score": sum(feedback_scores) / len(feedback_scores),
        "min_score": min(feedback_scores),
        "max_score": max(feedback_scores),
    }

# 使用
stats = get_feedback_stats("production-chatbot")
print(f"反馈统计：{stats}")

    13.7.2 敏感信息处理¶

    from langsmith import Client
from langsmith.run_helpers import trace

client = Client()

@trace
def safe_chat(user_input: str) -> str:
    """安全的聊天函数"""

    # 脱敏敏感信息
    sanitized_input = user_input

    # 移除可能的敏感信息
    import re
    sanitized_input = re.sub(r'\b\d{16}\b', '[CARD]', sanitized_input)
    sanitized_input = re.sub(r'\b\d{11}\b', '[PHONE]', sanitized_input)

    model = ChatOpenAI(model="gpt-5.5")
    response = model.invoke(sanitized_input)

    return response.content

# 不要在元数据中存储敏感信息
# ❌ 错误
metadata = {"user_password": "secret123"}

# ✅ 正确
metadata = {"user_id": "hashed_xxx"}

    13.7.3 成本控制¶

    from langsmith import Client

client = Client()

def estimate_cost(project_name: str) -> dict:
    """估算成本"""

    runs = list(client.list_runs(project_name=project_name, limit=1000))

    # 价格（每 1000 tokens）
    prices = {
        "gpt-5.5": {"input": 0.005, "output": 0.015},
        "gpt-4-turbo": {"input": 0.01, "output": 0.03},
        "gpt-5.4-mini": {"input": 0.0005, "output": 0.0015},
    }

    total_cost = 0

    for run in runs:
        if run.metadata and "token_usage" in run.metadata:
            usage = run.metadata["token_usage"]
            model = run.metadata.get("model_name", "gpt-5.5")

            price = prices.get(model, prices["gpt-5.5"])
            cost = (
                usage.get("prompt_tokens", 0) * price["input"] / 1000 +
                usage.get("completion_tokens", 0) * price["output"] / 1000
            )
            total_cost += cost

    return {
        "total_runs": len(runs),
        "estimated_cost_usd": total_cost,
    }

# 使用
cost = estimate_cost("production-chatbot")
print(f"估算成本：${cost['estimated_cost_usd']:.2f}")

    13.8 LangSmith Engine 与 Fleet¶

    13.8.1 LangSmith Engine¶

    Engine 是 LangSmith 的智能分析产品，可以：

    1. 自动检测问题：分析 Agent trace，识别工具调用错误、参数错误、逻辑问题
    2. 提议修复方案：给出具体的代码修改建议
    3. 创建 PR：直接从 Engine 页面创建 Pull Request 应用修复
    # 确保追踪已启用
import os
os.environ["LANGSMITH_TRACING"] = "true"
os.environ["LANGSMITH_API_KEY"] = "lsv2_..."

# 运行 Agent
from langchain.agents import create_agent

agent = create_agent(
    model="openai:gpt-5.5",
    tools=[my_tool],
    system_prompt="你是 helpful assistant",
)

result = agent.invoke({
    "messages": [{"role": "user", "content": "帮我分析数据"}]
})

# 在 LangSmith UI → Engine 页面：
# 1. 自动分析此次运行的 trace
# 2. 如果有问题，会显示修复建议
# 3. 点击"Create PR"即可应用修复

    Engine 工作流： 

    Agent 运行 → Trace 记录 → Engine 自动分析 → 发现问题 → 提议修复 → 创建 PR → 合并

    13.8.2 LangSmith Fleet¶

    Fleet 是 LangSmith 的无代码 Agent 构建器，适合快速创建和部署 Agent：

    • 模板：提供预构建的 Agent 模板（客服助手、数据分析、代码助手等）
    • 集成：直接连接工具和数据源
    • 日常管理：通过 UI 管理 Agent 的运行和配置
    • 零代码：不需要写代码即可创建功能完整的 Agent

    13.8.3 产品选择建议¶

    场景 推荐产品
    构建定制化 Agent，需要完整控制 LangChain (create_agent)
    快速构建功能完整的 Agent Deep Agents 或 Fleet
    调试和优化已有 Agent Engine + Tracing
    团队协作和版本管理 Fleet
    生产部署和长期运行 Deployment
