    02 Event Streaming — stream_events v3 API¶

    官方文档：https://docs.langchain.com/oss/python/langchain/event-streaming

    ⭐ 推荐：新应用优先使用 Event Streaming API (version="v3")，类型安全、统一事件格式。

    📚 本章目标¶

    学完本章后，你将掌握：

    • ✅ stream_events(version="v3") 基础用法
    • ✅ 类型安全的事件结构：AgentEvent / StreamEvent
    • ✅ 过滤事件：按类型、按节点、按工具名
    • ✅ 流式 Agent 进度、Token、工具调用、自定义更新
    • ✅ 异步流式输出到 Web (SSE)
    • ✅ v2 vs v3 对比

    02.1 为什么用 Event Streaming¶

    之前的 stream_mode 分支模式需要手动判断事件类型：

    # ❌ 旧方式 — 需要分支判断
for event in agent.stream(..., version="v2"):
    for key, value in event.items():
        if key == "model": ...
        elif key == "tool": ...

    v3 Event Streaming 提供统一的类型化事件：

    # ✅ 新方式 — 类型安全，统一结构
for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "你好"}]},
    version="v3",
):
    match event:
        case AgentEvent(type="model_start"): ...
        case AgentEvent(type="tool_result"): ...
        case AgentEvent(type="custom"): ...

    v3 优势¶

    特性 说明
    类型安全 AgentEvent 数据类，IDE 自动补全
    统一格式 所有事件同一结构，无需 key 分支
    过滤能力 按类型/节点/工具名过滤
    自定义事件 get_stream_writer() 发送的事件统一为 custom 类型

    02.2 基础用法¶

    最小示例¶

    from langchain.agents import create_agent

agent = create_agent(
    "openai:gpt-5.5",
    tools=[web_search, get_weather],
)

for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "北京明天天气怎么样？"}]},
    version="v3",
):
    print(event.type, event.data)

    事件类型¶

    事件类型 说明 触发时机
    model_start 模型调用开始 每次 LLM 调用
    model_chunk 模型输出生成 每个 token/chunk
    model_end 模型调用结束 LLM 返回完整响应
    tool_start 工具调用开始 工具执行前
    tool_chunk 工具输出生成 工具流式输出时
    tool_result 工具执行结束 工具返回结果
    custom 自定义事件 get_stream_writer()
    agent_end Agent 执行结束 最终回答完成

    02.3 打印完整 Agent 思考过程¶

    from langchain.agents import create_agent
from langchain.middleware import TodoListMiddleware

agent = create_agent(
    "openai:gpt-5.5",
    tools=[web_search, get_weather],
    middleware=[TodoListMiddleware()],
)


def print_thinking_v3(query: str):
    """使用 v3 API 打印 Agent 完整思考过程"""
    for event in agent.stream_events(
        {"messages": [{"role": "user", "content": query}]},
        version="v3",
    ):
        match event.type:
            case "model_start":
                print("🤔 模型正在思考...")

            case "model_chunk":
                # 流式输出模型文本
                if event.data.content:
                    print(event.data.content, end="", flush=True)

            case "model_end":
                if event.data.tool_calls:
                    for tc in event.data.tool_calls:
                        print(f"\n🔧 调用: {tc.name}({tc.args})")
                print()  # 换行

            case "tool_start":
                print(f"⏳ 执行 {event.data.tool_name}...")

            case "tool_result":
                preview = event.data.content[:80] + "..." if len(event.data.content) > 80 else event.data.content
                print(f"✅ {event.data.tool_name}: {preview}")

            case "custom":
                print(f"📊 {event.data}")

            case "agent_end":
                print(f"\n✅ Agent 执行完成")


print_thinking_v3("北京和上海明天哪个更热？")

    02.4 过滤事件¶

    按类型过滤¶

    # 只看模型输出，不看工具调用
for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "搜索并总结 AI 最新进展"}]},
    version="v3",
    include_types=["model_chunk", "model_end", "agent_end"],
):
    if event.type == "model_chunk" and event.data.content:
        print(event.data.content, end="", flush=True)
    elif event.type == "agent_end":
        print("\n✅ 完成")

    按节点过滤¶

    # 只看 "model" 节点的事件
for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "你好"}]},
    version="v3",
    include_nodes=["model"],
):
    print(event)

    按工具名过滤¶

    # 只关心特定工具的调用和结果
for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "查一下天气"}]},
    version="v3",
    include_names=["get_weather"],
):
    print(event)

    排除过滤¶

    # 排除特定事件
for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "你好"}]},
    version="v3",
    exclude_types=["model_start", "model_chunk"],  # 不看思考过程
):
    print(event)

    02.5 自定义事件 + v3¶

    get_stream_writer() 发送的自定义事件在 v3 中统一为 custom 类型：

    import time
from langchain.tools import tool
from langchain.tools.base import ToolRuntime
from langchain.agents import create_agent


@tool
def long_task(description: str, runtime: ToolRuntime) -> str:
    """执行长时间任务"""
    writer = runtime.get_stream_writer()

    writer({"step": "init", "progress": 0})
    time.sleep(1)

    writer({"step": "processing", "progress": 50})
    time.sleep(1)

    writer({"step": "done", "progress": 100})
    return f"任务完成: {description}"


agent = create_agent(
    "openai:gpt-5.5",
    tools=[long_task],
)

for event in agent.stream_events(
    {"messages": [{"role": "user", "content": "执行数据分析"}]},
    version="v3",
):
    match event.type:
        case "model_chunk":
            print(event.data.content, end="", flush=True)

        case "custom":
            # 工具发送的进度更新
            data = event.data
            print(f"\n📊 Step: {data['step']}, Progress: {data['progress']}%")

        case "agent_end":
            print("\n✅ 全部完成")

    async def async_stream():
    from langchain.agents import create_agent

    agent = create_agent(
        "anthropic:claude-sonnet-4-6",
        tools=[web_search],
    )

    async for event in agent.astream_events(

        version="v3",

            case "model_chunk":

            case "tool_result":
                print(f"\n🔧 {event.data.tool_name} 完成")
            case "agent_end":
                print("\n✅ 完成")


# 运行
import asyncio
asyncio.run(async_stream())

    02.7 SSE Web 输出¶

    from fastapi import FastAPI
from fastapi.responses import StreamingResponse
import json

app = FastAPI()

@app.post("/chat/stream")
async def stream_chat(prompt: str):
    async def generate():
        async for event in agent.astream_events(
            {"messages": [{"role": "user", "content": prompt}]},
            version="v3",
        ):
            match event.type:
                case "model_chunk":
                    if event.data.content:
                        yield f"data: {json.dumps({'type': 'token', 'content': event.data.content})}\n\n"
                case "tool_result":
                    yield f"data: {json.dumps({'type': 'tool', 'name': event.data.tool_name})}\n\n"
                case "agent_end":
                    yield f"data: {json.dumps({'type': 'done'})}\n\n"
                case "custom":
                    yield f"data: {json.dumps({'type': 'custom', 'data': event.data})}\n\n"

    return StreamingResponse(generate(), media_type="text/event-stream")

    前端接收：

    const response = await fetch("/chat/stream", {

while (true) {

            }
        }
    }
}

    02.8 v2 vs v3 对比¶

    特性 v2 (进度流) v3 (Event Streaming)
    API agent.stream(..., version="v2") agent.stream_events(..., version="v3")
    事件结构 dict with key branches AgentEvent 数据类
    类型安全 ❌ 需要 key 判断 ✅ match/case
    过滤 ❌ 手动过滤 ✅ include_types/names/nodes
    自定义事件 custom key custom 类型
    推荐场景 旧项目兼容 新项目首选

    何时用哪个？¶

    场景 推荐
    新项目 v3
    需要过滤事件 v3
    类型安全要求 v3
    旧项目快速迁移 v2（暂保留）
    需要底层 Pregel 模式 stream_mode（01-streaming.md）

    02.9 完整实战示例¶

    from langchain.agents import create_agent
from langchain.middleware import TodoListMiddleware, SummarizationMiddleware
from rich.console import Console
from rich.markdown import Markdown

console = Console()

agent = create_agent(
    "openai:gpt-5.5",
    tools=[web_search, file_reader, code_executor],
    middleware=[
        TodoListMiddleware(),
        SummarizationMiddleware(max_tokens=4000),
    ],
)


def chat_v3(user_input: str) -> str:
    """完整的 v3 流式对话界面"""
    console.print(f"\n👤 你: {user_input}", style="bold blue")
    console.print("🤖 Agent:", style="bold green")

    answer = ""

    for event in agent.stream_events(
        {"messages": [{"role": "user", "content": user_input}]},
        version="v3",
    ):
        match event.type:
            case "model_start":
                console.print("  🤔 思考中...", style="dim")

            case "model_chunk":
                if event.data.content:
                    console.print(event.data.content, end="")
                    answer += event.data.content

            case "model_end":
                if event.data.tool_calls:
                    for tc in event.data.tool_calls:
                        console.print(f"\n  🔧 {tc.name}({tc.args})", style="yellow")

            case "tool_start":
                console.print(f"  ⏳ {event.data.tool_name}...", style="cyan")

            case "tool_result":
                preview = event.data.content[:60] + "..." if len(event.data.content) > 60 else event.data.content
                console.print(f"  ✅ {event.data.tool_name}: {preview}", style="dim green")

            case "custom":
                console.print(f"  📊 {event.data}", style="dim")

    console.print()  # 换行
    return answer


# 对话循环
while True:
    user_input = console.input("\n👤 你: ")
    if user_input.lower() in ("quit", "exit", "q"):
        break
    chat_v3(user_input)

    📝 本章小结¶

    1. v3 是官方推荐的新 API，类型安全、统一格式、支持过滤
    2. 事件类型：model_start/chunk/end、tool_start/chunk/result、custom、agent_end
    3. 过滤能力：include_types/names/nodes + exclude_types
    4. 自定义事件：get_stream_writer() → custom 类型
    5. 异步支持：astream_events() 完美配合 FastAPI SSE
    6. v2 vs v3：新项目用 v3，旧项目可保留 v2
