    第 06 章：消息管理 - Messages 详解¶

    本章内容：消息格式、内容块、多模态处理

    官方文档：https://docs.langchain.com/oss/python/langchain/messages/

    学完本章，你将掌握：

    • ✅ 理解 LangChain 消息系统的核心概念

    • ✅ 消息对象属性：text / content / content_blocks / tool_calls / usage_metadata

    6.1 消息类型概述¶

    LangChain 提供了多种消息类型，每种类型都有明确的语义：

    from langchain.messages import (
    HumanMessage,      # 用户消息
    AIMessage,         # AI 响应
    SystemMessage,     # 系统指令
    ToolMessage,       # 工具调用结果
    FunctionMessage,   # 函数调用结果（旧版）
    ChatMessage,       # 通用消息
)

    注意：官方推荐使用 langchain.messages 作为标准导入路径（自动代理到 langchain_core.messages）。

    6.1.1 基础消息类型¶

    from langchain_core.messages import HumanMessage, AIMessage, SystemMessage

# SystemMessage - 系统指令（设定 AI 行为和角色）
system_msg = SystemMessage(content="你是一个专业的编程助手。")

# HumanMessage - 用户消息
human_msg = HumanMessage(content="如何用 Python 读取文件？")

# AIMessage - AI 响应
ai_msg = AIMessage(content="你可以使用 open() 函数或 pathlib 模块...")

# 查看消息属性
print(f"类型：{human_msg.type}")           # human
print(f"内容：{human_msg.content}")        # 如何用 Python 读取文件？
print(f"额外信息：{human_msg.additional_kwargs}")

    6.1.2 消息序列¶

    在实际对话中，消息通常以列表形式存在：

    from langchain_core.messages import HumanMessage, AIMessage, SystemMessage

messages = [
    SystemMessage(content="你是一个友好的助手。"),
    HumanMessage(content="你好！"),
    AIMessage(content="你好！有什么我可以帮助你的吗？"),
    HumanMessage(content="Python 怎么读取 JSON 文件？"),
]

# 将消息序列传递给模型
from langchain_openai import ChatOpenAI

model = ChatOpenAI(model="gpt-5.5")
response = model.invoke(messages)
print(response.content)

    6.2 内容块（Content Blocks）¶

    LangChain 支持复杂的内容块结构，可以包含文本、图片、文件等多种类型。

    6.2.0 标准内容块（Standard Content Blocks）¶

    LangChain 提供了一套标准内容块格式，跨 Provider 通用：

    from langchain.messages import HumanMessage

# 字符串内容（最简单）
msg = HumanMessage("你好，你好吗？")

# Provider 原生格式（如 OpenAI 格式）
msg = HumanMessage(content=[
    {"type": "text", "text": "你好"},
    {"type": "image_url", "image_url": {"url": "https://example.com/image.jpg"}},
])

# LangChain 标准内容块格式
msg = HumanMessage(content_blocks=[
    {"type": "text", "text": "你好"},
    {"type": "image", "url": "https://example.com/image.jpg"},
])

    提示：使用 content_blocks 初始化消息时，LangChain 会自动填充 content 字段， 同时提供类型安全的接口。

    跨 Provider 标准化¶

    不同 Provider 的推理/思考内容格式不同，但 LangChain 可以统一解析为标准 reasoning 块：

    # Anthropic 消息 → content_blocks 自动解析
from langchain.messages import AIMessage

anthropic_msg = AIMessage(
    content=[
        {"type": "thinking", "thinking": "...", "signature": "WaUjzkyp..."},
        {"type": "text", "text": "回答内容..."},
    ],
    response_metadata={"model_provider": "anthropic"},
)

print(anthropic_msg.content_blocks)
# [
#   {'type': 'reasoning', 'reasoning': '...', 'extras': {'signature': '...'}},
#   {'type': 'text', 'text': '回答内容...'}
# ]

# OpenAI 消息 → content_blocks 自动解析
openai_msg = AIMessage(
    content=[
        {"type": "reasoning", "id": "rs_abc123", "summary": [...]},
        {"type": "text", "text": "..."},
    ],
    response_metadata={"model_provider": "openai"},
)

print(openai_msg.content_blocks)
# [
#   {'type': 'reasoning', 'id': 'rs_abc123', 'reasoning': 'summary text...'},
#   {'type': 'text', 'text': '...'}
# ]

    序列化标准内容块¶

    如果你需要将消息传递给 LangChain 之外的应用，可以选择存储标准内容块格式：

    from langchain.chat_models import init_chat_model

# 方式 1：环境变量
import os
os.environ["LC_OUTPUT_VERSION"] = "v1"

# 方式 2：初始化时指定
model = init_chat_model("openai:gpt-5.5", output_version="v1")

    6.3.4 AIMessage 属性详解¶

    AIMessage 是模型返回的响应对象，包含丰富的属性：

    from langchain.messages import AIMessage

ai_msg = AIMessage(
    content="你好！我是助手。",
    tool_calls=[{
        "name": "get_weather",
        "args": {"city": "北京"},
        "id": "call_123",
    }],
    id="msg_abc123",
    response_metadata={"model_provider": "openai"},
)

# 主要属性
print(ai_msg.text)              # "你好！我是助手。"  — 纯文本内容
print(ai_msg.content)           # 原始内容（可能是字符串或列表）
print(ai_msg.content_blocks)    # 标准化内容块列表
print(ai_msg.tool_calls)        # 工具调用列表
print(ai_msg.id)                # 唯一消息 ID
print(ai_msg.usage_metadata)    # Token 使用统计
print(ai_msg.response_metadata) # Provider 特定元数据

    AIMessageChunk — 流式消息合并¶

    流式输出时，你会收到 AIMessageChunk 对象，它们可以合并成完整消息：

    chunks = []
full_message = None
for chunk in model.stream("写一首关于春天的诗"):
    chunks.append(chunk)
    print(chunk.text, end="", flush=True)
    full_message = chunk if full_message is None else full_message + chunk

# full_message 是合并后的完整 AIMessage
print(f"\n总 chunk 数：{len(chunks)}")
print(f"完整消息：{full_message.text}")

    6.4 工具消息（ToolMessage）¶

    当 Agent 调用工具时，需要使用 ToolMessage 传递结果：

    from langchain_core.messages import (
    AIMessage,
    ToolMessage,
    HumanMessage,
)
from langchain_core.tools import tool

# 定义工具
@tool
def get_weather(city: str) -> str:
    """获取城市天气"""
    weather_data = {
        "北京": "晴，25°C",
        "上海": "多云，28°C",
        "广州": "小雨，30°C",
    }
    return weather_data.get(city, "未知城市")

# 模拟工具调用流程
messages = [
    HumanMessage(content="北京天气怎么样？"),
    AIMessage(
        content="",
        tool_calls=[
            {
                "name": "get_weather",
                "args": {"city": "北京"},
                "id": "call_123",
            }
        ],
    ),
    ToolMessage(
        content="晴，25°C",
        tool_call_id="call_123",
        name="get_weather",  # 工具名称
    ),
]

# ToolMessage 的 artifact 字段 — 存储不发给模型的额外数据
tool_msg_with_artifact = ToolMessage(
    content="引用段落：这是最好的时代，这是最坏的时代...",
    tool_call_id="call_456",
    name="search_books",
    artifact={
        "document_id": "doc_123",
        "page": 0,
        "source_url": "https://example.com/book/123",
    },
)
# content → 发给模型
# artifact → 应用层可访问（用于渲染页面、追踪来源等）

# 继续对话
model = ChatOpenAI(model="gpt-5.5")
response = model.invoke(messages)
print(response.content)  # 北京今天天气晴朗，气温 25 度...

    6.5 消息历史管理¶

    6.5.1 手动管理消息历史¶

    from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langchain_openai import ChatOpenAI

model = ChatOpenAI(model="gpt-5.5")

# 初始化消息历史
message_history = [
    SystemMessage(content="你是一个乐于助人的编程助手。")
]

# 多轮对话
while True:
    user_input = input("你：")
    if user_input.lower() == "exit":
        break

    # 添加用户消息
    message_history.append(HumanMessage(content=user_input))

    # 获取 AI 响应
    response = model.invoke(message_history)
    print(f"AI: {response.content}")

    # 添加 AI 响应到历史
    message_history.append(AIMessage(content=response.content))

    # 可选：限制历史长度
    if len(message_history) > 20:
        # 保留系统消息和最近的对话
        message_history = [message_history[0]] + message_history[-18:]

    6.6.2 消息合并¶

    from langchain_core.messages import HumanMessage, AIMessage

# 合并连续的同类型消息
def merge_consecutive_messages(messages):
    if not messages:
        return []

    merged = []
    current_type = type(messages[0])
    current_content = [messages[0].content]

    for msg in messages[1:]:
        if type(msg) == current_type:
            current_content.append(msg.content)
        else:
            merged.append(current_type(content="\n".join(map(str, current_content))))
            current_type = type(msg)
            current_content = [msg.content]

    merged.append(current_type(content="\n".join(map(str, current_content))))
    return merged

# 使用示例
messages = [
    HumanMessage(content="第一段"),
    HumanMessage(content="第二段"),
    AIMessage(content="回复"),
    HumanMessage(content="继续问"),
]

merged = merge_consecutive_messages(messages)
print(f"原始消息数：{len(messages)}")
print(f"合并后消息数：{len(merged)}")

    6.6.3 消息截断¶

    from langchain_core.messages import HumanMessage, AIMessage, SystemMessage

def truncate_messages(messages, max_tokens=4000, system_first=True):
    """
    截断消息以适配模型上下文窗口

    Args:
        messages: 消息列表
        max_tokens: 最大 token 数
        system_first: 是否始终保留系统消息
    """
    if not messages:
        return []

    # 估算 token 数（简单估算：每 4 个字符约 1 个 token）
    def estimate_tokens(msg):
        return len(str(msg.content)) // 4

    # 分离系统消息
    system_messages = [m for m in messages if isinstance(m, SystemMessage)]
    other_messages = [m for m in messages if not isinstance(m, SystemMessage)]

    # 从后往前保留消息
    result = []
    total_tokens = 0

    for msg in reversed(other_messages):
        msg_tokens = estimate_tokens(msg)
        if total_tokens + msg_tokens <= max_tokens:
            result.insert(0, msg)
            total_tokens += msg_tokens
        else:
            break

    # 添加系统消息
    if system_first and system_messages:
        result.insert(0, system_messages[0])

    return result

# 使用示例
messages = [SystemMessage(content="系统指令")] + [
    HumanMessage(content=f"消息{i}") 
    for i in range(100)
]

truncated = truncate_messages(messages, max_tokens=500)
print(f"原始：{len(messages)} 条，截断后：{len(truncated)} 条")

    6.7 实战：构建多轮对话系统¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langchain_core.chat_history import InMemoryChatMessageHistory
from typing import Dict

class ChatBot:
    """简单的多轮对话机器人"""

    def __init__(self, system_prompt: str = None):
        self.model = ChatOpenAI(model="gpt-5.5", temperature=0.7)
        self.history = InMemoryChatMessageHistory()

        if system_prompt:
            self.history.add_message(SystemMessage(content=system_prompt))

    def chat(self, user_input: str) -> str:
        """发送消息并获取响应"""
        # 添加用户消息
        self.history.add_user_message(user_input)

        # 获取 AI 响应
        response = self.model.invoke(self.history.messages)

        # 添加 AI 响应到历史
        self.history.add_ai_message(response.content)

        return response.content

    def reset(self):
        """重置对话历史"""
        self.history.clear()

    def get_history(self):
        """获取对话历史"""
        return self.history.messages

    def export_history(self) -> list:
        """导出对话历史为字典列表"""
        return [
            {"role": msg.type, "content": msg.content}
            for msg in self.history.messages
        ]

    # 监控消息长度
def check_context_length(messages):
    """检查上下文长度"""
    total_chars = sum(len(str(m.content)) for m in messages)
    estimated_tokens = total_chars // 4

    print(f"字符数：{total_chars}")
    print(f"估算 token 数：{estimated_tokens}")

    # GPT-4o 上下文窗口：128K tokens
    if estimated_tokens > 100000:
        print("⚠️  警告：接近上下文窗口限制")

    return estimated_tokens

# 定期清理历史
def cleanup_history(messages, keep_last_n=20):
    """保留最近的 N 轮对话"""
    system_messages = [m for m in messages if isinstance(m, SystemMessage)]
    other_messages = messages[-keep_last_n:]

    return system_messages + other_messages

from PIL import Image
import io

def optimize_image(image_path, max_size=1024):
    """优化图片大小"""
    img = Image.open(image_path)

    # 调整大小
    if max(img.size) > max_size:
        ratio = max_size / max(img.size)
        new_size = (int(img.size[0] * ratio), int(img.size[1] * ratio))
        img = img.resize(new_size, Image.Resampling.LANCZOS)

    # 转换为 base64
    buffer = io.BytesIO()
    img.save(buffer, format="PNG")
    return base64.b64encode(buffer.getvalue()).decode("utf-8")
