    3.1 模型初始化¶

    为什么需要统一接口？¶

    问题：每个 Provider 的 API 不同

    # ❌ 不使用 LangChain - 每个 Provider API 不同

# OpenAI
import openai
response = openai.ChatCompletion.create(
    model="gpt-5.5",
    messages=[...],
    temperature=0.7
)

# Anthropic
import anthropic
client = anthropic.Client()
response = client.messages.create(
    model="claude-sonnet-4-6",
    max_tokens=1000,
    messages=[...]
)

# Google
import google.generativeai as genai
model = genai.GenerativeModel('gemini-3.5-flash')
response = model.generate_content("...")

    解决：LangChain 统一接口

    # ✅ 使用 LangChain - 统一接口

from langchain.agents import create_agent

# 所有 Provider 使用相同接口：provider:model-id 格式
agent = create_agent(
    model="openai:gpt-5.5",
    tools=[],
    system_prompt="You are a helpful assistant",
)
# model="anthropic:claude-sonnet-4-6"
# model="google_genai:gemini-3.5-flash"

result = agent.invoke({"messages": [{"role": "user", "content": "Hello"}]})

    init_chat_model（适用于非 Agent 场景）¶

    如果你只需要调用模型而不构建 Agent，这是最简单的初始化方式：

    💡 注意：官方推荐使用 create_agent 作为主要入口。init_chat_model 仍然可用， 适合纯模型调用场景（如分类、翻译等不需要工具循环的任务）。

    from langchain.chat_models import init_chat_model

# 方式 1: 自动识别 Provider
model = init_chat_model("gpt-5.5")  # OpenAI
model = init_chat_model("openai:gpt-5.5")  # Anthropic
model = init_chat_model("gemini-3.5-flash")  # Google

# 方式 2: 明确指定 Provider（推荐）
model = init_chat_model("openai:gpt-5.5")
model = init_chat_model("anthropic:claude-sonnet-4-6")
model = init_chat_model("google_genai:gemini-3.5-flash")

# 方式 3: 带参数
model = init_chat_model(
    "openai:gpt-5.5",
    temperature=0.7,
    max_tokens=1000,
    timeout=30
)

    优势：

    • ✅ 自动识别 Provider
    • ✅ 支持动态切换模型
    • ✅ 参数传递简单
    • ✅ 代码更简洁

    直接使用 Model Class¶

    传统方式，适合需要精细控制：

    # OpenAI
from langchain_openai import ChatOpenAI
model = ChatOpenAI(model="gpt-5.5")

# Anthropic
from langchain_anthropic import ChatAnthropic
model = ChatAnthropic(model="claude-sonnet-4-6")

# Google
from langchain_google_genai import ChatGoogleGenerativeAI
model = ChatGoogleGenerativeAI(model="gemini-3.5-flash")

# AWS Bedrock
from langchain_aws import ChatBedrock
model = ChatBedrock(model="anthropic.claude-3-5-sonnet-20240620-v1:0")

# Azure OpenAI
from langchain_openai import AzureChatOpenAI
model = AzureChatOpenAI(
    model="gpt-5.5",
    azure_deployment="my-deployment",
    api_version="2025-03-01-preview"
)

    适用场景：

    • 需要 Provider 特定参数
    • 需要自定义认证逻辑
    • 需要精细控制连接池

    3.3 模型参数配置¶

    标准参数¶

    from langchain.chat_models import init_chat_model

model = init_chat_model(
    "openai:gpt-5.5",

    # temperature - 控制随机性 (0-2)
    temperature=0.7,  # 0=确定，1=创意

    # max_tokens - 限制输出长度
    max_tokens=2000,

    # timeout - 超时时间（秒）
    timeout=30,

    # max_retries - 最大重试次数
    max_retries=6
)

    temperature 详解¶

    # temperature=0 - 最确定
model = init_chat_model("openai:gpt-5.5", temperature=0)
response = model.invoke("1+1=?")  # 每次都输出 2

# temperature=0.7 - 平衡（推荐）
model = init_chat_model("openai:gpt-5.5", temperature=0.7)
response = model.invoke("写一首诗")  # 有创意但不过分

# temperature=1.5 - 非常有创意
model = init_chat_model("openai:gpt-5.5", temperature=1.5)
response = model.invoke("编故事")  # 充满想象力

    推荐值：

    场景 temperature
    事实问答 0 - 0.3
    代码生成 0.2 - 0.5
    日常对话 0.5 - 0.8
    创意写作 0.8 - 1.2

    3.4 模型调用方式¶

    Invoke（同步调用）¶

    from langchain.chat_models import init_chat_model

model = init_chat_model("openai:gpt-5.5")

# 简单字符串
response = model.invoke("你好")

# 消息列表（对话历史）
from langchain.messages import HumanMessage, AIMessage, SystemMessage

messages = [
    SystemMessage("你是翻译助手"),
    HumanMessage("翻译：I love programming"),
    AIMessage("J'adore la programmation"),
    HumanMessage("翻译：I love building applications")
]

response = model.invoke(messages)

    Stream（流式输出）¶

    model = init_chat_model("openai:gpt-5.5")

# 基础流式
for chunk in model.stream("写一首长诗"):
    print(chunk.content, end="", flush=True)

# 累积完整消息
full_message = None
for chunk in model.stream("为什么天空是蓝色的？"):
    full_message = chunk if full_message is None else full_message + chunk

print(f"完整响应：{full_message.content}")

    Batch（批量处理）¶

    questions = [
    "什么是人工智能？",
    "量子计算是什么？",
    "如何学习编程？"
]

# 批量处理
responses = model.batch(questions)

# 带并发限制
responses = model.batch(
    questions,
    config={"max_concurrency": 5}
)

# 批量流式（按完成顺序）
for idx, response in model.batch_as_completed(questions):
    print(f"问题 {idx} 完成")

    3.5 工具调用（Tool Calling）¶

    定义工具¶

    from langchain.tools import tool

@tool
def get_weather(location: str) -> str:
    """获取指定城市的天气信息"""
    return f"{location}: 晴朗，25°C"

@tool
def search_web(query: str) -> str:
    """搜索网络信息"""
    return f"搜索结果：{query}"

    绑定工具¶

    model = init_chat_model("openai:gpt-5.5")
model_with_tools = model.bind_tools([get_weather, search_web])

# 调用
response = model_with_tools.invoke("北京天气如何？")

# 检查工具调用
if response.tool_calls:
    for tool_call in response.tool_calls:
        print(f"工具：{tool_call['name']}")
        print(f"参数：{tool_call['args']}")

    执行工具调用¶

    from langchain.messages import ToolMessage

def run_agent(user_input: str):
    messages = [{"role": "user", "content": user_input}]

    while True:
        # 1. 模型生成响应
        ai_msg = model_with_tools.invoke(messages)
        messages.append(ai_msg)

        # 2. 检查是否有工具调用
        if not ai_msg.tool_calls:
            return ai_msg.content  # 最终答案

        # 3. 执行工具
        for tool_call in ai_msg.tool_calls:
            result = get_weather.invoke(tool_call["args"])
            messages.append(ToolMessage(
                content=result,
                tool_call_id=tool_call["id"]
            ))

response = run_agent("北京天气如何？")

    3.6 结构化输出¶

    使用 Pydantic（推荐）¶

    from pydantic import BaseModel, Field

class Movie(BaseModel):
    """电影信息"""
    title: str = Field(description="电影标题")
    year: int = Field(description="上映年份")
    director: str = Field(description="导演")
    rating: float = Field(description="评分 (0-10)")

model = init_chat_model("openai:gpt-5.5")
model_with_structure = model.with_structured_output(Movie)

response = model_with_structure.invoke(
    "提供《盗梦空间》的详细信息"
)

print(f"电影：{response.title}")
print(f"年份：{response.year}")

    使用 TypedDict¶

    from typing_extensions import TypedDict, Annotated

class MovieDict(TypedDict):
    title: Annotated[str, ..., "电影标题"]
    year: Annotated[int, ..., "上映年份"]
    director: Annotated[str, ..., "导演"]

model_with_structure = model.with_structured_output(MovieDict)
response = model_with_structure.invoke("提供《盗梦空间》信息")
print(response['title'])

    嵌套结构¶

    from typing import List

class Actor(BaseModel):
    name: str
    role: str

class Movie(BaseModel):
    title: str
    year: int
    cast: List[Actor]
    genres: List[str]

model_with_structure = model.with_structured_output(Movie)
response = model_with_structure.invoke("提供电影信息，包括演员")
print(f"主演：{response.cast[0].name}")

    模型选择策略¶

    def get_model(task_complexity: str):
    if task_complexity == "simple":
        return init_chat_model("openai:gpt-5.4-mini")  # 便宜快速
    elif task_complexity == "medium":
        return init_chat_model("openai:gpt-5.5")  # 平衡
    else:
        return init_chat_model("openai:o3")  # 最强推理

    3.9 本章小结¶

    关键要点¶

    1. 模型初始化 - create_agent（推荐）或 init_chat_model 统一接口
    2. 模型格式 - provider:model-id 是官方统一格式
    3. 参数配置 - temperature、max_tokens、timeout
    4. 调用方式 - invoke（同步）、stream（流式）、batch（批量）
    5. 工具调用 - bind_tools + 执行循环
    6. 结构化输出 - Pydantic/TypedDict/JSON Schema
    7. 多模态 - 图像/音频/视频输入输出
    8. 性能优化 - 缓存、批处理、模型路由

    快速参考¶

    from langchain.chat_models import init_chat_model

# 初始化
model = init_chat_model("openai:gpt-5.5", temperature=0.7)

# 调用
response = model.invoke("你好")

# 流式
for chunk in model.stream("写诗"):
    print(chunk.content, end="")

# 批量
responses = model.batch(["问题 1", "问题 2"])

# 工具调用
model_with_tools = model.bind_tools([tool_func])

# 结构化输出
model_with_structure = model.with_structured_output(MySchema)
