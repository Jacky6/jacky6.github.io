    10.1 流式输出概述¶

    10.1.1 为什么需要流式输出¶

    from langchain_openai import ChatOpenAI

model = ChatOpenAI(model="gpt-4o")

# ❌ 非流式：等待完整响应
print("等待响应...")
response = model.invoke("写一篇 1000 字的文章")
print(response.content)  # 一次性输出所有内容

# ✅ 流式：逐字输出
print("开始输出：")
for chunk in model.stream("写一篇 1000 字的文章"):
    print(chunk.content, end="", flush=True)
# 可以立即看到输出，体验更好

    10.2.4 打字机效果¶

    from langchain_openai import ChatOpenAI
import time

model = ChatOpenAI(model="gpt-4o")

def typewriter_effect(text, speed=0.01):
    """打字机效果打印"""
    for char in text:
        print(char, end="", flush=True)
        time.sleep(speed)
    print()

# 结合流式输出和打字机效果
print("AI: ", end="")
for chunk in model.stream("写一段优美的文字"):
    # 每个 chunk 可能包含多个字符
    for char in chunk.content:
        print(char, end="", flush=True)
        time.sleep(0.01)  # 每个字符延迟 10ms
print()

    10.2.5 带进度指示的流式输出¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

model = ChatOpenAI(model="gpt-4o")

# 带进度统计
total_tokens = 0
total_chars = 0
start_time = time.time()

print("AI: ", end="")
for chunk in model.stream("写一篇关于人工智能的短文"):
    content = chunk.content
    total_chars += len(content)
    total_tokens += len(content) // 4  # 粗略估算

    print(content, end="", flush=True)

    # 每 100 个字符显示一次统计
    if total_chars % 100 < len(content):
        elapsed = time.time() - start_time
        speed = total_chars / elapsed if elapsed > 0 else 0
        print(f"\n[进度：{total_chars}字，{elapsed:.1f}s，{speed:.0f}字/秒]", end="", flush=True)

print(f"\n\n完成：{total_chars}字，{total_tokens}token")

    10.3.1 使用事件回调¶

    from langchain_openai import ChatOpenAI
from langchain_core.callbacks import StreamingStdOutCallbackHandler
from langchain_core.messages import HumanMessage

# 创建回调处理器
class MyCallbackHandler(StreamingStdOutCallbackHandler):
    def on_llm_new_token(self, token: str, **kwargs):
        """新 token 生成时调用"""
        print(f"[Token] {token}", end="", flush=True)

    def on_llm_start(self, serialized, prompts, **kwargs):
        """LLM 开始时调用"""
        print("\n[开始生成]")

    def on_llm_end(self, response, **kwargs):
        """LLM 结束时调用"""
        print("\n[生成完成]")

    def on_llm_error(self, error, **kwargs):
        """LLM 错误时调用"""
        print(f"\n[错误] {error}")

# 使用回调
model = ChatOpenAI(
    model="gpt-4o",
    streaming=True,
    callbacks=[MyCallbackHandler()]
)

response = model.invoke([HumanMessage(content="你好")])

    10.3.3 自定义事件处理器¶

    from langchain_core.callbacks import BaseCallbackHandler
from typing import Any, Dict, List

class EventCollector(BaseCallbackHandler):
    """事件收集器"""

    def __init__(self):
        self.events = []
        self.tokens = []
        self.start_time = None
        self.end_time = None

    def on_llm_start(self, serialized: Dict[str, Any], prompts: List[str], **kwargs):
        self.events.append({
            "type": "start",
            "time": time.time(),
            "model": serialized.get("name", "unknown"),
        })
        self.start_time = time.time()

    def on_llm_new_token(self, token: str, **kwargs):
        self.tokens.append(token)
        self.events.append({
            "type": "token",
            "time": time.time(),
            "token": token,
        })

    def on_llm_end(self, response: Any, **kwargs):
        self.end_time = time.time()
        self.events.append({
            "type": "end",
            "time": time.time(),
            "total_tokens": response.usage.get("total_tokens", 0) if hasattr(response, "usage") else 0,
        })

    def on_llm_error(self, error: Exception, **kwargs):
        self.events.append({
            "type": "error",
            "time": time.time(),
            "error": str(error),
        })

    def get_stats(self) -> Dict[str, Any]:
        """获取统计信息"""
        duration = (self.end_time - self.start_time) if self.end_time else 0
        return {
            "total_tokens": len(self.tokens),
            "duration": duration,
            "tokens_per_second": len(self.tokens) / duration if duration > 0 else 0,
            "events": len(self.events),
        }

# 使用
collector = EventCollector()
model = ChatOpenAI(
    model="gpt-4o",
    streaming=True,
    callbacks=[collector]
)

response = model.invoke("介绍 Python")
print(f"\n统计：{collector.get_stats()}")

    10.4 工具调用的流式输出¶

    10.4.1 流式工具调用¶

    from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage

# 定义工具
@tool
def get_weather(city: str) -> str:
    """获取天气"""
    return f"{city}今天晴朗，25°C"

@tool
def get_time() -> str:
    """获取时间"""
    from datetime import datetime
    return datetime.now().strftime("%H:%M:%S")

# 创建带工具的模型
model = ChatOpenAI(model="gpt-4o")
model_with_tools = model.bind_tools([get_weather, get_time])

# 流式调用
messages = [HumanMessage(content="北京天气怎么样？现在几点了？")]

for chunk in model_with_tools.stream(messages):
    if chunk.content:
        print(chunk.content, end="", flush=True)

    # 检查工具调用
    if chunk.tool_calls:
        for tool_call in chunk.tool_calls:
            print(f"\n[工具调用] {tool_call['name']}: {tool_call['args']}")

    10.4.2 处理工具调用结果¶

    from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage

@tool
def calculate(expression: str) -> str:
    """计算表达式"""
    try:
        result = eval(expression)
        return str(result)
    except Exception as e:
        return f"计算错误：{e}"

model = ChatOpenAI(model="gpt-4o")
model_with_tools = model.bind_tools([calculate])

# 完整工具调用流程
messages = [HumanMessage(content="计算 123 + 456 * 789")]

# 第一次调用（获取工具调用）
for chunk in model_with_tools.stream(messages):
    if chunk.tool_calls:
        for tool_call in chunk.tool_calls:
            print(f"调用工具：{tool_call['name']}")

            # 执行工具
            tool_result = calculate.invoke(tool_call['args'])
            print(f"工具结果：{tool_result}")

            # 添加工具结果到消息
            messages.append(AIMessage(content="", tool_calls=[tool_call]))
            messages.append(ToolMessage(content=tool_result, tool_call_id=tool_call['id']))

# 第二次调用（获取最终响应）
print("\n最终响应：")
for chunk in model_with_tools.stream(messages):
    print(chunk.content, end="", flush=True)

    10.4.3 流式工具调用事件¶

    from langchain_core.callbacks import BaseCallbackHandler
from typing import Any, Dict

class ToolCallHandler(BaseCallbackHandler):
    """工具调用处理器"""

    def __init__(self):
        self.tool_calls = []

    def on_llm_new_token(self, token: str, **kwargs):
        if "tool_calls" in kwargs:
            for tool_call in kwargs["tool_calls"]:
                self.tool_calls.append(tool_call)
                print(f"\n🔧 工具调用：{tool_call.get('name')}")
                print(f"   参数：{tool_call.get('args')}")

    def on_tool_start(self, serialized: Dict[str, Any], input_str: str, **kwargs):
        print(f"   → 执行中...")

    def on_tool_end(self, output: str, **kwargs):
        print(f"   ← 结果：{output[:100]}")

# 使用
handler = ToolCallHandler()
model = ChatOpenAI(
    model="gpt-4o",
    streaming=True,
    callbacks=[handler]
)
model_with_tools = model.bind_tools([calculate])

response = model_with_tools.invoke("计算 100 + 200")

    10.5 实战：流式对话界面¶

    10.5.1 命令行对话界面¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from typing import List
import sys

class StreamingChatCLI:
    """流式命令行对话界面"""

    def __init__(self, system_prompt: str = None):
        self.model = ChatOpenAI(model="gpt-4o", streaming=True)
        self.messages: List = []

        if system_prompt:
            self.messages.append(SystemMessage(content=system_prompt))

    def chat(self):
        """开始对话"""
        print("╔════════════════════════════════════════╗")
        print("║  流式对话助手                          ║")
        print("║  输入 'quit' 退出，'clear' 清空历史    ║")
        print("╚════════════════════════════════════════╝")

        while True:
            try:
                # 获取用户输入
                user_input = input("\n👤 你：").strip()

                if not user_input:
                    continue

                if user_input.lower() == "quit":
                    print("再见！")
                    break

                if user_input.lower() == "clear":
                    self.messages = self.messages[:1]  # 保留系统消息
                    print("对话历史已清空")
                    continue

                # 添加用户消息
                self.messages.append(HumanMessage(content=user_input))

                # 流式输出 AI 响应
                print("🤖 AI: ", end="")

                full_response = ""
                for chunk in self.model.stream(self.messages):
                    content = chunk.content
                    print(content, end="", flush=True)
                    full_response += content

                print()  # 换行

                # 添加 AI 响应到历史
                self.messages.append(AIMessage(content=full_response))

            except KeyboardInterrupt:
                print("\n\n中断退出")
                break
            except Exception as e:
                print(f"\n错误：{e}")

# 运行
if __name__ == "__main__":
    cli = StreamingChatCLI("你是一个友好的助手。")
    cli.chat()

    10.5.2 带进度条的流式输出¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

def stream_with_progress(prompt: str):
    """带进度条的流式输出"""
    model = ChatOpenAI(model="gpt-4o", streaming=True)

    total_chars = 0
    line_count = 0

    print("🤖 AI: ", end="")

    for chunk in model.stream([HumanMessage(content=prompt)]):
        content = chunk.content
        print(content, end="", flush=True)

        total_chars += len(content)
        line_count += content.count('\n')

        # 每行更新一次状态
        if '\n' in content:
            print(f"  [{line_count}行，{total_chars}字]", end="\r", flush=True)

    print(f"\n✅ 完成：{line_count}行，{total_chars}字")

# 使用
stream_with_progress("写一篇关于人工智能的短文，至少 500 字")

    10.5.3 Web 流式输出（Flask 示例）¶

    from flask import Flask, Response, stream_with_context
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage
import json

app = Flask(__name__)
model = ChatOpenAI(model="gpt-4o", streaming=True)

@app.route('/chat', methods=['POST'])
def chat():
    from flask import request

    user_input = request.json.get('message', '')
    messages = [HumanMessage(content=user_input)]

    def generate():
        for chunk in model.stream(messages):
            # Server-Sent Events 格式
            data = {
                "type": "token",
                "content": chunk.content,
            }
            yield f"data: {json.dumps(data)}\n\n"

        # 完成信号
        yield f"data: {json.dumps({'type': 'done'})}\n\n"

    return Response(
        stream_with_context(generate()),
        mimetype='text/event-stream'
    )

# HTML 前端示例
html_frontend = """
<!DOCTYPE html>
<html>
<head>
    <title>流式对话</title>
</head>
<body>
    <div id="chat"></div>
    <input id="input" placeholder="输入消息...">
    <button onclick="send()">发送</button>

    <script>
        async function send() {
            const input = document.getElementById('input');
            const chat = document.getElementById('chat');
            const message = input.value;

            chat.innerHTML += '<div><strong>你:</strong> ' + message + '</div>';
            input.value = '';

            const response = await fetch('/chat', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({message: message})
            });

            const reader = response.body.getReader();
            const decoder = new TextDecoder();

            chat.innerHTML += '<div><strong>AI:</strong> <span id="response"></span></div>';
            const responseSpan = document.getElementById('response');

            while (true) {
                const {done, value} = await reader.read();
                if (done) break;

                const text = decoder.decode(value);
                const lines = text.split('\\n');

                for (const line of lines) {
                    if (line.startsWith('data: ')) {
                        const data = JSON.parse(line.slice(6));
                        if (data.type === 'token') {
                            responseSpan.textContent += data.content;
                        }
                    }
                }
            }
        }
    </script>
</body>
</html>
"""

if __name__ == '__main__':
    app.run(debug=True, port=5000)

    10.5.4 异步流式对话¶

    import asyncio
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage

class AsyncStreamingChat:
    """异步流式对话"""

    def __init__(self):
        self.model = ChatOpenAI(model="gpt-4o")
        self.messages = []

    async def chat(self, user_input: str):
        """异步对话"""
        self.messages.append(HumanMessage(content=user_input))

        full_response = ""

        # 异步流式输出
        async for chunk in self.model.astream(self.messages):
            print(chunk.content, end="", flush=True)
            full_response += chunk.content

        print()

        self.messages.append(AIMessage(content=full_response))
        return full_response

    async def interactive(self):
        """交互式对话"""
        print("异步流式对话助手（输入 'quit' 退出）")

        while True:
            # 异步获取输入
            user_input = await asyncio.get_event_loop().run_in_executor(
                None, input, "\n你："
            )

            if user_input.lower() == "quit":
                break

            await self.chat(user_input)

# 运行
async def main():
    chat = AsyncStreamingChat()
    await chat.interactive()

asyncio.run(main())

    10.6.1 流式处理长文本¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

model = ChatOpenAI(model="gpt-4o")

# 分块处理长文本
def stream_long_text(prompt: str, chunk_size: int = 100):
    buffer = ""

    for chunk in model.stream([HumanMessage(content=prompt)]):
        buffer += chunk.content

        # 每 chunk_size 个字符输出一次
        while len(buffer) >= chunk_size:
            print(buffer[:chunk_size], end="", flush=True)
            buffer = buffer[chunk_size:]

    # 输出剩余内容
    if buffer:
        print(buffer, end="", flush=True)

stream_long_text("写一篇 2000 字的文章")

    10.6.2 流式中断¶

    import signal

class InterruptibleStream:
    """可中断的流式输出"""

    def __init__(self):
        self.interrupted = False
        signal.signal(signal.SIGINT, self._handler)

    def _handler(self, signum, frame):
        self.interrupted = True
        print("\n[用户中断]")

    def stream(self, model, messages):
        for chunk in model.stream(messages):
            if self.interrupted:
                print("\n[已停止]")
                break
            print(chunk.content, end="", flush=True)

# 使用
streamer = InterruptibleStream()
model = ChatOpenAI(model="gpt-4o")
streamer.stream(model, [HumanMessage(content="写一个长故事")])

    10.6.3 流式性能优化¶

    from langchain_openai import ChatOpenAI
import time

# 优化 1：使用合适的 chunk 大小
model = ChatOpenAI(
    model="gpt-4o",
    streaming=True,
    # 某些 provider 支持配置 chunk 大小
)

# 优化 2：批量处理输出
def batched_stream(model, messages, batch_size: int = 5):
    buffer = []

    for chunk in model.stream(messages):
        buffer.append(chunk.content)

        if len(buffer) >= batch_size:
            print("".join(buffer), end="", flush=True)
            buffer = []

    if buffer:
        print("".join(buffer), end="", flush=True)

# 优化 3：异步预取
async def prefetch_stream(model, messages):
    chunks = []

    async for chunk in model.astream(messages):
        chunks.append(chunk)
        # 可以在后台处理已接收的 chunks

    return chunks
