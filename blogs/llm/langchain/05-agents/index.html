    工具调用链¶

    from langchain.agents import create_agent

def get_city_coordinates(city: str) -> str:
    """Get coordinates for a city.

    Returns: "lat,lon" format
    """
    # 返回坐标
    return "39.9042,116.4074"

def get_weather_by_coords(lat: float, lon: float) -> str:
    """Get weather by coordinates."""
    return f"Weather at {lat},{lon}: Sunny"

# Agent 会自动串联这两个工具
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_city_coordinates, get_weather_by_coords],
    system_prompt="Use tools to get weather information.",
)

result = agent.invoke({
    "messages": [{"role": "user", "content": "北京天气"}]
})
# Agent 可能：
# 1. 调用 get_city_coordinates("北京") → "39.9042,116.4074"
# 2. 调用 get_weather_by_coords(39.9042, 116.4074) → "Sunny"
# 3. 整合回复

    工具错误处理¶

    def search_api(query: str) -> str:
    """Search an API that might fail."""
    try:
        response = requests.get(f"https://api.example.com/search?q={query}")
        response.raise_for_status()
        return response.text
    except requests.exceptions.Timeout:
        return "Error: API timeout, please try again"
    except requests.exceptions.HTTPError as e:
        return f"Error: API returned {e.response.status_code}"
    except Exception as e:
        return f"Error: {str(e)}"

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[search_api],
    system_prompt="""If a tool returns an error:
1. Inform the user about the error
2. Suggest trying again
3. Offer alternative approaches""",
)

    重试逻辑¶

    import time
from functools import wraps

def retry(max_attempts=3, delay=1):
    """Retry decorator for tools."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        return f"Error after {max_attempts} attempts: {e}"
                    time.sleep(delay * (attempt + 1))  # 指数退避
            return "Error: Unknown"
        return wrapper
    return decorator

@retry(max_attempts=3, delay=1)
def flaky_tool(data: str) -> str:
    """A tool that sometimes fails."""
    # 可能失败的实现
    pass

    Agent 评估¶

    测试 Agent¶

    from langchain.agents import create_agent

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[...],
)

# 测试用例
test_cases = [
    {
        "input": "北京天气如何？",
        "expected_tool": "get_weather",
    },
    {
        "input": "计算 123 * 456",
        "expected_tool": "calculator",
    },
    {
        "input": "你好",
        "expected_tool": None,  # 不需要工具
    },
]

# 运行测试
for test in test_cases:
    result = agent.invoke({
        "messages": [{"role": "user", "content": test["input"]}]
    })

    # 检查是否使用了正确的工具
    tool_calls = [m for m in result["messages"] if m.type == "tool"]

    if test["expected_tool"]:
        assert any(m.name == test["expected_tool"] for m in tool_calls)
        print(f"✅ {test['input']}")
    else:
        assert len(tool_calls) == 0
        print(f"✅ {test['input']} (no tool needed)")

    性能测试¶

    import time

def benchmark_agent(agent, queries, iterations=3):
    """Benchmark agent performance."""
    results = []

    for query in queries:
        times = []
        for _ in range(iterations):
            start = time.time()
            agent.invoke({"messages": [{"role": "user", "content": query}]})
            end = time.time()
            times.append(end - start)

        avg_time = sum(times) / len(times)
        results.append({
            "query": query,
            "avg_time": avg_time,
            "min_time": min(times),
            "max_time": max(times),
        })

    return results

# 使用
queries = ["简单问题", "复杂问题", "多步骤问题"]
results = benchmark_agent(agent, queries)

for r in results:
    print(f"{r['query']}: {r['avg_time']:.2f}s")

    ✅ 推荐¶

    # 1. 编写详细的系统提示
system_prompt = """You are a helpful assistant.
- Use tools for factual questions
- Be concise but thorough
- Ask for clarification if unsure"""

# 2. 保持对话历史
messages = []
while True:
    user_input = input("你：")
    messages.append({"role": "user", "content": user_input})
    result = agent.invoke({"messages": messages})
    messages.extend(result["messages"])

# 3. 限制上下文长度
MAX_MESSAGES = 20
if len(messages) > MAX_MESSAGES:
    messages = messages[-MAX_MESSAGES:]

# 4. 清晰的工具文档
def tool(param: str) -> str:
    """Clear description.

    Args:
        param: Detailed description

    Returns:
        What this returns
    """

# 5. 适当的错误处理
def tool(param: str) -> str:
    """A tool."""
    try:
        return do_something(param)
    except Exception as e:
        return f"Error: {e}"

    ❌ 避免¶

    # 1. 模糊的系统提示
system_prompt = "Be helpful"  # ❌ 太模糊

# 2. 忽略对话历史
messages = [{"role": "user", "content": user_input}]  # ❌ 没有上下文

# 3. 不限制上下文长度
# ❌ 可能导致 token 超限或成本过高

# 4. 工具文档不清晰
def tool(x):  # ❌ 没有文档
    return x

# 5. 不处理错误
def tool(param):  # ❌ 可能崩溃
    return api_call(param)
