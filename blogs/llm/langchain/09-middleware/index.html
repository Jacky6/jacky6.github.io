    9.1 中间件概述¶

    9.1.1 什么是中间件¶

    中间件是在模型调用前后插入的自定义逻辑，类似于 Web 框架中的中间件：

    ┌─────────────────────────────────────────────────────────┐
│                    用户请求                              │
└────────────────────┬────────────────────────────────────┘
                     │
                     ▼
┌─────────────────────────────────────────────────────────┐
│              before_model 中间件                         │
│  • 修改提示词                                            │
│  • 添加系统指令                                          │
│  • 检查权限                                              │
│  • 请求日志                                              │
└────────────────────┬────────────────────────────────────┘
                     │
                     ▼
┌─────────────────────────────────────────────────────────┐
│                    LLM 模型                              │
└────────────────────┬────────────────────────────────────┘
                     │
                     ▼
┌─────────────────────────────────────────────────────────┐
│              after_model 中间件                          │
│  • 转换响应格式                                          │
│  • 内容过滤                                              │
│  • 响应日志                                              │
│  • 缓存结果                                              │
└────────────────────┬────────────────────────────────────┘
                     │
                     ▼
┌─────────────────────────────────────────────────────────┐
│                    用户响应                              │
└─────────────────────────────────────────────────────────┘

    9.2 before_model 中间件¶

    9.2.1 基础用法¶

    from langchain_core.messages import HumanMessage, SystemMessage
from langchain_openai import ChatOpenAI
from langchain_core.middleware import before_model

# 创建模型
model = ChatOpenAI(model="gpt-4o")

# 定义 before_model 中间件
@before_model
def add_system_prompt(messages, config, **kwargs):
    """在请求前添加系统提示"""
    # 在消息列表开头添加系统消息
    system_message = SystemMessage(content="你是一个专业的编程助手，擅长 Python 和 Web 开发。")

    # 如果已经有系统消息，跳过
    if messages and isinstance(messages[0], SystemMessage):
        return messages

    return [system_message] + messages

# 应用中间件
model_with_middleware = model | add_system_prompt

# 使用
response = model_with_middleware.invoke([HumanMessage(content="如何读取文件？")])
print(response.content)

    9.2.2 修改用户输入¶

    from langchain_core.middleware import before_model
from langchain_core.messages import HumanMessage

@before_model
def enhance_input(messages, config, **kwargs):
    """增强用户输入"""
    if not messages:
        return messages

    last_message = messages[-1]
    if isinstance(last_message, HumanMessage):
        # 优化用户输入
        enhanced_content = f"请详细回答以下问题，并提供代码示例：\n\n{last_message.content}"

        # 创建新消息
        enhanced_message = HumanMessage(content=enhanced_content)

        # 替换最后一条消息
        return messages[:-1] + [enhanced_message]

    return messages

# 应用
model = ChatOpenAI(model="gpt-4o")
enhanced_model = model | enhance_input

response = enhanced_model.invoke([HumanMessage(content="Python 怎么读取 JSON？")])

    9.2.3 添加上下文信息¶

    from langchain_core.middleware import before_model
from langchain_core.messages import HumanMessage
from datetime import datetime

@before_model
def add_context(messages, config, **kwargs):
    """添加上下文信息（时间、用户等）"""
    # 获取当前时间
    current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    # 获取用户信息（从 config 或 kwargs）
    user_name = kwargs.get("user_name", "用户")
    user_location = kwargs.get("location", "未知")

    # 构建上下文
    context = f"""当前信息：
- 时间：{current_time}
- 用户：{user_name}
- 位置：{user_location}

请在回答时考虑以上上下文信息。

"""

    # 添加到用户消息
    if messages and isinstance(messages[-1], HumanMessage):
        original_content = messages[-1].content
        new_content = context + original_content
        messages[-1] = HumanMessage(content=new_content)

    return messages

# 使用
model = ChatOpenAI(model="gpt-4o")
context_model = model | add_context

response = context_model.invoke(
    [HumanMessage(content="现在适合做什么运动？")],
    user_name="张三",
    location="北京"
)

    9.2.4 输入验证和过滤¶

    from langchain_core.middleware import before_model
from langchain_core.messages import HumanMessage
import re

class InputValidationError(Exception):
    """输入验证错误"""
    pass

@before_model
def validate_input(messages, config, **kwargs):
    """验证和过滤用户输入"""
    if not messages:
        raise InputValidationError("消息不能为空")

    last_message = messages[-1]
    if isinstance(last_message, HumanMessage):
        content = last_message.content

        # 检查长度
        if len(content) > 10000:
            raise InputValidationError("输入内容过长，请限制在 10000 字以内")

        # 检查敏感词
        sensitive_words = ["密码", "secret", "token", "apikey"]
        for word in sensitive_words:
            if word.lower() in content.lower():
                raise InputValidationError(f"输入包含敏感词：{word}")

        # 检查是否为空
        if not content.strip():
            raise InputValidationError("输入不能为空")

        # 清理特殊字符
        cleaned_content = re.sub(r'[^\w\s\u4e00-\u9fff.,!?;:()\'"-]', '', content)
        messages[-1] = HumanMessage(content=cleaned_content)

    return messages

# 使用
model = ChatOpenAI(model="gpt-4o")
validated_model = model | validate_input

try:
    response = validated_model.invoke([HumanMessage(content="正常问题")])
except InputValidationError as e:
    print(f"验证失败：{e}")

    9.3.3 内容过滤¶

    from langchain_core.middleware import after_model
from langchain_core.messages import AIMessage
import re

@after_model
def filter_content(response, config, **kwargs):
    """过滤不适当内容"""
    if isinstance(response, AIMessage):
        content = response.content

        # 过滤敏感词
        sensitive_patterns = [
            r'\b(密码|secret|token|apikey)\b',
            r'\b\d{16}\b',  # 信用卡号
            r'\b\d{11}\b',  # 手机号
        ]

        for pattern in sensitive_patterns:
            content = re.sub(pattern, '[已过滤]', content)

        # 过滤过长的重复内容
        content = re.sub(r'(.)\1{10,}', r'\1\1\1', content)

        return AIMessage(content=content)

    return response

# 应用
model = ChatOpenAI(model="gpt-4o")
filtered_model = model | filter_content

    9.3.4 响应缓存¶

    from langchain_core.middleware import after_model, before_model
from langchain_core.messages import HumanMessage, AIMessage
import hashlib
import json

class ResponseCache:
    """响应缓存"""

    def __init__(self):
        self.cache = {}

    def _get_key(self, messages) -> str:
        """生成缓存键"""
        content = str([m.content for m in messages])
        return hashlib.md5(content.encode()).hexdigest()

    @before_model
    def cache_lookup(self, messages, config, **kwargs):
        """缓存查找"""
        key = self._get_key(messages)
        if key in self.cache:
            print(f"[缓存命中] {key}")
            # 返回缓存的响应
            raise CacheHitException(self.cache[key])
        print(f"[缓存未命中] {key}")
        return messages

    @after_model
    def cache_store(self, response, config, **kwargs):
        """存储缓存"""
        messages = kwargs.get("_original_messages", [])
        key = self._get_key(messages)
        self.cache[key] = response
        print(f"[缓存存储] {key}")
        return response

class CacheHitException(Exception):
    """缓存命中异常"""
    def __init__(self, response):
        self.response = response

# 使用
cache = ResponseCache()
model = ChatOpenAI(model="gpt-4o")
cached_model = model | cache.cache_lookup | cache.cache_store

# 第一次调用（缓存未命中）
response1 = cached_model.invoke("什么是 Python？")
print(f"响应 1: {response1.content[:50]}...")

# 第二次调用（缓存命中）
try:
    response2 = cached_model.invoke("什么是 Python？")
except CacheHitException as e:
    response2 = e.response
    print(f"响应 2 (缓存): {response2.content[:50]}...")

    9.4 wrap_tool_call 中间件¶

    9.4.1 基础用法¶

    from langchain_core.middleware import wrap_tool_call
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI

# 定义工具
@tool
def get_weather(city: str) -> str:
    """获取城市天气"""
    return f"{city}今天天气晴朗，25°C"

@tool
def get_time() -> str:
    """获取当前时间"""
    from datetime import datetime
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

# 定义 wrap_tool_call 中间件
@wrap_tool_call
def log_tool_calls(tool_call, config, **kwargs):
    """记录工具调用日志"""
    print(f"[工具调用] {tool_call['name']}({tool_call['args']})")

    # 继续执行工具
    result = tool_call.invoke()

    print(f"[工具结果] {result}")
    return result

# 应用
model = ChatOpenAI(model="gpt-4o")
model_with_tools = model.bind_tools([get_weather, get_time])
model_with_middleware = model_with_tools | log_tool_calls

response = model_with_middleware.invoke("北京天气怎么样？现在几点了？")

    9.4.2 工具调用拦截¶

    from langchain_core.middleware import wrap_tool_call
from langchain_core.tools import tool

# 定义受保护的工具列表
PROTECTED_TOOLS = ["delete_data", "execute_sql", "run_command"]

@wrap_tool_call
def protect_sensitive_tools(tool_call, config, **kwargs):
    """保护敏感工具"""
    tool_name = tool_call.get("name")

    # 检查是否是敏感工具
    if tool_name in PROTECTED_TOOLS:
        # 检查权限（从 config 或 kwargs）
        user_role = kwargs.get("user_role", "guest")
        if user_role not in ["admin", "superuser"]:
            raise PermissionError(f"无权调用工具：{tool_name}")

    # 记录审计日志
    print(f"[审计] 用户 {kwargs.get('user_id', 'unknown')} 调用工具 {tool_name}")

    # 继续执行
    return tool_call.invoke()

# 定义敏感工具
@tool
def delete_data(table: str, condition: str) -> str:
    """删除数据（敏感操作）"""
    return f"已删除 {table} 中满足 {condition} 的数据"

# 使用
model = ChatOpenAI(model="gpt-4o")
model_with_tools = model.bind_tools([delete_data])
protected_model = model_with_tools | protect_sensitive_tools

try:
    response = protected_model.invoke(
        "删除 users 表中 id=1 的数据",
        user_id="user_001",
        user_role="guest"
    )
except PermissionError as e:
    print(f"权限错误：{e}")

    9.4.3 工具调用重试¶

    from langchain_core.middleware import wrap_tool_call
from langchain_core.tools import tool
from tenacity import retry, stop_after_attempt, wait_exponential

@wrap_tool_call
def retry_on_failure(tool_call, config, **kwargs):
    """工具调用失败时重试"""

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=1, max=10)
    )
    def invoke_with_retry():
        return tool_call.invoke()

    try:
        return invoke_with_retry()
    except Exception as e:
        print(f"[工具调用失败] {tool_call['name']}: {e}")
        # 返回错误信息而不是抛出异常
        return f"工具调用失败：{str(e)}"

# 定义可能失败的工具
@tool
def fetch_external_api(url: str) -> str:
    """调用外部 API（可能失败）"""
    import requests
    response = requests.get(url, timeout=5)
    return response.text

# 使用
model = ChatOpenAI(model="gpt-4o")
model_with_tools = model.bind_tools([fetch_external_api])
resilient_model = model_with_tools | retry_on_failure

    9.4.4 工具调用超时¶

    from langchain_core.middleware import wrap_tool_call
import signal

class TimeoutError(Exception):
    pass

def timeout_handler(signum, frame):
    raise TimeoutError("工具调用超时")

@wrap_tool_call
def with_timeout(tool_call, config, **kwargs):
    """工具调用超时控制"""
    timeout_seconds = kwargs.get("tool_timeout", 30)

    # 设置超时信号
    signal.signal(signal.SIGALRM, timeout_handler)
    signal.alarm(timeout_seconds)

    try:
        result = tool_call.invoke()
        signal.alarm(0)  # 取消超时
        return result
    except TimeoutError:
        return f"工具调用超时（超过{timeout_seconds}秒）"
    finally:
        signal.alarm(0)

# 使用
model = ChatOpenAI(model="gpt-4o")

@tool
def slow_operation() -> str:
    """慢操作"""
    import time
    time.sleep(60)  # 模拟慢操作
    return "完成"

model_with_tools = model.bind_tools([slow_operation])
timeout_model = model_with_tools | with_timeout

response = timeout_model.invoke(
    "执行慢操作",
    tool_timeout=5  # 5 秒超时
)

    9.5 组合中间件¶

    9.5.1 中间件链¶

    from langchain_core.middleware import before_model, after_model
from langchain_openai import ChatOpenAI

# 定义多个中间件
@before_model
def add_timestamp(messages, config, **kwargs):
    from datetime import datetime
    print(f"[{datetime.now().strftime('%H:%M:%S')}] 收到请求")
    return messages

@before_model
def validate_input(messages, config, **kwargs):
    if messages and len(messages[-1].content) > 1000:
        print("⚠️  输入过长")
    return messages

@after_model
def log_response(response, config, **kwargs):
    from datetime import datetime
    print(f"[{datetime.now().strftime('%H:%M:%S')}] 返回响应")
    return response

@after_model
def add_footer(response, config, **kwargs):
    if hasattr(response, 'content'):
        response.content += "\n\n---\n_由 AI 生成_"
    return response

# 组合中间件链
model = ChatOpenAI(model="gpt-4o")
model_with_chain = (
    model
    | add_timestamp
    | validate_input
    | log_response
    | add_footer
)

response = model_with_chain.invoke("你好")

    9.5.2 条件中间件¶

    from langchain_core.middleware import before_model, after_model

def conditional_middleware(condition_fn, middleware_fn):
    """条件中间件"""
    def wrapper(*args, **kwargs):
        if condition_fn(*args, **kwargs):
            return middleware_fn(*args, **kwargs)
        return args[0]  # 返回原始输入

    wrapper.__name__ = middleware_fn.__name__
    return wrapper

# 只对特定用户启用中间件
def is_premium_user(config, **kwargs):
    return kwargs.get("user_tier") == "premium"

@before_model
def premium_enhance(messages, config, **kwargs):
    """为高级用户增强输入"""
    system_prompt = "你是一个高级助手，提供更详细和专业的回答。"
    from langchain_core.messages import SystemMessage
    return [SystemMessage(content=system_prompt)] + messages

# 条件应用
enhanced_middleware = conditional_middleware(is_premium_user, premium_enhance)

model = ChatOpenAI(model="gpt-4o")
conditional_model = model | enhanced_middleware

# 普通用户
response1 = conditional_model.invoke("问题", user_tier="free")

# 高级用户
response2 = conditional_model.invoke("问题", user_tier="premium")

    9.6 实战：完整的中间件系统¶

    9.6.1 企业级中间件栈¶

    from langchain_core.middleware import before_model, after_model, wrap_tool_call
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
import json
import time
import hashlib
from datetime import datetime
from typing import Dict, Any, List

class EnterpriseMiddleware:
    """企业级中间件系统"""

    def __init__(self, config: Dict[str, Any] = None):
        self.config = config or {}
        self.cache = {}
        self.metrics = {
            "requests": 0,
            "cache_hits": 0,
            "cache_misses": 0,
            "total_time": 0,
            "errors": 0,
        }

    # ========== before_model 中间件 ==========

    @before_model
    def request_logging(self, messages, config, **kwargs):
        """请求日志"""
        self.metrics["requests"] += 1
        start_time = time.time()

        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "message_count": len(messages),
            "last_message": messages[-1].content[:100] if messages else "",
            "user_id": kwargs.get("user_id", "unknown"),
        }

        print(f"[请求] {json.dumps(log_entry, ensure_ascii=False)}")

        # 存储开始时间
        config["__start_time"] = start_time

        return messages

    @before_model
    def input_validation(self, messages, config, **kwargs):
        """输入验证"""
        if not messages:
            raise ValueError("消息不能为空")

        last_message = messages[-1]
        if isinstance(last_message, HumanMessage):
            content = last_message.content

            # 长度检查
            if len(content) > 50000:
                raise ValueError("输入内容过长")

            # 敏感信息检查
            sensitive_patterns = [
                r'password\s*[=:]\s*\S+',
                r'secret\s*[=:]\s*\S+',
                r'api[_-]?key\s*[=:]\s*\S+',
            ]

            import re
            for pattern in sensitive_patterns:
                if re.search(pattern, content, re.IGNORECASE):
                    raise ValueError("输入包含敏感信息")

        return messages

    @before_model
    def cache_lookup(self, messages, config, **kwargs):
        """缓存查找"""
        if not self.config.get("enable_cache", True):
            return messages

        # 生成缓存键
        cache_key = self._generate_cache_key(messages)

        if cache_key in self.cache:
            self.metrics["cache_hits"] += 1
            print(f"[缓存命中] {cache_key[:16]}...")

            # 返回缓存响应
            cached_response = self.cache[cache_key]
            cached_response.additional_kwargs["from_cache"] = True
            raise CacheHitException(cached_response)

        self.metrics["cache_misses"] += 1
        return messages

    @before_model
    def add_system_context(self, messages, config, **kwargs):
        """添加系统上下文"""
        context_parts = []

        # 时间上下文
        context_parts.append(f"当前时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")

        # 用户上下文
        user_info = kwargs.get("user_info", {})
        if user_info.get("name"):
            context_parts.append(f"用户：{user_info['name']}")
        if user_info.get("role"):
            context_parts.append(f"角色：{user_info['role']}")

        if context_parts:
            system_content = "\n".join(context_parts)
            system_message = SystemMessage(content=system_content)

            # 如果已有系统消息，跳过
            if not (messages and isinstance(messages[0], SystemMessage)):
                messages = [system_message] + messages

        return messages

    # ========== after_model 中间件 ==========

    @after_model
    def response_logging(self, response, config, **kwargs):
        """响应日志"""
        end_time = time.time()
        start_time = config.get("__start_time", end_time)
        duration = end_time - start_time

        self.metrics["total_time"] += duration

        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "duration_ms": round(duration * 1000, 2),
            "response_length": len(response.content) if hasattr(response, 'content') else 0,
        }

        print(f"[响应] {json.dumps(log_entry, ensure_ascii=False)}")

        return response

    @after_model
    def cache_store(self, response, config, **kwargs):
        """存储缓存"""
        if not self.config.get("enable_cache", True):
            return response

        # 不缓存错误响应
        if response.additional_kwargs.get("from_cache"):
            return response

        # 生成缓存键
        messages = kwargs.get("_original_messages", [])
        cache_key = self._generate_cache_key(messages)

        # 存储缓存
        self.cache[cache_key] = response

        # 限制缓存大小
        if len(self.cache) > 1000:
            # 删除最旧的缓存
            oldest_key = next(iter(self.cache))
            del self.cache[oldest_key]

        return response

    @after_model
    def content_safety(self, response, config, **kwargs):
        """内容安全检查"""
        if not hasattr(response, 'content'):
            return response

        content = response.content

        # 检查并过滤敏感内容
        sensitive_patterns = [
            (r'\b\d{16}\b', '[信用卡号已隐藏]'),
            (r'\b\d{11}\b', '[手机号已隐藏]'),
        ]

        import re
        for pattern, replacement in sensitive_patterns:
            content = re.sub(pattern, replacement, content)

        response.content = content
        return response

    @after_model
    def add_metadata(self, response, config, **kwargs):
        """添加响应元数据"""
        if hasattr(response, 'additional_kwargs'):
            response.additional_kwargs.update({
                "generated_at": datetime.now().isoformat(),
                "model": self.config.get("model", "unknown"),
                "user_id": kwargs.get("user_id", "unknown"),
            })

        return response

    @after_model
    def response_timing(self, response, config, **kwargs):
        """添加响应时间信息"""
        if hasattr(response, 'content'):
            end_time = time.time()
            start_time = config.get("__start_time", end_time)
            duration = end_time - start_time

            timing_info = f"\n\n_响应时间：{duration:.2f}秒_"
            response.content += timing_info

        return response

    # ========== wrap_tool_call 中间件 ==========

    @wrap_tool_call
    def tool_logging(self, tool_call, config, **kwargs):
        """工具调用日志"""
        tool_name = tool_call.get("name", "unknown")
        tool_args = tool_call.get("args", {})

        print(f"[工具调用] {tool_name}({json.dumps(tool_args, ensure_ascii=False)})")

        start_time = time.time()
        try:
            result = tool_call.invoke()
            duration = time.time() - start_time
            print(f"[工具完成] {tool_name} ({duration:.2f}s)")
            return result
        except Exception as e:
            self.metrics["errors"] += 1
            print(f"[工具错误] {tool_name}: {e}")
            raise

    # ========== 辅助方法 ==========

    def _generate_cache_key(self, messages) -> str:
        """生成缓存键"""
        content = str([m.content for m in messages])
        return hashlib.md5(content.encode()).hexdigest()

    def get_metrics(self) -> Dict[str, Any]:
        """获取指标"""
        return {
            **self.metrics,
            "cache_size": len(self.cache),
            "cache_hit_rate": (
                self.metrics["cache_hits"] / 
                (self.metrics["cache_hits"] + self.metrics["cache_misses"])
                if (self.metrics["cache_hits"] + self.metrics["cache_misses"]) > 0
                else 0
            ),
            "avg_response_time": (
                self.metrics["total_time"] / self.metrics["requests"]
                if self.metrics["requests"] > 0
                else 0
            ),
        }

class CacheHitException(Exception):
    """缓存命中异常"""
    def __init__(self, response):
        self.response = response

    9.7.1 中间件设计原则¶

    # ✅ 好的中间件设计
class GoodMiddleware:
    # 单一职责
    @before_model
    def only_one_thing(self, messages, config, **kwargs):
        """只做一件事"""
        return messages

    # 无副作用
    @after_model
    def pure_function(self, response, config, **kwargs):
        """不修改外部状态"""
        return response

    # 错误处理
    @wrap_tool_call
    def handle_errors(self, tool_call, config, **kwargs):
        try:
            return tool_call.invoke()
        except Exception as e:
            # 优雅降级
            return f"操作失败：{e}"

# ❌ 不好的中间件设计
class BadMiddleware:
    @before_model
    def do_everything(self, messages, config, **kwargs):
        """做太多事情"""
        # 验证、日志、缓存、转换... 都在一起
        pass

    9.7.2 性能考虑¶

    # ✅ 高效的缓存
class EfficientCache:
    def __init__(self, max_size=1000):
        self.cache = {}
        self.max_size = max_size
        self.access_order = []

    @before_model
    def lookup(self, messages, config, **kwargs):
        key = self._hash(messages)
        if key in self.cache:
            # 更新访问顺序
            self.access_order.remove(key)
            self.access_order.append(key)
            raise CacheHitException(self.cache[key])
        return messages

    @after_model
    def store(self, response, config, **kwargs):
        key = self._hash(kwargs.get("_original_messages", []))

        # LRU 淘汰
        if len(self.cache) >= self.max_size:
            oldest = self.access_order.pop(0)
            del self.cache[oldest]

        self.cache[key] = response
        self.access_order.append(key)
        return response

# ❌ 低效的缓存
class InefficientCache:
    def __init__(self):
        self.cache = {}  # 无限制增长

    @after_model
    def store(self, response, config, **kwargs):
        key = str(kwargs.get("_original_messages", []))  # 低效的键生成
        self.cache[key] = response  # 永不清理
        return response

    9.7.3 调试技巧¶

    import logging
from functools import wraps

# 启用详细日志
logging.basicConfig(level=logging.DEBUG)
logger = logging.getLogger("middleware")

def debug_middleware(middleware_fn):
    """调试中间件装饰器"""
    @wraps(middleware_fn)
    def wrapper(*args, **kwargs):
        logger.debug(f"调用 {middleware_fn.__name__}")
        logger.debug(f"参数：{args}, {kwargs}")

        result = middleware_fn(*args, **kwargs)

        logger.debug(f"返回：{result}")
        return result

    return wrapper

# 使用
@debug_middleware
@before_model
def my_middleware(messages, config, **kwargs):
    return messages
