    14.2.1 简单的多代理系统¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from typing import List, Dict

class Agent:
    """基础代理"""

    def __init__(self, name: str, role: str, expertise: str):
        self.name = name
        self.role = role
        self.expertise = expertise
        self.model = ChatOpenAI(model="gpt-4o")

    def process(self, task: str, context: List[str] = None) -> str:
        """处理任务"""
        system_prompt = f"""你是{self.name}，{self.role}。
你的专业领域：{self.expertise}

请专业地完成任务。"""

        messages = [SystemMessage(content=system_prompt)]

        if context:
            for ctx in context:
                messages.append(AIMessage(content=ctx))

        messages.append(HumanMessage(content=task))

        response = self.model.invoke(messages)
        return response.content

class MultiAgentSystem:
    """多代理系统"""

    def __init__(self):
        self.agents: Dict[str, Agent] = {}
        self.history: List[str] = []

    def add_agent(self, agent: Agent):
        """添加代理"""
        self.agents[agent.name] = agent

    def delegate(self, task: str, agent_name: str) -> str:
        """分配任务给指定代理"""
        if agent_name not in self.agents:
            raise ValueError(f"未知代理：{agent_name}")

        agent = self.agents[agent_name]
        result = agent.process(task, self.history[-5:])  # 提供最近上下文

        self.history.append(f"{agent_name}: {result[:100]}...")
        return result

    def broadcast(self, task: str) -> Dict[str, str]:
        """广播任务给所有代理"""
        results = {}

        for name, agent in self.agents.items():
            print(f"分配给 {name}...")
            results[name] = self.delegate(task, name)

        return results

# 使用
system = MultiAgentSystem()

# 添加专业代理
system.add_agent(Agent(
    name="研究员",
    role="信息研究专家",
    expertise="收集、整理、分析信息"
))

system.add_agent(Agent(
    name="作家",
    role="内容创作专家",
    expertise="撰写文章、故事、报告"
))

system.add_agent(Agent(
    name="程序员",
    role="代码开发专家",
    expertise="编写、审查、优化代码"
))

# 分配任务
research_result = system.delegate("研究人工智能的最新进展", "研究员")
print(f"研究员：{research_result[:100]}...")

writing_result = system.delegate("根据研究结果写一篇科普文章", "作家")
print(f"作家：{writing_result[:100]}...")

    14.2.2 主管 - 员工模式¶

    from typing import List, Dict

class Supervisor:
    """主管代理"""

    def __init__(self):
        self.model = ChatOpenAI(model="gpt-4o")
        self.workers: Dict[str, Agent] = {}

    def add_worker(self, worker: Agent):
        """添加员工"""
        self.workers[worker.name] = worker

    def analyze_task(self, task: str) -> Dict[str, str]:
        """分析任务并分解"""
        prompt = f"""分析以下任务，并分解为子任务：

任务：{task}

请返回 JSON 格式：
{{
    "subtasks": [
        {{"name": "子任务 1", "assigned_to": "代理名", "description": "描述"}},
        {{"name": "子任务 2", "assigned_to": "代理名", "description": "描述"}}
    ]
}}
"""
        response = self.model.invoke([HumanMessage(content=prompt)])

        import json
        try:
            return json.loads(response.content)
        except:
            return {"subtasks": []}

    def execute(self, task: str) -> Dict[str, str]:
        """执行任务"""
        # 分解任务
        plan = self.analyze_task(task)
        print(f"任务分解：{plan}")

        results = {}

        # 分配和执行子任务
        for subtask in plan.get("subtasks", []):
            worker_name = subtask.get("assigned_to")

            if worker_name in self.workers:
                print(f"\n执行：{subtask['name']}")
                result = self.workers[worker_name].process(
                    subtask["description"]
                )
                results[subtask["name"]] = result
                print(f"完成：{result[:50]}...")

        # 汇总结果
        summary = self.summarize(results)
        return {
            "plan": plan,
            "results": results,
            "summary": summary,
        }

    def summarize(self, results: Dict[str, str]) -> str:
        """汇总结果"""
        results_text = "\n\n".join([
            f"## {name}\n{result}"
            for name, result in results.items()
        ])

        prompt = f"""汇总以下工作结果：

{results_text}

请提供一个简洁的总结。"""

        response = self.model.invoke([HumanMessage(content=prompt)])
        return response.content

# 使用
supervisor = Supervisor()

# 添加员工
supervisor.add_worker(Agent("研究员", "研究专家", "信息收集和分析"))
supervisor.add_worker(Agent("作家", "写作专家", "内容创作"))
supervisor.add_worker(Agent("编辑", "编辑专家", "内容审核和优化"))

# 执行复杂任务
result = supervisor.execute("创建一份关于量子计算的科普报告")
print(f"\n总结：{result['summary']}")

    14.3.1 消息传递机制¶

    from dataclasses import dataclass
from datetime import datetime
from typing import Optional

@dataclass
class Message:
    """代理消息"""
    sender: str
    receiver: str
    content: str
    timestamp: str = None
    message_id: str = None

    def __post_init__(self):
        if not self.timestamp:
            self.timestamp = datetime.now().isoformat()
        if not self.message_id:
            import uuid
            self.message_id = str(uuid.uuid4())

class MessageBus:
    """消息总线"""

    def __init__(self):
        self.messages: List[Message] = []
        self.subscribers: Dict[str, List] = {}

    def send(self, message: Message):
        """发送消息"""
        self.messages.append(message)
        print(f"[消息] {message.sender} → {message.receiver}: {message.content[:50]}...")

        # 通知订阅者
        if message.receiver in self.subscribers:
            for callback in self.subscribers[message.receiver]:
                callback(message)

    def subscribe(self, agent_name: str, callback):
        """订阅消息"""
        if agent_name not in self.subscribers:
            self.subscribers[agent_name] = []
        self.subscribers[agent_name].append(callback)

    def get_messages(self, agent_name: str) -> List[Message]:
        """获取发给指定代理的消息"""
        return [m for m in self.messages if m.receiver == agent_name]

    def broadcast(self, sender: str, content: str, receivers: List[str]):
        """广播消息"""
        for receiver in receivers:
            self.send(Message(sender=sender, receiver=receiver, content=content))

# 使用
bus = MessageBus()

def on_message(msg: Message):
    print(f"收到消息：{msg.content}")

bus.subscribe("作家", on_message)
bus.send(Message(sender="研究员", receiver="作家", content="研究完成，发现如下..."))

    14.3.2 共享黑板模式¶

    class SharedBlackboard:
    """共享黑板"""

    def __init__(self):
        self.content: Dict[str, any] = {}
        self.history: List[Dict] = []

    def write(self, key: str, value: any, author: str):
        """写入内容"""
        self.content[key] = value
        self.history.append({
            "action": "write",
            "key": key,
            "author": author,
            "timestamp": datetime.now().isoformat(),
        })
        print(f"[黑板] {author} 写入 {key}: {str(value)[:50]}...")

    def read(self, key: str) -> any:
        """读取内容"""
        return self.content.get(key)

    def get_all(self) -> Dict[str, any]:
        """获取所有内容"""
        return self.content.copy()

    def get_history(self) -> List[Dict]:
        """获取历史记录"""
        return self.history

class BlackboardAgent(Agent):
    """使用黑板的代理"""

    def __init__(self, name: str, role: str, expertise: str, blackboard: SharedBlackboard):
        super().__init__(name, role, expertise)
        self.blackboard = blackboard

    def contribute(self, key: str, content: str):
        """贡献内容到黑板"""
        self.blackboard.write(key, content, self.name)

    def read_context(self) -> str:
        """读取黑板上的上下文"""
        all_content = self.blackboard.get_all()
        return "\n\n".join([
            f"## {key}\n{value}"
            for key, value in all_content.items()
        ])

# 使用
blackboard = SharedBlackboard()

researcher = BlackboardAgent("研究员", "研究专家", "信息分析", blackboard)
writer = BlackboardAgent("作家", "写作专家", "内容创作", blackboard)
editor = BlackboardAgent("编辑", "编辑专家", "内容审核", blackboard)

# 协作流程
researcher.contribute("research", "量子计算研究结果...")
writer.contribute("draft", "基于研究的初稿...")
editor.contribute("final", "编辑后的最终版本...")

print(f"\n黑板内容：{blackboard.get_all()}")

    14.3.3 发布 - 订阅模式¶

    class EventBus:
    """事件总线"""

    def __init__(self):
        self.handlers: Dict[str, List] = {}

    def subscribe(self, event_type: str, handler):
        """订阅事件"""
        if event_type not in self.handlers:
            self.handlers[event_type] = []
        self.handlers[event_type].append(handler)

    def publish(self, event_type: str, data: any):
        """发布事件"""
        print(f"[事件] {event_type}: {str(data)[:50]}...")

        for handler in self.handlers.get(event_type, []):
            try:
                handler(data)
            except Exception as e:
                print(f"事件处理失败：{e}")

class EventDrivenAgent(Agent):
    """事件驱动的代理"""

    def __init__(self, name: str, role: str, expertise: str, event_bus: EventBus):
        super().__init__(name, role, expertise)
        self.event_bus = event_bus

    def publish_event(self, event_type: str, data: any):
        """发布事件"""
        self.event_bus.publish(event_type, {
            "from": self.name,
            "data": data,
        })

# 使用
event_bus = EventBus()

# 定义事件处理
def on_research_complete(data):
    print(f"研究完成事件：{data}")
    # 触发写作

def on_draft_ready(data):
    print(f"草稿就绪事件：{data}")
    # 触发编辑

event_bus.subscribe("research_complete", on_research_complete)
event_bus.subscribe("draft_ready", on_draft_ready)

# 发布事件
researcher = EventDrivenAgent("研究员", "研究专家", "分析", event_bus)
researcher.publish_event("research_complete", {"topic": "量子计算", "findings": "..."})

    14.4.1 任务队列¶

    from queue import PriorityQueue
from threading import Lock
import heapq

class Task:
    """任务"""

    def __init__(self, name: str, description: str, priority: int = 0, assigned_to: str = None):
        self.name = name
        self.description = description
        self.priority = priority
        self.assigned_to = assigned_to
        self.status = "pending"
        self.result = None

    def __lt__(self, other):
        return self.priority < other.priority  # 优先级高的先执行

class TaskQueue:
    """任务队列"""

    def __init__(self):
        self.queue = []
        self.lock = Lock()
        self.completed = []

    def add_task(self, task: Task):
        """添加任务"""
        with self.lock:
            heapq.heappush(self.queue, task)
            print(f"[任务队列] 添加：{task.name} (优先级：{task.priority})")

    def get_next_task(self, agent_name: str = None) -> Task:
        """获取下一个任务"""
        with self.lock:
            if not self.queue:
                return None

            # 如果有指定代理，查找匹配的任务
            if agent_name:
                for i, task in enumerate(self.queue):
                    if task.assigned_to == agent_name or task.assigned_to is None:
                        task = self.queue.pop(i)
                        heapq.heapify(self.queue)
                        task.status = "in_progress"
                        return task

            # 否则返回最高优先级任务
            task = heapq.heappop(self.queue)
            task.status = "in_progress"
            return task

    def complete_task(self, task: Task, result: str):
        """完成任务"""
        task.status = "completed"
        task.result = result
        self.completed.append(task)
        print(f"[任务队列] 完成：{task.name}")

    def get_status(self) -> Dict:
        """获取队列状态"""
        return {
            "pending": len(self.queue),
            "completed": len(self.completed),
            "tasks": [t.name for t in self.queue],
        }

# 使用
task_queue = TaskQueue()

# 添加任务
task_queue.add_task(Task("研究", "研究量子计算", priority=1))
task_queue.add_task(Task("写作", "撰写科普文章", priority=2, assigned_to="作家"))
task_queue.add_task(Task("编辑", "审核和优化", priority=3))

# 处理任务
while True:
    task = task_queue.get_next_task()
    if not task:
        break

    print(f"\n处理任务：{task.name}")
    # 模拟处理
    result = f"{task.name} 完成"
    task_queue.complete_task(task, result)

print(f"\n队列状态：{task_queue.get_status()}")

    14.4.2 工作流引擎¶

    from enum import Enum

class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"

class WorkflowNode:
    """工作流节点"""

    def __init__(self, name: str, agent: Agent, dependencies: List[str] = None):
        self.name = name
        self.agent = agent
        self.dependencies = dependencies or []
        self.status = TaskStatus.PENDING
        self.result = None
        self.inputs = {}

    def can_run(self, completed_nodes: set) -> bool:
        """检查是否可以运行"""
        return all(dep in completed_nodes for dep in self.dependencies)

    def run(self) -> str:
        """运行节点"""
        self.status = TaskStatus.RUNNING

        # 构建任务描述
        task = f"{self.name}: {self.inputs}"

        # 执行
        result = self.agent.process(task)

        self.result = result
        self.status = TaskStatus.COMPLETED
        return result

class WorkflowEngine:
    """工作流引擎"""

    def __init__(self):
        self.nodes: Dict[str, WorkflowNode] = {}
        self.completed = set()

    def add_node(self, node: WorkflowNode):
        """添加节点"""
        self.nodes[node.name] = node

    def set_inputs(self, node_name: str, inputs: dict):
        """设置节点输入"""
        if node_name in self.nodes:
            self.nodes[node_name].inputs = inputs

    def execute(self) -> Dict[str, str]:
        """执行工作流"""
        results = {}
        max_iterations = len(self.nodes) * 2  # 防止死循环
        iteration = 0

        while len(self.completed) < len(self.nodes) and iteration < max_iterations:
            iteration += 1

            # 找到可以运行的节点
            runnable = [
                node for node in self.nodes.values()
                if node.status == TaskStatus.PENDING and node.can_run(self.completed)
            ]

            if not runnable:
                if len(self.completed) < len(self.nodes):
                    print("⚠️  工作流阻塞")
                break

            # 并行运行所有可运行的节点
            for node in runnable:
                print(f"\n运行节点：{node.name}")
                result = node.run()
                results[node.name] = result
                self.completed.add(node.name)

                # 将结果传递给依赖此节点的节点
                for other_node in self.nodes.values():
                    if node.name in other_node.dependencies:
                        other_node.inputs[f"from_{node.name}"] = result

        return results

# 使用
engine = WorkflowEngine()

# 创建工作流
research_node = WorkflowNode("研究", Agent("研究员", "研究专家", "分析"))
writing_node = WorkflowNode("写作", Agent("作家", "写作专家", "创作"), dependencies=["研究"])
editing_node = WorkflowNode("编辑", Agent("编辑", "编辑专家", "审核"), dependencies=["写作"])
publishing_node = WorkflowNode("发布", Agent("发布员", "发布专家", "发布"), dependencies=["编辑"])

engine.add_node(research_node)
engine.add_node(writing_node)
engine.add_node(editing_node)
engine.add_node(publishing_node)

# 设置初始输入
engine.set_inputs("研究", {"topic": "量子计算"})

# 执行
results = engine.execute()
print(f"\n工作流完成：{list(results.keys())}")

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage

class ResearchAgent(Agent):

1. 收集相关信息
2. 分析关键点
3. 整理成结构化报告

返回 JSON 格式：
{
    "key_points": ["要点 1", "要点 2"],

class ReviewAgent(Agent):

1. 检查事实准确性

3. 提出改进建议

返回 JSON 格式：
{
    "accuracy_score": 0-10,

        messages = [SystemMessage(content=system_prompt)]
        if context:

class MultiAgentResearchSystem:

    def __init__(self):

        self.writer = WritingAgent("作家", "写作专家", "内容创作")
        self.reviewer = ReviewAgent("审核员", "审核专家", "质量把控")

        self.blackboard = SharedBlackboard()
        self.history = []

    def execute(self, topic: str) -> Dict:
        """执行研究任务"""
        print(f"\n📚 开始研究：{topic}")

        # 阶段 1：研究
        print("\n🔍 阶段 1: 研究")
        research_task = f"研究以下主题：{topic}"
        research_result = self.researcher.process(research_task)
        self.blackboard.write("research", research_result, "研究员")
        self.history.append(("研究", research_result))

        # 阶段 2：写作
        print("\n✍️  阶段 2: 写作")
        writing_task = f"""根据以下研究结果撰写文章：

{research_result}
"""
        writing_result = self.writer.process(
            writing_task,
            [f"研究员：{research_result[:200]}..."]
        )
        self.blackboard.write("draft", writing_result, "作家")
        self.history.append(("写作", writing_result))

        # 阶段 3：审核
        print("\n📋 阶段 3: 审核")
        review_task = f"""审核以下文章：

{writing_result}

基于研究：
{research_result}
"""
        review_result = self.reviewer.process(review_task)
        self.blackboard.write("review", review_result, "审核员")
        self.history.append(("审核", review_result))

        # 阶段 4：修订（如果需要）
        try:
            review_data = json.loads(review_result)
            if not review_data.get("approved", False):
                print("\n🔄 阶段 4: 修订")
                revision_task = f"""根据审核意见修订文章：

原文：{writing_result}

审核意见：{review_result}
"""
                revised_result = self.writer.process(revision_task)
                self.blackboard.write("final", revised_result, "作家")
                self.history.append(("修订", revised_result))
            else:
                self.blackboard.write("final", writing_result, "作家")
        except:
            self.blackboard.write("final", writing_result, "作家")

        return {
            "topic": topic,
            "research": research_result,
            "draft": writing_result,
            "review": review_result,
            "final": self.blackboard.read("final"),
        }

# 使用
system = MultiAgentResearchSystem()
result = system.execute("人工智能在医疗领域的应用")

print(f"\n{'='*50}")
print(f"研究完成")
print(f"{'='*50}")
print(f"最终成果：{result['final'][:500]}...")
