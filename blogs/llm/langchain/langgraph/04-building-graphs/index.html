    04. 构建图¶

    完整的图构建流程¶

    from typing import TypedDict
from langgraph.graph import StateGraph, START, END

# 1. 定义状态
class State(TypedDict):
    value: int

# 2. 创建图
graph = StateGraph(State)

# 3. 添加节点
graph.add_node("node_a", function_a)
graph.add_node("node_b", function_b)

# 4. 添加边
graph.add_edge(START, "node_a")
graph.add_edge("node_a", "node_b")
graph.add_edge("node_b", END)

# 5. 编译
app = graph.compile()

# 6. 运行
result = app.invoke({"value": 0})

    节点的高级用法¶

    节点装饰器¶

    from langgraph.graph import StateGraph

graph = StateGraph(State)

# 使用装饰器添加节点
@graph.add_node
def my_node(state: State) -> dict:
    return {"value": state["value"] + 1}

    节点配置¶

    # 添加节点时配置元数据
graph.add_node(
    "agent",
    agent_function,
    metadata={"description": "Main agent node"}
)

    访问运行时信息¶

    from langgraph.graph import StateGraph
from langgraph.config import get_config, get_store

def node_with_config(state: State) -> dict:
    # 获取当前配置
    config = get_config()
    thread_id = config.get("configurable", {}).get("thread_id")

    # 获取存储
    store = get_store()

    return {"value": state["value"] + 1}

    边的类型¶

    1. 普通边（Normal Edge）¶

    # 简单顺序执行
graph.add_edge("A", "B")
graph.add_edge("B", "C")

    2. 条件边（Conditional Edge）¶

    from typing import Literal

def router(state: State) -> Literal["path_a", "path_b"]:
    if state["value"] > 10:
        return "path_a"
    return "path_b"

graph.add_conditional_edges(
    source="decision_node",
    condition=router,
    # mapping 可选，默认返回什么节点名就去哪个节点
    mapping={
        "path_a": "node_a",
        "path_b": "node_b"
    }
)

    3. 并行边（Parallel Edges）¶

    # 从同一个节点到多个节点（并行执行）
graph.add_edge("start", "node_a")
graph.add_edge("start", "node_b")
graph.add_edge("start", "node_c")

# 所有节点执行完后，继续到下一个
graph.add_edge("node_a", "end")
graph.add_edge("node_b", "end")
graph.add_edge("node_c", "end")

    条件边实战：路由模式¶

    模式 1：基于内容的路由¶

    from typing import Literal
from langchain_core.messages import AIMessage, HumanMessage

class State(TypedDict):
    messages: list

def classify_intent(state: State) -> Literal["support", "sales", "general"]:
    last_message = state["messages"][-1].content

    if "help" in last_message or "problem" in last_message:
        return "support"
    elif "price" in last_message or "buy" in last_message:
        return "sales"
    return "general"

graph = StateGraph(State)
graph.add_node("classify", classify_intent)
graph.add_node("support", support_agent)
graph.add_node("sales", sales_agent)
graph.add_node("general", general_agent)

graph.add_edge(START, "classify")
graph.add_conditional_edges("classify", classify_intent)

    模式 2：基于工具调用的路由¶

    def route_after_tool(state: State) -> Literal["tool_node", "agent_node", END]:
    messages = state["messages"]
    last_message = messages[-1]

    # 如果有工具调用，去工具节点
    if last_message.tool_calls:
        return "tool_node"

    # 如果有工具结果，回 agent
    if hasattr(last_message, "tool_call_id"):
        return "agent_node"

    # 否则结束
    return END

graph.add_conditional_edges("agent", route_after_tool)

    循环模式¶

    模式 1：固定次数循环¶

    class State(TypedDict):
    value: int
    count: int

def increment(state: State) -> dict:
    return {"value": state["value"] + 1, "count": state["count"] + 1}

def check_limit(state: State) -> Literal["continue", "end"]:
    if state["count"] >= 5:
        return "end"
    return "continue"

graph = StateGraph(State)
graph.add_node("increment", increment)
graph.add_conditional_edges("increment", check_limit, {
    "continue": "increment",
    "end": END
})
graph.add_edge(START, "increment")

    模式 2：Agent 循环（ReAct 模式）¶

    def should_continue(state: State) -> Literal["tools", END]:
    messages = state["messages"]
    last_message = messages[-1]

    # 如果 AI 要求调用工具，继续
    if last_message.tool_calls:
        return "tools"

    # 否则结束
    return END

graph = StateGraph(State)
graph.add_node("agent", agent_function)
graph.add_node("tools", tool_function)

graph.add_edge(START, "agent")
graph.add_edge("tools", "agent")  # 工具执行后回到 agent
graph.add_conditional_edges("agent", should_continue)

    分支与合并¶

    扇出模式（Fan-out）¶

    # 一个节点触发多个并行节点
graph.add_edge("start", "process_a")
graph.add_edge("start", "process_b")
graph.add_edge("start", "process_c")

# 所有处理完成后合并
graph.add_edge("process_a", "merge")
graph.add_edge("process_b", "merge")
graph.add_edge("process_c", "merge")

    条件分支模式¶

    def branch_router(state: State) -> Literal["branch_a", "branch_b", "branch_c"]:
    # 根据条件选择分支
    if condition_a:
        return "branch_a"
    elif condition_b:
        return "branch_b"
    return "branch_c"

graph.add_conditional_edges("decision", branch_router)

# 所有分支汇聚到同一个结束节点
graph.add_edge("branch_a", "end")
graph.add_edge("branch_b", "end")
graph.add_edge("branch_c", "end")

    图的编译选项¶

    基础编译¶

    app = graph.compile()

    带检查点编译¶

    from langgraph.checkpoint.memory import MemorySaver

memory = MemorySaver()
app = graph.compile(checkpointer=memory)

    带中断点编译¶

    # 在特定节点前中断
app = graph.compile(
    checkpointer=memory,
    interrupt_before=["human_review"]  # 在 human_review 节点前中断
)

# 或在特定节点后中断
app = graph.compile(
    interrupt_after=["agent_step"]
)

    配置编译选项¶

    app = graph.compile(
    checkpointer=memory,
    interrupt_before=["human_review"],
    # 其他配置
    store=store,  # 状态存储
)

    运行图的不同方式¶

    invoke - 同步执行¶

    # 一次性执行完成
result = app.invoke({"value": 0})
print(result)

    stream - 流式输出¶

    # 逐个节点输出结果
for node, values in app.stream({"value": 0}):
    print(f"{node}: {values}")

# 输出示例：
# {'node_a': {'value': 1}}
# {'node_b': {'value': 2}}

    流式输出模式¶

    # values 模式：每个节点的输出
for step in app.stream(input, stream_mode="values"):
    print(step)  # 完整的当前状态

# updates 模式：节点的增量更新
for step in app.stream(input, stream_mode="updates"):
    print(step)  # 只包含更新的键

# messages 模式：消息流（用于对话）
for step in app.stream(input, stream_mode="messages"):
    print(step)

    astream - 异步流式¶

    async for node, values in app.astream({"value": 0}):
    print(f"{node}: {values}")

    实战：完整的对话机器人¶

    from typing import TypedDict, Annotated, List
from langchain_core.messages import BaseMessage, HumanMessage, AIMessage
from langgraph.graph import StateGraph, START, END, add_messages

class State(TypedDict):
    messages: Annotated[List[BaseMessage], add_messages]

def chatbot(state: State) -> dict:
    response = model.invoke(state["messages"])
    return {"messages": [response]}

graph = StateGraph(State)
graph.add_node("chatbot", chatbot)
graph.add_edge(START, "chatbot")
graph.add_edge("chatbot", END)

app = graph.compile()

# 使用
messages = [HumanMessage(content="Hello!")]
result = app.invoke({"messages": messages})
print(result["messages"])

    常见图模式总结¶

    模式 适用场景 关键特性
    线性链 简单流程 A → B → C
    条件分支 路由决策 if/else 逻辑
    循环 Agent、重试 带退出条件的循环
    扇出/扇入 并行处理 多节点并行后合并
    状态机 复杂工作流 多状态转换
