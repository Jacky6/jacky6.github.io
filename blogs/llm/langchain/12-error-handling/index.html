    12.2.1 使用 tenacity 实现重试¶

    from tenacity import (
    retry,
    stop_after_attempt,
    wait_exponential,
    wait_fixed,
    retry_if_exception_type,
    retry_if_result,
)
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

model = ChatOpenAI(model="gpt-4o")

# 基础重试
@retry(stop=stop_after_attempt(3))
def call_with_retry(prompt: str):
    return model.invoke([HumanMessage(content=prompt)])

# 高级重试配置
@retry(
    stop=stop_after_attempt(5),                    # 最多重试 5 次
    wait=wait_exponential(multiplier=1, min=1, max=60),  # 指数退避
    retry=retry_if_exception_type((TimeoutError, ConnectionError)),  # 只重试特定错误
)
def robust_call(prompt: str):
    return model.invoke([HumanMessage(content=prompt)])

# 使用
try:
    response = robust_call("你好")
    print(response.content)
except Exception as e:
    print(f"最终失败：{e}")

    12.2.2 自定义重试条件¶

    from tenacity import retry_if_exception, RetryError

def is_rate_limit_error(exception):
    """判断是否是速率限制错误"""
    return hasattr(exception, 'status_code') and exception.status_code == 429

def is_retryable_error(exception):
    """判断是否可重试"""
    retryable_codes = [429, 500, 502, 503, 504]
    return hasattr(exception, 'status_code') and exception.status_code in retryable_codes

@retry(
    stop=stop_after_attempt(5),
    wait=wait_exponential(multiplier=2, min=4, max=60),
    retry=retry_if_exception(is_retryable_error),
)
def call_with_smart_retry(prompt: str):
    response = model.invoke([HumanMessage(content=prompt)])

    # 检查响应质量
    if not response.content or len(response.content) < 10:
        raise ValueError("响应质量过低")

    return response

    12.3.2 自定义超时装饰器¶

    import signal
from functools import wraps

class TimeoutError(Exception):
    pass

def timeout(seconds: int):
    """超时装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            def handler(signum, frame):
                raise TimeoutError(f"函数执行超时（{seconds}秒）")

            # 设置信号
            signal.signal(signal.SIGALRM, handler)
            signal.alarm(seconds)

            try:
                result = func(*args, **kwargs)
            finally:
                signal.alarm(0)  # 取消闹钟

            return result
        return wrapper
    return decorator

# 使用
@timeout(30)
def call_with_timeout(prompt: str):
    return model.invoke([HumanMessage(content=prompt)])

try:
    response = call_with_timeout("长问题")
except TimeoutError:
    print("调用超时")

    from typing import Optional

class TimeoutConfig:
    """分级超时配置"""

    def __init__(self):
        self.timeouts = {
            "simple": 10,      # 简单问题 10 秒
            "normal": 30,      # 普通问题 30 秒
            "complex": 60,     # 复杂问题 60 秒
            "analysis": 120,   # 分析任务 120 秒
        }

    def get_timeout(self, complexity: str) -> int:
        return self.timeouts.get(complexity, self.timeouts["normal"])

timeout_config = TimeoutConfig()

def estimate_complexity(prompt: str) -> str:
    """估算问题复杂度"""
    length = len(prompt)

    if length < 50:
        return "simple"
    elif length < 200:
        return "normal"
    elif length < 500:
        return "complex"
    else:
        return "analysis"

def call_with_adaptive_timeout(prompt: str):
    complexity = estimate_complexity(prompt)
    timeout_seconds = timeout_config.get_timeout(complexity)

    print(f"复杂度：{complexity}, 超时：{timeout_seconds}秒")

    @timeout(timeout_seconds)
    def inner():
        return model.invoke([HumanMessage(content=prompt)])

    return inner()

    12.4.2 令牌桶限流¶

    import time
from threading import Lock

class TokenBucket:
    """令牌桶限流器"""

    def __init__(self, rate: float, capacity: int):
        """
        Args:
            rate: 每秒生成的令牌数
            capacity: 桶容量
        """
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_update = time.time()
        self._lock = Lock()

    def acquire(self, tokens: int = 1) -> bool:
        """获取令牌"""
        with self._lock:
            now = time.time()
            elapsed = now - self.last_update

            # 添加新令牌
            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
            self.last_update = now

            # 检查是否有足够令牌
            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False

    def wait_for_token(self, tokens: int = 1, timeout: float = None):
        """等待令牌"""
        start = time.time()

        while not self.acquire(tokens):
            if timeout and (time.time() - start) > timeout:
                raise TimeoutError("等待令牌超时")
            time.sleep(0.1)

# 使用
limiter = TokenBucket(rate=1.0, capacity=10)  # 每秒 1 个请求，最多 10 个突发

def rate_limited_call(prompt: str):
    limiter.wait_for_token()
    return model.invoke([HumanMessage(content=prompt)])

    12.4.3 请求队列¶

    from queue import Queue, Empty
from threading import Thread
import time

class RequestQueue:
    """请求队列"""

    def __init__(self, max_concurrent: int = 5):
        self.queue = Queue()
        self.max_concurrent = max_concurrent
        self.running = False
        self.results = {}

    def start(self):
        """启动队列处理器"""
        self.running = True
        for i in range(self.max_concurrent):
            thread = Thread(target=self._process, daemon=True)
            thread.start()

    def stop(self):
        """停止队列"""
        self.running = False

    def submit(self, request_id: str, prompt: str):
        """提交请求"""
        self.queue.put((request_id, prompt))
        self.results[request_id] = None
        return request_id

    def _process(self):
        """处理队列"""
        while self.running:
            try:
                request_id, prompt = self.queue.get(timeout=1)

                try:
                    result = model.invoke([HumanMessage(content=prompt)])
                    self.results[request_id] = {"success": True, "result": result.content}
                except Exception as e:
                    self.results[request_id] = {"success": False, "error": str(e)}

                self.queue.task_done()
            except Empty:
                continue

    def get_result(self, request_id: str, timeout: float = 30):
        """获取结果"""
        start = time.time()

        while time.time() - start < timeout:
            if self.results.get(request_id) is not None:
                return self.results[request_id]
            time.sleep(0.1)

        raise TimeoutError("获取结果超时")

# 使用
queue = RequestQueue(max_concurrent=3)
queue.start()

# 提交多个请求
request_ids = []
for i in range(10):
    rid = queue.submit(f"req_{i}", f"问题{i}")
    request_ids.append(rid)

# 获取结果
for rid in request_ids:
    result = queue.get_result(rid)
    print(f"{rid}: {result}")

queue.stop()

    from typing import List

class ModelFallback:
    """模型降级"""

    def __init__(self, models: List[str]):
        self.models = models  # 按优先级排序

    def invoke(self, prompt: str):
        """按优先级尝试模型"""
        last_error = None

        for model_name in self.models:
            try:
                print(f"尝试模型：{model_name}")
                fallback_model = ChatOpenAI(model=model_name)
                response = fallback_model.invoke([HumanMessage(content=prompt)])
                return response
            except Exception as e:
                print(f"模型 {model_name} 失败：{e}")
                last_error = e

        # 所有模型都失败
        raise Exception(f"所有模型都失败：{last_error}")

# 使用
fallback = ModelFallback([
    "gpt-4o",        # 首选
    "gpt-4-turbo",   # 降级 1

    "gpt-3.5-turbo", # 降级 3
])

    12.5.2 功能降级¶

    class GracefulDegradation:
    """功能降级"""

    def __init__(self):
        self.full_features = True
        self.cached_responses = {}

    def chat(self, prompt: str, use_cache: bool = True):
        """带降级的对话"""

        # 检查缓存
        if use_cache and prompt in self.cached_responses:
            return self.cached_responses[prompt]

        try:
            # 完整功能
            if self.full_features:
                response = model.invoke([HumanMessage(content=prompt)])

                # 结构化输出
                structured_model = model.with_structured_output({
                    "type": "object",
                    "properties": {
                        "answer": {"type": "string"},
                        "confidence": {"type": "number"},
                    }
                })

                result = structured_model.invoke(prompt)

                if use_cache:
                    self.cached_responses[prompt] = result

                return result

        except Exception as e:
            print(f"完整功能失败，降级：{e}")
            self.full_features = False

        # 降级：简单响应
        try:
            response = model.invoke([HumanMessage(content=f"简单回答：{prompt}")])
            return {"answer": response.content, "confidence": 0.5}
        except Exception as e:
            # 进一步降级：返回缓存或默认响应
            return {
                "answer": "抱歉，暂时无法回答您的问题。",
                "confidence": 0.0
            }

# 使用
degradation = GracefulDegradation()
response = degradation.chat("复杂问题")

    12.5.3 熔断器模式¶

    import time
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开（测试）

class CircuitBreaker:
    """熔断器"""

    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: int = 60,
        half_open_max_calls: int = 3,
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls

        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
        self.half_open_calls = 0

    def call(self, func, *args, **kwargs):
        """带熔断的调用"""

        if self.state == CircuitState.OPEN:
            # 检查是否可以进入半开状态
            if time.time() - self.last_failure_time >= self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
                print("进入半开状态")
            else:
                raise Exception("熔断器打开，拒绝调用")

        try:
            result = func(*args, **kwargs)

            # 成功
            if self.state == CircuitState.HALF_OPEN:
                self.half_open_calls += 1
                if self.half_open_calls >= self.half_open_max_calls:
                    self.state = CircuitState.CLOSED
                    self.failure_count = 0
                    print("熔断器关闭")

            return result

        except Exception as e:
            self.failure_count += 1
            self.last_failure_time = time.time()

            if self.state == CircuitState.HALF_OPEN:
                self.state = CircuitState.OPEN
                print("半开测试失败，重新熔断")
            elif self.failure_count >= self.failure_threshold:
                self.state = CircuitState.OPEN
                print(f"失败次数达到阈值，熔断器打开")

            raise e

    def get_state(self) -> str:
        return self.state.value

# 使用
breaker = CircuitBreaker(failure_threshold=3, recovery_timeout=30)

def fragile_call(prompt: str):
    return model.invoke([HumanMessage(content=prompt)])

for i in range(10):
    try:
        response = breaker.call(fragile_call, f"测试{i}")
        print(f"成功：{response.content[:20]}...")
    except Exception as e:
        print(f"失败（状态：{breaker.get_state()}）: {e}")

    12.6.1 结构化日志¶

    import logging
import json
from datetime import datetime

class StructuredLogger:
    """结构化日志"""

    def __init__(self, name: str = "llm"):
        self.logger = logging.getLogger(name)
        self.logger.setLevel(logging.INFO)

        # 文件处理器
        handler = logging.FileHandler(f"{name}.log", encoding="utf-8")
        formatter = logging.Formatter('%(message)s')
        handler.setFormatter(formatter)
        self.logger.addHandler(handler)

    def log_call(self, prompt: str, response: str = None, error: str = None, **kwargs):
        """记录调用日志"""
        entry = {
            "timestamp": datetime.now().isoformat(),
            "level": "error" if error else "info",
            "prompt": prompt[:200],
            "response": response[:200] if response else None,
            "error": error,
            **kwargs,
        }

        if error:
            self.logger.error(json.dumps(entry, ensure_ascii=False))
        else:
            self.logger.info(json.dumps(entry, ensure_ascii=False))

# 使用
logger = StructuredLogger()

try:
    response = model.invoke([HumanMessage(content="问题")])
    logger.log_call("问题", response.content, model="gpt-4o")
except Exception as e:
    logger.log_call("问题", error=str(e), model="gpt-4o")

    12.6.2 错误统计¶

    from collections import defaultdict
from datetime import datetime, timedelta

class ErrorStats:
    """错误统计"""

    def __init__(self):
        self.errors = defaultdict(list)
        self.calls = 0
        self.success = 0

    def record_call(self, success: bool, error_type: str = None):
        """记录调用"""
        self.calls += 1

        if success:
            self.success += 1
        else:
            self.errors[error_type].append(datetime.now())

    def get_error_rate(self, window_minutes: int = 60) -> float:
        """获取错误率"""
        cutoff = datetime.now() - timedelta(minutes=window_minutes)

        recent_errors = sum(
            len([t for t in times if t > cutoff])
            for times in self.errors.values()
        )

        recent_calls = self.calls  # 简化：假设所有调用都在窗口内

        return recent_errors / recent_calls if recent_calls > 0 else 0

    def get_top_errors(self, n: int = 5) -> list:
        """获取最常见的错误"""
        sorted_errors = sorted(
            self.errors.items(),
            key=lambda x: len(x[1]),
            reverse=True
        )
        return [(k, len(v)) for k, v in sorted_errors[:n]]

    def report(self) -> dict:
        """生成报告"""
        return {
            "total_calls": self.calls,
            "success": self.success,
            "error_rate": self.get_error_rate(),
            "top_errors": self.get_top_errors(),
        }

# 使用
stats = ErrorStats()

# 模拟记录
stats.record_call(True)
stats.record_call(False, "timeout")
stats.record_call(False, "timeout")
stats.record_call(False, "rate_limit")

print(stats.report())

    12.6.3 告警系统¶

    class AlertSystem:
    """告警系统"""

    def __init__(self):
        self.alert_thresholds = {
            "error_rate": 0.1,      # 错误率超过 10%
            "consecutive_failures": 5,  # 连续 5 次失败
            "response_time": 30,    # 响应时间超过 30 秒
        }
        self.consecutive_failures = 0
        self.alert_cooldown = {}  # 告警冷却

    def check(self, stats: dict):
        """检查是否需要告警"""
        alerts = []

        # 错误率告警
        if stats.get("error_rate", 0) > self.alert_thresholds["error_rate"]:
            if self._can_alert("error_rate"):
                alerts.append(f"错误率过高：{stats['error_rate']:.2%}")

        # 连续失败告警
        if self.consecutive_failures >= self.alert_thresholds["consecutive_failures"]:
            if self._can_alert("consecutive_failures"):
                alerts.append(f"连续失败：{self.consecutive_failures}次")

        return alerts

            return False

        self.alert_cooldown[alert_type] = now
        return True

    12.7 实战：健壮的 Agent 系统¶

    from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from tenacity import retry, stop_after_attempt, wait_exponential
from typing import Optional, Dict, Any
import time
import logging

class RobustAgent:
    """健壮的 Agent 系统"""

    def __init__(self, config: Dict[str, Any] = None):
        self.config = config or {}

        # 模型配置
        self.primary_model = ChatOpenAI(model="gpt-4o")
        self.fallback_model = ChatOpenAI(model="gpt-3.5-turbo")

        # 错误处理
        self.max_retries = self.config.get("max_retries", 3)
        self.timeout = self.config.get("timeout", 30)

        # 统计
        self.stats = {
            "calls": 0,
            "success": 0,
            "failures": 0,
            "fallbacks": 0,
        }

        # 日志
        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(__name__)

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=2, max=30),
    )
    def _call_with_retry(self, messages: list, model):
        """带重试的调用"""
        return model.invoke(messages)

    def chat(self, user_input: str) -> Optional[str]:
        """健壮的对话方法"""
        self.stats["calls"] += 1

        messages = [
            SystemMessage(content="你是一个有用的助手。"),
            HumanMessage(content=user_input),
        ]

        start_time = time.time()

        try:
            # 尝试主模型
            self.logger.info(f"使用主模型处理：{user_input[:50]}...")
            response = self._call_with_retry(messages, self.primary_model)

            self.stats["success"] += 1
            elapsed = time.time() - start_time

            self.logger.info(f"成功（{elapsed:.2f}s）: {response.content[:50]}...")
            return response.content

        except Exception as e:
            self.logger.warning(f"主模型失败：{e}")

            try:
                # 降级到备用模型
                self.logger.info("降级到备用模型")
                self.stats["fallbacks"] += 1

                response = self.fallback_model.invoke(messages)

                self.stats["success"] += 1
                return response.content

            except Exception as fallback_error:
                self.logger.error(f"备用模型也失败：{fallback_error}")
                self.stats["failures"] += 1

                return None

    def get_stats(self) -> Dict[str, Any]:
        """获取统计信息"""
        return {
            **self.stats,
            "success_rate": (
                self.stats["success"] / self.stats["calls"]
                if self.stats["calls"] > 0 else 0
            ),
        }

# 使用
agent = RobustAgent({
    "max_retries": 3,
    "timeout": 30,
})

# 对话
response = agent.chat("你好，请介绍一下自己")
print(f"响应：{response}")

# 统计
print(f"统计：{agent.get_stats()}")
