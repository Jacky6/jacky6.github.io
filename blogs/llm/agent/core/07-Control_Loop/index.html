    Control Loop 控制循环¶

    Agent 的"心跳"——驱动 Agent 从感知到行动的完整生命周期。

    1. 模块定位¶

    Control Loop 是 Agent 如何运转 的核心机制。它定义了 Agent 在每个时间步做什么、如何做、何时停止。

    ┌─────────────────────────────────────────────────────────┐
│              The Agent Control Loop                       │
├─────────────────────────────────────────────────────────┤
│                                                         │
│  ┌─────────┐    ─────────┐    ┌─────────┐             │
│  │ Observe │ →  │  Think  │ →  │   Act   │             │
│  │  感知   │    │  思考   │    │  行动   │             │
│  └─────────┘    ─────────┘    └────┬────┘             │
│       ↑                              │                  │
│       │          ┌─────────┐         │                  │
│       └───────── │Reflect  │ ←───────┘                  │
│                  │  反思   │                             │
│                  └─────────┘                             │
│                                                         │
│  持续循环，直到：                                         │
│  - 任务完成                                               │
│  - 达到最大迭代次数                                       │
│  - 用户中断                                               │
│  - 错误无法恢复                                           │
│                                                         │
└─────────────────────────────────────────────────────────┘

    2. 核心循环：Observe → Think → Act → Reflect¶

    2.1 从零实现¶

    from dataclasses import dataclass, field
from typing import Optional
import time

@dataclass
class AgentState:
    """Agent 运行时状态"""
    goal: str
    observations: list[str] = field(default_factory=list)
    thoughts: list[str] = field(default_factory=list)
    actions: list[dict] = field(default_factory=list)
    reflections: list[str] = field(default_factory=list)
    iteration: int = 0
    max_iterations: int = 10
    is_done: bool = False
    final_output: str = ""
    error: Optional[str] = None

class ControlLoop:
    """Agent 控制循环"""

    def __init__(self, llm, tools: dict, max_iterations: int = 10):
        self.llm = llm
        self.tools = tools
        self.max_iterations = max_iterations

    def run(self, goal: str) -> AgentState:
        """执行完整的控制循环"""
        state = AgentState(goal=goal, max_iterations=self.max_iterations)

        while not state.is_done and state.iteration < state.max_iterations:
            state.iteration += 1
            print(f"\n{'='*50}")
            print(f"  Iteration {state.iteration}/{state.max_iterations}")
            print(f"{'='*50}")

            # 1. Observe - 收集环境信息
            self._observe(state)

            # 2. Think - 推理下一步
            self._think(state)

            # 3. Act - 执行动作
            self._act(state)

            # 4. Reflect - 反思结果
            self._reflect(state)

            # 5. Check termination
            state.is_done = self._should_terminate(state)

        # 循环结束，生成最终输出
        if not state.error:
            state.final_output = self._generate_final_output(state)

        return state

    def _observe(self, state: AgentState):
        """Observe：收集当前环境状态"""
        # 对于纯 LLM Agent，observation 主要是上一步 action 的结果
        if state.actions:
            last_action = state.actions[-1]
            if last_action.get("result"):
                observation = f"Last action '{last_action['name']}' returned: {last_action['result']}"
                state.observations.append(observation)
        else:
            state.observations.append(f"Goal: {state.goal}")

    def _think(self, state: AgentState):
        """Think：推理下一步该做什么"""
        context = self._build_context(state)

        prompt = f"""
        You are working on this goal: {state.goal}

        Current state:
        - Iteration: {state.iteration}/{state.max_iterations}
        - Observations: {state.observations[-3:] if state.observations else 'None'}
        - Previous actions: {len(state.actions)}
        - Previous reflections: {state.reflections[-1] if state.reflections else 'None'}

        Context:
        {context}

        What should you do next? Choose one:
        1. Call a tool to gather more information or perform an action
        2. You have enough information - produce the final answer
        3. You need to reconsider your approach

        Return JSON:
        {{
            "reasoning": "your thinking process",
            "decision": "use_tool" | "final_answer" | "reconsider",
            "tool_name": "name if using a tool",
            "tool_input": {{"param": "value"}} if using a tool
        }}
        """

        response = json.loads(
            self.llm.invoke(
                prompt,
                response_format={"type": "json_object"}
            ).content
        )

        state.thoughts.append(response.get("reasoning", ""))
        state._current_decision = response.get("decision", "use_tool")
        state._current_tool = response.get("tool_name")
        state._current_tool_input = response.get("tool_input", {})

    def _act(self, state: AgentState):
        """Act：执行决定"""
        decision = getattr(state, "_current_decision", "use_tool")

        if decision == "final_answer":
            state.is_done = True
            state._current_result = "FINAL_ANSWER_MARKER"
            return

        if decision == "reconsider":
            state._current_result = "RECONSIDER_MARKER"
            return

        # 执行工具
        tool_name = getattr(state, "_current_tool", None)
        tool_input = getattr(state, "_current_tool_input", {})

        if not tool_name or tool_name not in self.tools:
            state._current_result = f"Error: Unknown tool '{tool_name}'"
            return

        try:
            result = self.tools[tool_name](**tool_input)
            state._current_result = str(result)
        except Exception as e:
            state._current_result = f"Error: {str(e)}"

        state.actions.append({
            "name": tool_name,
            "input": tool_input,
            "result": state._current_result,
            "iteration": state.iteration,
        })

    def _reflect(self, state: AgentState):
        """Reflect：评估当前进度"""
        result = getattr(state, "_current_result", "")

        if result in ("FINAL_ANSWER_MARKER", "RECONSIDER_MARKER"):
            return

        prompt = f"""
        Reflect on the current progress:

        Goal: {state.goal}
        Last action: {state.actions[-1]['name']}
        Result: {result[:500]}
        Iterations used: {state.iteration}/{state.max_iterations}

        Questions:
        1. Are we making progress toward the goal?
        2. Is the approach working?
        3. Should we continue, adjust, or stop?

        Return a brief reflection (2-3 sentences).
        """

        reflection = self.llm.invoke(prompt).content.strip()
        state.reflections.append(reflection)

    def _should_terminate(self, state: AgentState) -> bool:
        """判断是否应该终止"""
        # 检查是否已达到最终答案
        if getattr(state, "_current_result", "") == "FINAL_ANSWER_MARKER":
            return True

        # 检查是否连续出错
        recent_errors = sum(
            1 for a in state.actions[-3:]
            if a.get("result", "").startswith("Error:")
        )
        if recent_errors >= 3:
            state.error = "Too many consecutive errors"
            return True

        # 检查是否达到最大迭代
        if state.iteration >= state.max_iterations:
            state.error = "Max iterations reached"
            return True

        return False

    def _generate_final_output(self, state: AgentState) -> str:
        """生成最终输出"""
        context = self._build_context(state)

        prompt = f"""
        Generate the final answer for this goal:

        Goal: {state.goal}

        Work history:
        {context}

        Provide a clear, complete answer.
        """

        return self.llm.invoke(prompt).content.strip()

    def _build_context(self, state: AgentState) -> str:
        """构建当前上下文字符串"""
        parts = []

        if state.observations:
            parts.append("Observations:\n" + "\n".join(f"  - {o}" for o in state.observations[-5:]))

        if state.thoughts:
            parts.append("Thoughts:\n" + "\n".join(f"  - {t}" for t in state.thoughts[-5:]))

        if state.actions:
            parts.append("Actions:\n" + "\n".join(
                f"  - {a['name']}: {a['result'][:100]}" for a in state.actions[-5:]
            ))

        if state.reflections:
            parts.append("Reflections:\n" + "\n".join(f"  - {r}" for r in state.reflections[-3:]))

        return "\n\n".join(parts)

# ---- 使用 ----
tools = {
    "search": lambda query: f"Search results for '{query}'",
    "read_file": lambda path: f"Content of {path}",
    "calculator": lambda expr: str(eval(expr)),
}

loop = ControlLoop(llm=llm, tools=tools, max_iterations=8)
state = loop.run("查询北京今天的天气并计算温度平均值")

print(f"\n最终输出: {state.final_output}")
print(f"迭代次数: {state.iteration}")
print(f"动作数: {len(state.actions)}")

    3. LangGraph 中的 Control Loop¶

    LangGraph 的图结构 本身就是 一个 Control Loop。每个节点对应循环中的一个阶段。

    3.1 标准 ReAct 循环¶

    from typing import TypedDict, Literal
from langgraph.graph import StateGraph, END
from langgraph.prebuilt import ToolNode
from langchain_openai import ChatOpenAI

class AgentState(TypedDict):
    messages: list
    step_count: int

llm = ChatOpenAI(model="gpt-4o")
tools = [...]  # 工具列表
tool_node = ToolNode(tools)

def agent_step(state: AgentState) -> AgentState:
    """Think + Act 决策"""
    messages = state["messages"]
    response = llm.bind_tools(tools).invoke(messages)

    return {
        "messages": state["messages"] + [response],
        "step_count": state.get("step_count", 0) + 1,
    }

def should_continue(state: AgentState) -> Literal["tools", END]:
    """判断：继续调用工具 还是 结束"""
    last_msg = state["messages"][-1]

    # 有工具调用 → 继续
    if hasattr(last_msg, "tool_calls") and last_msg.tool_calls:
        return "tools"

    # 达到最大步数 → 结束
    if state.get("step_count", 0) >= 10:
        return END

    # 否则结束
    return END

# 构建循环
builder = StateGraph(AgentState)
builder.add_node("agent", agent_step)
builder.add_node("tools", tool_node)

builder.set_entry_point("agent")
builder.add_conditional_edges("agent", should_continue)
builder.add_edge("tools", "agent")  # ← 循环边

graph = builder.compile()

    这就是一个完整的 Control Loop： 

    agent → (有工具调用?) → tools → agent → (有工具调用?) → tools → ... → END

    3.2 带 Reflect 的循环¶

    class ReflectiveAgentState(TypedDict):
    messages: list
    step_count: int
    reflection: str

def agent_step(state: ReflectiveAgentState) -> ReflectiveAgentState:
    """Think + Act"""
    response = llm.bind_tools(tools).invoke(state["messages"])
    return {
        "messages": state["messages"] + [response],
        "step_count": state.get("step_count", 0) + 1,
    }

def reflect_step(state: ReflectiveAgentState) -> ReflectiveAgentState:
    """Reflect：评估当前进度"""
    last_result = state["messages"][-1].content if state["messages"] else ""

    prompt = f"""
    Review the current progress:

    Last action result: {last_result[:500]}
    Steps taken: {state.get('step_count', 0)}

    Are we making good progress? Should we continue or wrap up?
    Return "continue" or "wrap_up".
    """

    reflection = llm.invoke(prompt).content.strip().lower()

    return {**state, "reflection": reflection}

def route_after_reflect(state: ReflectiveAgentState) -> Literal["agent", END]:
    if "wrap_up" in state.get("reflection", ""):
        return END
    return "agent"

# 构建带反思的循环
builder = StateGraph(ReflectiveAgentState)
builder.add_node("agent", agent_step)
builder.add_node("tools", tool_node)
builder.add_node("reflect", reflect_step)

builder.set_entry_point("agent")
builder.add_conditional_edges("agent", lambda s: "tools" if hasattr(s["messages"][-1], "tool_calls") and s["messages"][-1].tool_calls else "reflect")
builder.add_edge("tools", "agent")
builder.add_conditional_edges("reflect", route_after_reflect)

reflective_graph = builder.compile()

    4. 循环控制策略¶

    4.0 循环防御体系（面试重点）¶

    面试官高频问题："Agent 陷入死循环，一直重复同一个操作怎么办？"

    标准回答：五层防御，层层兜底。

    第一层：最大迭代次数（10-15）     → 硬兜底
第二层：重复动作检测（deque窗口） → 识别循环模式
第三层：Token 预算控制            → 防止 token 爆炸
第四层：超时机制（60s）           → 防止永远跑不完
第五层：状态收敛检测              → 识别停滞状态

检测到循环后不直接报错，而是尝试恢复：
  换工具 → 问用户 → 给出最佳猜测

    为什么 Agent 会循环？

    原因 例子 对应防御
    工具调用失败后重试同一工具 参数不对→报错→不调整直接重试 重复动作检测
    工具返回信息不足以推进 返回为空，无法推导下一步 状态收敛检测
    Prompt 指令冲突 既要 A 又要 B，来回摇摆 任务前置验证
    没有明确终止条件 Agent 不知道什么时候该停 MAX_ITER
    多 Agent 互相等待 A 等 B，B 等 A 超时 + 死锁检测

    核心检测器实现：

    from collections import deque

class LoopDetector:
    """循环检测器——用 deque 滑动窗口识别重复模式"""
    def __init__(self, window_size: int = 3):
        self.window_size = window_size
        self.history: deque[tuple] = deque(maxlen=window_size)

    def add(self, action: str, action_input: str, observation: str = ""):
        self.history.append((action, action_input, observation[:200]))

    def is_looping(self) -> bool:
        if len(self.history) < self.window_size:
            return False
        # 检测 1：最近 N 次 action 全部相同
        actions = [h[0] for h in self.history]
        if len(set(actions)) == 1:
            return True
        # 检测 2：最近 2 次 (action, input) 对完全相同
        if len(self.history) >= 2:
            pairs = [(h[0], h[1]) for h in list(self.history)[-2:]]
            if len(set(pairs)) == 1:
                return True
        return False

    def get_summary(self) -> str:
        recent = list(self.history)
        actions = [h[0] for h in recent]
        return f"Loop detected: repeating '{actions[-1]}' for {len(recent)} steps"

    def recover_from_loop(history: str, loop_summary: str, llm) -> str:

    LangGraph 中的集成：

    def should_continue(state) -> str:
    # 防御 1：最大步数
    if state["step_count"] >= 15:
        return "max_steps_reached"

    # 防御 2：重复动作检测
    history = state.get("action_history", [])

    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END

# 兜底节点
def max_steps_handler(state):
    return {"messages": state["messages"] + [{
        "role": "assistant",
        "content": "Max steps reached. Could not complete the task.",
    }]}

def loop_handler(state):
    repeated = state["action_history"][-1]
    return {"messages": state["messages"] + [{
        "role": "assistant",
        "content": f"Detected loop: repeating '{repeated}'. Please rephrase or provide guidance.",
    }]}

    4.1 终止条件¶

    上面 4.0 的 LoopDetector 是轻量级方案，下面是更完整的 TerminationChecker， 整合了最大迭代、Token 预算、连续错误、无进展等多维度判断。

    class TerminationChecker:
    """多维度终止条件检查"""

    def __init__(self, max_iterations: int = 10, max_tokens: int = 8000):
        self.max_iterations = max_iterations
        self.max_tokens = max_tokens

    def check(self, state: AgentState) -> tuple[bool, str]:
        """检查是否应该终止

        Returns:
            (should_terminate, reason)
        """
        # 1. 最大迭代次数
        if state.iteration >= self.max_iterations:
            return True, "max_iterations"

        # 2. Token 预算
        total_tokens = sum(len(a.get("result", "")) for a in state.actions)
        if total_tokens > self.max_tokens:
            return True, "token_budget"

        # 3. 连续错误
        recent = state.actions[-5:]
        error_count = sum(1 for a in recent if a.get("result", "").startswith("Error"))
        if error_count >= 3:
            return True, "consecutive_errors"

        # 4. 无进展检测（连续多次相同动作）
        if len(state.actions) >= 4:
            last_4 = [a["name"] for a in state.actions[-4:]]
            if len(set(last_4)) == 1:
                return True, "no_progress"

        # 5. Agent 自己表示完成
        if state.thoughts and "final answer" in state.thoughts[-1].lower():
            return True, "agent_decided"

        return False, ""

    4.2 超时控制¶

    import asyncio
from functools import wraps

def with_timeout(seconds: int):
    """给控制循环添加超时"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            try:
                return await asyncio.wait_for(
                    func(*args, **kwargs),
                    timeout=seconds
                )
            except asyncio.TimeoutError:
                return AgentState(
                    goal=args[1] if len(args) > 1 else kwargs.get("goal", ""),
                    error=f"Timeout after {seconds}s",
                    is_done=True,
                )
        return wrapper
    return decorator

# 使用
class TimeoutControlLoop(ControlLoop):
    @with_timeout(seconds=60)
    async def run(self, goal: str) -> AgentState:
        return await super().run_async(goal)

    4.3 断点续跑¶

    import pickle
import os

class CheckpointControlLoop(ControlLoop):
    """支持断点续跑的控制循环"""

    def __init__(self, checkpoint_dir: str = "./checkpoints", **kwargs):
        super().__init__(**kwargs)
        self.checkpoint_dir = checkpoint_dir
        os.makedirs(checkpoint_dir, exist_ok=True)

    def save_checkpoint(self, state: AgentState, run_id: str):
        """保存检查点"""
        path = os.path.join(self.checkpoint_dir, f"{run_id}.pkl")
        with open(path, "wb") as f:
            pickle.dump(state, f)

    def load_checkpoint(self, run_id: str) -> Optional[AgentState]:
        """加载检查点"""
        path = os.path.join(self.checkpoint_dir, f"{run_id}.pkl")
        if os.path.exists(path):
            with open(path, "rb") as f:
                return pickle.load(f)
        return None

    def run_with_checkpoint(self, goal: str, run_id: str) -> AgentState:
        """带检查点的运行"""
        # 尝试恢复
        state = self.load_checkpoint(run_id)
        if state:
            print(f"Resuming from checkpoint (iteration {state.iteration})")
        else:
            state = AgentState(goal=goal, max_iterations=self.max_iterations)
            print(f"Starting new run")

        while not state.is_done and state.iteration < state.max_iterations:
            state.iteration += 1

            self._observe(state)
            self._think(state)
            self._act(state)
            self._reflect(state)
            state.is_done = self._should_terminate(state)

            # 每步后保存检查点
            self.save_checkpoint(state, run_id)

        if not state.error:
            state.final_output = self._generate_final_output(state)

        # 完成后删除检查点
        self._cleanup_checkpoint(run_id)

        return state

    def _cleanup_checkpoint(self, run_id: str):
        path = os.path.join(self.checkpoint_dir, f"{run_id}.pkl")
        if os.path.exists(path):
            os.remove(path)

    5. 循环可视化与调试¶

    5.1 执行 Trace¶

    class TraceControlLoop(ControlLoop):
    """带详细 Trace 的控制循环"""

    def run(self, goal: str) -> AgentState:
        state = AgentState(goal=goal, max_iterations=self.max_iterations)
        trace = []

        while not state.is_done and state.iteration < state.max_iterations:
            step_trace = {"iteration": state.iteration + 1}
            state.iteration += 1

            # Observe
            self._observe(state)
            step_trace["observation"] = state.observations[-1] if state.observations else None

            # Think
            self._think(state)
            step_trace["thought"] = state.thoughts[-1] if state.thoughts else None
            step_trace["decision"] = getattr(state, "_current_decision", None)

            # Act
            self._act(state)
            step_trace["action"] = state.actions[-1] if state.actions else None

            # Reflect
            self._reflect(state)
            step_trace["reflection"] = state.reflections[-1] if state.reflections else None

            trace.append(step_trace)

            state.is_done = self._should_terminate(state)

        state.trace = trace
        if not state.error:
            state.final_output = self._generate_final_output(state)

        return state

# 打印 Trace
def print_trace(state: AgentState):
    print(f"\n Execution Trace")
    print(f"{'='*60}")
    print(f"Goal: {state.goal}")
    print(f"Iterations: {state.iteration}")
    print(f"{'='*60}")

    for step in state.trace:
        print(f"\n Step {step['iteration']}:")
        print(f"   Thought: {step.get('thought', '')[:100]}")
        print(f"   Decision: {step.get('decision', '')}")
        action = step.get('action')
        if action:
            print(f"   Action: {action['name']}({action['input']})")
            print(f"   Result: {str(action['result'])[:100]}")
        reflection = step.get('reflection')
        if reflection:
            print(f"   Reflection: {reflection[:100]}")

    print(f"\n{'='*60}")
    print(f"Final: {state.final_output[:200] if state.final_output else state.error}")

    6. 控制循环模式总结¶

    模式 结构 适用场景
    简单循环 Observe → Think → Act → 判断终止 简单任务，快速验证
    ReAct 循环 Think → Act → Observe → 循环 需要工具交互
    Reflect 循环 Observe → Think → Act → Reflect → 判断 需要自我评估
    Plan-Execute 循环 Plan → Execute → Check → Replan → 循环 长链路复杂任务
    Supervisor 循环 Supervisor 决策 → 成员执行 → 回到 Supervisor 多体协作

    7. Control Loop Checklist¶

    • 核心循环：Observe → Think → Act → Reflect
    • 终止条件：最大迭代、Token 预算、连续错误、无进展
    • 超时控制：防止无限循环
    • 断点续跑：支持长任务中断后恢复
    • 执行 Trace：详细记录每步操作，便于调试
    • 可观测性：集成 LangSmith 等追踪工具
    • 循环模式选择：根据任务复杂度选择合适的循环模式
