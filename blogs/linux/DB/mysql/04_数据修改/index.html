    04_数据修改¶

    INSERT、UPDATE、DELETE、REPLACE INTO 详解

    1. INSERT 插入数据¶

    1.1 基本语法¶

    INSERT INTO table_name (column1, column2, ...)
VALUES (value1, value2, ...);

    1.2 插入单行¶

    -- 指定列
INSERT INTO users (username, email, password)
VALUES ('zhangsan', 'zhangsan@example.com', 'hashed_password');

-- 省略列名（必须包含所有列，按顺序）
INSERT INTO users VALUES (1, 'zhangsan', 'zhangsan@example.com', ...);

-- 使用 DEFAULT
INSERT INTO users (username, email, password, status)
VALUES ('lisi', 'lisi@example.com', 'password', DEFAULT);

    1.3 插入多行¶

    -- 批量插入
INSERT INTO users (username, email, password) VALUES
('zhangsan', 'zhangsan@example.com', 'pwd1'),
('lisi', 'lisi@example.com', 'pwd2'),
('wangwu', 'wangwu@example.com', 'pwd3');

-- 从子查询插入
INSERT INTO users_backup (id, username, email)
SELECT id, username, email FROM users WHERE status = 1;

    1.4 INSERT ... SET 语法¶

    INSERT INTO users SET
    username = 'zhangsan',
    email = 'zhangsan@example.com',
    password = 'hashed_password';

    1.5 INSERT ... ON DUPLICATE KEY UPDATE¶

    -- 如果唯一键冲突则更新
INSERT INTO users (username, email, password)
VALUES ('zhangsan', 'new@example.com', 'new_pwd')
ON DUPLICATE KEY UPDATE
    email = VALUES(email),
    password = VALUES(password),
    updated_at = NOW();

-- 使用 VALUES() 函数（8.0+ 已废弃，使用别名）
INSERT INTO users (username, email) VALUES ('zhangsan', 'new@email.com')
AS new
ON DUPLICATE KEY UPDATE email = new.email;

    1.6 INSERT IGNORE¶

    -- 忽略错误（如唯一键冲突）
INSERT IGNORE INTO users (username, email)
VALUES ('zhangsan', 'zhangsan@example.com');

-- 如果已存在则跳过，不报错

    2. UPDATE 更新数据¶

    2.1 基本语法¶

    UPDATE table_name
SET column1 = value1, column2 = value2, ...
[WHERE condition]
[ORDER BY column]
[LIMIT count];

    2.2 更新单行¶

    -- 带 WHERE
UPDATE users 
SET email = 'new@example.com' 
WHERE id = 1;

-- 更新多列
UPDATE users 
SET email = 'new@example.com', 
    phone = '13800138000',
    updated_at = NOW()
WHERE id = 1;

    2.3 批量更新¶

    -- 更新多行
UPDATE users 
SET status = 0 
WHERE created_at < '2023-01-01';

-- CASE 批量更新
UPDATE products 
SET status = CASE 
    WHEN stock = 0 THEN 0
    WHEN stock < 10 THEN 1
    ELSE 2
END;

-- 使用 FIELD 批量更新
UPDATE users 
SET status = 1 
WHERE id IN (1, 2, 3, 4, 5);

    2.4 带 LIMIT 的更新¶

    -- 只更新前 10 条
UPDATE users 
SET status = 0 
WHERE last_login_at < '2023-01-01'
LIMIT 10;

    2.5 多表更新¶

    -- 基于 JOIN 更新
UPDATE orders o
JOIN users u ON o.user_id = u.id
SET o.status = 3
WHERE u.status = 0;

-- 多表同时更新
UPDATE orders o
JOIN order_items oi ON o.id = oi.order_id
SET o.status = 3, oi.updated_at = NOW()
WHERE o.id = 1;

    2.6 使用子查询更新¶

    -- 单值子查询
UPDATE products 
SET price = (SELECT AVG(price) FROM products)
WHERE id = 1;

-- 相关子查询
UPDATE products p
SET price = (
    SELECT AVG(price) 
    FROM products 
    WHERE category_id = p.category_id
);

-- FROM 子查询（MySQL 特有）
UPDATE products p
JOIN (
    SELECT category_id, AVG(price) AS avg_price
    FROM products
    GROUP BY category_id
) AS stats ON p.category_id = stats.category_id
SET p.price = stats.avg_price;

    3. DELETE 删除数据¶

    3.1 基本语法¶

    DELETE FROM table_name
[WHERE condition]
[ORDER BY column]
[LIMIT count];

    3.2 删除单行¶

    -- 带 WHERE
DELETE FROM users WHERE id = 1;

-- 使用主键
DELETE FROM users WHERE id = 1;

    3.3 批量删除¶

    -- 删除多行
DELETE FROM users WHERE status = 0;

-- IN 删除
DELETE FROM users WHERE id IN (1, 2, 3);

-- 带条件删除
DELETE FROM orders 
WHERE created_at < '2023-01-01' AND status = 'cancelled';

    3.4 带 LIMIT 的删除¶

    -- 删除前 100 条
DELETE FROM logs ORDER BY created_at ASC LIMIT 100;

-- 分批删除大表
DELETE FROM logs 
WHERE created_at < '2023-01-01'
ORDER BY created_at ASC
LIMIT 1000;

    3.5 多表删除¶

    -- 删除关联表数据
DELETE o, oi
FROM orders o
JOIN order_items oi ON o.id = oi.order_id
WHERE o.user_id = 1;

-- 使用 USING
DELETE FROM o, oi
USING orders o
JOIN order_items oi ON o.id = oi.order_id
WHERE o.user_id = 1;

    3.6 TRUNCATE vs DELETE¶

    -- DELETE
DELETE FROM users;  -- 逐行删除，可回滚，保留自增

-- TRUNCATE
TRUNCATE TABLE users;  -- 删除表重建，不可回滚，重置自增

-- 区别：
-- DELETE: DML，可带 WHERE，可回滚，速度慢，保留自增
-- TRUNCATE: DDL，不能带 WHERE，不可回滚，速度快，重置自增

    4. REPLACE INTO¶

    4.1 基本语法¶

    REPLACE INTO table_name (column1, column2, ...)
VALUES (value1, value2, ...);

    4.2 REPLACE 工作原理¶

    -- 如果唯一键/主键冲突：
-- 1. 删除旧记录
-- 2. 插入新记录

REPLACE INTO users (id, username, email)
VALUES (1, 'zhangsan', 'new@example.com');

-- 等价于：
-- DELETE FROM users WHERE id = 1;
-- INSERT INTO users (id, username, email) VALUES (1, 'zhangsan', 'new@example.com');

    4.3 REPLACE 注意事项¶

    -- 注意：REPLACE 会删除再插入
-- 1. 自增 ID 会变化（如果不是主键冲突）
-- 2. 触发器会触发 DELETE 和 INSERT
-- 3. 外键约束可能失败

-- 推荐使用 INSERT ... ON DUPLICATE KEY UPDATE 替代
INSERT INTO users (id, username, email)
VALUES (1, 'zhangsan', 'new@example.com')
ON DUPLICATE KEY UPDATE
    username = VALUES(username),
    email = VALUES(email);

    5. 批量操作优化¶

    5.1 批量插入优化¶

    -- 方式 1：多 VALUES（推荐）
INSERT INTO users (username, email) VALUES
('u1', 'u1@example.com'),
('u2', 'u2@example.com'),
...
('u1000', 'u1000@example.com');

-- 方式 2：LOAD DATA（最快）
LOAD DATA LOCAL INFILE '/path/to/users.csv'
INTO TABLE users
FIELDS TERMINATED BY ','
LINES TERMINATED BY '\n'
(username, email);

-- 方式 3：事务包裹
START TRANSACTION;
INSERT INTO users ...;
INSERT INTO users ...;
...
COMMIT;

    5.2 批量更新优化¶

    -- 方式 1：CASE 语句
UPDATE products SET price = CASE id
    WHEN 1 THEN 100
    WHEN 2 THEN 200
    WHEN 3 THEN 300
    ELSE price
END
WHERE id IN (1, 2, 3);

-- 方式 2：临时表
CREATE TEMPORARY TABLE temp_prices (id INT, price DECIMAL(10,2));
INSERT INTO temp_prices VALUES (1, 100), (2, 200), (3, 300);

UPDATE products p
JOIN temp_prices t ON p.id = t.id
SET p.price = t.price;

DROP TEMPORARY TABLE temp_prices;

    5.3 批量删除优化¶

    -- 分批删除，避免长事务
DELETE FROM logs WHERE created_at < '2023-01-01' LIMIT 1000;
-- 重复执行直到影响行数为 0

-- 使用分区表（如果表已分区）
ALTER TABLE logs TRUNCATE PARTITION p2022;

    6. 事务中的 DML¶

    -- 开启事务
START TRANSACTION;
-- 或
BEGIN;

-- 执行 DML
INSERT INTO users (username, email) VALUES ('test', 'test@example.com');
UPDATE users SET status = 1 WHERE id = 1;
DELETE FROM logs WHERE created_at < '2023-01-01';

-- 提交事务
COMMIT;

-- 或回滚
ROLLBACK;

-- 设置保存点
START TRANSACTION;
INSERT INTO users ...;
SAVEPOINT sp1;
UPDATE users ...;
-- 回滚到保存点
ROLLBACK TO sp1;
-- 继续或提交
COMMIT;

    7. 实战示例¶

    7.1 用户注册¶

    -- 插入新用户
INSERT INTO users (username, email, password_hash, nickname)
VALUES ('zhangsan', 'zhangsan@example.com', 'hashed_pwd', '张三');

-- 获取自增 ID
SELECT LAST_INSERT_ID();

    7.2 用户信息更新¶

    -- 更新部分字段
UPDATE users 
SET nickname = '新昵称',
    avatar_url = 'https://example.com/avatar.jpg',
    updated_at = NOW()
WHERE id = 1;

-- 更新密码
UPDATE users 
SET password_hash = 'new_hashed_pwd',
    updated_at = NOW()
WHERE id = 1;

    7.3 下单流程¶

    START TRANSACTION;

-- 1. 创建订单
INSERT INTO orders (order_no, user_id, total_amount, status)
VALUES ('ORD202403250001', 1, 299.00, 0);

-- 获取订单 ID
SET @order_id = LAST_INSERT_ID();

-- 2. 插入订单详情
INSERT INTO order_items (order_id, product_id, quantity, unit_price, subtotal)
VALUES 
(@order_id, 1, 2, 99.00, 198.00),
(@order_id, 2, 1, 101.00, 101.00);

-- 3. 扣减库存
UPDATE products 
SET stock = stock - 2 
WHERE id = 1 AND stock >= 2;

UPDATE products 
SET stock = stock - 1 
WHERE id = 2 AND stock >= 1;

-- 4. 检查库存是否足够（ROW_COUNT() = 0 表示库存不足）
-- 如果库存不足，ROLLBACK

COMMIT;

    7.4 数据归档¶

    -- 创建归档表
CREATE TABLE orders_archive LIKE orders;

-- 移动旧数据
START TRANSACTION;

INSERT INTO orders_archive
SELECT * FROM orders 
WHERE created_at < '2023-01-01';

DELETE FROM orders 
WHERE created_at < '2023-01-01';

COMMIT;

    8. 本章小结¶

    核心知识点¶

    ✅ INSERT 插入数据
    ✅ UPDATE 更新数据
    ✅ DELETE 删除数据
    ✅ REPLACE INTO
    ✅ 批量操作优化
    ✅ 事务中的 DML

    最佳实践¶

    1. 批量插入：使用多 VALUES 或 LOAD DATA
    2. 批量更新：使用 CASE 或临时表
    3. 批量删除：分批执行，避免长事务
    4. 事务包裹：保证数据一致性
    5. 使用索引：WHERE 条件列添加索引
