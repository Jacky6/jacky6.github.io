    08 - Python 客户端实战¶

    客户端选择¶

    库 底层 特点 现状（2026-08）
    confluent-kafka librdkafka (C) 性能最好，功能最全，推荐 活跃维护
    kafka-python 纯 Python 安装简单，无 C 依赖 2.x 长期停更；3.0（2025 起）由原作者复活，内部网络层重写为 async substrate，高层 API 基本兼容并新增 cooperative rebalance 支持
    kafka-python-ng 纯 Python kafka-python 停更期间的社区维护分叉 老项目存量常见
    aiokafka asyncio 异步支持 活跃维护 
    # 推荐
pip install confluent-kafka

# 备选
pip install kafka-python   # 3.x 是新架构，老代码注意 API 变化
pip install aiokafka

    选型建议：生产环境优先 confluent-kafka（性能好、支持 Share Groups 等新特性， 2.12+ 对 KIP-848 新 rebalance 协议已生产可用；kafka-python / aiokafka 暂不支持）。

    confluent-kafka（推荐）¶

    Producer¶

    from confluent_kafka import Producer

conf = {
    'bootstrap.servers': 'broker1:9092,broker2:9092',
    'client.id': 'python-producer',
    'acks': 'all',
    'enable.idempotence': True,
    'batch.size': 16384,
    'linger.ms': 5,
    'compression.type': 'lz4',
}

producer = Producer(conf)


def delivery_report(err, msg):
    """消息投递回调"""
    if err is not None:
        print(f'Message delivery failed: {err}')
    else:
        print(f'Message delivered to {msg.topic()} [{msg.partition()}] @ offset {msg.offset()}')


# 发送消息
producer.produce(
    topic='orders',
    key='order-123',
    value='{"amount": 99.9}',
    callback=delivery_report,
)

# 触发回调（异步，需要 poll）
producer.poll(0)

# 等待所有消息投递完成
producer.flush()

    Consumer¶

    from confluent_kafka import Consumer

conf = {
    'bootstrap.servers': 'broker1:9092,broker2:9092',
    'group.id': 'order-service',
    'auto.offset.reset': 'earliest',
    'enable.auto.commit': False,
    'max.poll.interval.ms': 300000,
}

consumer = Consumer(conf)
consumer.subscribe(['orders'])

try:
    while True:
        msg = consumer.poll(timeout=1.0)

        if msg is None:
            continue
        if msg.error():
            print(f'Consumer error: {msg.error()}')
            continue

        print(f'Received: topic={msg.topic()}, '
              f'partition={msg.partition()}, '
              f'offset={msg.offset()}, '
              f'key={msg.key()}, '
              f'value={msg.value()}')

        # 手动提交
        consumer.commit(msg)

except KeyboardInterrupt:
    pass
finally:
    consumer.close()

messages = consumer.consume(num_messages=100, timeout=1.0)

    Admin Client¶

    from confluent_kafka.admin import AdminClient, NewTopic, NewPartitions

admin = AdminClient({'bootstrap.servers': 'broker1:9092'})

# 创建 Topic
new_topics = [
    NewTopic(topic='orders', num_partitions=3, replication_factor=3),
    NewTopic(topic='payments', num_partitions=6, replication_factor=3),
]
fs = admin.create_topics(new_topics)
for topic, f in fs.items():
    try:
        f.result()
        print(f'Topic {topic} created')
    except Exception as e:
        print(f'Failed to create topic {topic}: {e}')

# 查看 Topic 列表
metadata = admin.list_topics(timeout=10)
for topic_name in metadata.topics:
    print(f'Topic: {topic_name}')

# 增加 Partition
fs = admin.create_partitions([NewPartitions('orders', new_total_count=6)])

# 删除 Topic
fs = admin.delete_topics(['old-topic'])

    JSON 序列化¶

    import json
from confluent_kafka import SerializingProducer
from confluent_kafka.serialization import StringSerializer

def json_serializer(obj, ctx=None):
    return json.dumps(obj).encode('utf-8')

producer = SerializingProducer({
    'bootstrap.servers': 'broker1:9092',
    'key.serializer': StringSerializer(),
    'value.serializer': json_serializer,
})

order = {'id': 'order-123', 'amount': 99.9, 'status': 'created'}
producer.produce(topic='orders', key=order['id'], value=order)
producer.flush()

    Avro 序列化（需要 Schema Registry）¶

    pip install confluent-kafka[avro]
# 或
pip install confluent-kafka[jsonschema]

    from confluent_kafka import SerializingProducer
from confluent_kafka.schema_registry import SchemaRegistryClient
from confluent_kafka.schema_registry.avro import AvroSerializer

schema_registry = SchemaRegistryClient({'url': 'http://localhost:8081'})

order_schema = '''
{
  "type": "record",
  "name": "Order",
  "fields": [
    {"name": "id", "type": "string"},
    {"name": "amount", "type": "double"},
    {"name": "status", "type": "string"}
  ]
}
'''

avro_serializer = AvroSerializer(schema_registry, order_schema)

producer = SerializingProducer({
    'bootstrap.servers': 'broker1:9092',
    'key.serializer': lambda k, ctx: k.encode('utf-8'),
    'value.serializer': avro_serializer,
})

producer.produce(
    topic='orders',
    key='order-123',
    value={'id': 'order-123', 'amount': 99.9, 'status': 'created'},
)
producer.flush()

    事务¶

    from confluent_kafka import Producer

producer = Producer({
    'bootstrap.servers': 'broker1:9092',
    'transactional.id': 'python-tx-001',
})

producer.init_transactions()

try:
    producer.begin_transaction()

    producer.produce('account-a', value='debit:100')
    producer.produce('account-b', value='credit:100')
    producer.produce('notifications', value='transfer done')

    producer.commit_transaction()
    print('Transaction committed')

except Exception as e:
    print(f'Transaction failed: {e}')
    producer.abort_transaction()

    kafka-python¶

    from kafka import KafkaProducer, KafkaConsumer, KafkaAdminClient
from kafka.admin import NewTopic

# Producer
producer = KafkaProducer(
    bootstrap_servers=['broker1:9092', 'broker2:9092'],
    acks='all',
    value_serializer=lambda v: v.encode('utf-8'),
)

future = producer.send('orders', key=b'order-123', value=b'{"amount": 99.9}')
record_metadata = future.get(timeout=10)
print(f'Sent to partition {record_metadata.partition}')

# Consumer
consumer = KafkaConsumer(
    'orders',
    bootstrap_servers=['broker1:9092'],
    group_id='order-service',
    auto_offset_reset='earliest',
    enable_auto_commit=False,
)

for message in consumer:
    print(f'{message.partition}:{message.offset}: '
          f'{message.key} = {message.value}')
    consumer.commit()

    aiokafka（异步）¶

    import asyncio
from aiokafka import AIOKafkaProducer, AIOKafkaConsumer


async def produce():
    producer = AIOKafkaProducer(
        bootstrap_servers='broker1:9092',
        acks='all',
    )
    await producer.start()

    try:
        for i in range(100):
            await producer.send_and_wait(
                'orders',
                key=f'order-{i}'.encode(),
                value=f'{{"amount": {i * 10}}}'.encode(),
            )
    finally:
        await producer.stop()


async def consume():
    consumer = AIOKafkaConsumer(
        'orders',
        bootstrap_servers='broker1:9092',
        group_id='order-service',
        auto_offset_reset='earliest',
    )
    await consumer.start()

    try:
        async for msg in consumer:
            print(f'{msg.topic}:{msg.partition}:{msg.offset}: '
                  f'{msg.key} = {msg.value}')
    finally:
        await consumer.stop()   # 注意：stop() 是协程，必须 await


async def main():
    await asyncio.gather(produce(), consume())


asyncio.run(main())

    性能优化¶

    # 高吞吐 Producer（librdkafka 配置名与 Java 客户端有差异）
producer = Producer({
    'bootstrap.servers': 'broker1:9092',
    'batch.size': 65536,                 # 64KB（librdkafka 默认 1MB，比 Java 的 16KB 大很多）
    'linger.ms': 50,                     # 等 50ms 攒批
    'compression.type': 'lz4',
    'queue.buffering.max.kbytes': 65536, # 64MB 队列上限（单位 KB；Java 里叫 buffer.memory）
    'acks': 1,                           # 牺牲可靠性换吞吐
})

# 高吞吐 Consumer
consumer = Consumer({
    'bootstrap.servers': 'broker1:9092',
    'group.id': 'order-service',
    'fetch.min.bytes': 1048576,          # 1MB
    'fetch.wait.max.ms': 500,            # 注意：librdkafka 里是 fetch.wait.max.ms，
                                         # 与 Java 的 fetch.max.wait.ms 词序相反
    'max.partition.fetch.bytes': 1048576,
    # 注意：Java 的 max.poll.records 在 librdkafka 里只支持 share consumer
})

    常见坑¶

    1. 忘记 poll/flush¶

    # 错误：消息可能没发出去
producer.produce('topic', value='data')
# 缺少 producer.flush()

# 正确
producer.produce('topic', value='data')
producer.flush()

    2. Consumer 没有 poll 导致 Rebalance¶

    # 处理消息太慢，超过 max.poll.interval.ms
# 解决：减小 max.poll.records，或异步处理

    3. 启用 KIP-848 新协议（confluent-kafka >= 2.12 + broker 4.0+）¶

    from confluent_kafka import Consumer

conf = {
    'bootstrap.servers': 'broker1:9092',
    'group.id': 'order-service',
    'auto.offset.reset': 'earliest',
    'enable.auto.commit': False,

    # 开启 KIP-848 新 rebalance 协议
    'group.protocol': 'consumer',

    # 可选：选择服务端 assignor（uniform 默认 / range）
    # 'group.remote.assignor': 'uniform',

    # ⚠️ 以下配置在新协议下由 broker 端控制，客户端设置会被拒绝：
    # 'session.timeout.ms': 45000,      # → broker: group.consumer.session.timeout.ms
    # 'heartbeat.interval.ms': 5000,    # → broker: group.consumer.heartbeat.interval.ms
    # 'partition.assignment.strategy': 'cooperative-sticky',  # 已废弃
}


def on_assign(consumer, partitions):
    # KIP-848 行为差异：回调收到的是增量分区（不是全量列表），
    # 对应 incremental_assign / incremental_unassign
    print(f'incrementally assigning: {partitions}')
    consumer.incremental_assign(partitions)   # 不手动调也行，客户端自动处理


def on_revoke(consumer, partitions):
    consumer.commit()                         # 交出前提交进度（业务需要时才加）
    consumer.incremental_unassign(partitions)


consumer = Consumer(conf)
consumer.subscribe(['orders'], on_assign=on_assign, on_revoke=on_revoke)

try:
    while True:
        msg = consumer.poll(timeout=1.0)
        if msg is None:
            continue
        if msg.error():
            print(f'Consumer error: {msg.error()}')
            continue
        process(msg)              # 业务处理
        consumer.commit(msg)      # offset 提交方式不变
finally:
    consumer.close()

    # 版本自检
import confluent_kafka
print(confluent_kafka.version())     # Python 客户端版本，需要 >= 2.12
print(confluent_kafka.libversion())  # 底层 librdkafka 版本

    迁移/回滚：滚动重启逐个切 group.protocol=consumer，Coordinator 自动切换整组； 出问题时删掉该配置重启，最后一个新协议 consumer 离开后组自动回落经典协议。 kafka-python / aiokafka 是纯 Python 实现，暂不支持新协议。

    4. 序列化错误¶

    # confluent-kafka 要求 value 是 bytes 或 str
producer.produce('topic', value={'key': 'value'})  # 报错！
producer.produce('topic', value=json.dumps({'key': 'value'}))  # 正确
