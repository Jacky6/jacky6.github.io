    WebSocket 教程（三）：服务端实现¶

    多语言实现 WebSocket 服务端，涵盖 Python、Node.js、Go、Java

    一、Python 实现¶

    1.1 websockets 库（推荐）¶

    pip install websockets

    基础示例：

    import asyncio
import websockets

async def handler(websocket):
    """处理每个 WebSocket 连接"""
    print(f"新连接: {websocket.remote_address}")

    try:
        async for message in websocket:
            print(f"收到: {message}")

            # 回显消息
            await websocket.send(f"Echo: {message}")
    except websockets.exceptions.ConnectionClosed as e:
        print(f"连接关闭: code={e.code}, reason={e.reason}")
    finally:
        print(f"连接断开: {websocket.remote_address}")

async def main():
    async with websockets.serve(handler, "localhost", 8765):
        print("WebSocket 服务器启动在 ws://localhost:8765")
        await asyncio.Future()  # 永远运行

asyncio.run(main())

    1.2 带认证的示例¶

    import asyncio
import websockets
from urllib.parse import parse_qs

# 模拟用户数据库
VALID_TOKENS = {"token123": "alice", "token456": "bob"}

async def handler(websocket):
    """带认证的 WebSocket 处理器"""

    # 从 URL 参数获取 Token
    path = websocket.request.path
    query = parse_qs(path.split("?")[-1]) if "?" in path else {}
    token = query.get("token", [None])[0]

    # 验证 Token
    if token not in VALID_TOKENS:
        await websocket.close(code=4001, reason="Invalid token")
        return

    username = VALID_TOKENS[token]
    print(f"用户 {username} 已连接")

    try:
        async for message in websocket:
            data = json.loads(message)

            # 处理业务逻辑
            response = await process_message(username, data)
            await websocket.send(json.dumps(response))
    except websockets.exceptions.ConnectionClosed:
        pass
    finally:
        print(f"用户 {username} 已断开")

async def process_message(username, data):
    """处理消息"""
    msg_type = data.get("type")

    if msg_type == "chat":
        return {
            "type": "chat",
            "from": username,
            "content": data.get("content"),
            "timestamp": time.time(),
        }
    elif msg_type == "ping":
        return {"type": "pong"}
    else:
        return {"type": "error", "message": "Unknown type"}

async def main():
    async with websockets.serve(handler, "localhost", 8765):
        await asyncio.Future()

asyncio.run(main())

    1.3 广播聊天室¶

    import asyncio
import websockets
import json

# 存储所有连接
connected_clients = set()

async def handler(websocket):
    """聊天室处理器"""

    # 注册客户端
    connected_clients.add(websocket)
    print(f"当前在线: {len(connected_clients)} 人")

    # 广播新用户加入
    await broadcast({
        "type": "system",
        "content": f"新用户加入，当前在线 {len(connected_clients)} 人"
    })

    try:
        async for message in websocket:
            data = json.loads(message)

            # 广播消息给所有人
            await broadcast({
                "type": "chat",
                "from": websocket.remote_address[1],  # 用端口号当昵称
                "content": data.get("content"),
            })
    except websockets.exceptions.ConnectionClosed:
        pass
    finally:
        # 移除客户端
        connected_clients.discard(websocket)
        await broadcast({
            "type": "system",
            "content": f"用户离开，当前在线 {len(connected_clients)} 人"
        })

async def broadcast(message):
    """广播消息给所有连接的客户端"""
    if not connected_clients:
        return

    msg_str = json.dumps(message)

    # 并发发送
    await asyncio.gather(
        *[client.send(msg_str) for client in connected_clients],
        return_exceptions=True  # 忽略单个发送失败
    )

async def main():
    async with websockets.serve(handler, "localhost", 8765):
        await asyncio.Future()

asyncio.run(main())

    import asyncio

    1.5 FastAPI + WebSocket¶

    from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from typing import List

app = FastAPI()

class ConnectionManager:
    """WebSocket 连接管理器"""

    def __init__(self):
        self.active_connections: List[WebSocket] = []

    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)

    def disconnect(self, websocket: WebSocket):
        self.active_connections.remove(websocket)

    async def send_personal(self, message: dict, websocket: WebSocket):
        await websocket.send_json(message)

    async def broadcast(self, message: dict):
        for connection in self.active_connections:
            try:
                await connection.send_json(message)
            except RuntimeError:
                pass  # 连接已关闭

manager = ConnectionManager()

@app.websocket("/ws/{client_id}")
async def websocket_endpoint(websocket: WebSocket, client_id: int):
    await manager.connect(websocket)
    await manager.broadcast({
        "type": "system",
        "content": f"客户端 {client_id} 加入"
    })

    try:
        while True:
            data = await websocket.receive_json()

            # 处理消息
            await manager.broadcast({
                "type": "chat",
                "from": client_id,
                "content": data.get("content"),
            })
    except WebSocketDisconnect:
        manager.disconnect(websocket)
        await manager.broadcast({
            "type": "system",
            "content": f"客户端 {client_id} 离开"
        })

    const WebSocket = require("ws");

const wss = new WebSocket.Server({ port: 8080 });

// 广播给所有客户端
function broadcast(data, exclude = null) {
    const message = JSON.stringify(data);

    wss.clients.forEach((client) => {
        if (client !== exclude && client.readyState === WebSocket.OPEN) {
            client.send(message);
        }
    });
}

wss.on("connection", (ws, req) => {
    const clientId = req.socket.remotePort;

    // 广播新用户
    broadcast({ type: "system", content: `用户 ${clientId} 加入` });

    ws.on("message", (message) => {
        const data = JSON.parse(message);

        // 广播消息
        broadcast({
            type: "chat",
            from: clientId,
            content: data.content,
        });
    });

    ws.on("close", () => {
        broadcast({ type: "system", content: `用户 ${clientId} 离开` });
    });
});

    2.4 心跳保活¶

    const WebSocket = require("ws");

const wss = new WebSocket.Server({ port: 8080 });

// 心跳检测
const interval = setInterval(() => {
    wss.clients.forEach((ws) => {
        if (ws.isAlive === false) {
            console.log("心跳超时，关闭连接");
            return ws.terminate();
        }

        ws.isAlive = false;
        ws.ping();  // 发送 Ping
    });
}, 30000);  // 每 30 秒检查一次

wss.on("connection", (ws) => {
    ws.isAlive = true;

    ws.on("pong", () => {
        ws.isAlive = true;  // 收到 Pong，标记为存活
    });
});

wss.on("close", () => {
    clearInterval(interval);
});

    三、Go 实现¶

    3.1 gorilla/websocket（经典库）¶

    go get github.com/gorilla/websocket

    基础示例：

    package main

import (
    "fmt"
    "log"
    "net/http"
    "github.com/gorilla/websocket"
)

var upgrader = websocket.Upgrader{
    CheckOrigin: func(r *http.Request) bool {
        return true  // 允许所有来源（生产环境需要限制）
    },
}

func handler(w http.ResponseWriter, r *http.Request) {
    // 升级到 WebSocket
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
        log.Println("升级失败:", err)
        return
    }
    defer conn.Close()

    fmt.Println("新连接:", r.RemoteAddr)

    for {
        // 读取消息
        messageType, message, err := conn.ReadMessage()
        if err != nil {
            log.Println("读取失败:", err)
            break
        }

        fmt.Printf("收到: %s\n", message)

        // 回显消息
        if err := conn.WriteMessage(messageType, message); err != nil {
            log.Println("写入失败:", err)
            break
        }
    }
}

func main() {
    http.HandleFunc("/ws", handler)
    fmt.Println("WebSocket 服务器启动在 ws://localhost:8080/ws")
    log.Fatal(http.ListenAndServe(":8080", nil))
}

    3.2 广播聊天室¶

    package main

import (
    "fmt"
    "log"
    "net/http"
    "sync"
    "github.com/gorilla/websocket"
)

var upgrader = websocket.Upgrader{
    CheckOrigin: func(r *http.Request) bool { return true },
}

// 客户端管理器
type ClientManager struct {
    clients    map[*websocket.Conn]bool
    broadcast  chan []byte
    register   chan *websocket.Conn
    unregister chan *websocket.Conn
    mutex      sync.Mutex
}

var manager = &ClientManager{
    clients:    make(map[*websocket.Conn]bool),
    broadcast:  make(chan []byte),
    register:   make(chan *websocket.Conn),
    unregister: make(chan *websocket.Conn),
}

func (m *ClientManager) run() {
    for {
        select {
        case conn := <-m.register:
            m.mutex.Lock()
            m.clients[conn] = true
            m.mutex.Unlock()
            fmt.Printf("当前在线: %d 人\n", len(m.clients))

        case conn := <-m.unregister:
            m.mutex.Lock()
            if _, ok := m.clients[conn]; ok {
                conn.Close()
                delete(m.clients, conn)
            }
            m.mutex.Unlock()
            fmt.Printf("当前在线: %d 人\n", len(m.clients))

        case message := <-m.broadcast:
            m.mutex.Lock()
            for conn := range m.clients {
                if err := conn.WriteMessage(websocket.TextMessage, message); err != nil {
                    conn.Close()
                    delete(m.clients, conn)
                }
            }
            m.mutex.Unlock()
        }
    }
}

func handler(w http.ResponseWriter, r *http.Request) {
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
        log.Println("升级失败:", err)
        return
    }

    manager.register <- conn

    defer func() {
        manager.unregister <- conn
    }()

    for {
        _, message, err := conn.ReadMessage()
        if err != nil {
            break
        }

        // 广播消息
        manager.broadcast <- message
    }
}

func main() {
    go manager.run()

    http.HandleFunc("/ws", handler)
    fmt.Println("WebSocket 服务器启动在 ws://localhost:8080/ws")
    log.Fatal(http.ListenAndServe(":8080", nil))
}

    3.3 心跳保活¶

    func handler(w http.ResponseWriter, r *http.Request) {
    conn, _ := upgrader.Upgrade(w, r, nil)
    defer conn.Close()

    // 配置 Ping/Pong
    conn.SetPongHandler(func(string) error {
        conn.SetReadDeadline(time.Time{})  // 重置超时
        return nil
    })

    // 启动 Ping 发送协程
    go func() {
        ticker := time.NewTicker(30 * time.Second)
        defer ticker.Stop()

        for range ticker.C {
            if err := conn.WriteMessage(websocket.PingMessage, nil); err != nil {
                return
            }
        }
    }()

    // 设置读超时
    conn.SetReadDeadline(time.Now().Add(60 * time.Second))

    for {
        _, message, err := conn.ReadMessage()
        if err != nil {
            break
        }
        conn.WriteMessage(websocket.TextMessage, message)
    }
}

    @Configuration
@EnableWebSocket
public class WebSocketConfig implements WebSocketConfigurer {

    @Override
    public void registerWebSocketHandlers(WebSocketHandlerRegistry registry) {
        registry.addHandler(new ChatHandler(), "/ws/chat")
                .setAllowedOrigins("*");  // 生产环境需要限制
    }

public class ChatHandler extends TextWebSocketHandler {

    private final Set<WebSocketSession> sessions = 
        Collections.newSetFromMap(new ConcurrentHashMap<>());

    @Override
    public void afterConnectionEstablished(WebSocketSession session) {
        sessions.add(session);
        System.out.println("新连接: " + session.getId());
    }

    @Override
    protected void handleTextMessage(WebSocketSession session, TextMessage message) 
            throws Exception {
        String payload = message.getPayload();
        System.out.println("收到: " + payload);

        // 广播给所有人
        for (WebSocketSession s : sessions) {
            if (s.isOpen()) {
                s.sendMessage(new TextMessage("Echo: " + payload));
            }
        }
    }

    @Override
    public void afterConnectionClosed(WebSocketSession session, CloseStatus status) {
        sessions.remove(session);
        System.out.println("连接关闭: " + session.getId() + ", status: " + status);
    }
}

    5.1 单机优化¶

    1. 文件描述符
   - 调整 ulimit -n（最大文件描述符数）
   - 每个 WebSocket 连接占用一个文件描述符

2. 内存优化
   - 减少每个连接的内存占用
   - 使用对象池复用缓冲区

3. 异步 I/O
   - 使用 epoll/kqueue 等事件驱动模型
   - Python: asyncio
   - Node.js: 天然异步
   - Go: goroutine
   - Java: Netty

4. 心跳优化
   - 不要为每个连接创建定时器
   - 使用全局定时器批量检查

    5.3 Redis Pub/Sub 实现跨服务器广播¶

    import asyncio
import websockets
import redis.asyncio as redis

connected_clients = set()
redis_client = redis.Redis()

async def handler(websocket):
    connected_clients.add(websocket)

    # 订阅 Redis 频道
    pubsub = redis_client.pubsub()
    await pubsub.subscribe("chat")

    try:
        # 监听 Redis 消息和本地 WebSocket 消息
        async for message in pubsub.listen():
            if message["type"] == "message":
                # 广播给本地连接
                data = message["data"].decode()
                await asyncio.gather(
                    *[client.send(data) for client in connected_clients],
                    return_exceptions=True
                )

        async for message in websocket:
            # 发布到 Redis（其他服务器会收到）
            await redis_client.publish("chat", message)
    finally:
        connected_clients.discard(websocket)
        await pubsub.unsubscribe("chat")

    6.2 消息批处理¶

    # 不要每条消息都立即发送，攒一批再发
buffer = []

async def flush_buffer():
    global buffer
    if buffer:
        batch = json.dumps(buffer)
        await broadcast(batch)
        buffer.clear()

async def handler(websocket):
    global buffer

    async for message in websocket:
        buffer.append(message)

        # 每 100ms 批量发送一次
        await asyncio.sleep(0.1)
        await flush_buffer()

    6.3 背压控制¶

    # 当客户端处理不过来时，限制发送速率
async def send_with_backpressure(websocket, message):
    try:
        # 设置超时，防止无限等待
        await asyncio.wait_for(websocket.send(message), timeout=5.0)
    except asyncio.TimeoutError:
        # 客户端太慢，关闭连接
        await websocket.close(code=1008, reason="Send timeout")
