    一、浏览器原生 WebSocket API¶

    1.1 基础用法¶

    // 创建连接
const ws = new WebSocket("ws://localhost:8080");

// 连接成功
ws.onopen = () => {
    console.log("连接已建立");
    ws.send("Hello Server!");
};

// 接收消息
ws.onmessage = (event) => {
    console.log("收到消息:", event.data);
};

// 连接关闭
ws.onclose = (event) => {
    console.log(`连接关闭: code=${event.code}, reason=${event.reason}`);
};

// 连接错误
ws.onerror = (error) => {
    console.error("WebSocket 错误:", error);
};

    1.2 readyState 状态¶

    const ws = new WebSocket("ws://localhost:8080");

console.log(ws.readyState);

// WebSocket.CONNECTING = 0  正在连接
// WebSocket.OPEN = 1        连接成功
// WebSocket.CLOSING = 2     正在关闭
// WebSocket.CLOSED = 3      已关闭

// 发送前检查状态
function sendMessage(message) {
    if (ws.readyState === WebSocket.OPEN) {
        ws.send(message);
    } else {
        console.warn("WebSocket 未连接");
    }
}

    // 查看缓冲区中待发送的数据量（字节）

// 背压控制：缓冲区太大时暂停发送
function sendWithBackpressure(data) {
    if (ws.bufferedAmount > 1024 * 1024) {  // > 1MB
        console.warn("缓冲区积压，暂停发送");
        return false;
    }
    ws.send(data);
    return true;
}

    二、🎯 面试考点：自动重连实现¶

    面试官：WebSocket 断线后如何实现自动重连？

    2.1 指数退避重连¶

    class ReconnectingWebSocket {
    constructor(url, protocols = []) {
        this.url = url;
        this.protocols = protocols;
        this.reconnectInterval = 1000;    // 初始重连间隔 1 秒
        this.maxReconnectInterval = 30000; // 最大重连间隔 30 秒
        this.reconnectDecay = 1.5;        // 指数增长因子
        this.maxReconnectAttempts = 10;   // 最大重连次数
        this.reconnectAttempts = 0;
        this.shouldReconnect = true;

        this.ws = null;
        this.listeners = {
            open: [],
            message: [],
            close: [],
            error: [],
            reconnecting: [],
            max_reconnect_attempts: [],
        };

        this.connect();
    }

    connect() {
        this.ws = new WebSocket(this.url, this.protocols);

        this.ws.onopen = () => {
            this.reconnectAttempts = 0;
            this.reconnectInterval = 1000;
            this.listeners.open.forEach(fn => fn());
        };

        this.ws.onmessage = (event) => {
            this.listeners.message.forEach(fn => fn(event));
        };

        this.ws.onclose = (event) => {
            this.listeners.close.forEach(fn => fn(event));

            if (this.shouldReconnect) {
                this._reconnect();
            }
        };

        this.ws.onerror = (event) => {
            this.listeners.error.forEach(fn => fn(event));
        };
    }

    _reconnect() {
        if (this.reconnectAttempts >= this.maxReconnectAttempts) {
            this.listeners.max_reconnect_attempts.forEach(fn => fn());
            return;
        }

        this.reconnectAttempts++;

        // 通知监听器
        this.listeners.reconnecting.forEach(fn => fn({
            attempt: this.reconnectAttempts,
            interval: this.reconnectInterval,
        }));

        // 指数退避
        setTimeout(() => {
            this.reconnectInterval = Math.min(
                this.reconnectInterval * this.reconnectDecay,
                this.maxReconnectInterval
            );
            this.connect();
        }, this.reconnectInterval);
    }

    send(data) {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(data);
        } else {
            throw new Error("WebSocket 未连接");
        }
    }

    close() {
        this.shouldReconnect = false;
        this.ws.close();
    }

    // 事件监听
    on(event, callback) {
        if (this.listeners[event]) {
            this.listeners[event].push(callback);
        }
    }
}

// 使用
const ws = new ReconnectingWebSocket("ws://localhost:8080");

ws.on("open", () => console.log("连接成功"));
ws.on("message", (e) => console.log("收到:", e.data));
ws.on("reconnecting", (e) => {
    console.log(`第 ${e.attempt} 次重连，${e.interval}ms 后...`);
});
ws.on("max_reconnect_attempts", () => {
    console.log("重连次数已达上限");
});
ws.on("close", (e) => console.log("连接关闭:", e.code));

    2.2 带消息队列的重连¶

    class ReliableWebSocket {
    constructor(url) {
        this.url = url;
        this.ws = null;
        this.messageQueue = [];  // 消息队列
        this.isConnected = false;

        this.connect();
    }

    connect() {
        this.ws = new WebSocket(this.url);

        this.ws.onopen = () => {
            this.isConnected = true;
            console.log("连接成功");

            // 发送队列中的消息
            while (this.messageQueue.length > 0) {
                const msg = this.messageQueue.shift();
                this.ws.send(msg);
            }
        };

        this.ws.onmessage = (event) => {
            console.log("收到:", event.data);
        };

        this.ws.onclose = () => {
            this.isConnected = false;
            console.log("连接断开，3 秒后重连...");
            setTimeout(() => this.connect(), 3000);
        };

        this.ws.onerror = () => {
            console.error("连接错误");
        };
    }

    send(data) {
        if (this.isConnected) {
            this.ws.send(data);
        } else {
            // 连接断开，加入队列等待重连后发送
            console.log("连接断开，消息加入队列");
            this.messageQueue.push(data);
        }
    }

    close() {
        this.ws.close();
    }
}

// 使用
const ws = new ReliableWebSocket("ws://localhost:8080");

// 即使连接断开，消息也不会丢失
ws.send("hello");  // 如果断开，会缓存
ws.send("world");  // 重连后自动发送

    3.1 自定义 Hook¶

    import { useEffect, useRef, useState, useCallback } from "react";

function useWebSocket(url) {
    const [messages, setMessages] = useState([]);
    const [status, setStatus] = useState("connecting");
    const wsRef = useRef(null);
    const reconnectTimeoutRef = useRef(null);
    const reconnectAttemptsRef = useRef(0);

    const connect = useCallback(() => {
        const ws = new WebSocket(url);
        wsRef.current = ws;

        ws.onopen = () => {
            setStatus("connected");
            reconnectAttemptsRef.current = 0;
        };

        ws.onmessage = (event) => {
            setMessages((prev) => [...prev, event.data]);
        };

        ws.onclose = () => {
            setStatus("disconnected");

            // 指数退避重连
            const delay = Math.min(
                1000 * Math.pow(2, reconnectAttemptsRef.current),
                30000
            );
            reconnectAttemptsRef.current++;

            reconnectTimeoutRef.current = setTimeout(() => {
                setStatus("reconnecting");
                connect();
            }, delay);
        };

        ws.onerror = () => {
            setStatus("error");
        };
    }, [url]);

    useEffect(() => {
        connect();

        return () => {
            clearTimeout(reconnectTimeoutRef.current);
            if (wsRef.current) {
                wsRef.current.close();
            }
        };
    }, [connect]);

    const sendMessage = useCallback((message) => {
        if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
            wsRef.current.send(message);
        }
    }, []);

    return { messages, status, sendMessage };
}

// 使用
function ChatRoom() {
    const { messages, status, sendMessage } = useWebSocket("ws://localhost:8080");
    const [input, setInput] = useState("");

    return (
        <div>
            <p>状态: {status}</p>

            <div style={{ height: 300, overflow: "auto" }}>
                {messages.map((msg, i) => (
                    <div key={i}>{msg}</div>
                ))}
            </div>

            <input
                value={input}
                onChange={(e) => setInput(e.target.value)}
                onKeyDown={(e) => {
                    if (e.key === "Enter") {
                        sendMessage(input);
                        setInput("");
                    }
                }}
            />
        </div>
    );
}

    四、Vue 集成¶

    4.1 Composition API¶

    import { ref, onUnmounted } from "vue";

export function useWebSocket(url) {
    const messages = ref([]);
    const status = ref("connecting");
    const ws = ref(null);
    let reconnectTimer = null;
    let reconnectAttempts = 0;

    function connect() {
        ws.value = new WebSocket(url);

        ws.value.onopen = () => {
            status.value = "connected";
            reconnectAttempts = 0;
        };

        ws.value.onmessage = (event) => {
            messages.value.push(event.data);
        };

        ws.value.onclose = () => {
            status.value = "disconnected";

            const delay = Math.min(1000 * Math.pow(2, reconnectAttempts), 30000);
            reconnectAttempts++;

            reconnectTimer = setTimeout(() => {
                status.value = "reconnecting";
                connect();
            }, delay);
        };

        ws.value.onerror = () => {
            status.value = "error";
        };
    }

    connect();

    function send(data) {
        if (ws.value && ws.value.readyState === WebSocket.OPEN) {
            ws.value.send(data);
        }
    }

    onUnmounted(() => {
        clearTimeout(reconnectTimer);
        if (ws.value) {
            ws.value.close();
        }
    });

    return { messages, status, send };
}

    五、🎯 面试考点：心跳保活¶

    面试官：客户端如何实现心跳保活？

    5.1 客户端心跳实现¶

    class HeartbeatWebSocket {
    constructor(url, options = {}) {
        this.url = url;
        this.heartbeatInterval = options.heartbeatInterval || 30000;
        this.heartbeatTimeout = options.heartbeatTimeout || 10000;
        this.ws = null;
        this.heartbeatTimer = null;
        this.heartbeatTimeoutTimer = null;

        this.connect();
    }

    connect() {
        this.ws = new WebSocket(this.url);

        this.ws.onopen = () => {
            console.log("连接成功");
            this.startHeartbeat();
        };

        this.ws.onmessage = (event) => {
            const data = JSON.parse(event.data);

            if (data.type === "pong") {
                // 收到 Pong，清除超时定时器
                clearTimeout(this.heartbeatTimeoutTimer);
                return;
            }

            // 处理业务消息
            console.log("收到:", data);
        };

        this.ws.onclose = () => {
            this.stopHeartbeat();
            setTimeout(() => this.connect(), 3000);
        };
    }

    startHeartbeat() {
        this.heartbeatTimer = setInterval(() => {
            if (this.ws.readyState === WebSocket.OPEN) {
                this.ws.send(JSON.stringify({ type: "ping" }));

                // 设置超时检测
                this.heartbeatTimeoutTimer = setTimeout(() => {
                    console.warn("心跳超时，关闭连接");
                    this.ws.close();
                }, this.heartbeatTimeout);
            }
        }, this.heartbeatInterval);
    }

    stopHeartbeat() {
        clearInterval(this.heartbeatTimer);
        clearTimeout(this.heartbeatTimeoutTimer);
    }

    send(data) {
        if (this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(data);
        }
    }
}

// 使用
const ws = new HeartbeatWebSocket("ws://localhost:8080", {
    heartbeatInterval: 30000,  // 每 30 秒发一次心跳
    heartbeatTimeout: 10000,   // 10 秒没收到 Pong 就断开
});

    六、🎯 面试考点：消息确认机制¶

    面试官：如何保证 WebSocket 消息不丢失？

    WebSocket 本身不保证消息送达，需要应用层实现确认机制：

    6.1 ACK 确认¶

    class ReliableWebSocket {
    constructor(url) {
        this.url = url;
        this.ws = null;
        this.pendingMessages = new Map();  // 待确认消息
        this.messageId = 0;

        this.connect();
    }

    connect() {
        this.ws = new WebSocket(this.url);

        this.ws.onopen = () => {
            // 重连后重发未确认的消息
            for (const [id, msg] of this.pendingMessages) {
                this.ws.send(JSON.stringify(msg));
            }
        };

        this.ws.onmessage = (event) => {
            const data = JSON.parse(event.data);

            if (data.type === "ack") {
                // 收到确认，移除待确认消息
                this.pendingMessages.delete(data.messageId);
                console.log(`消息 ${data.messageId} 已确认`);
            } else {
                console.log("收到:", data);
            }
        };

        this.ws.onclose = () => {
            setTimeout(() => this.connect(), 3000);
        };
    }

    send(data) {
        const id = ++this.messageId;
        const message = { id, data, timestamp: Date.now() };

        // 加入待确认队列
        this.pendingMessages.set(id, message);

        // 发送消息
        this.ws.send(JSON.stringify(message));

        // 超时重发
        setTimeout(() => {
            if (this.pendingMessages.has(id)) {
                console.log(`消息 ${id} 超时未确认，重发`);
                this.ws.send(JSON.stringify(message));
            }
        }, 5000);
    }
}

    八、小结¶

    要点 内容
    基础 API onopen、onmessage、onclose、onerror
    readyState CONNECTING(0)、OPEN(1)、CLOSING(2)、CLOSED(3)
    自动重连 指数退避（1s → 2s → 4s → ... → 30s）
    消息队列 断线时缓存消息，重连后自动发送
    心跳保活 定期发 Ping，超时未收到 Pong 则断开
    消息确认 ACK 机制保证消息不丢失
    调试工具 Network 面板、wscat、Wireshark

    🎯 面试必备： - 自动重连实现（指数退避） - 消息队列（断线缓存） - 心跳保活（Ping/Pong + 超时检测） - 消息确认（ACK 机制） - 背压控制（bufferedAmount）
