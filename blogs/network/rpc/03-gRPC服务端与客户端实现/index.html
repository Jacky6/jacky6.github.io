    RPC 教程（三）：gRPC 服务端与客户端实现¶

    参考： gRPC 官方文档、gRPC Python 指南

    一、gRPC 简介¶

    gRPC（Google Remote Procedure Call） 是 Google 开源的高性能、通用 RPC 框架。

    1.1 核心特性¶

    核心特性：
- 基于 HTTP/2（多路复用、头部压缩、流式传输）
- 使用 Protocol Buffers 序列化（高效、跨语言）
- 支持四种调用模式（Unary / Server Stream / Client Stream / Bidirectional）
- 自动代码生成（通过 .proto 文件生成客户端/服务端代码）
- 支持拦截器（类似中间件，用于日志、认证、限流等）
- 支持 TLS 加密
- 支持负载均衡和重试

    1.2 🎯 面试考点：gRPC vs 传统 REST¶

    面试官：gRPC 和 REST 有什么区别？

    对比维度 gRPC REST
    协议 HTTP/2 HTTP/1.1
    数据格式 Protobuf（二进制） JSON（文本）
    性能 高（二进制序列化） 中（JSON 解析开销）
    流式支持 ✅ 四种模式 ❌ 不支持
    代码生成 ✅ 自动生成 ❌ 手动编写
    浏览器支持 ⚠️ 需要 gRPC-Web ✅ 原生支持
    可读性 ❌ 差（二进制） ✅ 好（JSON）
    适用场景 微服务内部调用 对外 API

    二、gRPC 的四种调用模式¶

    2.1 🎯 面试考点：Unary RPC（一元调用）¶

    面试官：gRPC 的 Unary RPC 是什么？

    Unary RPC 是最简单的调用模式：客户端发送一个请求，服务器返回一个响应。

    // user_service.proto
syntax = "proto3";

package example;

message GetUserRequest {
  int32 user_id = 1;
}

message GetUserResponse {
  int32 id = 1;
  string name = 2;
  string email = 3;
}

service UserService {
  // 一元调用
  rpc GetUser (GetUserRequest) returns (GetUserResponse);
}

    Python 服务端实现：

    # server.py
from concurrent import futures
import grpc
import user_service_pb2 as pb2
import user_service_pb2_grpc as pb2_grpc


class UserServiceServicer(pb2_grpc.UserServiceServicer):
    """实现 UserService 服务"""

    def GetUser(self, request, context):
        """处理一元调用请求"""
        # request 是 GetUserRequest 对象
        user_id = request.user_id

        # 模拟数据库查询
        if user_id == 1:
            return pb2.GetUserResponse(
                id=1,
                name="Alice",
                email="alice@example.com"
            )
        else:
            # 返回错误
            context.set_code(grpc.StatusCode.NOT_FOUND)
            context.set_details(f"User {user_id} not found")
            return pb2.GetUserResponse()


def serve():
    """启动 gRPC 服务器"""
    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))
    pb2_grpc.add_UserServiceServicer_to_server(UserServiceServicer(), server)
    server.add_insecure_port("[::]:50051")
    server.start()
    print("Server started on port 50051")
    server.wait_for_termination()


if __name__ == "__main__":
    serve()

    Python 客户端实现：

    # client.py
import grpc
import user_service_pb2 as pb2
import user_service_pb2_grpc as pb2_grpc


def run():
    """调用 gRPC 服务"""
    # 建立连接
    with grpc.insecure_channel("localhost:50051") as channel:
        # 创建 Stub（客户端代理）
        stub = pb2_grpc.UserServiceStub(channel)

        # 调用服务
        request = pb2.GetUserRequest(user_id=1)
        try:
            response = stub.GetUser(request)
            print(f"User: id={response.id}, name={response.name}, email={response.email}")
        except grpc.RpcError as e:
            print(f"Error: {e.code()} - {e.details()}")


if __name__ == "__main__":
    run()

    2.2 🎯 面试考点：Server Streaming RPC（服务端流式）¶

    面试官：gRPC 的 Server Streaming RPC 是什么？

    Server Streaming RPC：客户端发送一个请求，服务器返回一个流（多个响应）。

    service UserService {
  // 服务端流式调用
  rpc ListUsers (ListUsersRequest) returns (stream UserResponse);
}

message ListUsersRequest {
  int32 page_size = 1;
}

message UserResponse {
  int32 id = 1;
  string name = 2;
}

    Python 服务端实现：

    class UserServiceServicer(pb2_grpc.UserServiceServicer):

    def ListUsers(self, request, context):
        """处理服务端流式请求"""
        # 模拟数据库查询
        users = [
            (1, "Alice"),
            (2, "Bob"),
            (3, "Charlie"),
        ]

        # 逐个返回用户
        for user_id, name in users:
            yield pb2.UserResponse(id=user_id, name=name)

    Python 客户端实现：

    def list_users():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.UserServiceStub(channel)
        request = pb2.ListUsersRequest(page_size=10)

        # 接收流式响应
        for response in stub.ListUsers(request):
            print(f"User: id={response.id}, name={response.name}")

    2.3 🎯 面试考点：Client Streaming RPC（客户端流式）¶

    面试官：gRPC 的 Client Streaming RPC 是什么？

    Client Streaming RPC：客户端发送一个流（多个请求），服务器返回一个响应。

    service UserService {
  // 客户端流式调用
  rpc CreateUsers (stream CreateUserRequest) returns (CreateUsersResponse);
}

message CreateUserRequest {
  string name = 1;
  string email = 2;
}

message CreateUsersResponse {
  int32 count = 1;
  repeated int32 user_ids = 2;
}

    Python 服务端实现：

    class UserServiceServicer(pb2_grpc.UserServiceServicer):

    def CreateUsers(self, request_iterator, context):
        """处理客户端流式请求"""
        user_ids = []

        # 逐个接收客户端请求
        for request in request_iterator:
            print(f"Creating user: {request.name}")
            user_id = len(user_ids) + 1
            user_ids.append(user_id)

        # 返回汇总响应
        return pb2.CreateUsersResponse(
            count=len(user_ids),
            user_ids=user_ids
        )

    Python 客户端实现：

    def create_users():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.UserServiceStub(channel)

        # 生成请求流
        def generate_requests():
            for name, email in [("Alice", "alice@example.com"), 
                                ("Bob", "bob@example.com")]:
                yield pb2.CreateUserRequest(name=name, email=email)

        # 发送流式请求
        response = stub.CreateUsers(generate_requests())
        print(f"Created {response.count} users: {response.user_ids}")

    2.4 🎯 面试考点：Bidirectional Streaming RPC（双向流式）¶

    面试官：gRPC 的 Bidirectional Streaming RPC 是什么？

    Bidirectional Streaming RPC：客户端和服务器都可以发送流（双向通信）。

    service ChatService {
  // 双向流式调用
  rpc Chat (stream ChatMessage) returns (stream ChatMessage);
}

message ChatMessage {
  string sender = 1;
  string text = 2;
  int64 timestamp = 3;
}

    Python 服务端实现：

    class ChatServiceServicer(pb2_grpc.ChatServiceServicer):

    def Chat(self, request_iterator, context):
        """处理双向流式请求"""
        # 逐个接收客户端消息
        for request in request_iterator:
            print(f"Received: {request.sender} - {request.text}")

            # 回复消息
            yield pb2.ChatMessage(
                sender="Server",
                text=f"Echo: {request.text}",
                timestamp=int(time.time())
            )

    Python 客户端实现：

    import threading
import time

def chat():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.ChatServiceStub(channel)

        # 生成请求流
        def generate_messages():
            for i in range(5):
                yield pb2.ChatMessage(
                    sender="Client",
                    text=f"Message {i}",
                    timestamp=int(time.time())
                )
                time.sleep(1)

        # 接收响应流
        def receive_responses():
            for response in stub.Chat(generate_messages()):
                print(f"Received: {response.sender} - {response.text}")

        # 启动接收线程
        thread = threading.Thread(target=receive_responses)
        thread.start()
        thread.join()

    三、🎯 面试考点：gRPC 拦截器¶

    面试官：gRPC 的拦截器是什么？有什么作用？

    拦截器（Interceptor） 类似于 Web 框架的中间件，可以在请求/响应处理前后执行自定义逻辑。

    常见用途： - 日志记录 - 认证与鉴权 - 限流与熔断 - 监控与追踪 - 错误处理

    3.1 服务端拦截器¶

    # interceptor.py
import grpc
import time


class LoggingInterceptor(grpc.ServerInterceptor):
    """日志拦截器"""

    def intercept_service(self, continuation, handler_call_details):
        """拦截服务调用"""
        method = handler_call_details.method
        print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] Received request: {method}")

        # 继续处理请求
        return continuation(handler_call_details)


class AuthInterceptor(grpc.ServerInterceptor):
    """认证拦截器"""

    def __init__(self, allowed_tokens):
        self.allowed_tokens = allowed_tokens

    def intercept_service(self, continuation, handler_call_details):
        """拦截服务调用并验证 Token"""
        # 获取元数据（类似 HTTP Header）
        metadata = dict(handler_call_details.invocation_metadata)
        token = metadata.get("authorization", "")

        # 验证 Token
        if token not in self.allowed_tokens:
            # 返回未授权错误
            return grpc.unary_unary_rpc_method_handler(
                lambda request, context: context.abort(
                    grpc.StatusCode.UNAUTHENTICATED,
                    "Invalid token"
                )
            )

        # Token 有效，继续处理
        return continuation(handler_call_details)


# 使用拦截器
def serve():
    server = grpc.server(
        futures.ThreadPoolExecutor(max_workers=10),
        interceptors=[LoggingInterceptor(), AuthInterceptor(["token123"])]
    )
    pb2_grpc.add_UserServiceServicer_to_server(UserServiceServicer(), server)
    server.add_insecure_port("[::]:50051")
    server.start()
    server.wait_for_termination()

    3.2 客户端拦截器¶

    class ClientLoggingInterceptor(grpc.UnaryUnaryClientInterceptor):
    """客户端日志拦截器"""

    def intercept_unary_unary(self, continuation, client_call_details, request):
        """拦截客户端调用"""
        method = client_call_details.method
        print(f"[Client] Calling: {method}")

        start_time = time.time()
        response = continuation(client_call_details, request)
        duration = time.time() - start_time

        print(f"[Client] Received response in {duration:.3f}s")
        return response


# 使用拦截器
def run():
    channel = grpc.insecure_channel("localhost:50051")
    intercepted_channel = grpc.intercept_channel(
        channel,
        ClientLoggingInterceptor()
    )
    stub = pb2_grpc.UserServiceStub(intercepted_channel)
    response = stub.GetUser(pb2.GetUserRequest(user_id=1))
    print(response)

    四、🎯 面试考点：gRPC 元数据（Metadata）¶

    面试官：gRPC 如何传递额外的信息（如 Token、Trace ID）？

    元数据（Metadata） 类似于 HTTP Header，用于传递额外的键值对信息。

    4.1 客户端发送元数据¶

    def run():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.UserServiceStub(channel)

        # 创建元数据
        metadata = [
            ("authorization", "Bearer token123"),
            ("x-request-id", "req-12345"),
            ("x-trace-id", "trace-67890"),
        ]

        # 发送请求并附带元数据
        request = pb2.GetUserRequest(user_id=1)
        response = stub.GetUser(request, metadata=metadata)
        print(response)

    4.2 服务端接收元数据¶

    class UserServiceServicer(pb2_grpc.UserServiceServicer):

    def GetUser(self, request, context):
        """处理请求并读取元数据"""
        # 获取元数据
        metadata = dict(context.invocation_metadata())
        token = metadata.get("authorization", "")
        request_id = metadata.get("x-request-id", "")

        print(f"Token: {token}, Request ID: {request_id}")

        # 处理业务逻辑
        return pb2.GetUserResponse(id=1, name="Alice", email="alice@example.com")

    4.3 服务端发送元数据¶

    class UserServiceServicer(pb2_grpc.UserServiceServicer):

    def GetUser(self, request, context):
        """处理请求并发送元数据"""
        # 发送元数据（类似 HTTP Response Header）
        context.send_initial_metadata([
            ("x-response-time", "10ms"),
            ("x-server-version", "1.0.0"),
        ])

        # 发送尾元数据（在响应结束后发送）
        context.set_trailing_metadata([
            ("x-request-cost", "100"),
        ])

        return pb2.GetUserResponse(id=1, name="Alice", email="alice@example.com")

    4.4 客户端接收元数据¶

    def run():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.UserServiceStub(channel)
        request = pb2.GetUserRequest(user_id=1)

        # 调用服务并获取元数据
        response, call = stub.GetUser.with_call(request)

        # 读取初始元数据
        initial_metadata = dict(call.initial_metadata())
        print(f"Response Time: {initial_metadata.get('x-response-time')}")

        # 读取尾元数据
        trailing_metadata = dict(call.trailing_metadata())
        print(f"Request Cost: {trailing_metadata.get('x-request-cost')}")

    五、🎯 面试考点：gRPC 错误处理¶

    面试官：gRPC 如何处理错误？

    5.1 错误码¶

    gRPC 定义了标准的错误码（类似 HTTP 状态码）：

    错误码 名称 说明
    0 OK 成功
    1 CANCELLED 请求被取消
    2 UNKNOWN 未知错误
    3 INVALID_ARGUMENT 参数无效
    4 DEADLINE_EXCEEDED 超时
    5 NOT_FOUND 资源未找到
    6 ALREADY_EXISTS 资源已存在
    7 PERMISSION_DENIED 权限不足
    8 RESOURCE_EXHAUSTED 资源耗尽（如限流）
    9 FAILED_PRECONDITION 前置条件不满足
    10 ABORTED 操作被中止
    11 OUT_OF_RANGE 超出范围
    12 UNIMPLEMENTED 未实现
    13 INTERNAL 内部错误
    14 UNAVAILABLE 服务不可用
    15 DATA_LOSS 数据丢失
    16 UNAUTHENTICATED 未认证

    5.2 服务端返回错误¶

    class UserServiceServicer(pb2_grpc.UserServiceServicer):

    def GetUser(self, request, context):
        user_id = request.user_id

        # 参数校验
        if user_id <= 0:
            context.abort(grpc.StatusCode.INVALID_ARGUMENT, "Invalid user ID")

        # 查询用户
        user = db.get_user(user_id)
        if not user:
            context.abort(grpc.StatusCode.NOT_FOUND, f"User {user_id} not found")

        return pb2.GetUserResponse(
            id=user.id,
            name=user.name,
            email=user.email
        )

    5.3 客户端处理错误¶

    def run():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.UserServiceStub(channel)
        request = pb2.GetUserRequest(user_id=-1)

        try:
            response = stub.GetUser(request)
            print(response)
        except grpc.RpcError as e:
            # 获取错误码和详情
            print(f"Error: {e.code()} - {e.details()}")

            # 根据错误码处理
            if e.code() == grpc.StatusCode.NOT_FOUND:
                print("User not found")
            elif e.code() == grpc.StatusCode.INVALID_ARGUMENT:
                print("Invalid argument")
            elif e.code() == grpc.StatusCode.UNAVAILABLE:
                print("Service unavailable, retry later")

    六、🎯 面试考点：gRPC 超时与重试¶

    面试官：gRPC 如何处理超时和重试？

    6.1 设置超时¶

    import grpc
import time

def run():
    with grpc.insecure_channel("localhost:50051") as channel:
        stub = pb2_grpc.UserServiceStub(channel)
        request = pb2.GetUserRequest(user_id=1)

        # 设置超时时间（5 秒）
        try:
            response = stub.GetUser(request, timeout=5)
            print(response)
        except grpc.RpcError as e:
            if e.code() == grpc.StatusCode.DEADLINE_EXCEEDED:
                print("Request timed out")

    6.2 配置重试策略¶

    def run():
    # 配置重试策略
    service_config = {
        "methodConfig": [{
            "name": [{}],  # 应用到所有方法
            "retryPolicy": {
                "maxAttempts": 3,  # 最大重试次数
                "initialBackoff": "0.1s",  # 初始退避时间
                "maxBackoff": "1s",  # 最大退避时间
                "backoffMultiplier": 2,  # 退避倍数
                "retryableStatusCodes": [
                    "UNAVAILABLE",  # 服务不可用时重试
                    "DEADLINE_EXCEEDED",  # 超时时重试
                ],
            },
        }]
    }

    # 创建带重试策略的 Channel
    options = [("grpc.service_config", json.dumps(service_config))]
    channel = grpc.insecure_channel("localhost:50051", options=options)

    stub = pb2_grpc.UserServiceStub(channel)
    response = stub.GetUser(pb2.GetUserRequest(user_id=1))
    print(response)

    七、🎯 面试考点：gRPC TLS 加密¶

    面试官：gRPC 如何启用 TLS 加密？

    7.1 生成证书¶

    # 生成私钥
openssl genrsa -out server.key 2048

# 生成证书签名请求（CSR）
openssl req -new -key server.key -out server.csr

# 生成自签名证书
openssl x509 -req -days 365 -in server.csr -signkey server.key -out server.crt

    7.2 服务端启用 TLS¶

    import grpc

def serve():
    # 读取证书和私钥
    with open("server.crt", "rb") as f:
        server_cert = f.read()
    with open("server.key", "rb") as f:
        server_key = f.read()

    # 创建 SSL 凭据
    server_credentials = grpc.ssl_server_credentials(
        [(server_key, server_cert)]
    )

    # 启动安全服务器
    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))
    pb2_grpc.add_UserServiceServicer_to_server(UserServiceServicer(), server)
    server.add_secure_port("[::]:50051", server_credentials)
    server.start()
    server.wait_for_termination()

    7.3 客户端启用 TLS¶

    import grpc

def run():
    # 读取服务器证书
    with open("server.crt", "rb") as f:
        server_cert = f.read()

    # 创建 SSL 凭据
    credentials = grpc.ssl_channel_credentials(server_cert)

    # 创建安全 Channel
    channel = grpc.secure_channel("localhost:50051", credentials)

    stub = pb2_grpc.UserServiceStub(channel)
    response = stub.GetUser(pb2.GetUserRequest(user_id=1))
    print(response)

    八、🎯 面试考点：gRPC 健康检查¶

    面试官：gRPC 如何进行健康检查？

    gRPC 提供了标准的健康检查协议（Health Checking Protocol）。

    8.1 服务端实现健康检查¶

    from grpc_health.v1 import health_pb2 as health_pb2
from grpc_health.v1 import health_pb2_grpc as health_pb2_grpc


class HealthServicer(health_pb2_grpc.HealthServicer):
    """实现健康检查服务"""

    def __init__(self):
        self.status = {}

    def Check(self, request, context):
        """检查服务健康状态"""
        service_name = request.service

        # 返回服务状态
        status = self.status.get(service_name, health_pb2.HealthCheckResponse.SERVING)
        return health_pb2.HealthCheckResponse(status=status)

    def Watch(self, request, context):
        """监控服务健康状态（流式）"""
        service_name = request.service

        # 定期发送状态更新
        while True:
            status = self.status.get(service_name, health_pb2.HealthCheckResponse.SERVING)
            yield health_pb2.HealthCheckResponse(status=status)
            time.sleep(5)


def serve():
    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))

    # 注册业务服务
    pb2_grpc.add_UserServiceServicer_to_server(UserServiceServicer(), server)

    # 注册健康检查服务
    health_servicer = HealthServicer()
    health_pb2_grpc.add_HealthServicer_to_server(health_servicer, server)

    server.add_insecure_port("[::]:50051")
    server.start()
    server.wait_for_termination()

    from grpc_health.v1 import health_pb2 as health_pb2

def check_health():
    with grpc.insecure_channel("localhost:50051") as channel:
        health_stub = health_pb2_grpc.HealthStub(channel)

        # 检查服务健康状态
        request = health_pb2.HealthCheckRequest(service="UserService")
        try:
            response = health_stub.Check(request)
            if response.status == health_pb2.HealthCheckResponse.SERVING:
                print("Service is healthy")
            else:
                print("Service is not healthy")
        except grpc.RpcError as e:
            print(f"Health check failed: {e.code()}")

    九、小结¶

    要点 内容
    gRPC 是什么 基于 HTTP/2 和 Protobuf 的高性能 RPC 框架
    四种调用模式 Unary、Server Stream、Client Stream、Bidirectional
    拦截器 类似中间件，用于日志、认证、限流等
    元数据 类似 HTTP Header，传递额外信息（Token、Trace ID）
    错误处理 标准错误码（OK、NOT_FOUND、UNAVAILABLE 等）
    超时与重试 设置超时、配置重试策略（指数退避）
    TLS 加密 使用 SSL/TLS 加密通信
    健康检查 标准健康检查协议（Check、Watch）

    🎯 面试必备： - 四种调用模式的区别与适用场景 - 拦截器的作用与实现 - 元数据的传递方式 - 错误码与错误处理 - 超时与重试策略 - TLS 加密配置 - 健康检查协议
