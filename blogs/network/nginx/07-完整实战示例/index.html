    Nginx 教程（七）：完整实战示例¶

    参考： nginx.org/en/docs/

    一、个人博客/静态网站¶

    场景¶

    部署一个静态博客（如 Hugo、Hexo 生成的站点）。

    配置¶

    server {
    listen 80;
    server_name blog.example.com;

    # 网站根目录
    root /var/www/blog;
    index index.html;

    # 静态资源缓存
    location ~* \.(css|js|png|jpg|gif|ico|svg|woff|woff2)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # 自定义 404
    error_page 404 /404.html;

    # 访问日志
    access_log /var/log/nginx/blog.access.log;
    error_log  /var/log/nginx/blog.error.log;
}

    二、前后端分离项目¶

    场景¶

    • 前端：Vue/React 打包后的静态文件
    • 后端：Spring Boot / Flask / Node.js 运行在 localhost:8080

    配置¶

    server {
    listen 80;
    server_name app.example.com;

    # 前端静态文件
    location / {
        root /var/www/frontend;
        index index.html;
        try_files $uri $uri/ /index.html;  # SPA 路由回退
    }

    # 后端 API 代理
    location /api/ {
        proxy_pass http://localhost:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # WebSocket 支持
    location /ws/ {
        proxy_pass http://localhost:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
    }
}

    SPA 路由关键： try_files $uri $uri/ /index.html; 让所有找不到文件的路由都返回 index.html，交给前端路由处理。

    三、WordPress 站点¶

    场景¶

    部署 WordPress（PHP + MySQL）。

    配置¶

    server {
    listen 80;
    server_name www.example.com;
    root /var/www/wordpress;
    index index.php index.html;

    # 上传文件大小限制
    client_max_body_size 50m;

    # 主 location
    location / {
        try_files $uri $uri/ /index.php?$args;
    }

    # PHP 处理
    location ~ \.php$ {
        fastcgi_pass unix:/var/run/php/php-fpm.sock;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;
    }

    # 禁止访问敏感文件
    location ~* \.(htaccess|htpasswd|ini|log|sh|sql)$ {
        deny all;
    }

    # 静态资源缓存
    location ~* \.(css|js|png|jpg|gif|ico|svg)$ {
        expires 30d;
        add_header Cache-Control "public";
    }
}

    四、多后端负载均衡¶

    场景¶

    多台应用服务器，需要负载均衡和会话保持。

    配置¶

    http {
    upstream myapp {
        # 最少连接算法
        least_conn;

        # 后端服务器
        server 192.168.1.10:8080 weight=3 max_fails=3 fail_timeout=30s;
        server 192.168.1.11:8080 weight=2 max_fails=3 fail_timeout=30s;
        server 192.168.1.12:8080 max_fails=3 fail_timeout=30s;
        server 192.168.1.13:8080 backup;   # 备用
    }

    server {
        listen 80;
        server_name api.example.com;

        location / {
            proxy_pass http://myapp;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

            # 超时
            proxy_connect_timeout 10s;
            proxy_read_timeout 30s;
            proxy_send_timeout 30s;

            # 如果后端返回错误，尝试下一台
            proxy_next_upstream error timeout http_500 http_502 http_503 http_504;
        }
    }
}

    五、HTTPS 全站示例（生产级）¶

    场景¶

    生产环境，启用 HTTPS + HTTP 跳转 + 安全优化。

    配置¶

    worker_processes auto;

events {
    worker_connections 4096;
    multi_accept on;
}

http {
    include       mime.types;
    default_type  application/octet-stream;

    # 基础优化
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;

    # Gzip
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;

    # SSL 优化
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # 日志
    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent"';

    # HTTP → HTTPS 重定向
    server {
        listen 80;
        server_name www.example.com;
        return 301 https://$host$request_uri;
    }

    # HTTPS 主站点
    server {
        listen 443 ssl;
        server_name www.example.com;

        ssl_certificate     /etc/letsencrypt/live/www.example.com/fullchain.pem;
        ssl_certificate_key /etc/letsencrypt/live/www.example.com/privkey.pem;

        # HSTS
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

        # 网站根目录
        root /var/www/html;
        index index.html;

        # 静态资源
        location ~* \.(css|js|png|jpg|gif|ico)$ {
            expires 30d;
            add_header Cache-Control "public, immutable";
        }

        # API 代理
        location /api/ {
            proxy_pass http://localhost:8080;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # 日志
        access_log /var/log/nginx/www.access.log main;
        error_log  /var/log/nginx/www.error.log warn;
    }
}

    六、反向代理 Docker 容器¶

    场景¶

    多个 Docker 容器通过 Nginx 反向代理。

    配置¶

    http {
    upstream frontend {
        server 127.0.0.1:3000;  # React 容器
    }

    upstream backend {
        server 127.0.0.1:8000;  # Flask 容器
    }

    server {
        listen 80;
        server_name myapp.example.com;

        location / {
            proxy_pass http://frontend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
        }

        location /api/ {
            proxy_pass http://backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
        }
    }
}

    七、快速上手模板¶

    最小可用配置（复制即用）¶

    worker_processes auto;

events {
    worker_connections 1024;
}

http {
    include       mime.types;
    default_type  application/octet-stream;
    sendfile      on;
    keepalive_timeout 65;

    server {
        listen      80;
        server_name localhost;

        location / {
            root   /usr/share/nginx/html;
            index  index.html index.htm;
        }

        error_page 500 502 503 504 /50x.html;
        location = /50x.html {
            root /usr/share/nginx/html;
        }
    }
}
