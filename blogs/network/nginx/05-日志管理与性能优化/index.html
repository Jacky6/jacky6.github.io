    Nginx 教程（五）：日志管理与性能优化¶

    参考： nginx.org/en/docs/

    一、访问日志（Access Log）¶

    基本配置¶

    http {
    access_log /var/log/nginx/access.log;
}

    自定义日志格式¶

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                '$status $body_bytes_sent "$http_referer" '
                '"$http_user_agent" "$http_x_forwarded_for"';

access_log /var/log/nginx/access.log main;

    常用日志变量¶

    变量 含义
    $remote_addr 客户端 IP
    $remote_user 认证用户名
    $time_local 本地时间
    $request 请求行（GET /index.html HTTP/1.1）
    $status 响应状态码
    $body_bytes_sent 响应体大小（字节）
    $http_referer 来源页面
    $http_user_agent 客户端浏览器信息
    $request_time 请求处理时间（秒）
    $upstream_response_time 后端响应时间

    JSON 格式日志（方便 ELK 分析）¶

    log_format json escape=json '{'
    '"remote_addr":"$remote_addr",'
    '"time_local":"$time_local",'
    '"request":"$request",'
    '"status":$status,'
    '"body_bytes_sent":$body_bytes_sent,'
    '"http_referer":"$http_referer",'
    '"http_user_agent":"$http_user_agent",'
    '"request_time":$request_time,'
    '"upstream_response_time":"$upstream_response_time"'
'}';

access_log /var/log/nginx/access.json.log json;

    按域名分开日志¶

    server {
    server_name example.com;
    access_log /var/log/nginx/example.com.access.log;
    error_log  /var/log/nginx/example.com.error.log;
}

    条件日志（只记特定请求）¶

    # 只记录慢请求（> 1 秒）
map $request_time $log_slow {
    default    0;
    ~^[1-9]   1;  # >= 1 秒
}

access_log /var/log/nginx/slow_requests.log combined if=$log_slow;

    二、错误日志（Error Log）¶

    error_log /var/log/nginx/error.log;

    日志级别¶

    级别 说明 生产环境建议
    debug 调试信息 ❌ 不用
    info 一般信息 ❌ 不用
    notice 重要通知 ⚠️ 偶尔用
    warn 警告 ✅ 推荐
    error 错误（默认） ✅ 推荐
    crit 严重错误 ⚠️ 太粗
    alert 需要立即处理 ❌ 太少
    emerg 紧急 ❌ 太少 
    error_log /var/log/nginx/error.log warn;

    三、日志轮转（Log Rotation）¶

    使用 logrotate 防止日志文件无限增长：

    # /etc/logrotate.d/nginx
/var/log/nginx/*.log {
    daily
    missingok
    rotate 14
    compress
    delaycompress
    notifempty
    create 0640 nginx adm
    sharedscripts
    postrotate
        [ -f /var/run/nginx.pid ] && kill -USR1 $(cat /var/run/nginx.pid)
    endscript
}

    kill -USR1 通知 Nginx 重新打开日志文件，等同于 nginx -s reopen。

    四、性能优化¶

    4.1 Worker 进程优化¶

    worker_processes auto;              # 自动匹配 CPU 核心数
worker_connections 4096;            # 每个 Worker 最大连接数

events {
    use epoll;                      # Linux 推荐
    multi_accept on;                # 一次性接受多个连接
}

    4.2 sendfile 零拷贝¶

    http {
    sendfile on;       # 直接在内核空间传输文件数据，不经过用户空间
    tcp_nopush on;     # 和 sendfile 配合，减少网络报文数量
    tcp_nodelay on;    # 禁用 Nagle 算法，小包立即发送
}
    指令 作用
    sendfile on 零拷贝，高效发送静态文件
    tcp_nopush on 配合 sendfile，填满包再发
    tcp_nodelay on 禁用 Nagle 算法，降低延迟

    4.3 文件描述符¶

    worker_rlimit_nofile 65535;

    设置 Worker 进程可打开的最大文件数。

    4.4 Keepalive 优化¶

    http {
    keepalive_timeout 65s;          # 连接保持时间
    keepalive_requests 1000;        # 每个连接最大请求数
    keepalive_time 1h;              # 连接总时长上限
}

    4.5 缓冲区优化¶

    http {
    # 请求头缓冲
    client_header_buffer_size 1k;
    large_client_header_buffers 4 8k;

    # 请求体缓冲
    client_body_buffer_size 16k;
    client_max_body_size 10m;

    # 代理缓冲
    proxy_buffer_size 4k;
    proxy_buffers 8 16k;
    proxy_busy_buffers_size 32k;
}

    五、静态资源缓存¶

    expires 头¶

    location ~* \.(jpg|jpeg|png|gif|ico|css|js)$ {
    expires 30d;                    # 浏览器缓存 30 天
    add_header Cache-Control "public, immutable";
}

    禁用缓存（开发环境）¶

    location / {
    add_header Cache-Control "no-cache, no-store, must-revalidate";
    add_header Pragma "no-cache";
    add_header Expires 0;
}

    六、Gzip 压缩¶

    http {
    gzip on;
    gzip_vary on;                   # 添加 Vary: Accept-Encoding
    gzip_proxied any;               # 对所有代理请求压缩
    gzip_comp_level 6;              # 压缩级别 1-9
    gzip_min_length 256;            # 小于 256 字节不压缩
    gzip_types
        text/plain
        text/css
        text/xml
        text/javascript
        application/json
        application/javascript
        application/xml
        application/xml+rss
        application/atom+xml
        image/svg+xml;
}

    注意： 图片和 PDF 通常已压缩过，不需要 gzip。

    七、性能监控¶

    开启 stub_status 模块¶

    server {
    listen 80;
    server_name localhost;

    location /nginx_status {
        stub_status;
        allow 127.0.0.1;
        allow 192.168.1.0/24;
        deny all;
    }
}

    访问 http://localhost/nginx_status 得到：

    Active connections: 123
server accepts handled requests
 12345 12345 67890
Reading: 5 Writing: 10 Waiting: 108
    指标 说明
    Active connections 当前活跃连接数
    accepts 已接受的总连接数
    handled 已处理的总连接数
    requests 总请求数
    Reading 正在读取请求头的连接数
    Writing 正在发送响应的连接数
    Waiting 空闲的 keepalive 连接数

    八、优化检查清单¶

    项目 推荐值
    worker_processes auto
    worker_connections 4096+
    sendfile on
    tcp_nopush on
    tcp_nodelay on
    gzip on
    keepalive_timeout 65s
    log_format JSON（便于分析）
    ssl_session_cache shared:SSL:10m
