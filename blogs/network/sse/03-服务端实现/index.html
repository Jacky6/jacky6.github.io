    SSE 教程（三）：服务端实现¶

    多语言实现 SSE 服务端，涵盖 Python、Node.js、Go、Java

    一、Python 实现¶

    1.1 FastAPI（推荐）¶

    FastAPI 原生支持流式响应，代码简洁：

    from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
import asyncio

app = FastAPI()

async def event_generator():
    """生成 SSE 事件流"""
    for i in range(10):
        # 发送数据
        yield f"id: {i}\n"
        yield f"data: 消息 {i}\n\n"

        # 模拟处理延迟
        await asyncio.sleep(1)

@app.get("/events")
async def stream_events():
    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
        }
    )

    1.2 带事件类型的 FastAPI 实现¶

    async def typed_event_generator():
    """发送不同类型的事件"""
    events = [
        ("init", {"status": "connected"}),
        ("update", {"value": 42}),
        ("error", {"message": "something went wrong"}),
    ]

    for event_type, data in events:
        yield f"event: {event_type}\n"
        yield f"data: {json.dumps(data)}\n\n"
        await asyncio.sleep(1)

@app.get("/typed-events")
async def typed_events():
    return StreamingResponse(
        typed_event_generator(),
        media_type="text/event-stream"
    )

    1.3 Flask 实现¶

    from flask import Flask, Response
import time

def generate_events():

@app.route("/events")

    )

    1.4 断点续传实现¶

    from fastapi import Request

@app.get("/events/resume")
async def events_with_resume(request: Request):
    # 获取客户端上次收到的事件 ID
    last_id = request.headers.get("last-event-id")
    start_from = int(last_id) + 1 if last_id else 0

    async def generate():
        for i in range(start_from, 100):
            yield f"id: {i}\n"
            yield f"data: 消息 {i}\n\n"
            await asyncio.sleep(1)

    return StreamingResponse(
        generate(),
        media_type="text/event-stream"
    )

    二、Node.js 实现¶

    2.1 Express + SSE¶

    const express = require("express");
const app = express();

app.get("/events", (req, res) => {
    // 关键：设置正确的响应头
    res.setHeader("Content-Type", "text/event-stream");
    res.setHeader("Cache-Control", "no-cache");
    res.setHeader("Connection", "keep-alive");
    res.setHeader("X-Accel-Buffering", "no");

    // 发送心跳
    const heartbeat = setInterval(() => {
        res.write(": heartbeat\n\n");
    }, 15000);

    // 发送事件
    let id = 0;
    const interval = setInterval(() => {
        res.write(`id: ${id++}\n`);
        res.write(`data: 消息 ${id}\n\n`);
    }, 1000);

    // 客户端断开时清理
    req.on("close", () => {
        clearInterval(interval);
        clearInterval(heartbeat);
    });
});

app.listen(3000);

    2.2 使用 sse 库（更简洁）¶

    npm install sse

    const express = require("express");
const SSE = require("sse");
const app = express();

const server = app.listen(3000);
const sse = new SSE(server, { path: "/events" });

sse.on("connection", (client) => {
    let id = 0;

    // 发送初始化事件
    client.send({ event: "init", data: JSON.stringify({ status: "ok" }) });

    // 定时推送
    const interval = setInterval(() => {
        client.send({
            event: "update",
            id: id++,
            data: JSON.stringify({ value: Math.random() }),
        });
    }, 1000);

    client.on("close", () => {
        clearInterval(interval);
    });
});

    三、Go 实现¶

    3.1 标准库实现¶

    package main

import (
    "fmt"
    "net/http"
    "time"
)

func sseHandler(w http.ResponseWriter, r *http.Request) {
    // 设置响应头
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.Header().Set("X-Accel-Buffering", "no")

    // 获取客户端上下文
    ctx := r.Context()

    // 发送事件
    for i := 0; i < 10; i++ {
        select {
        case <-ctx.Done():
            return // 客户端断开
        default:
            fmt.Fprintf(w, "id: %d\n", i)
            fmt.Fprintf(w, "data: 消息 %d\n\n", i)
            w.(http.Flusher).Flush() // 关键：立即刷新
            time.Sleep(1 * time.Second)
        }
    }
}

func main() {
    http.HandleFunc("/events", sseHandler)
    http.ListenAndServe(":8080", nil)
}

    3.2 带断点续传的 Go 实现¶

    func sseResumeHandler(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")

    // 读取 Last-Event-ID
    lastID := r.Header.Get("Last-Event-ID")
    startFrom := 0
    if lastID != "" {
        fmt.Sscanf(lastID, "%d", &startFrom)
        startFrom++
    }

    for i := startFrom; i < 100; i++ {
        fmt.Fprintf(w, "id: %d\n", i)
        fmt.Fprintf(w, "data: 消息 %d\n\n", i)
        w.(http.Flusher).Flush()
        time.Sleep(1 * time.Second)
    }
}

    四、Java (Spring Boot) 实现¶

    4.1 SseEmitter¶

    @RestController
public class SSEController {

    @GetMapping(value = "/events", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
    public SseEmitter streamEvents() {
        SseEmitter emitter = new SseEmitter(0L); // 0 表示不超时

        ExecutorService executor = Executors.newSingleThreadExecutor();

        executor.execute(() -> {
            try {
                for (int i = 0; i < 10; i++) {
                    SseEmitter.SseEventBuilder event = SseEmitter.event()
                        .id(String.valueOf(i))
                        .name("update")
                        .data("消息 " + i);

                    emitter.send(event);
                    Thread.sleep(1000);
                }
                emitter.complete();
            } catch (Exception e) {
                emitter.completeWithError(e);
            }
        });

        // 超时和错误回调
        emitter.onTimeout(() -> System.out.println("超时"));
        emitter.onError((e) -> System.out.println("错误: " + e.getMessage()));
        emitter.onCompletion(() -> System.out.println("完成"));

        return emitter;
    }
}

    4.2 多客户端广播¶

    @RestController
public class BroadcastController {

    private final CopyOnWriteArrayList<SseEmitter> emitters = new CopyOnWriteArrayList<>();

    @GetMapping(value = "/subscribe", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
    public SseEmitter subscribe() {
        SseEmitter emitter = new SseEmitter(0L);
        emitters.add(emitter);

        emitter.onCompletion(() -> emitters.remove(emitter));
        emitter.onTimeout(() -> emitters.remove(emitter));

        return emitter;
    }

    // 调用此方法向所有客户端广播
    @PostMapping("/broadcast")
    public void broadcast(@RequestBody String message) {
        for (SseEmitter emitter : emitters) {
            try {
                emitter.send(SseEmitter.event()
                    .name("broadcast")
                    .data(message));
            } catch (IOException e) {
                emitter.completeWithError(e);
            }
        }
    }
}

    res.write(": heartbeat\n\n");

    六、性能优化建议¶

    6.1 连接数管理¶

    # 限制最大连接数
MAX_CONNECTIONS = 1000
current_connections = 0

@app.get("/events")
async def events():
    global current_connections
    if current_connections >= MAX_CONNECTIONS:
        raise HTTPException(status_code=503, detail="Too many connections")

    current_connections += 1
    try:
        return StreamingResponse(...)
    finally:
        current_connections -= 1

    6.2 事件合并¶

    如果事件产生太快，可以合并后批量发送：

    buffer = []

async def batch_sender():
    while True:
        await asyncio.sleep(0.1)  # 每 100ms 批量发送
        if buffer:
            batch = json.dumps(buffer)
            buffer.clear()
            yield f"data: {batch}\n\n"

    6.3 使用消息队列¶

    生产环境建议用 Redis/RabbitMQ 解耦：

    import redis

r = redis.Redis()

async def event_from_redis():
    pubsub = r.pubsub()
    pubsub.subscribe("events")

    for message in pubsub.listen():
        if message["type"] == "message":
            yield f"data: {message['data'].decode()}\n\n"

    七、小结¶

    语言 推荐方案 关键点
    Python FastAPI StreamingResponse 异步生成器 + text/event-stream
    Node.js Express + 手动设置响应头 res.write() + Flusher
    Go 标准库 http.ResponseWriter Flush() 必须调用
    Java Spring Boot SseEmitter 异步线程 + 回调管理

    通用要点：

    • 设置正确的 Content-Type: text/event-stream
    • 禁用缓冲（Nginx: proxy_buffering off）
    • 实现心跳保活
    • 处理客户端断开（清理资源）
    • 支持 Last-Event-ID 断点续传
