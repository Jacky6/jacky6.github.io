    暂存与清理¶

    暂时保存未完成的工作，保持工作区干净。

    1. Stash（暂存）¶

    当你需要切换分支但工作还没完成时，stash 可以帮你保存当前状态。

    # 暂存所有改动
git stash

# 暂存并添加说明
git stash save "WIP: login page"

# 查看暂存列表
git stash list
# stash@{0}: WIP: login page
# stash@{1}: WIP: fix header

# 恢复最近一次暂存（保留在列表中）
git stash apply

# 恢复指定暂存
git stash apply stash@{1}

# 恢复并删除（pop = apply + drop）
git stash pop

# 删除暂存
git stash drop stash@{1}

# 清空所有暂存
git stash clear

    高级用法¶

    # 只暂存未跟踪的文件
git stash -u

# 暂存包含未跟踪和已忽略的文件
git stash -a

# 从暂存创建新分支
git stash branch new-branch-name

# 查看暂存内容
git stash show
git stash show -p stash@{0}    # 显示具体 diff

# 暂存部分文件（交互式）
git stash -p

# 基于暂存创建提交
git stash push -m "commit message"

    Stash 工作流示例¶

    # 正在开发 feature，突然需要修 bug
git stash                    # 保存当前进度
git checkout -b hotfix main  # 创建修复分支
# ... 修复 bug ...
git add .
git commit -m "fix: critical bug"
git checkout main
git merge hotfix             # 合并修复
git checkout feature
git stash pop                # 恢复之前的工作，继续开发

    2. .gitignore¶

    告诉 Git 忽略哪些文件。

    # 项目根目录创建 .gitignore
touch .gitignore

    常用规则¶

    # 忽略所有 .log 文件
*.log

# 忽略 build 目录
build/

# 忽略根目录的 config.json（不忽略子目录的）
/config.json

# 不忽略特定文件（覆盖前面的规则）
!important.log

# 忽略所有 .tmp 但保留 /docs/ 下的
*.tmp
!/docs/*.tmp

# 忽略 macOS / Windows / Linux 系统文件
.DS_Store
Thumbs.db
*~

# 忽略 IDE 配置
.vscode/
.idea/
*.swp

# 忽略 Python
__pycache__/
*.pyc
.venv/

# 忽略 Node.js
node_modules/
dist/
.env

# 忽略编译产物
*.o
*.so
*.a
*.class

    全局忽略¶

    # 创建全局忽略文件
touch ~/.gitignore_global

# 配置 Git 使用它
git config --global core.excludesFile ~/.gitignore_global

    忽略已跟踪的文件¶

    .gitignore 对已跟踪的文件无效。如需停止跟踪但不删除：

    # 从缓存中移除（保留本地文件）
git rm --cached <file>

# 递归移除目录
git rm -r --cached <dir>

    3. Git 垃圾回收¶

    # 清理无效对象并优化仓库
git gc

# 强制垃圾回收
git gc --aggressive

# 检查仓库完整性
git fsck

# 查找大文件
git rev-list --objects --all \
  | git cat-file --batch-check='%(objecttype) %(objectname) %(objectsize) %(rest)' \
  | awk '/^blob/ {print substr($0,6)}' \
  | sort -rn -k2 \
  | head -20
