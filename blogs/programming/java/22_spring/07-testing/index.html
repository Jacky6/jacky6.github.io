    07. 测试¶

    Spring 测试概述¶

    Spring 提供了强大的测试支持，包括单元测试、集成测试和切片测试。

    测试依赖¶

    <dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-test</artifactId>
    <scope>test</scope>
</dependency>

    这个 Starter 包含了： - JUnit 5（测试框架） - Mockito（Mock 框架） - AssertJ（断言库） - Spring Test（Spring 测试支持） - JSON 断言

    单元测试¶

    测试 Service 层¶

    package com.example.demo.service;

import com.example.demo.entity.User;
import com.example.demo.repository.UserRepository;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;

import java.util.Optional;

import static org.assertj.core.api.Assertions.assertThat;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)  // 启用 Mockito
class UserServiceTest {

    @Mock  // 创建 Mock 对象
    private UserRepository userRepository;

    @InjectMocks  // 注入 Mock 到被测试类
    private UserService userService;

    @Test
    void testCreateUser() {
        // 准备数据
        User user = new User("Amy", "amy@example.com");
        user.setId(1L);

        // 定义 Mock 行为
        when(userRepository.save(any(User.class))).thenReturn(user);

        // 执行测试
        User saved = userService.create("Amy", "amy@example.com");

        // 断言
        assertThat(saved.getName()).isEqualTo("Amy");
        assertThat(saved.getEmail()).isEqualTo("amy@example.com");

        // 验证 Mock 调用
        verify(userRepository, times(1)).save(any(User.class));
    }

    @Test
    void testFindById() {
        // 准备数据
        User user = new User("Amy", "amy@example.com");
        user.setId(1L);

        // 定义 Mock 行为
        when(userRepository.findById(1L)).thenReturn(Optional.of(user));

        // 执行测试
        Optional<User> found = userService.findById(1L);

        // 断言
        assertThat(found).isPresent();
        assertThat(found.get().getName()).isEqualTo("Amy");
    }

    @Test
    void testFindByIdNotFound() {
        // 定义 Mock 行为
        when(userRepository.findById(999L)).thenReturn(Optional.empty());

        // 执行测试
        Optional<User> found = userService.findById(999L);

        // 断言
        assertThat(found).isEmpty();
    }
}

    常用 Mockito 注解¶

    @Mock  // 创建 Mock 对象
@Mock(name = "customName")  // 指定名称
@Spy  // 部分 Mock（保留真实行为）
@InjectMocks  // 自动注入 Mock 到目标类
@Captor  // 捕获参数用于验证

    常用 Mockito 方法¶

    // 定义行为
when(mock.method(arg)).thenReturn(result);
when(mock.method(any())).thenThrow(new RuntimeException());
doNothing().when(mock).voidMethod();
doReturn(value).when(mock).method();

// 验证
verify(mock).method(arg);
verify(mock, times(2)).method();
verify(mock, never()).method();
verify(mock, atLeastOnce()).method();

// 参数匹配器
any(), anyString(), anyLong(), anyList()
eq(value), argThat(condition)

    集成测试¶

    @SpringBootTest¶

    package com.example.demo;

import com.example.demo.entity.User;
import com.example.demo.repository.UserRepository;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.transaction.annotation.Transactional;

import static org.assertj.core.api.Assertions.assertThat;

@SpringBootTest  // 加载完整应用上下文
@Transactional   // 测试后自动回滚
class UserServiceIntegrationTest {

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private UserService userService;

    @Test
    void testCreateAndFind() {
        // 创建用户
        User saved = userService.create("Amy", "amy@example.com");

        // 验证
        assertThat(saved.getId()).isNotNull();
        assertThat(saved.getName()).isEqualTo("Amy");

        // 从数据库查询
        User found = userService.findById(saved.getId()).get();
        assertThat(found.getEmail()).isEqualTo("amy@example.com");
    }

    @Test
    void testDelete() {
        // 创建用户
        User user = userService.create("Bob", "bob@example.com");

        // 删除
        userService.delete(user.getId());

        // 验证已删除
        assertThat(userService.findById(user.getId())).isEmpty();
    }
}

    Web 层测试¶

    @WebMvcTest - 控制器测试¶

    package com.example.demo.controller;

import com.example.demo.entity.User;
import com.example.demo.service.UserService;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.WebMvcTest;
import org.springframework.boot.test.mock.mockito.MockBean;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

import java.util.Arrays;
import java.util.Optional;

import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.*;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.*;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;

@WebMvcTest(UserController.class)  // 只加载 Web 层相关 Bean
class UserControllerTest {

    @Autowired
    private MockMvc mockMvc;  // Mock MVC 测试工具

    @MockBean  // Mock Service
    private UserService userService;

    @Autowired
    private ObjectMapper objectMapper;  // JSON 序列化

    @Test
    void testGetAllUsers() throws Exception {
        // 准备数据
        User user1 = new User("Amy", "amy@example.com");
        user1.setId(1L);
        User user2 = new User("Bob", "bob@example.com");
        user2.setId(2L);

        when(userService.findAll()).thenReturn(Arrays.asList(user1, user2));

        // 执行请求并断言
        mockMvc.perform(get("/api/users"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.length()").value(2))
            .andExpect(jsonPath("$[0].name").value("Amy"))
            .andExpect(jsonPath("$[1].name").value("Bob"));

        verify(userService, times(1)).findAll();
    }

    @Test
    void testGetUserById() throws Exception {
        User user = new User("Amy", "amy@example.com");
        user.setId(1L);

        when(userService.findById(1L)).thenReturn(Optional.of(user));

        mockMvc.perform(get("/api/users/1"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.name").value("Amy"));
    }

    @Test
    void testGetUserNotFound() throws Exception {
        when(userService.findById(999L)).thenReturn(Optional.empty());

        mockMvc.perform(get("/api/users/999"))
            .andExpect(status().isNotFound());
    }

    @Test
    void testCreateUser() throws Exception {
        User user = new User("Amy", "amy@example.com");
        user.setId(1L);

        when(userService.create(any(), any())).thenReturn(user);

        mockMvc.perform(post("/api/users")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(
                    Map.of("name", "Amy", "email", "amy@example.com"))))
            .andExpect(status().isCreated())
            .andExpect(header().exists("Location"))
            .andExpect(jsonPath("$.name").value("Amy"));

        verify(userService, times(1)).create(any(), any());
    }
}

    @DataJpaTest - Repository 测试¶

    package com.example.demo.repository;

import com.example.demo.entity.User;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.orm.jpa.DataJpaTest;
import org.springframework.test.context.ActiveProfiles;

import java.util.List;
import java.util.Optional;

import static org.assertj.core.api.Assertions.assertThat;

@DataJpaTest  // 只加载 JPA 相关 Bean，使用内存数据库
@ActiveProfiles("test")
class UserRepositoryTest {

    @Autowired
    private TestEntityManager entityManager;  // 测试专用的 EntityManager

    @Autowired
    private UserRepository userRepository;

    @Test
    void testSave() {
        User user = new User("Amy", "amy@example.com");
        User saved = userRepository.save(user);

        assertThat(saved.getId()).isNotNull();
        assertThat(saved.getName()).isEqualTo("Amy");
    }

    @Test
    void testFindById() {
        User user = new User("Amy", "amy@example.com");
        entityManager.persistAndFlush(user);

        Optional<User> found = userRepository.findById(user.getId());

        assertThat(found).isPresent();
        assertThat(found.get().getEmail()).isEqualTo("amy@example.com");
    }

    @Test
    void testFindByEmail() {
        User user = new User("Amy", "amy@example.com");
        entityManager.persistAndFlush(user);

        Optional<User> found = userRepository.findByEmail("amy@example.com");

        assertThat(found).isPresent();
    }

    @Test
    void testFindAll() {
        entityManager.persistAndFlush(new User("Amy", "amy@example.com"));
        entityManager.persistAndFlush(new User("Bob", "bob@example.com"));

        List<User> users = userRepository.findAll();

        assertThat(users).hasSize(2);
    }
}

    断言库¶

    AssertJ 断言¶

    import static org.assertj.core.api.Assertions.*;

// 基本断言
assertThat(value).isEqualTo(expected);
assertThat(value).isNotEqualTo(expected);
assertThat(value).isNull();
assertThat(value).isNotNull();
assertThat(value).isTrue();
assertThat(value).isFalse();

// 字符串断言
assertThat(text).contains("substring");
assertThat(text).startsWith("prefix");
assertThat(text).endsWith("suffix");
assertThat(text).hasSize(10);

// 集合断言
assertThat(list).hasSize(3);
assertThat(list).contains("a", "b");
assertThat(list).doesNotContain("z");
assertThat(list).isEmpty();
assertThat(list).isNotEmpty();

// Optional 断言
assertThat(optional).isPresent();
assertThat(optional).isEmpty();
assertThat(optional).hasValue(expected);

// 异常断言
assertThatThrownBy(() -> {
    throw new RuntimeException("error");
}).isInstanceOf(RuntimeException.class)
  .hasMessage("error");

// 对象断言
assertThat(user)
    .hasFieldOrProperty("name")
    .hasFieldOrPropertyWithValue("name", "Amy")
    .extracting("name", "email")
    .containsExactly("Amy", "amy@example.com");

    测试最佳实践¶

    ✅ 推荐¶

    // 1. 测试命名清晰
@Test
void testCreateUser_WhenEmailExists_ThrowsException() { ... }

// 2. 使用 AAA 模式（Arrange-Act-Assert）
@Test
void testCreateUser() {
    // Arrange - 准备数据
    User user = new User("Amy", "amy@example.com");
    when(repository.save(any())).thenReturn(user);

    // Act - 执行操作
    User saved = service.create("Amy", "amy@example.com");

    // Assert - 断言结果
    assertThat(saved.getName()).isEqualTo("Amy");
}

// 3. 测试边界条件
@Test
void testCreateUser_WithNullEmail_ThrowsException() { ... }

@Test
void testCreateUser_WithVeryLongName_ThrowsException() { ... }

// 4. 使用描述性的断言
assertThat(user.getName())
    .as("User name should be set")
    .isEqualTo("Amy");

// 5. 保持测试独立
// 每个测试应该独立运行，不依赖其他测试的状态

    ❌ 避免¶

    // 1. 避免测试逻辑过于复杂
@Test
void testEverything() {  // ❌ 测试太多内容
    // 50 行代码...
}
// 应该拆分为多个小测试

// 2. 避免硬编码
assertThat(result).isEqualTo("magic-string");  // ❌
// 应该使用有意义的常量或变量

// 3. 避免测试私有方法
// 应该通过公共方法间接测试

// 4. 避免依赖测试顺序
@Test
@Order(1)  // ❌ 不应该依赖执行顺序
void testFirst() { ... }

// 5. 避免吞掉异常
@Test
void testWithException() {
    try {
        // 测试代码
    } catch (Exception e) {
        // 空 catch ❌
    }
}
// 应该使用 assertThrows
assertThatThrownBy(() -> { ... }).isInstanceOf(ExpectedException.class);
