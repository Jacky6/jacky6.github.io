    附录 2：PEP 8 代码风格指南¶

    📐 代码布局¶

    缩进¶

    # ✅ 使用 4 个空格
def function():
    if True:
        print("正确")

# ❌ 避免制表符
def function():
    if True:  # 制表符
        print("错误")

    行长度¶

    # ✅ 每行不超过 79 字符
result = (
    some_long_variable
    + another_long_variable
    - third_variable
)

# ❌ 避免过长行
result = some_long_variable + another_long_variable + third_variable + fourth_variable

    空行¶

    # ✅ 函数之间空两行
def function1():
    pass


def function2():
    pass

# ✅ 类方法之间空一行
class MyClass:
    def method1(self):
        pass

    def method2(self):
        pass

    变量和函数¶

    # ✅ 小写 + 下划线
user_name = "Alice"
calculate_total = lambda x: x * 2

# ❌ 避免
userName = "Alice"  # 驼峰
UserName = "Alice"  # 帕斯卡

    类名¶

    # ✅ 帕斯卡命名法
class UserProfile:
    pass

class HTTPClient:
    pass

    常量¶

    # ✅ 全大写 + 下划线
MAX_SIZE = 100
PI = 3.14159
DEFAULT_TIMEOUT = 30

    私有属性¶

    class MyClass:
    def __init__(self):
        self.public = 1      # 公开
        self._private = 2    # 约定私有
        self.__mangled = 3   # 名称修饰

    💬 引号使用¶

    # ✅ 双引号用于文档字符串
def function():
    """这是文档字符串"""
    pass

# ✅ 单引号用于普通字符串
name = 'Alice'

# ✅ 包含单引号时用双引号
text = "It's a beautiful day"

# ✅ 包含双引号时用单引号
text = 'He said "Hello"'

    📦 导入语句¶

    # ✅ 标准库在前
import os
import sys

# ✅ 第三方库在中间
import requests
import numpy as np

# ✅ 本地模块在后
from mymodule import my_function

# ✅ 每个导入一行
import os
import sys

# ❌ 避免
import os, sys

    运算符¶

    # ✅ 运算符两侧加空格
x = 1 + 2
if x > 0 and y < 10:
    pass

    函数调用¶

    # ✅ 正确
result = function(arg1, arg2)
result = function(arg1, arg2, kwarg1=value)

# ❌ 避免
result = function( arg1, arg2 )
result = function(arg1,arg2)

    # ✅ 正确

items[1:5]
items[::2]

items[1: 5]

    📋 列表和字典¶

    # ✅ 正确
colors = ["red", "green", "blue"]
matrix = [
    [1, 2, 3],
    [4, 5, 6],
]

person = {
    "name": "Alice",
    "age": 25,

    💡 条件语句¶

    # ✅ 正确
if condition:
    pass
elif other_condition:
    pass
else:
    pass

# ✅ 真值测试
if items:  # 检查非空
    pass

if not items:  # 检查空
    pass

# ❌ 避免
if items == []:  # 冗长
    pass

if len(items) > 0:  # 冗长
    pass

    🔄 循环¶

    # ✅ 正确
for item in items:
    process(item)

for i, item in enumerate(items):
    print(f"{i}: {item}")

# ❌ 避免
i = 0
for item in items:
    process(item)
    i += 1

    ✅ 好的代码¶

    """用户管理模块"""

from typing import List, Optional


class User:
    """用户类"""

    def __init__(self, user_id: int, name: str) -> None:
        self.user_id = user_id
        self.name = name
        self.email: Optional[str] = None

    def set_email(self, email: str) -> None:
        """设置邮箱"""
        if "@" not in email:
            raise ValueError("无效的邮箱")
        self.email = email


def find_user(users: List[User], name: str) -> Optional[User]:
    """查找用户"""
    for user in users:
        if user.name == name:
            return user
    return None
