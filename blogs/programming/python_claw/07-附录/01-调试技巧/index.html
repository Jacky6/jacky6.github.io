    附录 1：调试技巧¶

    🐛 调试方法¶

    1. print 调试¶

    最简单直接的调试方式：

    def calculate_total(items):
    print(f"DEBUG: items = {items}")

    total = 0
    for item in items:
        print(f"DEBUG: processing {item}")
        total += item

    print(f"DEBUG: total = {total}")
    return total

# 使用
calculate_total([1, 2, 3, 4, 5])

    2. 使用 logging 模块¶

    import logging

# 配置日志
logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s - %(levelname)s - %(message)s'
)

def process_data(data):
    logging.debug(f"开始处理数据：{data}")

    if not data:
        logging.warning("数据为空")
        return None

    result = sum(data)
    logging.info(f"处理完成，结果：{result}")

    return result

# 日志级别
logging.debug("调试信息")    # 详细信息
logging.info("一般信息")     # 确认运行正常
logging.warning("警告信息")  # 潜在问题
logging.error("错误信息")    # 错误发生
logging.critical("严重错误") # 严重问题

    3. 使用 pdb 调试器¶

    import pdb

def buggy_function(x, y):
    result = x + y
    pdb.set_trace()  # 设置断点
    result = result * 2
    return result

# 运行后会进入交互调试模式

    pdb 常用命令¶

    n (next)      - 执行下一行
s (step)      - 进入函数内部
c (continue)  - 继续执行到下一个断点
l (list)      - 显示当前代码
p (print)     - 打印变量值
q (quit)      - 退出调试器
h (help)      - 显示帮助

    4. 使用 breakpoint() (Python 3.7+)¶

    def process(items):
    total = 0
    for item in items:
        total += item
        breakpoint()  # 自动进入调试器
    return total

    🔍 常见错误排查¶

    1. NameError¶

    # ❌ 错误
print(x)  # NameError: name 'x' is not defined

# ✅ 解决
x = 10
print(x)

    2. TypeError¶

    # ❌ 错误
result = "5" + 3  # TypeError

# ✅ 解决
result = int("5") + 3
# 或
result = "5" + str(3)

    3. IndexError¶

    # ❌ 错误
items = [1, 2, 3]
print(items[5])  # IndexError

# ✅ 解决
if len(items) > 5:
    print(items[5])
else:
    print("索引超出范围")

    4. KeyError¶

    # ❌ 错误
data = {"a": 1}
print(data["b"])  # KeyError

# ✅ 解决
print(data.get("b", "默认值"))
# 或
if "b" in data:
    print(data["b"])

    5. AttributeError¶

    # ❌ 错误
text = "hello"
text.appnd("x")  # AttributeError (拼写错误)

# ✅ 解决
text = "hello"
# 字符串不可变，需要创建新字符串
text = text + "x"

    3. 使用 timeit 性能分析¶

    import timeit

# 测试代码执行时间
time = timeit.timeit(
    "sum(range(1000))",
    number=10000
)
print(f"执行时间：{time:.4f}秒")

    4. 使用 cProfile 性能分析¶

    import cProfile
import pstats

def slow_function():
    total = 0
    for i in range(1000000):
        total += i
    return total

# 性能分析
profiler = cProfile.Profile()
profiler.enable()
slow_function()
profiler.disable()

# 打印统计
stats = pstats.Stats(profiler)
stats.sort_stats('cumulative')
stats.print_stats(10)

    ✅ 推荐¶

    # 1. 使用有意义的变量名
def calculate_user_total(user_id, items):  # 清晰
    pass

def calc(uid, i):  # 不清晰
    pass

# 2. 添加文档字符串
def process_data(data):
    """处理数据并返回结果"""
    pass

# 3. 使用断言验证假设
def divide(a, b):
    assert b != 0, "除数不能为零"
    return a / b

# 4. 记录关键信息
import logging

def process(order):
    logging.info(f"处理订单：{order.id}")
    # ...

    ❌ 避免¶

    # 1. 过度使用 print
def func():
    print("1")
    print("2")
    print("3")
    # ... 太多调试输出

# 2. 忽略异常
try:
    risky_operation()
except:
    pass  # 静默失败

# 3. 过长的函数
def long_function():
    # 100 行代码...
    # 难以调试
    pass

    🎯 调试检查清单¶

    • 变量是否正确初始化？
    • 函数参数类型是否正确？
    • 循环边界是否正确？
    • 条件判断是否完整？
    • 异常是否被正确处理？
    • 返回值是否符合预期？
    • 是否有副作用影响其他代码？

    调试是编程的重要技能，多练习会越来越好！
