    11 Context 与并发控制¶

    context 的设计、超时与取消、传值、最佳实践。

    1. 为什么需要 context¶

    请求进入 → 中间件 → 业务逻辑 → DB/HTTP 调用 → 返回

问题：客户端断开 / 超时后，下游调用如何及时停止？
答案：context 贯穿调用链，取消信号自动传播。

    context 是 Go 并发的"取消与元数据传递标准"。

    2. 基本用法¶

    import "context"

// 1. 背景 context（根，main/请求入口）
ctx := context.Background()

// 2. TODO（占位，尚未决定用哪个）
ctx := context.TODO()

// 3. 超时：到时自动取消
ctx, cancel := context.WithTimeout(ctx, 2*time.Second)
defer cancel()   // 必须调用！否则泄漏 timer

// 4. 截止时间
ctx, cancel := context.WithDeadline(ctx, time.Now().Add(5*time.Second))
defer cancel()

// 5. 取消信号
ctx, cancel := context.WithCancel(parent)
defer cancel()

// 6. 传值
ctx = context.WithValue(ctx, userIDKey, 42)

    3. 取消传播¶

    // 父 context 取消 → 所有子 context 自动取消
parent, cancel := context.WithCancel(context.Background())
child, _ := context.WithCancel(parent)
grandson, _ := context.WithTimeout(child, time.Hour)

cancel()   // parent 取消 → child、grandson 全部取消

// 检查取消
select {
case <-ctx.Done():   // 返回 struct{} 类型的 channel
    return ctx.Err() // context.Canceled 或 context.DeadlineExceeded
default:
    // 继续工作
}

    3.1 在函数中响应取消¶

    // ✅ 标准模式：ctx 作为第一个参数
func FetchData(ctx context.Context, url string) ([]byte, error) {
    req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
    if err != nil {
        return nil, err
    }
    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    return io.ReadAll(resp.Body)
}

// 长循环中检查
func Process(ctx context.Context, items []Item) error {
    for _, item := range items {
        if err := ctx.Err(); err != nil {   // 轻量检查
            return err
        }
        process(item)
    }
    return nil
}

// 阻塞操作用 select
func WaitResult(ctx context.Context, ch <-chan Result) (Result, error) {
    select {
    case r := <-ch:
        return r, nil
    case <-ctx.Done():
        return Result{}, ctx.Err()
    }
}

    4. WithValue¶

    // key 必须是自定义类型（避免冲突）
type ctxKey int

const userIDKey ctxKey = 0

ctx = context.WithValue(ctx, userIDKey, 42)
v, ok := ctx.Value(userIDKey).(int)

// ❌ 用字符串 key（可能与其他包冲突）
ctx = context.WithValue(ctx, "userID", 42)

    原则：

    • 只传请求作用域的数据（trace ID、用户身份、认证 token）
    • ❌ 不要传业务参数、配置、可选参数
    • context 不是依赖注入容器

    5. 常见错误¶

    // ❌ 1. 忘记 cancel → 泄漏
ctx, cancel := context.WithTimeout(ctx, time.Second)
// defer cancel() 缺失！

// ❌ 2. 把 ctx 存到结构体
type Service struct {
    ctx context.Context   // 生命周期混乱
}
// ✅ ctx 作为方法参数传递
func (s *Service) Do(ctx context.Context) {}

// ❌ 3. 在 goroutine 里 cancel
go func() {
    ctx, cancel := context.WithTimeout(parent, time.Second)
    defer cancel()
    work(ctx)
}()
// ✅ 在创建处 defer cancel

// ❌ 4. ctx 放在非第一个参数
func Do(url string, ctx context.Context)   // 不符合惯例

// ❌ 5. 用 ctx 传业务参数
ctx = context.WithValue(ctx, "pageSize", 20)   // 应该是函数参数

    6. 实战：HTTP 服务超时控制¶

    func handler(w http.ResponseWriter, r *http.Request) {
    ctx := r.Context()   // 请求自带 ctx，客户端断开自动取消

    ctx, cancel := context.WithTimeout(ctx, 3*time.Second)
    defer cancel()

    data, err := FetchData(ctx, "https://api.example.com")
    if err != nil {
        if errors.Is(err, context.DeadlineExceeded) {
            http.Error(w, "timeout", http.StatusGatewayTimeout)
            return
        }
        http.Error(w, err.Error(), 500)
        return
    }
    w.Write(data)
}

    7. 实战：优雅退出¶

    func main() {
    ctx, stop := signal.NotifyContext(context.Background(),
        syscall.SIGINT, syscall.SIGTERM)
    defer stop()

    srv := &http.Server{Addr: ":8080", Handler: routes()}

    go func() {
        if err := srv.ListenAndServe(); err != http.ErrServerClosed {
            log.Fatal(err)
        }
    }()

    <-ctx.Done()   // 等待退出信号
    log.Println("shutting down...")

    shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()
    srv.Shutdown(shutdownCtx)   // 等待处理中的请求完成
}

    8. 底层实现速览¶

    // context 是一个接口
type Context interface {
    Deadline() (deadline time.Time, ok bool)
    Done() <-chan struct{}
    Err() error
    Value(key any) any
}

// WithCancel 返回 *cancelCtx：内部维护 children 集合
// cancel() 时关闭 done channel 并级联取消所有子 context
// WithTimeout = WithDeadline = 定时器触发 cancel

    9. 小结¶

    • context 的核心价值：取消传播 + 超时控制
    • ctx 永远是第一个参数，不存结构体
    • WithTimeout/WithCancel 必须配 defer cancel()
    • ctx.Done() + ctx.Err() 是响应取消的标准姿势
    • WithValue 只传请求元数据，key 用自定义类型
