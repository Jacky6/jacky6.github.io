    12 标准库实战¶

    fmt、strings、time、encoding/json、net/http、io、os、log/slog。

    1. fmt¶

    // 动词（verbs）
fmt.Printf("%d\n", 42)        // 整数
fmt.Printf("%s\n", "str")     // 字符串
fmt.Printf("%q\n", "str")     // 带引号 "str"
fmt.Printf("%v\n", value)     // 默认格式
fmt.Printf("%+v\n", struct)   // 结构体带字段名
fmt.Printf("%#v\n", value)    // Go 语法表示
fmt.Printf("%T\n", value)     // 类型
fmt.Printf("%t\n", true)      // bool
fmt.Printf("%f\n", 3.14)      // 小数
fmt.Printf("%.2f\n", 3.14159) // 保留 2 位
fmt.Printf("%x\n", 255)       // 十六进制 ff
fmt.Printf("%p\n", &x)        // 指针地址
fmt.Printf("%w\n", err)       // 错误包装（Errorf）

// 宽度与对齐
fmt.Printf("%10s|\n", "go")   // 右对齐
fmt.Printf("%-10s|\n", "go")  // 左对齐

// 输出到 io.Writer
fmt.Fprintf(w, "hello %s", name)

// 拼接（返回字符串）
s := fmt.Sprintf("%d-%d", a, b)

// 扫描输入
fmt.Scan(&name)
fmt.Sscanf("2024-01-01", "%d-%d-%d", &y, &m, &d)

    3. time¶

    // 当前时间

// 格式化：Go 用参考时间（不是 %Y-%m-%d！）
// 2006-01-02 15:04:05  ← 记法：1月2日 3点4分5秒 2006年

now.Format("2006/01/02")

// 解析
t, err := time.Parse("2006-01-02", "2024-01-15")
t, err := time.Parse(time.RFC3339, "2024-01-01T12:00:00Z")

// 时区
loc, _ := time.LoadLocation("Asia/Shanghai")
t := now.In(loc)

// 时长
d := 2 * time.Second
d.Hours(), d.Minutes(), d.Seconds()
time.Millisecond, time.Microsecond, time.Nanosecond

// 加减
later := now.Add(24 * time.Hour)
diff := t2.Sub(t1)            // time.Duration

// 比较

    4. encoding/json¶

    type User struct {
    Name  string `json:"name"`
    Age   int    `json:"age,omitempty"`   // 零值时省略
    Email string `json:"email,omitempty"`
    Pass  string `json:"-"`               // 永不序列化
}

// 序列化
u := User{Name: "Alice", Age: 30}
data, err := json.Marshal(u)
data, err := json.MarshalIndent(u, "", "  ")   // 带缩进

// 反序列化
var u2 User
err := json.Unmarshal(data, &u2)

// 流式（大文件/HTTP body）
err := json.NewDecoder(r).Decode(&u)
err := json.NewEncoder(w).Encode(u)

// 任意 JSON
var m map[string]any
json.Unmarshal([]byte(`{"a":1,"b":[1,2]}`), &m)
// m["a"] 是 float64！
n := m["a"].(float64)

    4.1 自定义序列化¶

    // 实现 Marshaler / Unmarshaler 接口
func (u *User) MarshalJSON() ([]byte, error) {
    return json.Marshal(struct {
        Name string `json:"name"`
    }{u.Name})
}

// 时间自定义格式
type MyTime time.Time

func (t MyTime) MarshalJSON() ([]byte, error) {
    return json.Marshal(time.Time(t).Format("2006-01-02"))
}

    5.2 中间件¶

    // 中间件 = 包装 Handler 的函数
type Middleware func(http.Handler) http.Handler

func Logging(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        next.ServeHTTP(w, r)
        log.Printf("%s %s %v", r.Method, r.URL.Path, time.Since(start))
    })
}

func Chain(h http.Handler, mws ...Middleware) http.Handler {
    for i := len(mws) - 1; i >= 0; i-- {
        h = mws[i](h)
    }
    return h
}

    6. io¶

    // 核心接口
io.Reader   // Read(p []byte) (n int, err error)
io.Writer   // Write(p []byte) (n int, err error)
io.Closer   // Close() error
io.Seeker   // Seek(offset int64, whence int) (int64, error)

// 常用函数
io.ReadAll(r)              // 读全部（注意内存）
io.Copy(dst, src)          // 流式拷贝（返回字节数）
io.CopyN(dst, src, n)      // 拷贝 n 字节
io.WriteString(w, "hello")
io.LimitReader(r, 1024)    // 最多读 1KB
io.MultiReader(r1, r2)     // 串联
io.TeeReader(r, w)         // 读的同时写副本
io.Pipe()                  // 内存管道（同步）

// bufio：缓冲 IO
br := bufio.NewReader(r)
line, err := br.ReadString('\n')
scanner := bufio.NewScanner(r)
for scanner.Scan() {
    fmt.Println(scanner.Text())   // 按行读
}
bw := bufio.NewWriter(w)
bw.WriteString("x")
bw.Flush()   // 必须刷新

    8. log/slog（Go 1.21+）¶

    // 结构化日志：官方现代方案
import "log/slog"

// 默认 logger
slog.Info("user logged in", "user_id", 42, "ip", "1.2.3.4")
slog.Error("db error", "err", err, "op", "query")

// JSON 输出
logger := slog.New(slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{
    Level: slog.LevelDebug,
}))
slog.SetDefault(logger)

logger.Info("request",
    slog.String("method", "GET"),
    slog.Int("status", 200),
    slog.Duration("elapsed", time.Since(start)),
)

// 分组
logger = logger.With("service", "api")   // 附加公共字段

// 子 logger
child := logger.WithGroup("db")
child.Info("connected")   // {"db":{"service":"api"},...}

    9. 小结¶

    • fmt 动词：%v 万能、%+v 调试结构体、%w 包装错误
    • 时间格式化用参考时间 2006-01-02 15:04:05
    • JSON：tag 控制、omitempty 省略零值、any 数字是 float64
    • net/http 标准库足以构建生产级服务（1.22 路由增强）
    • slog 是官方结构化日志答案，新项目首选
