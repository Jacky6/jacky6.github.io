    命令¶

    go run main.go        # 编译并运行
go build ./...        # 编译全部
go install            # 安装到 $GOPATH/bin
go mod init <path>    # 初始化模块
go mod tidy           # 同步依赖（最常用）
go get pkg@v1.2.3     # 添加/升级依赖
go get -u ./...       # 升级全部
go test ./...         # 测试
go test -race -v      # 竞态 + 详细
go test -bench=. -benchmem
go test -coverprofile=c.out && go tool cover -html=c.out
go fmt ./...          # 格式化
go vet ./...          # 静态检查
go doc fmt.Println    # 文档
go env -w GOPROXY=https://goproxy.cn,direct
go work init ./a ./b  # 工作区
go tool pprof http://localhost:6060/debug/pprof/profile?seconds=30

    陷阱速查¶

    // 1. nil map 写入 panic
var m map[string]int
m["k"] = 1   // panic

// 2. for range 闭包（Go 1.22 前共享变量）
for _, v := range s { go func() { print(v) }() }

// 3. append 不接收返回值
append(s, x)        // ❌
s = append(s, x)    // ✅

// 4. 循环内 defer 不执行
for ... { defer f.Close() }   // ❌ 封装成函数

// 5. 接口 nil 陷阱
var p *T = nil
var i any = p
i == nil   // false！

// 6. 切片共享底层数组
b := a[1:3]; b[0] = 1   // 修改了 a

// 7. JSON 数字是 float64
var m map[string]any
json.Unmarshal(d, &m)
m["n"].(int)   // panic，应为 float64

// 8. 忘记 cancel
ctx, cancel := context.WithTimeout(...)
defer cancel()   // 必须！

// 9. time.After 循环泄漏（Go 1.23 前）
for { select { case <-time.After(time.Second): } }

// 10. map 并发读写 fatal（无法 recover）
// 用 sync.RWMutex 或 sync.Map
