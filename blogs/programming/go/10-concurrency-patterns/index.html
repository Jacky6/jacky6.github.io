    10 并发模式与 sync 包¶

    WaitGroup、Mutex、Once、原子操作、常见并发模式。

    1. sync 包¶

    1.1 WaitGroup¶

    var wg sync.WaitGroup

for i := 0; i < 5; i++ {
    wg.Add(1)          // 计数 +1（必须在 go 之前）
    go func() {
        defer wg.Done() // 计数 -1
        work()
    }()
}
wg.Wait()              // 阻塞直到计数归零

// ❌ 常见错误
wg.Add(1)
go func() {
    work()
    wg.Done()
    // panic 时 Done 不会执行 → 用 defer wg.Done()
}()

// ❌ 复制 WaitGroup
func bad(wg sync.WaitGroup) {}   // 值传递！必须传 *sync.WaitGroup

    1.2 Mutex / RWMutex¶

    var mu sync.Mutex
counter := 0

mu.Lock()
counter++
mu.Unlock()

// ❌ 忘记 Unlock → 死锁；用 defer 保证
func inc() {
    mu.Lock()
    defer mu.Unlock()
    counter++
}

// RWMutex：读多写少
var rw sync.RWMutex
rw.Lock()    // 写锁（独占）
rw.Unlock()
rw.RLock()   // 读锁（共享）
rw.RUnlock()

// 注意：RWMutex 写锁等待时会阻塞新读者（避免写饥饿）

    1.3 sync.Once¶

    var once sync.Once
var instance *DB

func GetDB() *DB {
    once.Do(func() {
        instance = connectDB()   // 只执行一次，并发安全
    })
    return instance
}

    1.4 sync.Map¶

    // 读多写少、key 稳定的场景
var m sync.Map

m.Store("k", 1)
v, ok := m.Load("k")
m.Delete("k")
m.Range(func(k, v any) bool { return true })   // 返回 false 停止
v, loaded := m.LoadOrStore("k", 1)             // 不存在才存

// 大多数场景：普通 map + RWMutex 更快、类型安全
// sync.Map 的 key/value 是 any，失去编译期类型检查

    1.5 sync.Pool¶

    // 对象池：复用临时对象，减少 GC 压力
var bufPool = sync.Pool{
    New: func() any {
        return new(bytes.Buffer)
    },
}

func Process() {
    buf := bufPool.Get().(*bytes.Buffer)
    defer func() {
        buf.Reset()
        bufPool.Put(buf)
    }()
    // 使用 buf
}

    注意：Pool 内容随时可能被 GC 清空，不保证存活。

    1.6 sync.WaitGroup 之外的同步原语¶

    // sync.Cond：条件变量（生产者-消费者）
cond := sync.NewCond(&sync.Mutex)

cond.L.Lock()
for !ready() {          // 必须用 for，防止虚假唤醒
    cond.Wait()         // 释放锁并等待
}
cond.L.Unlock()

cond.Broadcast()        // 唤醒所有
cond.Signal()           // 唤醒一个

    2. 原子操作¶

    import "sync/atomic"

var n atomic.Int64      // Go 1.19+ 泛型类型
n.Add(1)
n.Load()
n.Store(10)
n.CompareAndSwap(10, 20)

var p atomic.Pointer[Config]   // 泛型指针
p.Store(&cfg)
c := p.Load()

// 老式函数 API（仍可用）
atomic.AddInt64(&count, 1)
atomic.LoadInt64(&count)
atomic.Value{}               // 接口类型原子操作

    经验：简单计数器用 atomic，复杂临界区用 Mutex。

    3. 经典并发模式¶

    3.1 生成器¶

    func Generate(nums ...int) <-chan int {
    out := make(chan int)
    go func() {
        defer close(out)
        for _, n := range nums {
            out <- n
        }
    }()
    return out
}

for v := range Generate(1, 2, 3) {
    fmt.Println(v)
}

    3.2 扇入¶

    // 合并多个 channel
func FanIn[T any](chans ...<-chan T) <-chan T {
    out := make(chan T)
    var wg sync.WaitGroup
    for _, ch := range chans {
        wg.Add(1)
        go func(c <-chan T) {
            defer wg.Done()
            for v := range c {
                out <- v
            }
        }(ch)
    }
    go func() {
        wg.Wait()
        close(out)   // 所有输入关闭后才关闭输出
    }()
    return out
}

    3.3 扇出¶

    // 多个 worker 消费同一个 channel
func FanOut[T any](in <-chan T, workers int, fn func(T)) {
    var wg sync.WaitGroup
    for i := 0; i < workers; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for v := range in {
                fn(v)
            }
        }()
    }
    wg.Wait()
}

    3.4 管道¶

    // 生成 → 过滤 → 汇聚
func pipeline() {
    gen := Generate(1, 2, 3, 4, 5, 6)
    even := Filter(gen, func(n int) bool { return n%2 == 0 })
    sq := Map(even, func(n int) int { return n * n })

    for v := range sq {
        fmt.Println(v)   // 4 16 36
    }
}

func Filter[T any](in <-chan T, pred func(T) bool) <-chan T {
    out := make(chan T)
    go func() {
        defer close(out)
        for v := range in {
            if pred(v) {
                out <- v
            }
        }
    }()
    return out
}

    3.5 worker pool¶

    func WorkerPool(jobs <-chan Job, workers int) <-chan Result {
    results := make(chan Result)
    var wg sync.WaitGroup
    for i := 0; i < workers; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for j := range jobs {
                results <- process(j)
            }
        }()
    }
    go func() {
        wg.Wait()
        close(results)
    }()
    return results
}

    3.6 限速（semaphore）¶

    // 用带缓冲 channel 实现信号量
sem := make(chan struct{}, 10)   // 最多 10 并发

for _, task := range tasks {
    sem <- struct{}{}   // 获取
    go func(t Task) {
        defer func() { <-sem }()   // 释放
        process(t)
    }(task)
}

// 官方扩展包
import "golang.org/x/sync/semaphore"
ctx := context.Background()
sem := semaphore.NewWeighted(10)
sem.Acquire(ctx, 1)
defer sem.Release(1)

    3.7 errgroup：并发 + 错误收集¶

    import "golang.org/x/sync/errgroup"

g, ctx := errgroup.WithContext(context.Background())
g.SetLimit(10)   // 限制并发数

for _, url := range urls {
    g.Go(func() error {
        return fetch(ctx, url)   // 返回 error 自动收集
    })
}
if err := g.Wait(); err != nil {   // 等待全部完成，返回第一个错误
    log.Fatal(err)
}
// ctx 在首个错误发生时自动取消 → 其他 goroutine 感知退出

    4. 并发安全检查¶

    # 运行时检测数据竞争（开销 ~10x，CI 必开）
go test -race ./...
go run -race main.go
go build -race

# 输出示例：
# WARNING: DATA RACE
# Read at 0x00c000014128 by goroutine 6:
#   main.main.func1()

    5. 小结¶

    • WaitGroup 等待、Mutex 互斥、Once 单例、Pool 复用
    • 简单计数用 atomic，复杂状态用 Mutex
    • 管道/扇入/扇出/worker pool 是 channel 的标准组合
    • errgroup 是并发任务 + 错误处理 + 取消的现代答案
    • -race 检测器是并发 bug 的终极武器，测试必开
