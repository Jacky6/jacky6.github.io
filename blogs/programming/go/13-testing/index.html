    13 测试与基准¶

    单元测试、表驱动测试、基准测试、模糊测试、mock 与覆盖率。

    1. 测试基础¶

    文件约定：xxx_test.go，与被测代码同包。

    // math.go
func Add(a, b int) int { return a + b }

// math_test.go
package math

import "testing"

func TestAdd(t *testing.T) {
    got := Add(1, 2)
    want := 3
    if got != want {
        t.Errorf("Add(1, 2) = %d, want %d", got, want)
    }
}

    go test                    # 当前包
go test ./...              # 全部包
go test -v                 # 显示每个测试
go test -run TestAdd       # 只跑匹配的测试
go test -count=1           # 禁用缓存
go test -race              # 开启竞态检测
go test -cover             # 覆盖率
go test -coverprofile=c.out && go tool cover -html=c.out   # 可视化
go test -bench=.           # 跑基准
go test -shuffle=on        # 随机顺序（防依赖）

    2. 表驱动测试（Go 招牌模式）¶

    func TestAdd(t *testing.T) {
    tests := []struct {
        name string
        a, b int
        want int
    }{
        {"positive", 1, 2, 3},
        {"negative", -1, -2, -3},
        {"zero", 0, 0, 0},
        {"mixed", -1, 1, 0},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {   // 子测试
            got := Add(tt.a, tt.b)
            if got != tt.want {
                t.Errorf("Add(%d, %d) = %d, want %d", tt.a, tt.b, got, tt.want)
            }
        })
    }
}

    go test -run TestAdd -v
# === RUN   TestAdd
# === RUN   TestAdd/positive
# === RUN   TestAdd/negative
# ...

    3. 常用测试工具¶

    // t.Helper()：标记辅助函数，报错定位到调用处
func assertEqual(t *testing.T, got, want int) {
    t.Helper()
    if got != want {
        t.Errorf("got %d, want %d", got, want)
    }
}

// t.Cleanup：测试结束后清理（替代 defer 链）
t.Cleanup(func() { db.Close() })

// t.Parallel：并行执行
func TestX(t *testing.T) {
    t.Parallel()
    // ...
}

// t.Skip / t.Skipf
if testing.Short() {
    t.Skip("skipping in short mode")
}

// t.Fatal vs t.Error
t.Errorf(...)   // 标记失败，继续执行
t.Fatalf(...)   // 标记失败，立即终止当前测试

// t.TempDir：自动清理的临时目录
dir := t.TempDir()

// 子测试控制
t.Run("case", func(t *testing.T) { ... })

    4. 基准测试¶

    func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {   // b.N 由框架自动调整
        Add(1, 2)
    }
}

// 带参数
func BenchmarkConcat(b *testing.B) {
    for i := 0; i < b.N; i++ {
        s := "a" + "b"
        _ = s
    }
}

    go test -bench=.
# BenchmarkAdd-8   1000000000   0.5 ns/op
#                          ↑ 8 核，每次操作 0.5 纳秒

go test -bench=. -benchmem
# BenchmarkAdd-8   1000000000   0.5 ns/op   0 B/op   0 allocs/op

    4.1 防止编译器优化¶

    var result int   // 包级变量，防止被优化掉

func BenchmarkAdd(b *testing.B) {
    var r int
    for i := 0; i < b.N; i++ {
        r = Add(1, 2)
    }
    result = r   // 赋值给包级变量
}

    4.2 b.ResetTimer / b.ReportAllocs¶

    func BenchmarkBig(b *testing.B) {
    setup()              // 昂贵的准备工作
    b.ResetTimer()       // 重置计时
    b.ReportAllocs()     // 报告内存分配
    for i := 0; i < b.N; i++ {
        work()
    }
}

    5. 模糊测试（Go 1.18+）¶

    // FuzzXxx + f.Fuzz
func FuzzAdd(f *testing.F) {
    f.Add(1, 2)          // 种子语料
    f.Add(-1, 1)

    f.Fuzz(func(t *testing.T, a, b int) {
        got := Add(a, b)
        if got != a+b {
            t.Errorf("Add(%d, %d) = %d", a, b, got)
        }
    })
}

    go test -fuzz=FuzzAdd -fuzztime=10s
# 发现的崩溃语料自动存入 testdata/fuzz/，之后普通 go test 会复现

    7. Mock 与依赖¶

    Go 官方不提供 mock 框架，惯用接口 + 手写 fake：

    // 依赖接口
type UserStore interface {
    Get(ctx context.Context, id string) (*User, error)
}

// 手写 fake（测试文件中）
type fakeStore struct {
    users map[string]*User
}

func (f *fakeStore) Get(ctx context.Context, id string) (*User, error) {
    u, ok := f.users[id]
    if !ok {
        return nil, ErrNotFound
    }
    return u, nil
}

// 测试中使用
func TestHandler(t *testing.T) {
    store := &fakeStore{users: map[string]*User{"1": {Name: "a"}}}
    h := NewHandler(store)
    // ...
}

    第三方选择：gomock（官方维护，go.uber.org/mock）、testify（断言库）。

    // testify 断言
import "github.com/stretchr/testify/assert"
import "github.com/stretchr/testify/require"

assert.Equal(t, 3, got)      // 失败继续
require.NoError(t, err)      // 失败立即终止

    8. httptest¶

    import "net/http/httptest"

// 测试 HTTP handler
func TestHandler(t *testing.T) {
    req := httptest.NewRequest("GET", "/users/1", nil)
    w := httptest.NewRecorder()

    handler(w, req)

    resp := w.Result()
    assert.Equal(t, 200, resp.StatusCode)
}

// mock HTTP 服务端
srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
    w.Write([]byte(`{"ok":true}`))
}))
defer srv.Close()
// 用 srv.URL 作为目标地址测试客户端

    9. 测试组织最佳实践¶

    project/
├── math.go
├── math_test.go          # 同包：可测未导出函数
├── internal/
│   └── store/
│       ├── store.go
│       └── store_test.go
└── math_test/            # 外部测试包（package math_test）
    └── api_test.go       # 只测导出 API（黑盒）
    • 表驱动 + 子测试是默认选择
    • 覆盖率目标 80%+，但关键路径比数字重要
    • -race 在 CI 中必开
    • 集成测试用 build tag 隔离：//go:build integration
    //go:build integration

func TestDB(t *testing.T) { ... }

// go test -tags=integration ./...

    10. 小结¶

    • 表驱动测试是 Go 文化核心
    • t.Helper()/t.Cleanup()/t.Parallel() 提升质量
    • 基准测试注意防优化、-benchmem 看分配
    • 模糊测试自动探索边界输入
    • 接口 + 手写 fake 优于重型 mock 框架
