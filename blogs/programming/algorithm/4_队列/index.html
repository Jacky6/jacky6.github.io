    队列（Queue）知识点¶

    1. 定义¶

    队列是一种先进先出（FIFO - First In First Out）的线性数据结构。

    • 只允许在一端插入，在另一端删除
    • 允许插入的一端称为队尾（Rear/Tail）
    • 允许删除的一端称为队头（Front/Head）
    • 不含元素的空表称为空队列
    入队 → [ A ] [ B ] [ C ] → 出队
       ↑                   ↑
     队头                队尾
   (Front)              (Rear)

A 最先入队，也最先出队
C 最后入队，要等 A、B 出队后才能出队

    队列 vs 栈¶

    栈（LIFO）：后进先出          队列（FIFO）：先进先出

    ↑ 栈顶                      出队 ← [A][B][C] ← 入队
    │                                 队头    队尾
  ┌───┐
  │ C │  最后进，最先出           A 最先进，最先出
  ├───┤
  │ B │
  ├───┤
  │ A │  最先进，最后出           C 最后进，最后出
  └───┘
    ↓ 栈底

    2. 基本操作¶

    操作 描述 时间复杂度
    enqueue(x) / push(x) 将元素 x 加入队尾 O(1)
    dequeue() / pop() 从队头移除元素 O(1)
    front() / peek() 返回队头元素（不移除） O(1)
    isEmpty() 判断队列是否为空 O(1)
    size() 返回队列中元素个数 O(1)

    3. 实现方式¶

    3.1 数组实现（顺序队列）¶

    （1）普通数组队列（有缺陷）¶

    class ArrayQueue:
    def __init__(self, capacity=100):
        self.data = [None] * capacity
        self.capacity = capacity
        self.front = 0  # 队头索引
        self.rear = 0   # 队尾索引
        self._size = 0

    def enqueue(self, x):
        if self.is_full():
            raise Exception("Queue overflow")
        self.data[self.rear] = x
        self.rear = (self.rear + 1) % self.capacity
        self._size += 1

    def dequeue(self):
        if self.is_empty():
            raise Exception("Queue underflow")
        val = self.data[self.front]
        self.data[self.front] = None
        self.front = (self.front + 1) % self.capacity
        self._size -= 1
        return val

    def front(self):
        if self.is_empty():
            raise Exception("Queue is empty")
        return self.data[self.front]

    def is_empty(self):
        return self._size == 0

    def is_full(self):
        return self._size == self.capacity

    def size(self):
        return self._size

    （2）循环队列（推荐）⭐¶

    问题：普通数组队列出队后，前面的空间无法再利用（"假溢出"）

    解决：使用循环数组，队尾到达末尾后回到开头

    循环队列示意图：

  ┌───┬───┬───┬───┬───┬───┬───┬───┐
  │   │   │ D │ E │ F │   │   │   │
  └───┴───┴───┴───┴───┴───┴───┴───┘
        ↑               ↑
      front           rear
      (队头)          (队尾)

  元素：D, E, F
  出队方向 → → →
  入队方向 → → →

    class CircularQueue:
    """
    循环队列实现

    关键点：
    - 使用取模运算实现循环
    - rear 指向下一个可插入位置
    - 判断满：(rear + 1) % capacity == front（留一个空位）
    """

    def __init__(self, capacity=100):
        self.capacity = capacity + 1  # 多留一个位置判断满
        self.data = [None] * self.capacity
        self.front = 0
        self.rear = 0

    def enqueue(self, x):
        """入队"""
        if self.is_full():
            raise Exception("Queue overflow")
        self.data[self.rear] = x
        self.rear = (self.rear + 1) % self.capacity
        return True

    def dequeue(self):
        """出队"""
        if self.is_empty():
            raise Exception("Queue underflow")
        val = self.data[self.front]
        self.data[self.front] = None
        self.front = (self.front + 1) % self.capacity
        return val

    def front(self):
        """获取队头"""
        if self.is_empty():
            return None
        return self.data[self.front]

    def is_empty(self):
        return self.front == self.rear

    def is_full(self):
        # 留一个空位，避免 front == rear 无法区分空/满
        return (self.rear + 1) % self.capacity == self.front

    def size(self):
        return (self.rear - self.front + self.capacity) % self.capacity


# 使用示例
queue = CircularQueue(capacity=5)
queue.enqueue('A')
queue.enqueue('B')
queue.enqueue('C')
print(queue.dequeue())  # A
queue.enqueue('D')
print(queue.dequeue())  # B

    3.2 链表实现（链式队列）¶

    class Node:
    def __init__(self, val):
        self.val = val
        self.next = None

class LinkedQueue:
    """
    链表实现队列

    优点：
    - 无需预分配容量
    - 不会溢出（除非内存耗尽）

    缺点：
    - 每个节点需要额外指针空间
    - 缓存局部性不如数组
    """

    def __init__(self):
        self.head = None  # 队头
        self.tail = None  # 队尾
        self._size = 0

    def enqueue(self, x):
        """入队：在队尾添加"""
        node = Node(x)
        if self.is_empty():
            self.head = self.tail = node
        else:
            self.tail.next = node
            self.tail = node
        self._size += 1
        return True

    def dequeue(self):
        """出队：从队头移除"""
        if self.is_empty():
            raise Exception("Queue underflow")
        val = self.head.val
        self.head = self.head.next
        if self.head is None:  # 队列变空
            self.tail = None
        self._size -= 1
        return val

    def front(self):
        """获取队头"""
        if self.is_empty():
            return None
        return self.head.val

    def is_empty(self):
        return self.head is None

    def size(self):
        return self._size


# 使用示例
queue = LinkedQueue()
queue.enqueue('A')
queue.enqueue('B')
queue.enqueue('C')
print(queue.dequeue())  # A
print(queue.dequeue())  # B
print(queue.front())    # C

    3.3 双端队列（Deque）⭐¶

    双端队列：两端都可以入队和出队

    ← 左出队   [A][B][C][D]   右出队 →
← 左入队                  右入队 →

    from collections import deque

# Python 标准库实现
queue = deque()

# 右端操作（普通队列）
queue.append('A')      # 右入队
queue.append('B')
queue.popleft()        # 左出队 → 'A'

# 左端操作（双端特性）
queue.appendleft('X')  # 左入队
queue.pop()            # 右出队 → 'B'

print(queue)  # deque(['X', 'C'])

    class MyDeque:
    """手写双端队列"""

    def __init__(self, capacity=100):
        self.data = [None] * capacity
        self.capacity = capacity
        self.front = 0
        self.rear = 0
        self._size = 0

    def push_left(self, x):
        """左端入队"""
        if self.is_full():
            raise Exception("Deque overflow")
        self.front = (self.front - 1 + self.capacity) % self.capacity
        self.data[self.front] = x
        self._size += 1

    def push_right(self, x):
        """右端入队"""
        if self.is_full():
            raise Exception("Deque overflow")
        self.data[self.rear] = x
        self.rear = (self.rear + 1) % self.capacity
        self._size += 1

    def pop_left(self):
        """左端出队"""
        if self.is_empty():
            raise Exception("Deque underflow")
        val = self.data[self.front]
        self.front = (self.front + 1) % self.capacity
        self._size -= 1
        return val

    def pop_right(self):
        """右端出队"""
        if self.is_empty():
            raise Exception("Deque underflow")
        self.rear = (self.rear - 1 + self.capacity) % self.capacity
        val = self.data[self.rear]
        self._size -= 1
        return val

    def is_empty(self):
        return self._size == 0

    def is_full(self):
        return self._size == self.capacity

    操作 数组实现 链表实现 双端队列

    5.1 BFS 广度优先搜索 ✅⭐¶

    核心思想：用队列实现层序遍历

    from collections import deque

def bfs(graph, start):
    """
    广度优先搜索

    graph: 邻接表表示的图 {node: [neighbors]}
    start: 起始节点
    """
    visited = set()
    queue = deque([start])
    visited.add(start)
    result = []

    while queue:
        node = queue.popleft()
        result.append(node)

        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

    return result


# 示例
graph = {
    'A': ['B', 'C'],
    'B': ['A', 'D', 'E'],
    'C': ['A', 'F'],
    'D': ['B'],
    'E': ['B', 'F'],
    'F': ['C', 'E']
}

print(bfs(graph, 'A'))  # ['A', 'B', 'C', 'D', 'E', 'F']

    BFS 遍历过程：

        A              队列：[A]
      /   \            出队 A，加入 B、C
     B     C           队列：[B, C]
    / \   /            出队 B，加入 D、E
   D   E F             队列：[C, D, E]
                       出队 C，加入 F
                       队列：[D, E, F]
                       ...

    5.2 层序遍历二叉树 ✅¶

    from collections import deque
from typing import List, Optional

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def levelOrder(root: Optional[TreeNode]) -> List[List[int]]:
    """
    二叉树层序遍历
    LeetCode 102
    """
    if not root:
        return []

    result = []
    queue = deque([root])

    while queue:
        level_size = len(queue)
        level = []

        for _ in range(level_size):
            node = queue.popleft()
            level.append(node.val)

            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)

        result.append(level)

    return result


# 示例
#       3
#      / \
#     9  20
#       /  \
#      15   7

root = TreeNode(3)
root.left = TreeNode(9)
root.right = TreeNode(20, TreeNode(15), TreeNode(7))

print(levelOrder(root))  # [[3], [9, 20], [15, 7]]

    5.3 滑动窗口最大值 ✅⭐¶

    from collections import deque
from typing import List

def maxSlidingWindow(nums: List[int], k: int) -> List[int]:
    """
    滑动窗口最大值
    LeetCode 239

    使用单调递减双端队列：
    - 队头始终是当前窗口最大值
    - 新元素加入前，移除所有比它小的元素
    """
    if not nums or k <= 0:
        return []

    result = []
    dq = deque()  # 存储索引，保持单调递减

    for i, num in enumerate(nums):
        # 移除超出窗口的元素
        while dq and dq[0] <= i - k:
            dq.popleft()

        # 维护单调性：移除比当前元素小的
        while dq and nums[dq[-1]] < num:
            dq.pop()

        dq.append(i)

        # 窗口形成后，记录最大值
        if i >= k - 1:
            result.append(nums[dq[0]])

    return result


# 示例
nums = [1, 3, -1, -3, 5, 3, 6, 7]
k = 3
print(maxSlidingWindow(nums, k))  # [3, 3, 5, 5, 6, 7]

# 窗口位置                最大值
# ────────────────────────
# [1  3  -1] -3  5  3  6  7   3
#  1 [3  -1  -3] 5  3  6  7   3
#  1  3 [-1  -3  5] 3  6  7   5
#  1  3  -1 [-3  5  3] 6  7   5
#  1  3  -1  -3 [5  3  6] 7   6
#  1  3  -1  -3  5 [3  6  7]  7

    5.4 消息队列 ✅¶

    import redis
import json
from typing import Any, Optional
from collections import deque

class SimpleMessageQueue:
    """
    基于内存的消息队列（生产环境用 Redis/RabbitMQ/Kafka）
    """

    def __init__(self):
        self.queue = deque()

    def publish(self, message: Any):
        """生产者：发送消息"""
        self.queue.append({
            'id': len(self.queue) + 1,
            'data': message,
            'timestamp': __import__('time').time()
        })

    def consume(self) -> Optional[Any]:
        """消费者：接收消息"""
        if self.queue:
            return self.queue.popleft()
        return None

    def size(self) -> int:
        return len(self.queue)


# 使用示例
mq = SimpleMessageQueue()

# 生产者
mq.publish({'task': 'send_email', 'to': 'user@example.com'})
mq.publish({'task': 'generate_report', 'id': 1001})
mq.publish({'task': 'cleanup', 'target': '/tmp'})

# 消费者
while mq.size() > 0:
    msg = mq.consume()
    print(f"处理任务：{msg['data']['task']}")

    5.5 打印机任务队列 ✅¶

    from collections import deque
from dataclasses import dataclass
from typing import Optional

@dataclass
class PrintJob:
    job_id: int
    user: str
    document: str
    pages: int
    priority: int = 0  # 优先级（数字越大优先级越高）

class PrinterQueue:
    """
    打印机任务队列

    特点：
    - 先来先服务（FIFO）
    - 可支持优先级（高优先级插队）
    """

    def __init__(self):
        self.queue = deque()
        self.job_counter = 0

    def add_job(self, user: str, document: str, pages: int, priority: int = 0):
        """添加打印任务"""
        self.job_counter += 1
        job = PrintJob(self.job_counter, user, document, pages, priority)

        if priority > 0:
            # 高优先级：插入到队列前面
            self.queue.appendleft(job)
        else:
            # 普通优先级：加入队尾
            self.queue.append(job)

        print(f"✓ 任务 {job.job_id} 已加入队列：{user} - {document}")
        return job.job_id

    def process_next(self) -> Optional[PrintJob]:
        """处理下一个任务"""
        if self.queue:
            job = self.queue.popleft()
            print(f"🖨️  打印中：{job.document} ({job.pages}页)")
            return job
        print("队列为空")
        return None

    def queue_length(self) -> int:
        return len(self.queue)


# 使用示例
printer = PrinterQueue()

# 普通任务
printer.add_job("Alice", "report.pdf", 10)
printer.add_job("Bob", "notes.docx", 5)
printer.add_job("Charlie", "thesis.pdf", 50)

# 紧急任务（插队）
printer.add_job("Boss", "urgent_contract.pdf", 2, priority=1)

# 处理队列
print(f"\n队列长度：{printer.queue_length()}\n")
while printer.queue_length() > 0:
    printer.process_next()

    5.6 最近请求次数 ✅¶

    from collections import deque

class RecentCounter:
    """
    计算最近 3000ms 内的请求数
    LeetCode 933
    """

    def __init__(self):
        self.queue = deque()

    def ping(self, t: int) -> int:
        """
        在时间 t 添加新请求，返回过去 3000ms 内的请求数

        Args:
            t: 当前时间（毫秒）
        """
        self.queue.append(t)

        # 移除 3000ms 之前的请求
        while self.queue[0] < t - 3000:
            self.queue.popleft()

        return len(self.queue)


# 使用示例
counter = RecentCounter()
print(counter.ping(1))     # 1  → [1]
print(counter.ping(100))   # 2  → [1, 100]
print(counter.ping(3001))  # 3  → [1, 100, 3001]
print(counter.ping(3002))  # 3  → [100, 3001, 3002]

    6.1 用栈实现队列 ⭐⭐¶

    class MyQueue:
    """
    LeetCode 232：用两个栈实现队列

    思路：
    - stack_in：负责入队
    - stack_out：负责出队
    - 出队时如果 stack_out 为空，将 stack_in 全部倒入 stack_out
    """

    def __init__(self):
        self.stack_in = []
        self.stack_out = []

    def push(self, x: int):
        """入队"""
        self.stack_in.append(x)

    def pop(self) -> int:
        """出队"""
        self._dump()
        return self.stack_out.pop()

    def peek(self) -> int:
        """获取队头"""
        self._dump()
        return self.stack_out[-1]

    def empty(self) -> bool:
        return not self.stack_in and not self.stack_out

    def _dump(self):
        """将 stack_in 倒入 stack_out"""
        if not self.stack_out:
            while self.stack_in:
                self.stack_out.append(self.stack_in.pop())


# 使用示例
q = MyQueue()
q.push(1)
q.push(2)
q.push(3)
print(q.pop())   # 1
print(q.peek())  # 2
print(q.empty()) # False

    6.2 用队列实现栈 ⭐⭐¶

    from collections import deque

class MyStack:
    """
    LeetCode 225：用一个队列实现栈

    思路：
    - 入栈时，将新元素加入队尾
    - 然后将前面的所有元素依次出队再入队
    - 这样新元素就到了队头
    """

    def __init__(self):
        self.queue = deque()

    def push(self, x: int):
        """入栈"""
        size = len(self.queue)
        self.queue.append(x)
        # 将前面的元素重新加入队尾
        for _ in range(size):
            self.queue.append(self.queue.popleft())

    def pop(self) -> int:
        """出栈"""
        return self.queue.popleft()

    def top(self) -> int:
        """获取栈顶"""
        return self.queue[0]

    def empty(self) -> bool:
        return len(self.queue) == 0


# 使用示例
s = MyStack()
s.push(1)
s.push(2)
s.push(3)
print(s.pop())   # 3
print(s.top())   # 2

    6.3 设计循环队列 ⭐⭐¶

    class MyCircularQueue:
    """
    LeetCode 622：设计循环队列
    """

    def __init__(self, k: int):
        self.data = [None] * k
        self.capacity = k
        self.front = 0
        self.rear = 0
        self._size = 0

    def enQueue(self, value: int) -> bool:
        if self.isFull():
            return False
        self.data[self.rear] = value
        self.rear = (self.rear + 1) % self.capacity
        self._size += 1
        return True

    def deQueue(self) -> bool:
        if self.isEmpty():
            return False
        self.data[self.front] = None
        self.front = (self.front + 1) % self.capacity
        self._size -= 1
        return True

    def Front(self) -> int:
        if self.isEmpty():
            return -1
        return self.data[self.front]

    def Rear(self) -> int:
        if self.isEmpty():
            return -1
        return self.data[(self.rear - 1 + self.capacity) % self.capacity]

    def isEmpty(self) -> bool:
        return self._size == 0

    def isFull(self) -> bool:
        return self._size == self.capacity

    7. 队列 vs 栈 对比¶

    特性 队列（Queue） 栈（Stack）
    原则 FIFO（先进先出） LIFO（后进先出）
    操作端 两端（队头/队尾） 一端（栈顶）
