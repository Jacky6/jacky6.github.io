    🪄 什么是特殊方法？¶

    特殊方法（魔术方法）是以双下划线开头和结尾的方法，让对象表现得像内置类型。

    知识点：

    • 特殊方法（魔术方法）以 __ 开头和结尾
    • 让自定义对象支持内置语法：print()、len()、for、+ 等
    • ⭐面试重点：特殊方法是 Python 数据模型（data model）的核心
    class MyClass:
    def __init__(self):    # 构造方法
        pass

    def __str__(self):     # 字符串表示
        pass

    def __len__(self):     # len() 调用
        pass

    📝 字符串表示¶

    知识点：

    • str：面向用户的可读表示，print()/str() 调用
    • repr：面向开发者的表示，调试用；无 str 时 print 回退到 repr
    • ⭐面试重点：最佳实践是 repr 尽量能重建对象（如 Person('Alice', 25)）
    class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def __str__(self):
        """用户友好的字符串表示"""
        return f"{self.name}, {self.age}岁"

    def __repr__(self):
        """开发者友好的表示，用于调试"""
        return f"Person('{self.name}', {self.age})"

p = Person("Alice", 25)
print(str(p))   # Alice, 25 岁
print(repr(p))  # Person('Alice', 25)
print(p)        # Alice, 25 岁 (自动调用 __str__)

    📏 长度和迭代¶

    知识点：

    • len 支持 len()；getitem 支持 obj[i] 和切片
    • iter 支持 for 循环；contains 支持 in 判断
    • setitem/delitem 支持赋值和删除
    • ⭐面试重点：实现 getitem 后切片自动可用
    class MyList:
    def __init__(self, *items):
        self.items = list(items)

    def __len__(self):
        return len(self.items)

    def __getitem__(self, index):
        return self.items[index]

    def __setitem__(self, index, value):
        self.items[index] = value

    def __delitem__(self, index):
        del self.items[index]

    def __iter__(self):
        return iter(self.items)

    def __contains__(self, item):
        return item in self.items

lst = MyList(1, 2, 3, 4, 5)
print(len(lst))      # 5
print(lst[0])        # 1
print(lst[2:4])      # [3, 4] (切片也有效!)
print(3 in lst)      # True

for item in lst:
    print(item)      # 1 2 3 4 5

    ➕ 算术运算¶

    知识点：

    • add/sub/mul 对应 + - * 运算符
    • neg 取反、abs 取绝对值、eq 相等比较
    • 运算应返回新对象而非修改原对象
    • ⚠️易错点：不支持的操作数类型应返回 NotImplemented
    class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __add__(self, other):
        """向量加法"""
        return Vector(self.x + other.x, self.y + other.y)

    def __sub__(self, other):
        """向量减法"""
        return Vector(self.x - other.x, self.y - other.y)

    def __mul__(self, scalar):
        """标量乘法"""
        return Vector(self.x * scalar, self.y * scalar)

    def __eq__(self, other):
        """相等比较"""
        return self.x == other.x and self.y == other.y

    def __neg__(self):
        """取反"""
        return Vector(-self.x, -self.y)

    def __abs__(self):
        """向量长度"""
        return (self.x ** 2 + self.y ** 2) ** 0.5

    def __str__(self):
        return f"Vector({self.x}, {self.y})"

v1 = Vector(3, 4)
v2 = Vector(1, 2)

print(v1 + v2)   # Vector(4, 6)
print(v1 - v2)   # Vector(2, 2)
print(v1 * 2)    # Vector(6, 8)
print(v1 == v1)  # True
print(abs(v1))   # 5.0
print(-v1)       # Vector(-3, -4)

    📊 比较运算¶

    知识点：

    • eq/lt/le/gt/ge 对应 == < <= > >=
    • 实现比较后 sorted() 可直接排序对象列表
    • ⭐面试重点：sorted 默认升序依赖 lt
    class Student:
    def __init__(self, name, score):
        self.name = name
        self.score = score

    def __eq__(self, other):
        return self.score == other.score

    def __lt__(self, other):
        return self.score < other.score

    def __le__(self, other):
        return self.score <= other.score

    def __gt__(self, other):
        return self.score > other.score

    def __ge__(self, other):
        return self.score >= other.score

students = [
    Student("Alice", 85),
    Student("Bob", 92),
    Student("Charlie", 78),
]

# 排序
sorted_students = sorted(students)
for s in sorted_students:
    print(f"{s.name}: {s.score}")

# 比较
print(students[0] < students[1])  # True (85 < 92)

    📖 上下文管理器¶

    知识点：

    • enter 进入 with 块时调用，返回值绑定到 as 后的变量
    • exit 退出时调用，负责清理（如关闭文件）
    • exit 返回 True 可抑制异常
    • ⭐面试重点：with 语句的执行流程（enter → 块 → exit，异常也会执行 exit）
    class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None

    def __enter__(self):
        print(f"打开文件：{self.filename}")
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        print("关闭文件")
        self.file.close()
        # 返回 True 可以抑制异常
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello, World!")

    🗂️ 常用特殊方法¶

    对象创建和销毁¶

    知识点：

    • 生命周期：new 创建实例 → init 初始化 → del 销毁
    • 属性访问：getattr（属性不存在时）、setattr、getattribute（所有访问，慎用）
    • call 让实例像函数一样调用
    • int/float/bool/str 支持类型转换
    • ⚠️易错点：setattr 内直接 self.xxx 赋值会无限递归，要写 self.dict
    class MyClass:
    def __new__(cls, *args, **kwargs):
        """创建实例（很少重写）"""
        print("__new__ 被调用")
        return super().__new__(cls)

    def __init__(self, value):
        """初始化实例"""
        print("__init__ 被调用")
        self.value = value

    def __del__(self):
        """析构方法（对象销毁时调用）"""
        print("__del__ 被调用")

    属性访问¶

    class MyClass:
    def __init__(self):
        self._data = {}

    def __getattr__(self, name):
        """访问不存在的属性时调用"""
        return f"属性 {name} 不存在"

    def __setattr__(self, name, value):
        """设置属性时调用"""
        self._data[name] = value

    def __getattribute__(self, name):
        """访问任何属性时都调用（谨慎使用）"""
        return object.__getattribute__(self, name)

    可调用对象¶

    class Multiplier:
    def __init__(self, factor):
        self.factor = factor

    def __call__(self, x):
        """使实例可以像函数一样调用"""
        return x * self.factor

double = Multiplier(2)
print(double(5))  # 10 (调用 __call__)
print(double(10)) # 20

    类型转换¶

    class Temperature:
    def __init__(self, celsius):
        self.celsius = celsius

    def __int__(self):
        return int(self.celsius)

    def __float__(self):
        return float(self.celsius)

    def __str__(self):
        return f"{self.celsius}°C"

    def __bool__(self):
        return self.celsius != 0

temp = Temperature(36.5)
print(int(temp))    # 36
print(float(temp))  # 36.5
print(bool(temp))   # True

    🎯 实际例子¶

    1. 不可变列表¶

    知识点：

    • 不可变列表：内部用 tuple 存储，只实现读取类方法
    • 矩阵：getitem 接收 (row, col) 元组实现二维索引
    • 可调用计数器：call + int 组合
    • ⭐面试重点：按需实现最小方法集，不必全部实现
    class ImmutableList:
    def __init__(self, *items):
        self._data = tuple(items)

    def __getitem__(self, index):
        return self._data[index]

    def __len__(self):
        return len(self._data)

    def __iter__(self):
        return iter(self._data)

    def __contains__(self, item):
        return item in self._data

    def __repr__(self):
        return f"ImmutableList{self._data}"

    def __add__(self, other):
        return ImmutableList(*self._data, *other._data)

lst = ImmutableList(1, 2, 3)
print(lst[0])      # 1
print(len(lst))    # 3
print(2 in lst)    # True
new_lst = lst + ImmutableList(4, 5)
print(new_lst)     # ImmutableList(1, 2, 3, 4, 5)

    2. 多维向量¶

    class Matrix:
    def __init__(self, rows, cols, data=None):
        self.rows = rows
        self.cols = cols
        if data:
            self.data = data
        else:
            self.data = [[0] * cols for _ in range(rows)]

    def __getitem__(self, index):
        row, col = index
        return self.data[row][col]

    def __setitem__(self, index, value):
        row, col = index
        self.data[row][col] = value

    def __str__(self):
        return "\n".join(str(row) for row in self.data)

    def __repr__(self):
        return f"Matrix({self.rows}x{self.cols})"

    def __add__(self, other):
        if self.rows != other.rows or self.cols != other.cols:
            raise ValueError("矩阵维度不匹配")
        result = Matrix(self.rows, self.cols)
        for i in range(self.rows):
            for j in range(self.cols):
                result[i, j] = self[i, j] + other[i, j]
        return result

m1 = Matrix(2, 2, [[1, 2], [3, 4]])
m2 = Matrix(2, 2, [[5, 6], [7, 8]])
m3 = m1 + m2
print(m3)
# [6, 8]
# [10, 12]

    3. 计数器¶

    class Counter:
    def __init__(self, start=0):
        self.count = start

    def __call__(self, increment=1):
        """调用时增加计数"""
        self.count += increment
        return self.count

    def __int__(self):
        return self.count

    def __str__(self):
        return f"Counter({self.count})"

    def __repr__(self):
        return f"Counter({self.count})"

c = Counter(10)
print(c())      # 11
print(c(5))     # 16
print(int(c))   # 16

    ⚠️ 注意事项¶

    1. str vs repr¶

    知识点：

    • str 给用户看，repr 给开发者看，优先实现 repr
    • 算术运算返回新对象，不要修改操作数
    • 不支持的类型组合返回 NotImplemented（不是抛异常）
    • ⭐面试重点：NotImplemented 与 NotImplementedError 的区别
    # __str__ - 用户友好，用于 print
# __repr__ - 开发者友好，用于调试

# 最佳实践：__repr__ 应该能用于重建对象
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

    def __str__(self):
        return f"({self.x}, {self.y})"

    2. 不要修改不可变对象¶

    class MyInt:
    def __init__(self, value):
        self.value = value

    def __add__(self, other):
        # ✅ 返回新对象
        return MyInt(self.value + other.value)

    3. 处理 NotImplemented¶

    def __add__(self, other):
    if not isinstance(other, MyClass):
        return NotImplemented
    return MyClass(self.value + other.value)

    📝 练习题¶

    1. 实现一个 Fraction 类，支持分数运算
    2. 实现一个 Stack 类，支持 push/pop/len 操作
    3. 实现一个简单的 DataFrame 类，支持行列访问
    4. 实现一个 Timer 上下文管理器，计算代码块执行时间

    示例答案： 

    # 练习 1
class Fraction:
    def __init__(self, numerator, denominator=1):
        if denominator == 0:
            raise ValueError("分母不能为零")
        self.numerator = numerator
        self.denominator = denominator
        self._simplify()

    def _simplify(self):
        import math
        gcd = math.gcd(self.numerator, self.denominator)
        self.numerator //= gcd
        self.denominator //= gcd

    def __add__(self, other):
        num = self.numerator * other.denominator + other.numerator * self.denominator
        den = self.denominator * other.denominator
        return Fraction(num, den)

    def __mul__(self, other):
        return Fraction(self.numerator * other.numerator, 
                       self.denominator * other.denominator)

    def __str__(self):
        return f"{self.numerator}/{self.denominator}"

    def __repr__(self):
        return f"Fraction({self.numerator}, {self.denominator})"

f1 = Fraction(1, 2)
f2 = Fraction(1, 3)
print(f1 + f2)  # 5/6
print(f1 * f2)  # 1/6

    本章小结¶

    核心知识点¶

    ✅ 特殊方法让自定义对象支持内置语法：print、len、[]、for、+、with ✅ str 用户友好，repr 开发者友好且尽量可重建对象 ✅ 容器协议：len + getitem + iter + contains ✅ 上下文管理器：enter 返回值给 as，exit 保证清理执行 ✅ call 让实例可调用；new 创建、init 初始化 ✅ 运算符方法应返回新对象，类型不匹配返回 NotImplemented
