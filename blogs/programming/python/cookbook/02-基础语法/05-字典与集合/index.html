    📖 字典（Dictionary）¶

    知识点： - 字典是键值对集合，用 {} 或 dict() 创建，支持字典推导式 - ⭐ get(key, 默认值) 安全访问：键不存在返回默认值，d[key] 直接访问会抛 KeyError - 方法：keys()/values()/items() 视图，pop 删除并返回，update 批量更新，setdefault 设默认值 - ⭐ 推导式 {x: x**2 for x in range(5)}，交换键值 {v: k for k, v in d.items()}

    字典是键值对（key-value）的集合，用于存储映射关系。

    创建字典¶

    # 空字典
empty = {}
empty2 = dict()

# 直接创建
person = {
    "name": "Alice",
    "age": 25,
    "city": "Beijing"
}

# 使用 dict()
person2 = dict(name="Bob", age=30)

# 从键值对列表创建
pairs = [("a", 1), ("b", 2)]
d = dict(pairs)  # {'a': 1, 'b': 2}

# 字典推导式
squares = {x: x**2 for x in range(5)}
print(squares)  # {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

    访问元素¶

    person = {"name": "Alice", "age": 25, "city": "Beijing"}

# 通过键访问
print(person["name"])  # Alice
print(person["age"])   # 25

# 修改值
person["age"] = 26
person["city"] = "Shanghai"

# 添加新键值对
person["email"] = "alice@example.com"

# 使用 get() - 安全访问
print(person.get("name"))      # Alice
print(person.get("salary"))    # None (不会报错)
print(person.get("salary", 0)) # 0 (提供默认值)

# ❌ 直接访问不存在的键会报错
# print(person["salary"])  # KeyError!

    字典方法¶

    person = {"name": "Alice", "age": 25, "city": "Beijing"}

# 获取所有键
keys = person.keys()
print(keys)  # dict_keys(['name', 'age', 'city'])

# 获取所有值
values = person.values()
print(values)  # dict_values(['Alice', 25, 'Beijing'])

# 获取所有键值对
items = person.items()
print(items)  # dict_items([('name', 'Alice'), ('age', 25), ('city', 'Beijing')])

# 删除
age = person.pop("age")        # 删除并返回值
print(age)                     # 25

last = person.popitem()        # 删除并返回最后一个键值对
print(last)                    # ('city', 'Beijing')

del person["name"]             # 删除指定键

person.clear()                 # 清空字典

# 复制
copy1 = person.copy()
copy2 = dict(person)

# 更新
person = {"name": "Alice", "age": 25}
person.update({"age": 26, "city": "Beijing"})
print(person)  # {'name': 'Alice', 'age': 26, 'city': 'Beijing'}

# 设置默认值
person.setdefault("email", "default@example.com")
print(person)  # 如果 email 不存在则添加

# 检查键是否存在
print("name" in person)   # True
print("salary" in person) # False

    遍历字典¶

    person = {"name": "Alice", "age": 25, "city": "Beijing"}

# 遍历键
for key in person:
    print(key)

# 遍历键（显式）
for key in person.keys():
    print(key)

# 遍历值
for value in person.values():
    print(value)

# 遍历键值对
for key, value in person.items():
    print(f"{key}: {value}")

# 带索引遍历
for i, (key, value) in enumerate(person.items()):
    print(f"{i}. {key} = {value}")

    字典推导式¶

    # 基本用法
squares = {x: x**2 for x in range(5)}
print(squares)  # {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}

# 带条件
evens = {x: x**2 for x in range(10) if x % 2 == 0}
print(evens)  # {0: 0, 2: 4, 4: 16, 6: 36, 8: 64}

# 转换
names = ["Alice", "Bob", "Charlie"]
name_lengths = {name: len(name) for name in names}
print(name_lengths)  # {'Alice': 5, 'Bob': 3, 'Charlie': 7}

# 交换键值
original = {"a": 1, "b": 2, "c": 3}
swapped = {v: k for k, v in original.items()}
print(swapped)  # {1: 'a', 2: 'b', 3: 'c'}

    🔢 集合（Set）¶

    知识点： - 集合是无序、不重复的元素集合，set(列表) 自动去重 - ⚠️ {} 是空字典，空集合必须用 set() 创建 - ⭐ 集合运算：并集 |、交集 &、差集 -、对称差集 ^ - remove 不存在报错、discard 不存在不报错；frozenset 不可变、可作字典键

    集合是无序、不重复的元素集合。

    创建集合¶

    # 空集合（注意：{} 是空字典！）
empty = set()

# 创建集合
fruits = {"apple", "banana", "orange"}
numbers = {1, 2, 3, 4, 5}

# 从列表创建（自动去重）
nums = [1, 2, 2, 3, 3, 3]
unique = set(nums)
print(unique)  # {1, 2, 3}

# 集合推导式
squares = {x**2 for x in range(5)}
print(squares)  # {0, 1, 4, 9, 16}

    集合操作¶

    A = {1, 2, 3, 4, 5}
B = {4, 5, 6, 7, 8}

# 添加
A.add(6)
print(A)  # {1, 2, 3, 4, 5, 6}

# 删除
A.remove(6)      # 不存在会报错
A.discard(6)     # 不存在不会报错
elem = A.pop()   # 随机删除并返回一个元素
A.clear()        # 清空

# 并集
print(A | B)           # {1, 2, 3, 4, 5, 6, 7, 8}
print(A.union(B))      # 同上

# 交集
print(A & B)           # {4, 5}
print(A.intersection(B))  # 同上

# 差集
print(A - B)           # {1, 2, 3}
print(A.difference(B)) # 同上

# 对称差集（只在其中一个集合中的元素）
print(A ^ B)                 # {1, 2, 3, 6, 7, 8}
print(A.symmetric_difference(B))  # 同上

# 子集/超集检查
print({1, 2} <= A)       # True (子集)
print(A >= {1, 2})       # True (超集)
print({1, 2} < A)        # True (真子集)
print(A.isdisjoint(B))   # False (是否有交集)

    集合方法¶

    s = {1, 2, 3}

# 添加多个元素
s.update([4, 5, 6])
print(s)  # {1, 2, 3, 4, 5, 6}

# 交集更新
s = {1, 2, 3, 4}
s.intersection_update({3, 4, 5})
print(s)  # {3, 4}

# 差集更新
s = {1, 2, 3, 4}
s.difference_update({3, 4})
print(s)  # {1, 2}

# 对称差集更新
s = {1, 2, 3}
s.symmetric_difference_update({2, 3, 4})
print(s)  # {1, 4}

# 检查
print(2 in s)    # True
print(len(s))    # 集合大小

    冻结集合（Frozen Set）¶

    不可变的集合：

    frozen = frozenset([1, 2, 3, 4, 5])

# frozen.add(6)  # ❌ AttributeError

# 可用作字典的键
d = {frozen: "value"}

    🎯 实用例子¶

    知识点： - ⭐ 词频统计：dict.get(word, 0) + 1 累加，或 collections.Counter 一行搞定 - 去重并保持顺序：seen 集合 + 结果列表配合 - ⭐ 集合应用：& 求共同好友，| 求全部好友，- 求独有好友 - 配置合并：{default_config, user_config} 后者覆盖前者

    1. 词频统计¶

    text = "apple banana apple orange banana apple"
words = text.split()

# 方法 1: 使用字典
word_count = {}
for word in words:
    word_count[word] = word_count.get(word, 0) + 1
print(word_count)  # {'apple': 3, 'banana': 2, 'orange': 1}

# 方法 2: 使用 Counter（更简洁）
from collections import Counter
word_count = Counter(words)
print(word_count)  # Counter({'apple': 3, 'banana': 2, 'orange': 1})

    2. 去重并保持顺序¶

    def unique_preserve_order(items):
    seen = set()
    result = []
    for item in items:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

nums = [1, 2, 3, 2, 4, 1, 5, 3]
print(unique_preserve_order(nums))  # [1, 2, 3, 4, 5]

    3. 学生成绩管理¶

    students = {
    "Alice": {"math": 90, "english": 85, "science": 88},
    "Bob": {"math": 78, "english": 92, "science": 80},
    "Charlie": {"math": 95, "english": 88, "science": 92},
}

# 计算每个学生的平均分
for name, scores in students.items():
    avg = sum(scores.values()) / len(scores)
    print(f"{name}: {avg:.1f}")

# 找出每门课的最高分
for subject in ["math", "english", "science"]:
    best_student = max(students.keys(), key=lambda s: students[s][subject])
    best_score = students[best_student][subject]
    print(f"{subject}: {best_student} - {best_score}")

    4. 共同好友¶

    friends_alice = {"Bob", "Charlie", "David", "Eve"}
friends_bob = {"Alice", "Charlie", "Eve", "Frank"}

# 共同好友
common = friends_alice & friends_bob
print(f"共同好友：{common}")  # {'Charlie', 'Eve'}

# 所有好友
all_friends = friends_alice | friends_bob
print(f"所有好友：{all_friends}")

# Alice 独有的好友
alice_only = friends_alice - friends_bob
print(f"Alice 独有的好友：{alice_only}")

    5. 配置管理¶

    # 默认配置
default_config = {
    "host": "localhost",
    "port": 8080,
    "debug": False,
    "timeout": 30,
}

# 用户配置
user_config = {
    "port": 9000,
    "debug": True,

    ⚠️ 常见错误¶

    知识点： - ⚠️ 键必须可哈希：列表不能作键（unhashable），元组可以 - ⚠️ 遍历字典时删除键会抛 RuntimeError，先收集待删键再删 - ⚠️ copy() 是浅拷贝，嵌套值被修改会互相影响，需 deepcopy

    1. 可变类型不能作为键¶

    # ❌ 错误
d = {[1, 2]: "value"}  # TypeError: unhashable type: 'list'

# ✅ 正确
d = {(1, 2): "value"}  # 元组可以作为键

    2. 修改遍历中的字典¶

    # ❌ 错误
d = {"a": 1, "b": 2, "c": 3}
for key in d:
    if key == "b":
        del d[key]  # RuntimeError

# ✅ 正确
d = {"a": 1, "b": 2, "c": 3}
keys_to_delete = [k for k in d if k == "b"]
for key in keys_to_delete:
    del d[key]

    3. 浅拷贝问题¶

    # ❌ 错误
d1 = {"a": [1, 2, 3]}
d2 = d1.copy()
d2["a"].append(4)
print(d1)  # {'a': [1, 2, 3, 4]} (原字典也被修改!)

# ✅ 正确 - 深拷贝
import copy
d3 = copy.deepcopy(d1)

    📊 数据结构选择指南¶

    知识点： - ⭐ 选型口诀：有序可重复选列表，不可变选元组，键值映射选字典，去重/集合运算选集合 - 快速查找（成员判断）用字典/集合，比列表快得多 - 字典 3.7+ 保持插入顺序

    需求 选择
    有序、可重复 列表 (List)
    有序、不可变 元组 (Tuple)
    键值映射 字典 (Dict)
    去重、集合运算 集合 (Set)
    快速查找 字典/集合
    保持插入顺序 列表/字典 (3.7+)

    本章小结¶

    核心知识点¶

    ✅ 字典：键值对映射，get() 安全访问避免 KeyError，3.7+ 保持插入顺序
    ✅ 集合：无序去重，| & - ^ 四大集合运算，{} 是空字典不是空集合
    ✅ 键的限制：键必须可哈希，列表不能作键，元组可以
    ✅ 遍历安全：遍历字典/列表时不要直接删除元素
    ✅ 选型：映射用 dict、去重用 set、有序可重复用 list、固定数据用 tuple
