    5.5 文件操作¶

    本章知识点清单¶

    # 知识点 一句话结论 重要度
    1 open + with with open(...) 自动关闭文件，是打开文件的标准姿势 ⭐⭐⭐⭐
    2 文件模式 r/w/a/x 读写模式 + b 二进制 / t 文本 + + 读写 ⭐⭐⭐
    3 读取方式 read 全量、readline 单行、readlines 全部行，逐行迭代最省内存 ⭐⭐⭐
    4 写入方式 write 写字符串、writelines 写列表，'a' 模式追加 ⭐⭐
    5 pathlib.Path 面向对象的路径操作：read_text/write_text、exists/is_file、name/stem/suffix ⭐⭐⭐
    6 目录操作 iterdir 遍历、rglob 递归匹配、mkdir/rename/unlink、shutil.move/rmtree ⭐⭐
    7 实际场景 JSON 配置、批量重命名、备份、日志、CSV、大文件逐块处理 ⭐⭐⭐
    8 最佳实践 总是 with、显式 encoding、异常处理三件套 ⭐⭐⭐

    📁 打开文件¶

    知识点：

    • open() 返回文件对象，用完必须 close；with 语句自动关闭
    • ⭐ 面试重点：with open 是推荐方式——即使读写中抛异常文件也会被关闭
    # 基本方式
file = open("file.txt", "r")
content = file.read()
file.close()

# 推荐方式（自动关闭）
with open("file.txt", "r") as file:
    content = file.read()

    📝 文件模式¶

    知识点：

    • 基本模式：'r' 只读（默认）、'w' 覆盖写、'a' 追加、'x' 独占创建（存在则报错）
    • 修饰符：'b' 二进制、't' 文本（默认）、'+' 读写（如 'r+'）
    • ⚠️ 'w' 模式会清空原文件内容，覆盖写之前注意备份
    模式 说明
    'r' 只读（默认）
    'w' 写入（覆盖）
    'a' 追加
    'x' 独占创建
    'b' 二进制模式
    't' 文本模式（默认）
    '+' 读写模式 
    open("file.txt", "r")   # 只读
open("file.txt", "w")   # 写入（会覆盖）
open("file.txt", "a")   # 追加
open("file.txt", "r+")  # 读写
open("file.txt", "rb")  # 二进制读取

    📖 读取文件¶

    知识点：

    • read() 读全部或指定字符数，readline() 读一行，readlines() 读所有行成列表
    • ⭐ 面试重点：大文件用 for line in f 逐行迭代，避免一次性载入内存
    • ⚠️ 读取文本建议显式 encoding="utf-8"，避免平台默认编码差异
    with open("file.txt", "r", encoding="utf-8") as f:
    # 读取全部内容
    content = f.read()

    # 读取指定字节
    content = f.read(100)

    # 读取一行
    line = f.readline()

    # 读取所有行
    lines = f.readlines()

    # 逐行迭代（推荐）
    for line in f:
        print(line.strip())

    ✍️ 写入文件¶

    知识点：

    • write() 写字符串（需自行加 \n），writelines() 写字符串列表
    • 'w' 覆盖、'a' 追加，同一文件可多次 with 打开续写
    # 写入文件（覆盖）
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("Hello, World!\n")
    f.write("第二行\n")

    # 写入多行
    lines = ["line1\n", "line2\n", "line3\n"]
    f.writelines(lines)

# 追加文件
with open("output.txt", "a", encoding="utf-8") as f:
    f.write("追加的内容\n")

    📂 使用 pathlib¶

    知识点：

    • Path 用 / 运算符拼接路径；read_text()/write_text() 一步读写文本
    • 检查：exists()/is_file()/is_dir()；信息：name/stem/suffix/parent/absolute()
    • mkdir(parents=True, exist_ok=True) 递归创建目录；unlink() 删文件
    • ⭐ 面试重点：pathlib 是比字符串拼接更现代的路径处理方式
    from pathlib import Path

# 创建路径
p = Path("data/file.txt")
p = Path.home() / "documents" / "file.txt"

# 读取
content = p.read_text()
content = p.read_text(encoding="utf-8")

# 写入
p.write_text("Hello")
p.write_text("Hello", encoding="utf-8")

# 追加
with p.open("a") as f:
    f.write("追加")

# 检查
print(p.exists())     # 是否存在
print(p.is_file())    # 是否文件
print(p.is_dir())     # 是否目录

# 获取信息
print(p.name)         # 文件名
print(p.stem)         # 不含扩展名
print(p.suffix)       # 扩展名
print(p.parent)       # 父目录
print(p.absolute())   # 绝对路径

# 创建目录
p.parent.mkdir(parents=True, exist_ok=True)

# 删除
p.unlink()            # 删除文件
p.parent.rmdir()      # 删除空目录

    🗂️ 目录操作¶

    知识点：

    • iterdir() 列目录，rglob("*.py") 递归匹配文件
    • rename() 重命名/移动，shutil.move 跨目录移动，shutil.rmtree 递归删除目录
    • ⚠️ rmtree 是危险操作，递归删除前务必确认路径
    import os
from pathlib import Path

# 当前目录
cwd = Path.cwd()
print(cwd)

# 遍历目录
for item in Path(".").iterdir():
    print(item.name)

# 递归遍历
for item in Path(".").rglob("*.py"):
    print(item)

# 创建目录
Path("new_dir").mkdir()
Path("a/b/c").mkdir(parents=True)

# 重命名
Path("old.txt").rename("new.txt")

# 移动
import shutil
shutil.move("src.txt", "dst/dir/")

# 删除
Path("file.txt").unlink()           # 删除文件
shutil.rmtree("dir")                # 递归删除目录

    🎯 实际例子¶

    知识点：

    • JSON 配置：json.load/dump 配合 exists() 检查；dump 时 ensure_ascii=False 保留中文
    • 批量重命名：iterdir 遍历 + str.replace + rename；备份：时间戳文件名 + shutil.copy2
    • 日志：追加模式写入带时间戳的行；CSV：csv.DictReader/DictWriter 按字典读写
    • ⭐ 大文件：二进制按 chunk_size 逐块 read，或文本逐行迭代处理

    1. 读取配置文件¶

    import json
from pathlib import Path

def load_config(filename="config.json"):
    path = Path(filename)
    if not path.exists():
        return {}

    with path.open("r", encoding="utf-8") as f:
        return json.load(f)

def save_config(config, filename="config.json"):
    path = Path(filename)
    with path.open("w", encoding="utf-8") as f:
        json.dump(config, f, indent=2, ensure_ascii=False)

# 使用
config = load_config()
config["debug"] = True
save_config(config)

    2. 批量重命名¶

    from pathlib import Path

def batch_rename(directory, pattern, replacement):
    dir_path = Path(directory)

    for file in dir_path.iterdir():
        if file.is_file():
            new_name = file.name.replace(pattern, replacement)
            file.rename(file.parent / new_name)
            print(f"{file.name} → {new_name}")

# 使用
batch_rename("./photos", "IMG_", "Photo_")

    3. 文件备份¶

    import shutil
from pathlib import Path
from datetime import datetime

def backup_file(filename):
    src = Path(filename)
    if not src.exists():
        raise FileNotFoundError(f"文件不存在：{filename}")

    # 生成备份文件名
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_name = f"{src.stem}_backup_{timestamp}{src.suffix}"
    dst = src.parent / backup_name

    # 复制文件
    shutil.copy2(src, dst)
    print(f"备份完成：{dst}")
    return dst

# 使用
backup_file("important.docx")

    4. 日志记录¶

    from pathlib import Path
from datetime import datetime

class SimpleLogger:
    def __init__(self, log_file="app.log"):
        self.log_file = Path(log_file)
        self.log_file.parent.mkdir(parents=True, exist_ok=True)

    def _write(self, level, message):
        timestamp = datetime.now().isoformat()
        line = f"[{timestamp}] [{level}] {message}\n"

        with self.log_file.open("a", encoding="utf-8") as f:
            f.write(line)

    def info(self, message):
        self._write("INFO", message)

    def warning(self, message):
        self._write("WARNING", message)

    def error(self, message):
        self._write("ERROR", message)

# 使用
logger = SimpleLogger()
logger.info("应用启动")
logger.error("发生错误")

    5. CSV 处理¶

    import csv
from pathlib import Path

def read_csv(filename):
    path = Path(filename)
    with path.open("r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        return list(reader)

def write_csv(filename, data, fieldnames=None):
    path = Path(filename)
    if not data:
        return

    if fieldnames is None:
        fieldnames = data[0].keys()

    with path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(data)

# 使用
data = [
    {"name": "Alice", "age": 25},
    {"name": "Bob", "age": 30},
]
write_csv("users.csv", data)
users = read_csv("users.csv")

    6. 大文件处理¶

    def process_large_file(input_file, output_file, chunk_size=8192):
    """逐块处理大文件"""
    with open(input_file, "rb") as src, open(output_file, "wb") as dst:
        while True:
            chunk = src.read(chunk_size)
            if not chunk:
                break
            # 处理数据块
            processed = chunk.upper()  # 示例处理
            dst.write(processed)

# 逐行处理大文本文件
def process_large_text(input_file, output_file):
    with open(input_file, "r", encoding="utf-8") as src, \
         open(output_file, "w", encoding="utf-8") as dst:
        for line in src:
            processed = line.strip().upper()
            dst.write(processed + "\n")

    ⚠️ 最佳实践¶

    知识点：

    • 总是用 with 打开文件，避免忘记 close
    • 显式指定 encoding="utf-8"，不依赖平台默认编码
    • 先 exists() 检查或直接捕获 FileNotFoundError/PermissionError/UnicodeDecodeError

    1. 总是使用 with¶

    # ✅ 推荐
with open("file.txt") as f:
    content = f.read()

# ❌ 避免
f = open("file.txt")
content = f.read()
f.close()

    2. 指定编码¶

    # ✅ 推荐
with open("file.txt", "r", encoding="utf-8") as f:
    content = f.read()

# ❌ 避免（平台相关）
with open("file.txt") as f:
    content = f.read()

    3. 检查文件存在¶

    from pathlib import Path

path = Path("file.txt")
if path.exists():
    content = path.read_text()
else:
    print("文件不存在")

    4. 处理异常¶

    from pathlib import Path

try:
    content = Path("file.txt").read_text(encoding="utf-8")
except FileNotFoundError:
    print("文件不存在")
except PermissionError:
    print("没有权限")
except UnicodeDecodeError:
    print("编码错误")

    📝 练习题¶

    1. 创建一个函数，统计文本文件的行数、单词数、字符数
    2. 创建一个函数，合并多个文本文件
    3. 创建一个简单的文件浏览器，列出目录内容
    4. 实现一个函数，查找文件中包含特定字符串的行

    示例答案： 

    # 练习 1
from pathlib import Path

def count_file_stats(filename):
    path = Path(filename)
    content = path.read_text(encoding="utf-8")

    lines = content.splitlines()
    words = content.split()
    chars = len(content)

    return {
        "lines": len(lines),
        "words": len(words),
        "chars": chars
    }

stats = count_file_stats("README.md")
print(f"行数：{stats['lines']}, 单词：{stats['words']}, 字符：{stats['chars']}")

# 练习 4
def search_in_file(filename, search_term):
    path = Path(filename)
    results = []

    with path.open("r", encoding="utf-8") as f:
        for i, line in enumerate(f, 1):
            if search_term in line:
                results.append((i, line.strip()))

    return results

matches = search_in_file("README.md", "Python")
for line_num, content in matches:
    print(f"第{line_num}行：{content}")

    本章小结¶

    核心知识点¶

    ✅ 标准姿势：with open(...) as f，自动关闭文件 ✅ 文件模式：r/w/a/x + b/t/+，'w' 会覆盖原内容 ✅ 读取：逐行迭代 for line in f 最省内存，显式指定 encoding ✅ pathlib：Path 用 / 拼路径，read_text/write_text、exists/is_file、mkdir/unlink ✅ 目录操作：iterdir/rglob 遍历，shutil.move/rmtree 移动与递归删除 ✅ 大文件：按 chunk 逐块读或逐行迭代，避免一次性载入内存
