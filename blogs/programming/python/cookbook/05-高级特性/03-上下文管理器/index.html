    5.3 上下文管理器¶

    本章知识点清单¶

    # 知识点 一句话结论 重要度
    1 with 语句 with 自动完成资源的设置与清理，异常时也不泄漏资源 ⭐⭐⭐⭐
    2 类实现协议 实现 enter() 和 exit() 两个魔术方法即可成为上下文管理器 ⭐⭐⭐⭐
    3 exit 返回值 返回 True 抑制异常，返回 False/None 让异常继续传播 ⭐⭐⭐
    4 @contextmanager 生成器 + contextlib.contextmanager 装饰器，yield 前是进入逻辑、后是清理逻辑 ⭐⭐⭐
    5 典型应用场景 数据库连接、临时目录、锁、事务、性能分析、输出重定向 ⭐⭐⭐
    6 contextlib 工具 nullcontext 表示"无操作"上下文，ExitStack 动态管理任意多个上下文 ⭐⭐
    7 最佳实践 总是用 with 管理资源，清理逻辑放 finally，异常该重抛就重抛 ⭐⭐⭐

    🎯 什么是上下文管理器？¶

    知识点：

    • 上下文管理器用于设置和清理资源，配合 with 语句使用
    • 对比传统 try/finally 手动关闭：with 保证即使发生异常也会清理资源
    • ⭐ 面试重点：with open(...) 退出时自动关闭文件，无需手动 close

    上下文管理器用于设置和清理资源，使用 with 语句：

    # 传统方式
file = open("test.txt", "w")
try:
    file.write("Hello")
finally:
    file.close()

# 使用上下文管理器
with open("test.txt", "w") as file:
    file.write("Hello")
# 自动关闭文件

    📦 实现上下文管理器¶

    知识点：

    • 类实现协议：enter() 进入时执行并返回绑定给 as 后变量的值，exit() 退出时执行清理
    • exit(exc_type, exc_val, exc_tb) 接收异常三要素，无异常时均为 None
    • ⭐ 面试重点：exit 返回 True 会抑制异常，返回 False/None 则异常继续传播
    • ⚠️ enter 的返回值决定 as 变量绑定的内容，忘记 return 会绑定 None

    使用类实现¶

    class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None

    def __enter__(self):
        print(f"打开文件：{self.filename}")
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        print("关闭文件")
        self.file.close()
        # 返回 True 可以抑制异常
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello, World!")

    exit 参数说明¶

    def __exit__(self, exc_type, exc_val, exc_tb):
    """
    exc_type: 异常类型（如果没有异常则为 None）
    exc_val: 异常值
    exc_tb: 异常追踪信息
    """
    if exc_type is not None:
        print(f"发生异常：{exc_type.__name__}: {exc_val}")
    # 返回 True 抑制异常，False 或 None 传播异常
    return False

    🎨 使用 contextlib¶

    知识点：

    • @contextmanager 把生成器函数变成上下文管理器：yield 之前是进入逻辑，yield 值是 as 绑定值
    • ⚠️ 清理逻辑必须放在 try/finally 中，保证异常时也能执行
    • yield 可以不带值（如计时器），此时 as 变量无意义

    contextmanager 装饰器¶

    from contextlib import contextmanager

@contextmanager
def file_manager(filename, mode):
    f = open(filename, mode)
    try:
        yield f
    finally:
        f.close()

# 使用
with file_manager("test.txt", "w") as f:
    f.write("Hello")

    带参数的上下文管理器¶

    from contextlib import contextmanager

@contextmanager
def timer(name="操作"):
    import time
    start = time.time()
    try:
        yield
    finally:
        end = time.time()
        print(f"{name} 耗时：{end - start:.4f}秒")

with timer("计算"):
    total = sum(range(1000000))

with timer("排序"):
    sorted(range(1000000), reverse=True)

    🎯 实际例子¶

    知识点：

    • 数据库连接/事务：进入时开启，正常退出提交，异常时回滚后重新 raise
    • 临时目录：tempfile.mkdtemp() 创建，finally 中 shutil.rmtree() 清理
    • 锁管理：acquire 失败抛 TimeoutError，finally 中 release 保证解锁
    • 抑制异常：except 指定类型元组后 pass；重定向输出：替换 sys.stdout 并在 finally 还原

    1. 数据库连接¶

    from contextlib import contextmanager

@contextmanager
def database_connection(conn_string):
    """数据库连接上下文管理器"""
    conn = None
    try:
        print(f"连接到数据库：{conn_string}")
        # conn = create_connection(conn_string)
        conn = {"connected": True}
        yield conn
    except Exception as e:
        print(f"数据库错误：{e}")
        raise
    finally:
        if conn:
            print("关闭数据库连接")
            # conn.close()

# 使用
with database_connection("mysql://localhost/db") as conn:
    print(f"连接状态：{conn}")

    2. 临时目录¶

    from contextlib import contextmanager
import os
import tempfile
import shutil

@contextmanager
def temporary_directory():
    """临时目录上下文管理器"""
    temp_dir = tempfile.mkdtemp()
    try:
        yield temp_dir
    finally:
        shutil.rmtree(temp_dir)
        print(f"清理临时目录：{temp_dir}")

# 使用
with temporary_directory() as temp_dir:
    # 在 temp_dir 中创建临时文件
    test_file = os.path.join(temp_dir, "test.txt")
    with open(test_file, "w") as f:
        f.write("临时数据")
    print(f"临时文件：{test_file}")
# 退出时自动清理

    3. 锁管理¶

    from contextlib import contextmanager
import threading

class LockManager:
    def __init__(self):
        self.lock = threading.Lock()

    @contextmanager
    def acquire_lock(self, timeout=None):
        acquired = self.lock.acquire(timeout=timeout)
        if not acquired:
            raise TimeoutError("获取锁超时")
        try:
            yield
        finally:
            self.lock.release()

# 使用
manager = LockManager()
with manager.acquire_lock():
    # 临界区代码
    print("持有锁")

    4. 事务管理¶

    from contextlib import contextmanager

@contextmanager
def transaction(db):
    """事务上下文管理器"""
    try:
        print("开始事务")
        # db.begin()
        yield db
        print("提交事务")
        # db.commit()
    except Exception as e:
        print(f"回滚事务：{e}")
        # db.rollback()
        raise

# 使用
db = {"connected": True}
with transaction(db):
    # 数据库操作
    print("执行 SQL...")

    5. 性能分析¶

    from contextlib import contextmanager
import time
import cProfile
import pstats
from io import StringIO

@contextmanager
def profile(output_file=None):
    """性能分析上下文管理器"""
    profiler = cProfile.Profile()
    profiler.enable()
    try:
        yield
    finally:
        profiler.disable()

        if output_file:
            stats = pstats.Stats(profiler)
            stats.sort_stats('cumulative')
            stats.dump_stats(output_file)
            print(f"性能数据已保存到：{output_file}")
        else:
            s = StringIO()
            stats = pstats.Stats(profiler, stream=s)
            stats.sort_stats('cumulative')
            stats.print_stats(10)
            print(s.getvalue())

# 使用
with profile():
    # 要分析的代码
    total = 0
    for i in range(1000000):
        total += i

    6. 抑制异常¶

    from contextlib import contextmanager

@contextmanager
def suppress_exception(*exceptions):
    """抑制指定异常"""
    try:
        yield
    except exceptions:
        pass

# 使用
with suppress_exception(FileNotFoundError, ValueError):
    with open("not_exist.txt") as f:
        content = f.read()
    int("not a number")
# 异常被抑制，程序继续执行

    7. 重定向输出¶

    from contextlib import contextmanager
import sys
import os

@contextmanager
def redirect_output(filename):
    """重定向标准输出到文件"""
    original_stdout = sys.stdout
    with open(filename, "w") as f:
        sys.stdout = f
        try:
            yield
        finally:
            sys.stdout = original_stdout

# 使用
with redirect_output("output.txt"):
    print("这行会写入文件")
    print("不会显示在控制台")

print("这行显示在控制台")

    📚 contextlib 工具¶

    知识点：

    • nullcontext：可空的上下文管理器，用于"可能需要上下文也可能不需要"的分支场景
    • ExitStack：用 enter_context() 动态注册任意数量的上下文，退出时按后进先出顺序统一清理

    nullcontext¶

    from contextlib import nullcontext

# 可选的上下文管理器
def process_file(filename, use_temp=False):
    if use_temp:
        ctx = temporary_file()
    else:
        ctx = nullcontext(filename)

    with ctx as f:
        process(f)

    ExitStack¶

    from contextlib import ExitStack

# 动态管理多个上下文
def process_files(filenames):
    with ExitStack() as stack:
        files = [stack.enter_context(open(f)) for f in filenames]
        # 所有文件会在退出时自动关闭
        for f in files:
            process(f)

    ⚠️ 最佳实践¶

    知识点：

    • 总是使用 with 管理资源：手动 close 在出错时可能不会执行
    • 清理逻辑放 finally，确保任何路径都会释放资源
    • 异常处理要明确：捕获后要么处理完重新 raise，要么刻意抑制

    1. 总是使用 with¶

    # ✅ 推荐
with open("file.txt") as f:
    content = f.read()

# ❌ 避免
f = open("file.txt")
content = f.read()
f.close()  # 如果中间出错可能不会执行

    2. 清理资源¶

    @contextmanager
def resource_manager():
    resource = acquire_resource()
    try:
        yield resource
    finally:
        release_resource(resource)  # 确保清理

    3. 异常处理¶

    @contextmanager
def safe_operation():
    try:
        yield
    except SpecificError as e:
        handle_error(e)
        raise  # 重新抛出或返回 False

    示例答案： 

    # 练习 3
import os
from contextlib import contextmanager

@contextmanager
def temporary_env(**env_vars):
    """临时设置环境变量"""
    original = {}
    for key, value in env_vars.items():
        original[key] = os.environ.get(key)
        os.environ[key] = value

    try:
        yield
    finally:
        for key, original_value in original.items():
            if original_value is None:
                del os.environ[key]
            else:
                os.environ[key] = original_value

# 使用
print(os.environ.get("TEST_VAR"))  # None
with temporary_env(TEST_VAR="hello"):
    print(os.environ.get("TEST_VAR"))  # hello
print(os.environ.get("TEST_VAR"))  # None

    本章小结¶

    核心知识点¶

    ✅ with 语句：自动设置和清理资源，异常时也不泄漏 ✅ 类实现：enter 返回值绑定 as 变量，exit 返回 True 可抑制异常 ✅ @contextmanager：生成器式写法，yield 前进入、后清理，清理必须放 finally ✅ 典型场景：数据库连接/事务、临时目录、锁、性能分析、抑制异常、重定向输出 ✅ contextlib 工具：nullcontext（可空上下文）、ExitStack（动态管理多个上下文） ✅ 最佳实践：总是用 with，清理放 finally，异常该重抛就重抛
