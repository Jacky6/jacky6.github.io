    5.2 装饰器¶

    本章知识点清单¶

    # 知识点 一句话结论 重要度
    1 装饰器本质 装饰器是接收函数、返回新函数的函数，在不改原函数代码的前提下增强行为 ⭐⭐⭐⭐
    2 @ 语法糖 @decorator 等价于 func = decorator(func) ⭐⭐⭐⭐
    3 无参 vs 带参装饰器 无参两层嵌套，带参三层嵌套（外层接收参数） ⭐⭐⭐
    4 functools.wraps @wraps(func) 保留原函数的 name 和 doc，写装饰器必加 ⭐⭐⭐
    5 常用装饰器 日志、缓存 memoize、权限检查、重试、类型检查五大实用模板 ⭐⭐⭐
    6 类装饰器 类实现 call() 即可当装饰器，还能维护状态（如调用次数） ⭐⭐
    7 类方法装饰器 @staticmethod 免 self、@classmethod 收 cls、@property 把方法变属性 ⭐⭐⭐
    8 装饰器链 多个装饰器自下而上应用，执行时自上而下进入 ⭐⭐
    9 常见错误 忘记 return 结果、忘记 @wraps、带参装饰器少写一层 ⭐⭐⭐

    🎨 什么是装饰器？¶

    知识点： - 装饰器是"修改或增强函数行为的函数"，无需修改原函数代码 - 内部定义 wrapper 包裹原函数，在其前后插入额外逻辑，最后返回 wrapper - ⭐ 面试重点：装饰器本质是高阶函数（接收函数并返回函数）

    装饰器是修改或增强函数行为的函数，无需修改原函数代码。

    def my_decorator(func):
    def wrapper():
        print("函数执行前")
        func()
        print("函数执行后")
    return wrapper

@my_decorator
def say_hello():
    print("Hello!")

say_hello()
# 函数执行前
# Hello!
# 函数执行后

    📦 装饰器语法¶

    知识点： - @decorator 只是语法糖：等价于先定义函数再执行 func = decorator(func) - ⭐ 面试重点：@ 后面的表达式先求值，其返回值被调用并传入原函数

    # 以下两种写法等价
@decorator
def func():
    pass

# 等价于
def func():
    pass
func = decorator(func)

    🔧 装饰器基础¶

    知识点： - 无参装饰器两层嵌套：decorator(func) 内定义 wrapper 并返回 - 带参装饰器三层嵌套：最外层接收参数，返回真正的 decorator - wrapper 用 args, *kwargs 接收任意参数并转发，才能装饰任意签名的函数 - ⚠️ 忘记 @wraps(func) 会丢失原函数的 name 与 doc

    1. 无参数装饰器¶

    def timer(func):
    """计算函数执行时间"""
    import time

    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 耗时：{end - start:.4f}秒")
        return result

    return wrapper

@timer
def slow_function():
    import time
    time.sleep(1)

slow_function()  # 会打印耗时

    2. 带参数的装饰器¶

    def repeat(times):
    """重复执行函数"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")
# Hello, Alice!
# Hello, Alice!
# Hello, Alice!

    3. 保留函数信息¶

    from functools import wraps

def my_decorator(func):
    @wraps(func)  # 保留原函数的名称和文档
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

@my_decorator
def greet(name):
    """打招呼"""
    print(f"Hello, {name}!")

print(greet.__name__)  # greet (不是 wrapper)
print(greet.__doc__)   # 打招呼

    🎯 常用装饰器¶

    知识点： - 日志装饰器：打印调用参数与返回值；缓存装饰器：用 dict 存 args→结果，避免重复计算 - 权限检查：未认证抛 PermissionError；重试装饰器：失败后 delay 重试，末次失败 raise - 类型检查：依据 func.annotations 校验参数与返回值类型（跳过 'return' 键） - ⭐ 面试重点：memoize 让指数级递归的 fibonacci(50) 瞬间完成

    1. 日志装饰器¶

    from functools import wraps

def logger(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"调用 {func.__name__}，参数：{args}, {kwargs}")
        result = func(*args, **kwargs)
        print(f"{func.__name__} 返回：{result}")
        return result
    return wrapper

@logger
def add(a, b):
    return a + b

add(3, 5)
# 调用 add，参数：(3, 5), {}
# add 返回：8

    2. 缓存装饰器¶

    from functools import wraps

def memoize(func):
    """缓存函数结果"""
    cache = {}

    @wraps(func)
    def wrapper(*args):
        if args in cache:
            return cache[args]
        result = func(*args)
        cache[args] = result
        return result

    return wrapper

@memoize
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

print(fibonacci(50))  # 快速计算

    3. 权限检查¶

    from functools import wraps

def require_auth(func):
    @wraps(func)
    def wrapper(user, *args, **kwargs):
        if not user.get("is_authenticated"):
            raise PermissionError("需要登录")
        return func(user, *args, **kwargs)
    return wrapper

@require_auth
def view_profile(user):
    return f"欢迎，{user['name']}"

user = {"name": "Alice", "is_authenticated": True}
print(view_profile(user))  # 欢迎，Alice

    4. 重试装饰器¶

    from functools import wraps
import time

def retry(max_attempts=3, delay=1):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"失败，{delay}秒后重试...")
                    time.sleep(delay)
            return None
        return wrapper
    return decorator

@retry(max_attempts=3, delay=2)
def unstable_api():
    import random
    if random.random() < 0.7:
        raise ConnectionError("连接失败")
    return "成功"

    5. 类型检查¶

    from functools import wraps

def type_check(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        annotations = func.__annotations__

        # 检查参数类型
        for i, (name, expected) in enumerate(annotations.items()):
            if name == 'return':
                continue
            if i < len(args):
                if not isinstance(args[i], expected):
                    raise TypeError(f"{name} 应该是 {expected}")

        result = func(*args, **kwargs)

        # 检查返回类型
        if 'return' in annotations:
            if not isinstance(result, annotations['return']):
                raise TypeError(f"返回值应该是 {annotations['return']}")

        return result
    return wrapper

@type_check
def add(a: int, b: int) -> int:
    return a + b

print(add(3, 5))      # 8
# add(3, "5")         # TypeError!

    🏗️ 类装饰器¶

    知识点： - 类实现 call() 后其实例可像函数一样被调用，因此能当装饰器 - 相比闭包装饰器，类装饰器可以用实例属性维护状态（如 self.count 统计调用次数）

    class CountCalls:
    """统计函数调用次数"""

    def __init__(self, func):
        self.func = func
        self.count = 0

    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"第 {self.count} 次调用 {self.func.__name__}")
        return self.func(*args, **kwargs)

@CountCalls
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")  # 第 1 次调用 greet
greet("Bob")    # 第 2 次调用 greet

    🎯 类方法装饰器¶

    知识点： - @staticmethod：静态方法，不需要 self，可通过类直接调用 - @classmethod：类方法，第一个参数是 cls，可访问类本身 - @property：把方法包装成属性，obj.prop 无需括号访问 - ⭐ 面试重点：三者的区别是高频面试题

    class MyClass:
    @staticmethod
    def static_method():
        """静态方法 - 不需要 self"""
        return "static"

    @classmethod
    def class_method(cls):
        """类方法 - 需要 cls"""
        return f"class: {cls.__name__}"

    @property
    def property_method(self):
        """属性方法 - 像属性一样访问"""
        return "property"

print(MyClass.static_method())  # static
print(MyClass.class_method())   # class: MyClass
obj = MyClass()
print(obj.property_method)      # property

    🔗 装饰器链¶

    知识点： - 多个装饰器堆叠时自下而上应用：@decorator1 @decorator2 等价于 decorator1(decorator2(func)) - ⚠️ 执行顺序与应用顺序相反：先执行 decorator1 的前置逻辑，最后执行其后置逻辑

    def decorator1(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print("装饰器 1 - 前")
        result = func(*args, **kwargs)
        print("装饰器 1 - 后")
        return result
    return wrapper

def decorator2(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print("装饰器 2 - 前")
        result = func(*args, **kwargs)
        print("装饰器 2 - 后")
        return result
    return wrapper

@decorator1
@decorator2
def greet(name):
    print(f"Hello, {name}!")
    return name

greet("Alice")
# 装饰器 1 - 前
# 装饰器 2 - 前
# Hello, Alice!
# 装饰器 2 - 后
# 装饰器 1 - 后

    ⚠️ 常见错误¶

    知识点： - ⚠️ wrapper 忘记 return func(...) 会让被装饰函数返回 None - ⚠️ 忘记 @wraps 导致函数元信息丢失 - ⚠️ 带参装饰器必须三层嵌套，少写一层会把参数误当作 func

    1. 忘记返回结果¶

    # ❌ 错误
def bad_decorator(func):
    def wrapper(*args, **kwargs):
        print("before")
        func(*args, **kwargs)  # 忘记 return!
        print("after")
    return wrapper

# ✅ 正确
def good_decorator(func):
    def wrapper(*args, **kwargs):
        print("before")
        return func(*args, **kwargs)  # 返回结果
        print("after")
    return wrapper

    2. 忘记 @wraps¶

    # ❌ 错误
def decorator(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

# ✅ 正确
from functools import wraps

def decorator(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

    3. 装饰器参数错误¶

    # ❌ 错误
def repeat(times):
    def wrapper(func):  # 缺少一层
        # ...
    return wrapper

# ✅ 正确
def repeat(times):
    def decorator(func):
        def wrapper(*args, **kwargs):
            # ...
        return wrapper
    return decorator

    📝 练习题¶

    1. 创建一个装饰器，限制函数最大执行时间
    2. 创建一个装饰器，将函数结果缓存到文件
    3. 创建一个装饰器，记录函数的调用历史
    4. 使用装饰器实现单例模式

    示例答案： 

    # 练习 1
from functools import wraps
import signal

def timeout(seconds):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            def handler(signum, frame):
                raise TimeoutError("函数执行超时")

            signal.signal(signal.SIGALRM, handler)
            signal.alarm(seconds)

            try:
                result = func(*args, **kwargs)
            finally:
                signal.alarm(0)

            return result
        return wrapper
    return decorator

@timeout(5)
def slow_func():
    import time
    time.sleep(10)

# slow_func()  # TimeoutError!

# 练习 4
def singleton(cls):
    instances = {}
    @wraps(cls)
    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    return get_instance

@singleton
class Database:
    def __init__(self):
        print("初始化数据库连接")

db1 = Database()  # 初始化数据库连接
db2 = Database()  # 不会再次初始化
print(db1 is db2)  # True

    本章小结¶

    核心知识点¶

    ✅ 本质：装饰器 = 接收函数并返回新函数的高阶函数，@decorator 等价于 func = decorator(func) ✅ 两种形态：无参两层嵌套，带参三层嵌套；wrapper 用 args/*kwargs 透传参数 ✅ @wraps(func)：保留原函数 name/doc，写装饰器必加 ✅ 实用模板：日志、缓存 memoize、权限检查、重试、类型检查 ✅ 进阶：类装饰器（call 维护状态）、装饰器链（自下而上应用） ✅ 易错点：忘记 return、忘记 @wraps、带参装饰器少一层嵌套
