    🌐 urllib - 内置 HTTP 库¶

    知识点：

    • urllib.request 是标准库自带的 HTTP 客户端，无需安装
    • 返回的响应是字节流，要 decode 成字符串；状态码用 response.status（旧版本 response.code）
    • ⚠️ 只适合简单请求，复杂场景（session、重试）用 requests

    基本请求¶

    import urllib.request
import urllib.parse

# GET 请求
response = urllib.request.urlopen("https://www.example.com")
html = response.read().decode("utf-8")
print(html)

# 带参数的 GET
params = urllib.parse.urlencode({"q": "python", "page": 1})
url = f"https://search.example.com?{params}"
response = urllib.request.urlopen(url)

# POST 请求
data = urllib.parse.urlencode({"username": "user", "password": "pass"}).encode()
request = urllib.request.Request("https://api.example.com/login", data=data)
response = urllib.request.urlopen(request)

    设置请求头¶

    import urllib.request

url = "https://api.example.com/data"
headers = {
    "User-Agent": "Mozilla/5.0",
    "Accept": "application/json",
}

request = urllib.request.Request(url, headers=headers)
response = urllib.request.urlopen(request)
data = response.read().decode("utf-8")

    处理错误¶

    import urllib.request
import urllib.error

try:
    response = urllib.request.urlopen("https://api.example.com/data")
    data = response.read().decode("utf-8")
except urllib.error.HTTPError as e:
    print(f"HTTP 错误：{e.code}")
except urllib.error.URLError as e:
    print(f"网络错误：{e.reason}")

    📦 requests - 第三方 HTTP 库¶

    知识点：

    • ⭐ requests 是 Python HTTP 请求的事实标准（需 pip install requests）
    • 响应对象：response.text（文本）、response.json()（解析 JSON）、response.status_code
    • ⚠️ requests 默认不抛异常（404/500 也正常返回），用 raise_for_status() 检查
    • ⚠️ 所有请求加 timeout 参数，防止服务器无响应时永久阻塞

    安装¶

    pip install requests

    基本使用¶

    import requests

# GET 请求
response = requests.get("https://api.github.com/users/github")
print(response.status_code)  # 200
print(response.json())       # 解析 JSON

# 带参数
params = {"q": "python", "sort": "stars"}
response = requests.get("https://api.github.com/search/repositories", params=params)

# POST 请求
data = {"title": "New Post", "body": "Content"}
response = requests.post("https://api.example.com/posts", json=data)

# 带请求头
headers = {"Authorization": "Bearer token123"}
response = requests.get("https://api.example.com/protected", headers=headers)

    会话管理¶

    import requests

# 登录

    "username": "user",

    🎯 实际例子¶

    知识点：

    • 调用第三方 API 的通用套路：组装参数 → GET → 检查状态 → 解析 JSON → 异常兜底
    • API key 等敏感信息不要硬编码，用环境变量

    1. 天气查询¶

    import requests

def get_weather(city, api_key):
    """查询天气"""
    url = "https://api.openweathermap.org/data/2.5/weather"
    params = {
        "q": city,
        "appid": api_key,
        "units": "metric",
        "lang": "zh_cn"
    }

    response = requests.get(url, params=params)
    response.raise_for_status()
    data = response.json()

    return {
        "city": data["name"],
        "temperature": data["main"]["temp"],
        "description": data["weather"][0]["description"],
        "humidity": data["main"]["humidity"],
    }

# 使用

    2. 汇率转换¶

    import requests

def get_exchange_rates(base="USD"):
    """获取汇率"""
    url = f"https://api.exchangerate-api.com/v4/latest/{base}"
    response = requests.get(url)
    response.raise_for_status()
    return response.json()["rates"]

def convert_currency(amount, from_currency, to_currency):
    """货币转换"""
    rates = get_exchange_rates(from_currency)
    rate = rates.get(to_currency)
    if not rate:
        raise ValueError(f"不支持的货币：{to_currency}")
    return amount * rate

# 使用
# result = convert_currency(100, "USD", "CNY")
# print(f"100 USD = {result:.2f} CNY")

    import requests
from bs4 import BeautifulSoup

def scrape_titles(url):
    """爬取网页标题"""
    response = requests.get(url)
    response.raise_for_status()

    soup = BeautifulSoup(response.text, "html.parser")

    titles = []
    for tag in soup.find_all("h2"):
        titles.append(tag.get_text(strip=True))

    return titles

    4. API 客户端¶

    import requests
from typing import Optional, Dict, Any

class APIClient:
    def __init__(self, base_url: str, api_key: Optional[str] = None):
        self.base_url = base_url
        self.session = requests.Session()

        if api_key:
            self.session.headers.update({"Authorization": f"Bearer {api_key}"})

    def request(self, method: str, endpoint: str, **kwargs) -> Dict[str, Any]:
        """发送请求"""
        url = f"{self.base_url}/{endpoint}"
        response = self.session.request(method, url, **kwargs)
        response.raise_for_status()
        return response.json()

    def get(self, endpoint: str, **kwargs) -> Dict[str, Any]:
        return self.request("GET", endpoint, **kwargs)

    def post(self, endpoint: str, **kwargs) -> Dict[str, Any]:
        return self.request("POST", endpoint, **kwargs)

    def put(self, endpoint: str, **kwargs) -> Dict[str, Any]:
        return self.request("PUT", endpoint, **kwargs)

    def delete(self, endpoint: str, **kwargs) -> Dict[str, Any]:
        return self.request("DELETE", endpoint, **kwargs)

# 使用 GitHub API
github = APIClient("https://api.github.com")

    5. 批量 URL 检查¶

    import requests
from concurrent.futures import ThreadPoolExecutor

def check_url(url):
    """检查 URL 状态"""
    try:
        response = requests.get(url, timeout=5)
        return {
            "url": url,
            "status": response.status_code,
            "ok": response.ok
        }
    except requests.exceptions.RequestException as e:
        return {
            "url": url,
            "status": None,
            "ok": False,
            "error": str(e)
        }

def check_urls(urls, max_workers=5):
    """批量检查 URL"""
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        results = list(executor.map(check_url, urls))
    return results

# 使用
urls = [
    "https://www.google.com",
    "https://www.github.com",
    "https://invalid-url.example.com",
]

results = check_urls(urls)
for result in results:
    status = "✅" if result["ok"] else "❌"
    print(f"{status} {result['url']}: {result.get('status', '错误')}")

    6. RSS 订阅阅读器¶

    import requests
import xml.etree.ElementTree as ET

def parse_rss(url):
    """解析 RSS 订阅"""
    response = requests.get(url)
    response.raise_for_status()

    root = ET.fromstring(response.content)

    # 查找 channel
    channel = root.find("channel")
    title = channel.find("title").text
    link = channel.find("link").text

    # 解析条目
    items = []
    for item in channel.findall("item"):
        items.append({
            "title": item.find("title").text,
            "link": item.find("link").text,
            "description": item.find("description").text if item.find("description") is not None else "",
            "pub_date": item.find("pubDate").text if item.find("pubDate") is not None else "",
        })

    return {
        "title": title,
        "link": link,
        "items": items
    }

    ⚠️ 最佳实践¶

    知识点：

    • ⭐ 所有请求必须设 timeout：requests 默认无限等待，服务器无响应时程序会永久阻塞
    • timeout 单值同时作用于连接（connect）和读取（read）；传元组 (connect, read) 分开设置
    • ⚠️ timeout 不是整体下载时限——是"两次收到数据之间"的最大等待，超时抛 Timeout 异常
    • 错误处理按具体异常分层捕获，最后用 RequestException 兜底

    1. 设置超时¶

    # ✅ 推荐
response = requests.get(url, timeout=10)

# ✅ 连接超时 3 秒、读取超时 27 秒（元组形式）
response = requests.get(url, timeout=(3.05, 27))

# ❌ 避免
response = requests.get(url)  # 可能永远等待

# 超时异常：ConnectTimeout / ReadTimeout，都是 Timeout 的子类
try:
    requests.get(url, timeout=1)
except requests.exceptions.Timeout:
    print("连接或读取超时")

    2. 错误处理¶

    try:
    response = requests.get(url)
    response.raise_for_status()  # 检查 HTTP 错误
    data = response.json()
except requests.exceptions.HTTPError as e:
    print(f"HTTP 错误：{e}")
except requests.exceptions.ConnectionError as e:
    print(f"连接错误：{e}")
except requests.exceptions.Timeout as e:
    print(f"超时：{e}")
except requests.exceptions.RequestException as e:
    print(f"请求错误：{e}")

    3. 使用会话¶

    # ✅ 推荐
with requests.Session() as session:
    session.headers.update({"User-Agent": "MyApp"})
    response = session.get(url)

# ❌ 避免（每次创建新连接）
for url in urls:
    response = requests.get(url)

    🔁 重试与指数退避¶

    知识点：

    • 网络抖动（超时、连接重置、5xx）很常见，自动重试能显著提高成功率
    • 指数退避：第 n 次重试前等待 delay × 2^n 秒，避免失败时疯狂重试打垮服务器
    • ⭐ 面试重点：为什么用指数退避而不是固定间隔——大量客户端同时重试会形成"重试风暴"
    • 进阶：HTTPAdapter + urllib3 的 Retry 可在连接层自动重试指定状态码
    import time
import requests

def fetch_with_retry(url, max_attempts=3, delay=1.0):
    """带指数退避的请求"""
    for attempt in range(max_attempts):
        try:
            response = requests.get(url, timeout=5)
            response.raise_for_status()
            return response
        except (requests.exceptions.ConnectionError,
                requests.exceptions.Timeout) as e:
            if attempt == max_attempts - 1:
                raise  # 重试用尽，抛出最后一次异常
            wait = delay * (2 ** attempt)  # 1s, 2s, 4s...
            print(f"第 {attempt + 1} 次失败，{wait:.1f}s 后重试...")
            time.sleep(wait)

# 使用（与 5.2 装饰器章的 retry 装饰器思路一致）
# response = fetch_with_retry("https://api.example.com/data")

    连接层自动重试（无需手写循环）：

    from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

# 最多重试 3 次，遇到 429/5xx 状态码自动重试，间隔按 backoff_factor 指数增长
retry = Retry(total=3, backoff_factor=0.5,
              status_forcelist=[429, 500, 502, 503, 504])
session = requests.Session()
session.mount("https://", HTTPAdapter(max_retries=retry))
session.mount("http://", HTTPAdapter(max_retries=retry))

    🤖 爬虫礼貌性¶

    知识点：

    • 爬虫必须设 User-Agent 表明身份（requests 默认 UA 容易被网站拒绝）
    • 每次请求之间 time.sleep 限速，别把目标服务器打挂；并发爬取更要克制
    • 先看目标网站的 robots.txt（如 https://example.com/robots.txt），遵守爬取规则
    • ⚠️ 商用爬虫还要注意服务条款与数据合规问题
    import time
import requests

session = requests.Session()
session.headers.update({
    "User-Agent": "MyBot/1.0 (+https://mysite.example/bot)"
})

def crawl_politely(urls, delay=1.0):
    """礼貌爬取：限速 + 错误容忍"""
    results = []
    for url in urls:
        try:
            response = session.get(url, timeout=10)
            response.raise_for_status()
            results.append((url, response.text))
        except requests.exceptions.RequestException as e:
            print(f"跳过 {url}: {e}")
        time.sleep(delay)  # 限速，做个好爬虫
    return results

    # 练习 3
import requests
from pathlib import Path

def download_images(urls, output_dir="downloads"):
    """批量下载图片"""
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)

    for i, url in enumerate(urls):
        try:
            response = requests.get(url, timeout=10)
            response.raise_for_status()

            # 生成文件名
            filename = f"image_{i+1}.jpg"
            filepath = output_path / filename

            # 保存
            with open(filepath, "wb") as f:
                f.write(response.content)

            print(f"下载：{filename}")
        except Exception as e:
            print(f"下载失败 {url}: {e}")

# 使用
urls = [
    "https://picsum.photos/200/300",
    "https://picsum.photos/400/300",
    "https://picsum.photos/300/400",
]
download_images(urls)

    本章小结¶

    核心知识点¶

    ✅ urllib：标准库 HTTP 客户端，适合简单请求
    ✅ requests：第三方事实标准，API 友好，复杂场景首选
    ✅ Session：复用连接、保持 cookie，多次请求必用
    ✅ 状态码：raise_for_status() 主动检查 4xx/5xx
    ✅ timeout：所有请求必须设超时（单值或 (connect, read) 元组），防止永久阻塞
    ✅ 健壮性：指数退避重试应对网络抖动，爬虫要设 UA、限速、遵守 robots.txt
