查找算法（Searching）知识点

1. 概述

查找是在数据集合中寻找特定元素的操作。

查找算法对比

算法 时间复杂度（平均） 空间复杂度 适用场景 数据要求
顺序查找 O(n) O(1) 任意 无序
二分查找 O(log n) O(1) 数组 有序 ⭐
哈希查找 O(1) O(n) 任意 哈希表
二叉搜索树 O(log n) O(n) 动态 BST
分块查找 O(√n) O(1) 分块有序 分块有序

2. 顺序查找

def linear_search(arr: List[int], target: int) -> int:
    """
    顺序查找

    原理：从头到尾遍历，找到返回索引，否则返回 -1

    时间复杂度：O(n)
    空间复杂度：O(1)
    """
    for i, val in enumerate(arr):
        if val == target:
            return i
    return -1


# 示例
arr = [64, 34, 25, 12, 22, 11, 90]
print(linear_search(arr, 25))  # 2
print(linear_search(arr, 100)) # -1

3. 二分查找 ⭐⭐⭐

3.1 基础二分查找

def binary_search(arr: List[int], target: int) -> int:
    """
    二分查找（闭区间版本）

    原理：每次将查找范围缩小一半

    时间复杂度：O(log n)
    空间复杂度：O(1)

    前提：数组必须有序
    """
    left, right = 0, len(arr) - 1

    while left <= right:  # 闭区间，所以是 <=
        mid = left + (right - left) // 2  # 防止溢出

        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1

    return -1


# 示例
arr = [1, 3, 5, 7, 9, 11, 13]
print(binary_search(arr, 7))   # 3
print(binary_search(arr, 10))  # -1

查找过程（找 7）：
[1, 3, 5, 7, 9, 11, 13]
 ↑           ↑        ↑
left       mid      right

arr[mid]=7，找到！

3.2 二分查找（开区间版本）

def binary_search_open(arr: List[int], target: int) -> int:
    """
    二分查找（开区间版本）

    区间定义：[left, right)
    """
    left, right = 0, len(arr)

    while left < right:  # 开区间，所以是 <
        mid = left + (right - left) // 2

        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid

    return -1

3.3 二分查找递归版本

def binary_search_recursive(arr: List[int], target: int) -> int:
    """二分查找 - 递归"""
    def helper(left, right):
        if left > right:
            return -1

        mid = left + (right - left) // 2

        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            return helper(mid + 1, right)
        else:
            return helper(left, mid - 1)

    return helper(0, len(arr) - 1)

4. 二分查找进阶 ⭐⭐⭐

4.1 查找第一个等于 target 的位置

def binary_search_first(arr: List[int], target: int) -> int:
    """
    查找第一个等于 target 的位置

    LeetCode 34：在排序数组中查找元素的第一个位置
    """
    left, right = 0, len(arr) - 1
    result = -1

    while left <= right:
        mid = left + (right - left) // 2

        if arr[mid] == target:
            result = mid
            right = mid - 1  # 继续在左边找
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1

    return result


# 示例
arr = [1, 2, 2, 2, 3, 4, 5]
print(binary_search_first(arr, 2))  # 1（第一个 2 的位置）

4.3 查找第一个大于等于 target 的位置

def binary_search_lower_bound(arr: List[int], target: int) -> int:
    """
    查找第一个大于等于 target 的位置（lower_bound）

    如果不存在，返回应该插入的位置
    """
    left, right = 0, len(arr) - 1
    result = len(arr)

    while left <= right:
        mid = left + (right - left) // 2

        if arr[mid] >= target:
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result


# 示例
arr = [1, 3, 5, 7, 9]
print(binary_search_lower_bound(arr, 4))  # 2（5 的位置）
print(binary_search_lower_bound(arr, 6))  # 3（7 的位置）
print(binary_search_lower_bound(arr, 10)) # 5（应该插入的位置）

5. 二分查找应用场景 ⭐⭐⭐

5.1 搜索旋转排序数组

def search(nums: List[int], target: int) -> int:
    """
    LeetCode 33：搜索旋转排序数组

    思路：至少有一半是有序的，判断 target 是否在有序的那一半
    """
    left, right = 0, len(nums) - 1

    while left <= right:
        mid = left + (right - left) // 2

        if nums[mid] == target:
            return mid

        # 判断哪一半有序
        if nums[left] <= nums[mid]:  # 左半有序
            if nums[left] <= target < nums[mid]:
                right = mid - 1
            else:
                left = mid + 1
        else:  # 右半有序
            if nums[mid] < target <= nums[right]:
                left = mid + 1
            else:
                right = mid - 1

    return -1


# 示例
nums = [4, 5, 6, 7, 0, 1, 2]
print(search(nums, 0))  # 4
print(search(nums, 3))  # -1

5.2 寻找峰值

def findPeakElement(nums: List[int]) -> int:
    """
    LeetCode 162：寻找峰值

    思路：二分查找，往高的方向走
    """
    left, right = 0, len(nums) - 1

    while left < right:
        mid = left + (right - left) // 2

        if nums[mid] > nums[mid + 1]:
            # 峰值在左边（包括 mid）
            right = mid
        else:
            # 峰值在右边
            left = mid + 1

    return left


# 示例
nums = [1, 2, 3, 1]
print(findPeakElement(nums))  # 2（索引 2，值为 3）

5.3 寻找两个正序数组的中位数

def findMedianSortedArrays(nums1: List[int], nums2: List[int]) -> float:
    """
    LeetCode 4：寻找两个正序数组的中位数

    思路：二分查找第 k 小的数
    时间复杂度：O(log(m+n))
    """
    def find_kth(k):
        i = j = 0
        while True:
            if i == len(nums1):
                return nums2[j + k - 1]
            if j == len(nums2):
                return nums1[i + k - 1]
            if k == 1:
                return min(nums1[i], nums2[j])

            # 比较中间值
            mid = k // 2
            ni = i + mid - 1 if i + mid - 1 < len(nums1) else len(nums1) - 1
            nj = j + mid - 1 if j + mid - 1 < len(nums2) else len(nums2) - 1

            if nums1[ni] < nums2[nj]:
                k -= ni - i + 1
                i = ni + 1
            else:
                k -= nj - j + 1
                j = nj + 1

    total = len(nums1) + len(nums2)
    if total % 2 == 1:
        return find_kth(total // 2 + 1)
    else:
        return (find_kth(total // 2) + find_kth(total // 2 + 1)) / 2

5.4 爱吃香蕉的珂珂

import math

def minEatingSpeed(piles: List[int], h: int) -> int:
    """
    LeetCode 875：爱吃香蕉的珂珂

    思路：二分答案
    在 [1, max(piles)] 范围内二分搜索最小速度
    """
    def can_finish(speed):
        hours = sum(math.ceil(p / speed) for p in piles)
        return hours <= h

    left, right = 1, max(piles)

    while left < right:
        mid = left + (right - left) // 2
        if can_finish(mid):
            right = mid
        else:
            left = mid + 1

    return left


# 示例
piles = [3, 6, 7, 11]
h = 8
print(minEatingSpeed(piles, h))  # 4

# 两数之和（哈希查找经典应用）
def twoSum(nums: List[int], target: int) -> List[int]:
    seen = {}
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:  # O(1) 查找
            return [seen[complement], i]
        seen[num] = i
    return []

7. 分块查找

import math

def block_search(arr: List[int], target: int, block_size: int = None) -> int:
    """
    分块查找

    原理：
    1. 将数组分成若干块
    2. 块间有序，块内无序
    3. 先确定 target 在哪一块，再在块内顺序查找

    时间复杂度：O(√n)
    """
    n = len(arr)
    if block_size is None:
        block_size = int(math.sqrt(n))

    # 找到可能包含 target 的块
    block_idx = -1
    for i in range(0, n, block_size):
        block_end = min(i + block_size, n)
        if arr[i] <= target <= arr[block_end - 1]:
            block_idx = i
            break

    if block_idx == -1:
        return -1

    # 在块内顺序查找
    block_end = min(block_idx + block_size, n)
    for i in range(block_idx, block_end):
        if arr[i] == target:
            return i

    return -1


# 示例（块间有序）
arr = [1, 3, 5, 2, 4, 6, 7, 9, 8]
print(block_search(arr, 4))  # 4

8. 经典算法题

8.1 搜索二维矩阵

def searchMatrix(matrix: List[List[int]], target: int) -> bool:
    """
    LeetCode 74：搜索二维矩阵

    思路：将二维矩阵视为一维有序数组进行二分查找
    """
    if not matrix or not matrix[0]:
        return False

    m, n = len(matrix), len(matrix[0])
    left, right = 0, m * n - 1

    while left <= right:
        mid = left + (right - left) // 2
        row, col = mid // n, mid % n
        mid_val = matrix[row][col]

        if mid_val == target:
            return True
        elif mid_val < target:
            left = mid + 1
        else:
            right = mid - 1

    return False

8.2 在排序数组中查找元素的第一个和最后一个位置

def searchRange(nums: List[int], target: int) -> List[int]:
    """
    LeetCode 34：在排序数组中查找元素的第一个和最后一个位置
    """
    def find_left():
        left, right = 0, len(nums) - 1
        result = -1
        while left <= right:
            mid = left + (right - left) // 2
            if nums[mid] >= target:
                if nums[mid] == target:
                    result = mid
                right = mid - 1
            else:
                left = mid + 1
        return result

    def find_right():
        left, right = 0, len(nums) - 1
        result = -1
        while left <= right:
            mid = left + (right - left) // 2
            if nums[mid] <= target:
                if nums[mid] == target:
                    result = mid
                left = mid + 1
            else:
                right = mid - 1
        return result

    return [find_left(), find_right()]


# 示例
nums = [5, 7, 7, 8, 8, 10]
target = 8
print(searchRange(nums, target))  # [3, 4]

9. 本章小结

核心知识点

✅ 顺序查找：O(n)，简单通用
✅ 二分查找：O(log n)，有序数组 ⭐
✅ 哈希查找：O(1)，哈希表
✅ 二分进阶：lower_bound、upper_bound

重点掌握

  1. 二分查找模板：闭区间 vs 开区间
  2. 二分变体：查找第一个/最后一个
  3. 二分答案：求最小值/最大值问题
  4. 旋转数组：判断哪一半有序
