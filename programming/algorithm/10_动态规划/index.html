动态规划（Dynamic Programming）知识点

1. 概述

动态规划是一种将复杂问题分解为更小子问题的优化方法。

核心思想

  • 分治思想：大问题 → 小问题
  • 记忆化：保存已计算的结果，避免重复计算
  • 状态转移：从小问题推导大问题
动态规划 vs 递归 vs 分治

递归：自顶向下，可能有重复计算
    f(5)
   /    \
 f(4)   f(3)
 /  \   /  \
f(3) f(2) ...  ← f(3) 重复计算！

动态规划：自底向上或记忆化，无重复
    f(5)
      ↑
    f(4)
      ↑
    f(3) ← 只计算一次
      ↑
    f(2)

适用场景

✅ 最优子结构：大问题的最优解包含子问题的最优解
✅ 重叠子问题：子问题会被重复计算
✅ 无后效性：未来与过去无关，只与现在有关

2. DP 解题步骤 ⭐⭐⭐

步骤 1：定义状态（dp 数组的含义）
步骤 2：确定状态转移方程
步骤 3：初始化边界条件
步骤 4：确定遍历顺序
步骤 5：举例推导 dp 数组

示例：爬楼梯

# 问题：每次可以爬 1 或 2 阶，爬到 n 阶有多少种方法？

# 步骤 1：定义状态
# dp[i] = 爬到第 i 阶的方法数

# 步骤 2：状态转移方程
# dp[i] = dp[i-1] + dp[i-2]
# 最后一步可以是爬 1 阶或 2 阶

# 步骤 3：初始化
# dp[0] = 1（地面）
# dp[1] = 1（1 阶）

# 步骤 4：遍历顺序
# 从前往后

# 步骤 5：推导
# dp[0]=1, dp[1]=1, dp[2]=2, dp[3]=3, dp[4]=5...

def climbStairs(n: int) -> int:
    if n <= 1:
        return 1

    dp = [0] * (n + 1)
    dp[0] = 1
    dp[1] = 1

    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]

    return dp[n]


# 空间优化（只需要前两个状态）
def climbStairs_optimized(n: int) -> int:
    if n <= 1:
        return 1

    prev2, prev1 = 1, 1
    for i in range(2, n + 1):
        curr = prev1 + prev2
        prev2 = prev1
        prev1 = curr

    return prev1

3. 基础 DP 问题 ⭐

3.1 斐波那契数列

def fib(n: int) -> int:
    """
    LeetCode 509：斐波那契数列

    F(0) = 0, F(1) = 1
    F(n) = F(n-1) + F(n-2)
    """
    if n <= 1:
        return n

    dp = [0] * (n + 1)
    dp[0] = 0
    dp[1] = 1

    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]

    return dp[n]


# 空间优化版
def fib_optimized(n: int) -> int:
    if n <= 1:
        return n

    prev2, prev1 = 0, 1
    for _ in range(2, n + 1):
        curr = prev1 + prev2
        prev2 = prev1
        prev1 = curr

    return prev1

3.2 最小路径和

def minPathSum(grid: List[List[int]]) -> int:
    """
    LeetCode 64：最小路径和

    从左上到右下，每次只能向下或向右
    求路径上数字和的最小值
    """
    m, n = len(grid), len(grid[0])

    # dp[i][j] = 到达 (i,j) 的最小路径和
    dp = [[0] * n for _ in range(m)]

    # 初始化
    dp[0][0] = grid[0][0]

    # 第一行（只能从左边来）
    for j in range(1, n):
        dp[0][j] = dp[0][j - 1] + grid[0][j]

    # 第一列（只能从上面来）
    for i in range(1, m):
        dp[i][0] = dp[i - 1][0] + grid[i][0]

    # 填充其余位置
    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j]

    return dp[m - 1][n - 1]


# 空间优化（只需要一行）
def minPathSum_optimized(grid: List[List[int]]) -> int:
    m, n = len(grid), len(grid[0])
    dp = [0] * n

    dp[0] = grid[0][0]
    for j in range(1, n):
        dp[j] = dp[j - 1] + grid[0][j]

    for i in range(1, m):
        dp[0] += grid[i][0]
        for j in range(1, n):
            dp[j] = min(dp[j], dp[j - 1]) + grid[i][j]

    return dp[n - 1]

3.3 不同路径

def uniquePaths(m: int, n: int) -> int:
    """
    LeetCode 62：不同路径

    从左上到右下，每次只能向下或向右
    求有多少种不同路径
    """
    # dp[i][j] = 到达 (i,j) 的路径数
    dp = [[1] * n for _ in range(m)]

    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = dp[i - 1][j] + dp[i][j - 1]

    return dp[m - 1][n - 1]


# 空间优化
def uniquePaths_optimized(m: int, n: int) -> int:
    dp = [1] * n

    for _ in range(1, m):
        for j in range(1, n):
            dp[j] += dp[j - 1]

    return dp[n - 1]

4. 背包问题 ⭐⭐⭐

4.1 0-1 背包

def knapsack_01(weights: List[int], values: List[int], capacity: int) -> int:
    """
    0-1 背包问题

    n 个物品，每个重量 weight[i]，价值 value[i]
    背包容量 capacity，求最大价值
    每个物品只能选 0 次或 1 次

    dp[i][j] = 前 i 个物品，容量为 j 时的最大价值
    """
    n = len(weights)
    dp = [[0] * (capacity + 1) for _ in range(n + 1)]

    for i in range(1, n + 1):
        for j in range(capacity + 1):
            # 不选第 i 个物品
            dp[i][j] = dp[i - 1][j]

            # 选第 i 个物品（如果容量够）
            if j >= weights[i - 1]:
                dp[i][j] = max(dp[i][j], 
                              dp[i - 1][j - weights[i - 1]] + values[i - 1])

    return dp[n][capacity]


# 空间优化（一维数组，倒序遍历）
def knapsack_01_optimized(weights: List[int], values: List[int], capacity: int) -> int:
    n = len(weights)
    dp = [0] * (capacity + 1)

    for i in range(n):
        # 倒序遍历（避免重复使用同一物品）
        for j in range(capacity, weights[i] - 1, -1):
            dp[j] = max(dp[j], dp[j - weights[i]] + values[i])

    return dp[capacity]


# 示例
weights = [2, 3, 4, 5]
values = [3, 4, 5, 6]
capacity = 8
print(knapsack_01(weights, values, capacity))  # 10

0-1 背包关键点：
- 倒序遍历容量：保证每个物品只用一次
- 状态转移：dp[j] = max(dp[j], dp[j-w] + v)

4.2 完全背包

def knapsack_complete(weights: List[int], values: List[int], capacity: int) -> int:
    """
    完全背包问题

    每个物品可以选无限次

    关键：正序遍历容量（允许重复使用）
    """
    n = len(weights)
    dp = [0] * (capacity + 1)

    for i in range(n):
        # 正序遍历（可以重复使用）
        for j in range(weights[i], capacity + 1):
            dp[j] = max(dp[j], dp[j - weights[i]] + values[i])

    return dp[capacity]


# 示例（零钱兑换问题）
def coinChange(coins: List[int], amount: int) -> int:
    """
    LeetCode 322：零钱兑换

    完全背包应用
    求凑成金额的最少硬币数
    """
    INF = float('inf')
    dp = [INF] * (amount + 1)
    dp[0] = 0

    for coin in coins:
        for j in range(coin, amount + 1):
            dp[j] = min(dp[j], dp[j - coin] + 1)

    return dp[amount] if dp[amount] != INF else -1


# 示例
coins = [1, 2, 5]
amount = 11
print(coinChange(coins, amount))  # 3 (5+5+1)

4.3 组合总和 IV

def combinationSum4(nums: List[int], target: int) -> int:
    """
    LeetCode 377：组合总和 IV

    求排列数（考虑顺序）

    关键：外层遍历容量，内层遍历物品
    """
    dp = [0] * (target + 1)
    dp[0] = 1

    # 外层容量，内层物品 → 排列数
    for j in range(1, target + 1):
        for num in nums:
            if j >= num:
                dp[j] += dp[j - num]

    return dp[target]


# 对比：求组合数（不考虑顺序）
# 外层物品，内层容量 → 组合数

5. 子序列问题 ⭐⭐⭐

5.1 最长递增子序列（LIS）

def lengthOfLIS(nums: List[int]) -> int:
    """
    LeetCode 300：最长递增子序列

    dp[i] = 以 nums[i] 结尾的最长递增子序列长度
    """
    if not nums:
        return 0

    n = len(nums)
    dp = [1] * n

    for i in range(1, n):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)

    return max(dp)


# 贪心 + 二分优化 O(n log n)
def lengthOfLIS_optimized(nums: List[int]) -> int:
    tails = []

    for num in nums:
        # 二分查找第一个 >= num 的位置
        left, right = 0, len(tails)
        while left < right:
            mid = (left + right) // 2
            if tails[mid] < num:
                left = mid + 1
            else:
                right = mid

        if left == len(tails):
            tails.append(num)
        else:
            tails[left] = num

    return len(tails)


# 示例
nums = [10, 9, 2, 5, 3, 7, 101, 18]
print(lengthOfLIS(nums))  # 4 ([2, 3, 7, 18])

5.2 最长公共子序列（LCS）

def longestCommonSubsequence(text1: str, text2: str) -> int:
    """
    LeetCode 1143：最长公共子序列

    dp[i][j] = text1[:i] 和 text2[:j] 的 LCS 长度
    """
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])

    return dp[m][n]


# 示例
text1 = "abcde"
text2 = "ace"
print(longestCommonSubsequence(text1, text2))  # 3 ("ace")

5.3 最长回文子序列

def longestPalindromeSubseq(s: str) -> int:
    """
    LeetCode 516：最长回文子序列

    dp[i][j] = s[i:j+1] 的最长回文子序列长度
    """
    n = len(s)
    dp = [[0] * n for _ in range(n)]

    # 单个字符都是回文
    for i in range(n):
        dp[i][i] = 1

    # 从下往上，从左往右
    for i in range(n - 2, -1, -1):
        for j in range(i + 1, n):
            if s[i] == s[j]:
                dp[i][j] = dp[i + 1][j - 1] + 2
            else:
                dp[i][j] = max(dp[i + 1][j], dp[i][j - 1])

    return dp[0][n - 1]


# 示例
s = "bbbab"
print(longestPalindromeSubseq(s))  # 4 ("bbbb")

6. 区间 DP ⭐

6.1 矩阵链乘法

def matrixChainOrder(p: List[int]) -> int:
    """
    矩阵链乘法

    p = [30, 35, 15, 5, 10, 20, 25]
    表示矩阵：A1(30×35), A2(35×15), A3(15×5), A4(5×10), A5(10×20), A6(20×25)

    求最小标量乘法次数
    """
    n = len(p) - 1
    dp = [[0] * n for _ in range(n)]

    # length = 链长度
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')

            for k in range(i, j):
                cost = dp[i][k] + dp[k + 1][j] + p[i] * p[k + 1] * p[j + 1]
                dp[i][j] = min(dp[i][j], cost)

    return dp[0][n - 1]

6.2 戳气球

def maxCoins(nums: List[int]) -> int:
    """
    LeetCode 312：戳气球

    区间 DP
    """
    # 添加边界
    nums = [1] + nums + [1]
    n = len(nums)
    dp = [[0] * n for _ in range(n)]

    # 区间长度
    for length in range(2, n):
        for left in range(n - length):
            right = left + length

            # 枚举最后戳破的气球
            for k in range(left + 1, right):
                coins = nums[left] * nums[k] * nums[right]
                coins += dp[left][k] + dp[k][right]
                dp[left][right] = max(dp[left][right], coins)

    return dp[0][n - 1]

7. 经典算法题

7.1 打家劫舍

def rob(nums: List[int]) -> int:
    """
    LeetCode 198：打家劫舍

    不能偷相邻的房屋，求最大金额
    """
    if not nums:
        return 0
    if len(nums) == 1:
        return nums[0]

    dp = [0] * len(nums)
    dp[0] = nums[0]
    dp[1] = max(nums[0], nums[1])

    for i in range(2, len(nums)):
        dp[i] = max(dp[i - 1], dp[i - 2] + nums[i])

    return dp[-1]


# 空间优化
def rob_optimized(nums: List[int]) -> int:
    prev2, prev1 = 0, 0

    for num in nums:
        curr = max(prev1, prev2 + num)
        prev2 = prev1
        prev1 = curr

    return prev1

7.2 编辑距离

def minDistance(word1: str, word2: str) -> int:
    """
    LeetCode 72：编辑距离

    可以将一个单词转换为另一个单词的操作：
    - 插入一个字符
    - 删除一个字符
    - 替换一个字符

    求最少操作数
    """
    m, n = len(word1), len(word2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    # 初始化
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j

    # 状态转移
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if word1[i - 1] == word2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = min(
                    dp[i - 1][j] + 1,      # 删除
                    dp[i][j - 1] + 1,      # 插入
                    dp[i - 1][j - 1] + 1   # 替换
                )

    return dp[m][n]

类型 特征 代表题
