树（Tree）知识点

1. 定义

树是一种非线性的层次数据结构，由 n (n≥0) 个有限节点组成。

  • 根节点（Root）：没有父节点的节点
  • 叶子节点（Leaf）：没有子节点的节点
  • 边（Edge）：连接两个节点的线段
  • 路径（Path）：从一个节点到另一个节点的边序列
        A       ← 根节点 (Root)
       / \
      B   C     ← 内部节点
     / \   \
    D   E   F   ← 叶子节点 (Leaf)

    A 是 B、C 的父节点
    B、C 是 A 的子节点
    D、E 是兄弟节点

重要概念

术语 说明
节点的度 节点拥有的子树个数
树的度 树中所有节点的度的最大值
节点的深度 从根到该节点的边数
节点的高度 从该节点到叶子节点的最长路径边数
树的高度 根节点的高度

2. 二叉树（Binary Tree）⭐

2.1 定义

每个节点最多有两个子节点（左子树、右子树）。

      1
     / \
    2   3
   / \   \
  4   5   6

2.2 特殊二叉树

类型 说明 示例
满二叉树 所有非叶子节点都有 2 个子节点，所有叶子在同一层 完美三角形
完全二叉树 除最后一层外，其他层都满，最后一层从左到右填充 堆
二叉搜索树 左 < 根 < 右 有序查找 
满二叉树：          完全二叉树：        二叉搜索树：
     1                   1                  4
    / \                 / \                / \
   2   3               2   3              2   6
  / \ / \             / \ /               / \ / \
 4  5 6  7           4  5 6              1  3 5  7

2.3 二叉树性质

  1. 第 i 层最多有 2^(i-1) 个节点（i≥1）
  2. 深度为 k 的树最多有 2^k - 1 个节点
  3. 叶子节点数 = 度为 2 的节点数 + 1
  4. 完全二叉树深度 = ⌊log₂n⌋ + 1

3.1 节点定义

class TreeNode:
    """二叉树节点"""
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

3.2 二叉树类

from typing import List, Optional
from collections import deque

class BinaryTree:
    """二叉树类"""

    def __init__(self):
        self.root = None

    def build_from_level_order(self, values: List[Optional[int]]):
        """
        从层序遍历数组构建二叉树

        示例：[1, 2, 3, None, 4, 5, 6]
             1
            / \
           2   3
            \ / \
            4 5  6
        """
        if not values:
            return None

        self.root = TreeNode(values[0])
        queue = deque([self.root])
        i = 1

        while queue and i < len(values):
            node = queue.popleft()

            # 左子节点
            if i < len(values) and values[i] is not None:
                node.left = TreeNode(values[i])
                queue.append(node.left)
            i += 1

            # 右子节点
            if i < len(values) and values[i] is not None:
                node.right = TreeNode(values[i])
                queue.append(node.right)
            i += 1

        return self.root

    def level_order(self) -> List[List[int]]:
        """层序遍历"""
        if not self.root:
            return []

        result = []
        queue = deque([self.root])

        while queue:
            level_size = len(queue)
            level = []

            for _ in range(level_size):
                node = queue.popleft()
                level.append(node.val)

                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)

            result.append(level)

        return result


# 使用示例
tree = BinaryTree()
tree.build_from_level_order([1, 2, 3, None, 4, 5, 6])
print(tree.level_order())  # [[1], [2, 3], [4, 5, 6]]

4. 树的遍历 ⭐⭐⭐

4.1 深度优先遍历（DFS）

       1
      / \
     2   3
    / \   \
   4   5   6

（1）前序遍历（Preorder）：根 → 左 → 右

def preorder_recursive(root: Optional[TreeNode]) -> List[int]:
    """前序遍历 - 递归"""
    result = []

    def dfs(node):
        if not node:
            return
        result.append(node.val)      # 根
        dfs(node.left)               # 左
        dfs(node.right)              # 右

    dfs(root)
    return result

# 结果：[1, 2, 4, 5, 3, 6]


def preorder_iterative(root: Optional[TreeNode]) -> List[int]:
    """前序遍历 - 迭代（栈）"""
    if not root:
        return []

    result = []
    stack = [root]

    while stack:
        node = stack.pop()
        result.append(node.val)

        # 先右后左（栈是 LIFO）
        if node.right:
            stack.append(node.right)
        if node.left:
            stack.append(node.left)

    return result

# 结果：[1, 2, 4, 5, 3, 6]

（2）中序遍历（Inorder）：左 → 根 → 右

def inorder_recursive(root: Optional[TreeNode]) -> List[int]:
    """中序遍历 - 递归"""
    result = []

    def dfs(node):
        if not node:
            return
        dfs(node.left)               # 左
        result.append(node.val)      # 根
        dfs(node.right)              # 右

    dfs(root)
    return result

# 结果：[4, 2, 5, 1, 3, 6]


def inorder_iterative(root: Optional[TreeNode]) -> List[int]:
    """中序遍历 - 迭代（栈）"""
    result = []
    stack = []
    current = root

    while current or stack:
        # 一直向左
        while current:
            stack.append(current)
            current = current.left

        # 弹出栈顶
        current = stack.pop()
        result.append(current.val)

        # 转向右子树
        current = current.right

    return result

# 结果：[4, 2, 5, 1, 3, 6]

（3）后序遍历（Postorder）：左 → 右 → 根

def postorder_recursive(root: Optional[TreeNode]) -> List[int]:
    """后序遍历 - 递归"""
    result = []

    def dfs(node):
        if not node:
            return
        dfs(node.left)               # 左
        dfs(node.right)              # 右
        result.append(node.val)      # 根

    dfs(root)
    return result

# 结果：[4, 5, 2, 6, 3, 1]


def postorder_iterative(root: Optional[TreeNode]) -> List[int]:
    """后序遍历 - 迭代（双栈）"""
    if not root:
        return []

    stack1 = [root]
    stack2 = []
    result = []

    while stack1:
        node = stack1.pop()
        stack2.append(node)

        if node.left:
            stack1.append(node.left)
        if node.right:
            stack1.append(node.right)

    while stack2:
        result.append(stack2.pop().val)

    return result

# 结果：[4, 5, 2, 6, 3, 1]

4.2 广度优先遍历（BFS）：层序遍历

def level_order(root: Optional[TreeNode]) -> List[List[int]]:
    """
    层序遍历
    LeetCode 102
    """
    if not root:
        return []

    result = []
    queue = deque([root])

    while queue:
        level_size = len(queue)
        level = []

        for _ in range(level_size):
            node = queue.popleft()
            level.append(node.val)

            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)

        result.append(level)

    return result

# 结果：[[1], [2, 3], [4, 5, 6]]

4.3 遍历对比

遍历方式 顺序 应用场景
前序 根左右 复制树、序列化
中序 左根右 BST 获取有序序列
后序 左右根 计算树高、删除树
层序 从上到下 BFS、层级处理

5. 二叉搜索树（BST）⭐⭐

5.1 定义

  • 左子树所有节点 < 根节点
  • 右子树所有节点 > 根节点
  • 左右子树也是 BST
      4
     / \
    2   6
   / \ / \
  1  3 5  7

5.2 BST 操作

class BST:
    """二叉搜索树"""

    def __init__(self):
        self.root = None

    def search(self, val: int) -> Optional[TreeNode]:
        """查找"""
        current = self.root
        while current:
            if val == current.val:
                return current
            elif val < current.val:
                current = current.left
            else:
                current = current.right
        return None

    def insert(self, val: int):
        """插入"""
        if not self.root:
            self.root = TreeNode(val)
            return

        current = self.root
        while True:
            if val < current.val:
                if not current.left:
                    current.left = TreeNode(val)
                    break
                current = current.left
            else:
                if not current.right:
                    current.right = TreeNode(val)
                    break
                current = current.right

    def delete(self, val: int):
        """删除（LeetCode 450）"""
        self.root = self._delete_node(self.root, val)

    def _delete_node(self, node: Optional[TreeNode], val: int) -> Optional[TreeNode]:
        if not node:
            return None

        if val < node.val:
            node.left = self._delete_node(node.left, val)
        elif val > node.val:
            node.right = self._delete_node(node.right, val)
        else:
            # 找到要删除的节点
            if not node.left:
                return node.right
            if not node.right:
                return node.left

            # 有两个子节点：找右子树最小值
            min_node = self._find_min(node.right)
            node.val = min_node.val
            node.right = self._delete_node(node.right, min_node.val)

        return node

    def _find_min(self, node: TreeNode) -> TreeNode:
        while node.left:
            node = node.left
        return node

    def inorder(self) -> List[int]:
        """中序遍历（得到有序序列）"""
        result = []

        def dfs(node):
            if not node:
                return
            dfs(node.left)
            result.append(node.val)
            dfs(node.right)

        dfs(self.root)
        return result


# 使用示例
bst = BST()
for val in [4, 2, 6, 1, 3, 5, 7]:
    bst.insert(val)

print(bst.search(5))      # TreeNode(5)
print(bst.inorder())      # [1, 2, 3, 4, 5, 6, 7]
bst.delete(4)
print(bst.inorder())      # [1, 2, 3, 5, 6, 7]

6. 堆（Heap）⭐⭐

6.1 定义

堆是一种完全二叉树，分为：

  • 大顶堆：父节点 ≥ 子节点
  • 小顶堆：父节点 ≤ 子节点
大顶堆：              小顶堆：
     9                    1
    / \                  / \
   7   8                3   2
  / \                  / \
 5   6                5   4

6.2 Python 实现

import heapq

# 小顶堆（默认）
heap = []
heapq.heappush(heap, 3)
heapq.heappush(heap, 1)
heapq.heappush(heap, 2)
print(heapq.heappop(heap))  # 1（最小）

# 大顶堆（取反）
max_heap = []
heapq.heappush(max_heap, -3)
heapq.heappush(max_heap, -1)
heapq.heappush(max_heap, -2)
print(-heapq.heappop(max_heap))  # 3（最大）

# 前 K 个元素
nums = [1, 3, 5, 2, 4, 6]
top3 = heapq.nlargest(3, nums)  # [6, 5, 4]
bottom3 = heapq.nsmallest(3, nums)  # [1, 2, 3]

6.3 手写堆

class MinHeap:
    """小顶堆"""

    def __init__(self):
        self.heap = []

    def push(self, val: int):
        """插入"""
        self.heap.append(val)
        self._sift_up(len(self.heap) - 1)

    def pop(self) -> int:
        """弹出最小值"""
        if not self.heap:
            raise IndexError("Heap is empty")

        result = self.heap[0]
        last = self.heap.pop()

        if self.heap:
            self.heap[0] = last
            self._sift_down(0)

        return result

    def _sift_up(self, index: int):
        """上浮"""
        while index > 0:
            parent = (index - 1) // 2
            if self.heap[index] < self.heap[parent]:
                self.heap[index], self.heap[parent] = self.heap[parent], self.heap[index]
                index = parent
            else:
                break

    def _sift_down(self, index: int):
        """下沉"""
        n = len(self.heap)
        while True:
            smallest = index
            left = 2 * index + 1
            right = 2 * index + 2

            if left < n and self.heap[left] < self.heap[smallest]:
                smallest = left
            if right < n and self.heap[right] < self.heap[smallest]:
                smallest = right

            if smallest != index:
                self.heap[index], self.heap[smallest] = self.heap[smallest], self.heap[index]
                index = smallest
            else:
                break

    def min(self) -> int:
        if not self.heap:
            raise IndexError("Heap is empty")
        return self.heap[0]

    def __len__(self):
        return len(self.heap)

7. 经典算法题

7.1 二叉树的最大深度 ✅

def maxDepth(root: Optional[TreeNode]) -> int:
    """
    LeetCode 104：二叉树的最大深度

    方法 1：DFS 递归
    """
    if not root:
        return 0
    return 1 + max(maxDepth(root.left), maxDepth(root.right))


def maxDepth_bfs(root: Optional[TreeNode]) -> int:
    """方法 2：BFS 层序遍历"""
    if not root:
        return 0

    depth = 0
    queue = deque([root])

    while queue:
        depth += 1
        for _ in range(len(queue)):
            node = queue.popleft()
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)

    return depth

7.2 验证二叉搜索树 ✅⭐

def isValidBST(root: Optional[TreeNode]) -> bool:
    """
    LeetCode 98：验证二叉搜索树

    思路：中序遍历应该是严格递增的
    """
    def dfs(node, min_val, max_val):
        if not node:
            return True

        if not (min_val < node.val < max_val):
            return False

        return (dfs(node.left, min_val, node.val) and
                dfs(node.right, node.val, max_val))

    return dfs(root, float('-inf'), float('inf'))

7.3 对称二叉树 ✅

def isSymmetric(root: Optional[TreeNode]) -> bool:
    """
    LeetCode 101：对称二叉树
    """
    def is_mirror(t1, t2):
        if not t1 and not t2:
            return True
        if not t1 or not t2:
            return False
        return (t1.val == t2.val and
                is_mirror(t1.left, t2.right) and
                is_mirror(t1.right, t2.left))

    return is_mirror(root, root)

7.4 路径总和 ✅

def hasPathSum(root: Optional[TreeNode], targetSum: int) -> bool:
    """
    LeetCode 112：路径总和
    """
    if not root:
        return False

    if not root.left and not root.right:
        return targetSum == root.val

    return (hasPathSum(root.left, targetSum - root.val) or
            hasPathSum(root.right, targetSum - root.val))

7.5 二叉树的最近公共祖先 ✅⭐

def lowestCommonAncestor(root: 'TreeNode', p: 'TreeNode', q: 'TreeNode') -> 'TreeNode':
    """
    LeetCode 236：二叉树的最近公共祖先
    """
    if not root or root == p or root == q:
        return root

    left = lowestCommonAncestor(root.left, p, q)
    right = lowestCommonAncestor(root.right, p, q)

    if left and right:
        return root
    return left if left else right
