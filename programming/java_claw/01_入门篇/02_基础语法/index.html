02_基础语法

1. 变量与常量

1.1 变量定义

// 语法：数据类型 变量名 = 值;
int age = 25;
String name = "张三";
double salary = 50000.50;
boolean isStudent = false;

// 先声明后赋值
int count;
count = 10;

// 多个同类型变量
int a = 1, b = 2, c = 3;

1.2 命名规则

// ✅ 正确的命名
int studentAge;
String userName;
double totalAmount;

// ❌ 错误的命名
int 1stPlace;      // 不能以数字开头
String user-name;  // 不能有连字符
int class;         // 不能使用关键字

1.3 常量

// 使用 final 关键字
final double PI = 3.14159;
final int MAX_SIZE = 100;

// 常量命名规范：全大写，下划线分隔
final String DEFAULT_NAME = "Unknown";

// 常量一旦赋值不能修改
// PI = 3.14;  // ❌ 编译错误

2. 数据类型

2.1 八大基本数据类型

类型 关键字 大小 范围 默认值
字节型 byte 8 位 -128 ~ 127 0
短整型 short 16 位 -32768 ~ 32767 0
整型 int 32 位 -2³¹ ~ 2³¹-1 0
长整型 long 64 位 -2⁶³ ~ 2⁶³-1 0L
浮点型 float 32 位 3.4E-38 ~ 3.4E38 0.0f
双精度 double 64 位 4.9E-324 ~ 1.8E308 0.0d
字符型 char 16 位 '\u0000' ~ '\uffff' '\u0000'
布尔型 boolean - true/false false 
// 整型
byte b = 127;
short s = 32767;
int i = 2147483647;
long l = 9223372036854775807L;  // 必须加 L

// 浮点型
float f = 3.14f;    // 必须加 f/F
double d = 3.14159;

// 字符型
char c1 = 'A';
char c2 = '\u0041';  // Unicode

// 布尔型
boolean flag = true;

2.2 引用数据类型

// 类
String name = "Java";
Person person = new Person();

// 数组
int[] numbers = new int[10];

// 接口
List<String> list = new ArrayList<>();

2.3 类型转换

自动类型转换（隐式）

// 小范围 → 大范围
int i = 100;
long l = i;      // int → long
double d = l;    // long → double

// 表达式自动提升
int a = 10;
double b = 3.0;
double result = a + b;  // int 自动转为 double

强制类型转换（显式）

// 大范围 → 小范围
double d = 3.14;
int i = (int) d;  // 3（丢失小数部分）

long l = 100L;
int j = (int) l;

// 注意精度丢失
int x = 1000;
byte y = (byte) x;  // -24（超出 byte 范围）

String 与其他类型转换

// 其他类型 → String
int num = 123;
String s1 = String.valueOf(num);
String s2 = num + "";

// String → 其他类型
String str = "123";
int i = Integer.parseInt(str);
double d = Double.parseDouble(str);
long l = Long.parseLong(str);
boolean b = Boolean.parseBoolean(str);

3. 运算符

3.1 算术运算符

int a = 10, b = 3;

// 基本运算
int sum = a + b;      // 13
int diff = a - b;     // 7
int product = a * b;  // 30
int quotient = a / b; // 3（整数除法）
int remainder = a % b; // 1（取模）

// 自增自减
int x = 5;
x++;  // 6
++x;  // 7

// 前缀 vs 后缀
int m = 5;
int n = ++m;  // n=6, m=6（先加后赋值）
int p = m++;  // p=6, m=7（先赋值后加）

3.2 关系运算符

int a = 10, b = 5;

boolean r1 = a == b;   // false
boolean r2 = a != b;   // true
boolean r3 = a > b;    // true
boolean r4 = a < b;    // false

3.3 逻辑运算符

boolean a = true, b = false;

boolean r1 = a && b;   // false（逻辑与）
boolean r2 = a || b;   // true（逻辑或）
boolean r3 = !a;       // false（逻辑非）

// 短路特性
int x = 0;
if (false && (x = 5) > 0) {
    // 不会执行
}
System.out.println(x);  // 0（右边未执行）

3.4 三元运算符

// 语法：条件？值 1 : 值 2
int age = 20;
String status = age >= 18 ? "成年" : "未成年";

// 嵌套
int score = 85;
String grade = score >= 90 ? "A" : 
               score >= 80 ? "B" : "C";

// 求最大值
int a = 10, b = 20;
int max = a > b ? a : b;

4. 流程控制

4.1 if-else 语句

int score = 85;

// 简单 if
if (score >= 60) {
    System.out.println("及格");
}

// if-else
if (score >= 60) {
    System.out.println("及格");
} else {
    System.out.println("不及格");
}

// if-else if-else
if (score >= 90) {
    System.out.println("优秀");
} else if (score >= 80) {
    System.out.println("良好");
} else if (score >= 70) {
    System.out.println("中等");
} else {
    System.out.println("不及格");
}

4.2 switch 语句

int day = 3;
String dayName;

switch (day) {
    case 1:
        dayName = "星期一";
        break;
    case 2:
        dayName = "星期二";
        break;
    case 3:
        dayName = "星期三";
        break;
    default:
        dayName = "周末";
        break;
}

// Java 14+ 新语法
String dayName2 = switch (day) {
    case 1 -> "星期一";
    case 2 -> "星期二";
    case 3 -> "星期三";
    default -> "周末";
};

4.3 for 循环

// 基本 for 循环
for (int i = 0; i < 5; i++) {
    System.out.println(i);  // 0, 1, 2, 3, 4
}

// 增强 for 循环（for-each）
int[] numbers = {1, 2, 3, 4, 5};
for (int num : numbers) {
    System.out.println(num);
}

// 嵌套循环（九九乘法表）
for (int i = 1; i <= 9; i++) {
    for (int j = 1; j <= i; j++) {
        System.out.print(j + "*" + i + "=" + (i*j) + " ");
    }
    System.out.println();
}

4.4 while 循环

// while 循环
int i = 0;
while (i < 5) {
    System.out.println(i);
    i++;
}

// do-while 循环（至少执行一次）
int j = 0;
do {
    System.out.println(j);
    j++;
} while (j < 5);

4.5 跳转语句

// break - 跳出循环
for (int i = 0; i < 10; i++) {
    if (i == 5) {
        break;  // 跳出整个循环
    }
    System.out.println(i);  // 0, 1, 2, 3, 4
}

// continue - 跳过本次
for (int i = 0; i < 10; i++) {
    if (i % 2 == 0) {
        continue;  // 跳过偶数
    }
    System.out.println(i);  // 1, 3, 5, 7, 9
}

5. 经典练习题

5.1 求 1-100 的和

int sum = 0;
for (int i = 1; i <= 100; i++) {
    sum += i;
}
System.out.println("和：" + sum);  // 5050

5.2 判断素数

public static boolean isPrime(int n) {
    if (n <= 1) return false;
    for (int i = 2; i <= Math.sqrt(n); i++) {
        if (n % i == 0) return false;
    }
    return true;
}

5.3 斐波那契数列

int a = 1, b = 1;
System.out.print(a + " " + b + " ");
for (int i = 3; i <= 20; i++) {
    int c = a + b;
    System.out.print(c + " ");
    a = b;
    b = c;
}

5.4 水仙花数

for (int i = 100; i <= 999; i++) {
    int hundreds = i / 100;
    int tens = i / 10 % 10;
    int units = i % 10;

    if (hundreds*hundreds*hundreds + 
        tens*tens*tens + 
        units*units*units == i) {
        System.out.println(i);
    }
}
// 输出：153, 370, 371, 407

6. 常见错误

6.1 变量未初始化

int x;
System.out.println(x);  // ❌ 编译错误

int y = 0;
System.out.println(y);  // ✅ 正确

6.2 整数除法

int a = 10, b = 3;
double result = a / b;  // ❌ 结果是 3.0

double result2 = (double) a / b;  // ✅ 3.333...

6.3 数组越界

int[] arr = new int[5];
arr[5] = 10;  // ❌ ArrayIndexOutOfBoundsException
              // 有效索引：0-4

7. 本章小结

核心知识点

✅ 变量与常量：定义、命名规范、final
✅ 数据类型：8 种基本类型、引用类型、类型转换
✅ 运算符：算术、关系、逻辑、三元
✅ 流程控制：if-else、switch、for、while、break/continue

最佳实践

  1. 变量命名：见名知意，使用驼峰命名
  2. 常量命名：全大写，下划线分隔
  3. 避免魔法数字：使用常量代替
  4. 循环选择：知道次数用 for，不知道用 while
