05_建造者模式

1. 建造者模式概述

1.1 什么是建造者模式

建造者模式（Builder Pattern）将复杂对象的构建与表示分离，使同样的构建过程可以创建不同的表示。

适用场景： - ✅ 对象参数很多（超过 4 个） - ✅ 部分参数可选 - ✅ 对象创建过程复杂 - ✅ 需要不可变对象

1.2 传统创建方式的问题

// 问题 1：构造函数参数过多
public class Person {
    public Person(String name, int age, String address, String phone, 
                  String email, String city, String district, int zipCode) {
        // 难以阅读和维护
    }
}

// 问题 2：Setter 方法导致对象可变
public class Person {
    public Person(String name, int age) {
        this.name = name;
        this.age = age;
    }

    public void setAddress(String address) { this.address = address; }
    public void setPhone(String phone) { this.phone = phone; }
    // ...
}

Person p = new Person("张三", 25);
p.setAddress("北京");
p.setPhone("13800000000");
// 对象状态可能被意外修改

2. 实现方式

2.1 静态内部类 Builder

public class Person {
    // 最终字段
    private final String name;
    private final int age;
    private final String address;
    private final String phone;
    private final String email;

    // 私有构造
    private Person(Builder builder) {
        this.name = builder.name;
        this.age = builder.age;
        this.address = builder.address;
        this.phone = builder.phone;
        this.email = builder.email;
    }

    // 静态内部类 Builder
    public static class Builder {
        private String name;
        private int age;
        private String address;
        private String phone;
        private String email;

        public Builder setName(String name) {
            this.name = name;
            return this;
        }

        public Builder setAge(int age) {
            this.age = age;
            return this;
        }

        public Builder setAddress(String address) {
            this.address = address;
            return this;
        }

        public Builder setPhone(String phone) {
            this.phone = phone;
            return this;
        }

        public Builder setEmail(String email) {
            this.email = email;
            return this;
        }

        public Person build() {
            // 可以在这里进行验证
            if (name == null || name.isEmpty()) {
                throw new IllegalArgumentException("name cannot be null");
            }
            if (age < 0 || age > 150) {
                throw new IllegalArgumentException("invalid age");
            }
            return new Person(this);
        }
    }

    // getter 方法
    public String getName() { return name; }
    public int getAge() { return age; }
    // ...
}

// 使用
Person person = new Person.Builder()
    .setName("张三")
    .setAge(25)
    .setAddress("北京")
    .setPhone("13800000000")
    .setEmail("zhangsan@example.com")
    .build();

2.2 抽象 Builder

// 产品
public class Computer {
    private String CPU;
    private String RAM;
    private String storage;
    private String GPU;
    private String powerSupply;

    // getter/setter
}

// 抽象 Builder
public abstract class ComputerBuilder {
    protected Computer computer = new Computer();

    public abstract void buildCPU();
    public abstract void buildRAM();
    public abstract void buildStorage();
    public abstract void buildGPU();
    public abstract void buildPowerSupply();

    public Computer getComputer() {
        return computer;
    }
}

// 具体 Builder
public class GamingComputerBuilder extends ComputerBuilder {
    @Override
    public void buildCPU() {
        computer.setCPU("Intel i9");
    }

    @Override
    public void buildRAM() {
        computer.setRAM("32GB DDR4");
    }

    @Override
    public void buildStorage() {
        computer.setStorage("1TB NVMe SSD");
    }

    @Override
    public void buildGPU() {
        computer.setGPU("RTX 3080");
    }

    @Override
    public void buildPowerSupply() {
        computer.setPowerSupply("750W");
    }
}

public class OfficeComputerBuilder extends ComputerBuilder {
    @Override
    public void buildCPU() {
        computer.setCPU("Intel i5");
    }

    @Override
    public void buildRAM() {
        computer.setRAM("16GB DDR4");
    }

    @Override
    public void buildStorage() {
        computer.setStorage("512GB SSD");
    }

    @Override
    public void buildGPU() {
        computer.setGPU("Integrated");
    }

    @Override
    public void buildPowerSupply() {
        computer.setPowerSupply("400W");
    }
}

// 指挥者
public class Director {
    private ComputerBuilder builder;

    public void setBuilder(ComputerBuilder builder) {
        this.builder = builder;
    }

    public Computer construct() {
        builder.buildCPU();
        builder.buildRAM();
        builder.buildStorage();
        builder.buildGPU();
        builder.buildPowerSupply();
        return builder.getComputer();
    }
}

// 使用
public class Main {
    public static void main(String[] args) {
        Director director = new Director();

        // 构建游戏电脑
        director.setBuilder(new GamingComputerBuilder());
        Computer gaming = director.construct();

        // 构建办公电脑
        director.setBuilder(new OfficeComputerBuilder());
        Computer office = director.construct();
    }
}

3. Lombok Builder

import lombok.Builder;

@Builder
public class Person {
    private String name;
    private int age;
    private String address;
    private String phone;
    private String email;
}

// 使用
Person person = Person.builder()
    .name("张三")
    .age(25)
    .address("北京")
    .build();

4. 实战示例

4.1 HTTP 请求构建器

public class HttpRequest {
    private final String url;
    private final String method;
    private final Map<String, String> headers;
    private final byte[] body;
    private final int timeout;

    private HttpRequest(Builder builder) {
        this.url = builder.url;
        this.method = builder.method;
        this.headers = builder.headers;
        this.body = builder.body;
        this.timeout = builder.timeout;
    }

    public static class Builder {
        private String url;
        private String method = "GET";
        private Map<String, String> headers = new HashMap<>();
        private byte[] body;
        private int timeout = 30000;

        public Builder url(String url) {
            this.url = url;
            return this;
        }

        public Builder method(String method) {
            this.method = method;
            return this;
        }

        public Builder header(String key, String value) {
            this.headers.put(key, value);
            return this;
        }

        public Builder body(byte[] body) {
            this.body = body;
            return this;
        }

        public Builder timeout(int timeout) {
            this.timeout = timeout;
            return this;
        }

        public HttpRequest build() {
            if (url == null || url.isEmpty()) {
                throw new IllegalArgumentException("URL is required");
            }
            return new HttpRequest(this);
        }
    }
}

4.2 SQL 查询构建器

public class QueryBuilder {
    private StringBuilder sql = new StringBuilder();
    private List<Object> params = new ArrayList<>();

    public QueryBuilder select(String... columns) {
        sql.append("SELECT ").append(String.join(", ", columns));
        return this;
    }

    public QueryBuilder from(String table) {
        sql.append(" FROM ").append(table);
        return this;
    }

    public QueryBuilder where(String condition, Object param) {
        sql.append(" WHERE ").append(condition);
        params.add(param);
        return this;
    }

    public QueryBuilder orderBy(String column) {
        sql.append(" ORDER BY ").append(column);
        return this;
    }

    public QueryBuilder limit(int limit) {
        sql.append(" LIMIT ").append(limit);
        return this;
    }

    public Query build() {
        return new Query(sql.toString(), params);
    }
}

// 使用
Query query = new QueryBuilder()
    .select("id", "name", "email")
    .from("users")
    .where("age > ?", 18)
    .orderBy("created_at")
    .limit(10)
    .build();

5. 优缺点

5.1 优点

  • ✅ 封装性好：构建过程封装在 Builder 中
  • ✅ 不可变对象：最终对象可以是不可变的
  • ✅ 链式调用：代码简洁易读
  • ✅ 参数验证：可以在 build() 中验证
  • ✅ 灵活：可以创建不同的表示

5.2 缺点

  • ❌ 代码量大：需要额外的 Builder 类
  • ❌ 对象创建复杂：对于简单对象过于繁琐

6. 本章小结

核心知识点

✅ 静态内部类 Builder：最常用的实现方式
✅ 抽象 Builder：支持不同表示
✅ Lombok Builder：简化代码
✅ 链式调用：流畅 API

最佳实践

  1. 参数超过 4 个用 Builder：提高可读性
  2. 不可变对象：使用 final 字段
  3. 参数验证：在 build() 中验证
  4. 默认值：提供合理的默认值
