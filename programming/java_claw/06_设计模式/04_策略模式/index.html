    35_策略模式¶

    1. 策略模式概述¶

    1.1 什么是策略模式¶

    策略模式（Strategy Pattern）定义一系列算法，将每个算法封装起来，并使它们可以互换。

    核心思想： - ✅ 封装变化：算法独立变化 - ✅ 组合优于继承：运行时切换算法 - ✅ 开闭原则：新增策略无需修改客户端

    应用场景： - ✅ 多种支付方式 - ✅ 多种排序算法 - ✅ 多种折扣策略 - ✅ 多种登录方式

    1.2 角色¶

    角色 说明
    Strategy（策略） 算法接口
    ConcreteStrategy 具体算法实现
    Context（上下文） 使用策略的类

    2. 实现方式¶

    2.1 策略接口¶

    // 策略接口
public interface PaymentStrategy {
    void pay(BigDecimal amount);
}

// 具体策略 1：支付宝
@Component
public class AlipayStrategy implements PaymentStrategy {
    @Override
    public void pay(BigDecimal amount) {
        System.out.println("使用支付宝支付：" + amount);
        // 支付宝支付逻辑
    }
}

// 具体策略 2：微信
@Component
public class WechatPayStrategy implements PaymentStrategy {

    }
}

// 具体策略 3：信用卡

    2.2 上下文¶

    // 上下文
@Component
public class PaymentContext {

    private PaymentStrategy strategy;

    // 注入所有策略
    @Autowired
    private Map<String, PaymentStrategy> strategies;

    // 根据类型获取策略
    public PaymentStrategy getStrategy(String type) {
        return strategies.get(type + "Strategy");
    }

    // 设置策略
    public void setStrategy(PaymentStrategy strategy) {
        this.strategy = strategy;
    }

    // 执行支付
    public void executePayment(BigDecimal amount) {
        if (strategy == null) {
            throw new IllegalStateException("Payment strategy not set");
        }
        strategy.pay(amount);
    }
}

// 使用
@Service
public class OrderService {
    @Autowired
    private PaymentContext paymentContext;

    public void checkout(Long orderId, String paymentType) {
        Order order = getOrder(orderId);

        // 动态选择策略
        PaymentStrategy strategy = paymentContext.getStrategy(paymentType);
        paymentContext.setStrategy(strategy);

        // 执行支付
        paymentContext.executePayment(order.getAmount());
    }
}

    3. 实战示例¶

    3.1 折扣策略¶

    // 策略接口
public interface DiscountStrategy {
    BigDecimal calculate(BigDecimal price);
}

// 策略 1：无折扣
@Component
public class NoDiscountStrategy implements DiscountStrategy {
    @Override
    public BigDecimal calculate(BigDecimal price) {
        return price;
    }
}

// 策略 2：满减
@Component
public class MoneyOffStrategy implements DiscountStrategy {
    @Override
    public BigDecimal calculate(BigDecimal price) {
        if (price.compareTo(new BigDecimal("100")) >= 0) {
            return price.subtract(new BigDecimal("20"));
        }
        return price;
    }
}

// 策略 3：打折
@Component
public class PercentageOffStrategy implements DiscountStrategy {
    private final BigDecimal discount;

    public PercentageOffStrategy(BigDecimal discount) {
        this.discount = discount;
    }

    @Override
    public BigDecimal calculate(BigDecimal price) {
        return price.multiply(discount);
    }
}

// 策略 4：会员折扣
@Component
public class VipDiscountStrategy implements DiscountStrategy {
    @Override
    public BigDecimal calculate(BigDecimal price) {
        return price.multiply(new BigDecimal("0.8"));
    }
}

// 上下文
@Component
public class DiscountContext {
    @Autowired
    private Map<String, DiscountStrategy> strategies;

    public BigDecimal calculate(String type, BigDecimal price) {
        DiscountStrategy strategy = strategies.get(type + "Strategy");
        if (strategy == null) {
            throw new IllegalArgumentException("Unknown discount type: " + type);
        }
        return strategy.calculate(price);
    }
}

// 使用
@Service
public class ProductService {
    @Autowired
    private DiscountContext discountContext;

    public BigDecimal getFinalPrice(String discountType, BigDecimal price) {
        return discountContext.calculate(discountType, price);
    }
}

    3.2 排序策略¶

    // 策略接口
public interface SortStrategy {
    <T extends Comparable<T>> void sort(List<T> list);
}

// 策略 1：冒泡排序
public class BubbleSortStrategy implements SortStrategy {
    @Override
    public <T extends Comparable<T>> void sort(List<T> list) {
        int n = list.size();
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - i - 1; j++) {
                if (list.get(j).compareTo(list.get(j + 1)) > 0) {
                    Collections.swap(list, j, j + 1);
                }
            }
        }
    }
}

// 策略 2：快速排序
public class QuickSortStrategy implements SortStrategy {
    @Override
    public <T extends Comparable<T>> void sort(List<T> list) {
        quickSort(list, 0, list.size() - 1);
    }

    private <T extends Comparable<T>> void quickSort(List<T> list, int low, int high) {
        if (low < high) {
            int pi = partition(list, low, high);
            quickSort(list, low, pi - 1);
            quickSort(list, pi + 1, high);
        }
    }

    private <T extends Comparable<T>> int partition(List<T> list, int low, int high) {
        T pivot = list.get(high);
        int i = low - 1;
        for (int j = low; j < high; j++) {
            if (list.get(j).compareTo(pivot) <= 0) {
                i++;
                Collections.swap(list, i, j);
            }
        }
        Collections.swap(list, i + 1, high);
        return i + 1;
    }
}

// 策略 3：归并排序
public class MergeSortStrategy implements SortStrategy {
    @Override
    public <T extends Comparable<T>> void sort(List<T> list) {
        if (list.size() <= 1) return;

        int mid = list.size() / 2;
        List<T> left = new ArrayList<>(list.subList(0, mid));
        List<T> right = new ArrayList<>(list.subList(mid, list.size()));

        sort(left);
        sort(right);
        merge(list, left, right);
    }

    private <T extends Comparable<T>> void merge(List<T> list, List<T> left, List<T> right) {
        int i = 0, j = 0, k = 0;
        while (i < left.size() && j < right.size()) {
            if (left.get(i).compareTo(right.get(j)) <= 0) {
                list.set(k++, left.get(i++));
            } else {
                list.set(k++, right.get(j++));
            }
        }
        while (i < left.size()) list.set(k++, left.get(i++));
        while (j < right.size()) list.set(k++, right.get(j++));
    }
}

// 上下文
public class Sorter {
    private SortStrategy strategy;

    public void setStrategy(SortStrategy strategy) {
        this.strategy = strategy;
    }

    public <T extends Comparable<T>> void sort(List<T> list) {
        if (strategy == null) {
            throw new IllegalStateException("Sort strategy not set");
        }
        strategy.sort(list);
    }
}

// 使用
public class Main {
    public static void main(String[] args) {
        List<Integer> numbers = Arrays.asList(5, 2, 8, 1, 9);

        Sorter sorter = new Sorter();

        // 使用快速排序
        sorter.setStrategy(new QuickSortStrategy());
        sorter.sort(numbers);

        // 使用归并排序
        sorter.setStrategy(new MergeSortStrategy());
        sorter.sort(numbers);
    }
}

    3.3 登录策略¶

    // 策略接口
public interface LoginStrategy {
    boolean authenticate(String username, String password);
    User login(String credential);
}

// 策略 1：账号密码登录
@Component
public class PasswordLoginStrategy implements LoginStrategy {
    @Autowired
    private UserRepository userRepository;

    @Override
    public boolean authenticate(String username, String password) {
        User user = userRepository.findByUsername(username);
        return user != null && passwordEncoder.matches(password, user.getPassword());
    }

    @Override
    public User login(String username) {
        return userRepository.findByUsername(username);
    }
}

// 策略 2：手机验证码登录
@Component
public class SmsLoginStrategy implements LoginStrategy {
    @Autowired
    private SmsService smsService;

    @Override
    public boolean authenticate(String phone, String code) {
        return smsService.verifyCode(phone, code);
    }

    @Override
    public User login(String phone) {
        return userRepository.findByPhone(phone);
    }
}

// 策略 3：微信登录
@Component
public class WechatLoginStrategy implements LoginStrategy {
    @Autowired
    private WechatService wechatService;

    @Override
    public boolean authenticate(String code, String unused) {
        return wechatService.verifyAuthCode(code);
    }

    @Override
    public User login(String openId) {
        User user = userRepository.findByWechatOpenId(openId);
        if (user == null) {
            user = wechatService.createUser(openId);
        }
        return user;
    }
}

// 上下文
@Component
public class LoginContext {
    @Autowired
    private Map<String, LoginStrategy> strategies;

    public User login(String type, String credential, String extra) {
        LoginStrategy strategy = strategies.get(type + "LoginStrategy");
        if (strategy == null) {
            throw new IllegalArgumentException("Unsupported login type: " + type);
        }

        if (!strategy.authenticate(credential, extra)) {
            throw new AuthenticationException("Login failed");
        }

        return strategy.login(credential);
    }
}

    4. 优缺点¶

    4.1 优点¶

    • ✅ 避免多重条件：消除 if-else 或 switch
    • ✅ 符合开闭原则：新增策略无需修改客户端
    • ✅ 符合单一职责：每个策略类只做一件事
    • ✅ 算法可互换：运行时切换算法

    4.2 缺点¶

    • ❌ 策略类增多：每个策略一个类
    • ❌ 客户端需了解策略：知道有哪些策略可选
    • ❌ 策略暴露：所有策略对客户端可见

    5. 与其他模式结合¶

    5.1 策略 + 工厂¶

    // 工厂创建策略
public class StrategyFactory {
    public static PaymentStrategy createStrategy(String type) {
        return switch (type) {
            case "alipay" -> new AlipayStrategy();
            case "wechat" -> new WechatPayStrategy();
            case "credit" -> new CreditCardStrategy();
            default -> throw new IllegalArgumentException("Unknown type");
        };
    }
}

    5.2 策略 + Spring¶

    // Spring 自动注入所有策略
@Component
public class PaymentService {
    @Autowired
    private Map<String, PaymentStrategy> strategies;

    public void pay(String type, BigDecimal amount) {
        PaymentStrategy strategy = strategies.get(type + "Strategy");
        strategy.pay(amount);
    }
}

    6. 本章小结¶

    核心知识点¶

    ✅ 策略模式结构：Strategy、Context
    ✅ 实现方式：接口、具体策略、上下文
    ✅ 实战应用：支付、折扣、排序、登录
    ✅ 与 Spring 结合：自动注入策略

    最佳实践¶

    1. 使用接口定义策略：清晰明确
    2. 结合工厂模式：创建策略
    3. 使用 Spring 注入：自动管理策略
    4. 策略命名规范：统一后缀（Strategy）
    5. 文档化策略：说明每个策略的用途
