07_包装类

1. 包装类概述

1.1 什么是包装类

包装类（Wrapper Class）是将基本数据类型包装成对象的类。

八大基本类型对应包装类：

基本类型 包装类 基本类型 包装类
byte Byte boolean Boolean
short Short char Character
int Integer float Float
long Long double Double

1.2 为什么需要包装类

// 泛型需要对象
List<int> list = new ArrayList<>();  // ❌ 编译错误
List<Integer> list = new ArrayList<>();  // ✅ 正确

// 集合存储
Collection<Integer> collection = new HashSet<>();

// null 值表示
Integer num = null;  // 可以表示"无值"
int num2 = 0;  // 0 可能是有效值

// 工具方法
int max = Integer.MAX_VALUE;
String binary = Integer.toBinaryString(10);

2. 自动装箱和拆箱 ⭐⭐⭐

2.1 自动装箱

// 基本类型 → 包装类（自动装箱）
int primitive = 10;
Integer boxed = primitive;  // 自动装箱：Integer.valueOf(primitive)

// 等价于
Integer boxed2 = Integer.valueOf(10);

2.2 自动拆箱

// 包装类 → 基本类型（自动拆箱）
Integer boxed = 10;
int primitive = boxed;  // 自动拆箱：boxed.intValue()

// 等价于
int primitive2 = boxed.intValue();

2.3 运算中的自动转换

Integer a = 10;
Integer b = 20;

// 自动拆箱后运算
int sum = a + b;  // 30

// 包装类比较
System.out.println(a == b);  // false（比较引用）
System.out.println(a.equals(b));  // false（值不同）

Integer c = 10;
Integer d = 10;
System.out.println(c == d);  // true（缓存 -128~127）

2.4 缓存机制

// Integer 缓存 -128 到 127
Integer a = 127;
Integer b = 127;
System.out.println(a == b);  // true（缓存）

Integer c = 128;
Integer d = 128;
System.out.println(c == d);  // false（新对象）

// 其他包装类没有缓存
Double e = 1.0;
Double f = 1.0;
System.out.println(e == f);  // false

3. 常用方法

3.1 Integer

// 转 int
Integer num = 100;
int value = num.intValue();

// 转 String
String str = num.toString();  // "100"
String str2 = Integer.toString(100);

// String 转 int
int parsed = Integer.parseInt("123");  // 123
int withDefault = Integer.parseInt("abc", 0);  // 抛异常

// 进制转换
String binary = Integer.toBinaryString(10);  // "1010"
String octal = Integer.toOctalString(10);    // "12"
String hex = Integer.toHexString(255);       // "ff"

// 解析不同进制
int fromBinary = Integer.parseInt("1010", 2);    // 10
int fromHex = Integer.parseInt("ff", 16);        // 255

// 最大值最小值
int max = Integer.MAX_VALUE;  // 2147483647
int min = Integer.MIN_VALUE;  // -2147483648

// 比较
int cmp = Integer.compare(10, 20);  // -1

3.2 Double

Double d = 3.14;

// 转 double
double value = d.doubleValue();

// 转 String
String str = d.toString();  // "3.14"

// String 转 double
double parsed = Double.parseDouble("3.14");

// 特殊值
boolean isNaN = Double.isNaN(d);
boolean isInfinite = Double.isInfinite(Double.POSITIVE_INFINITY);

// 最大值最小值
double max = Double.MAX_VALUE;
double min = Double.MIN_VALUE;

3.3 Boolean

Boolean b1 = true;
Boolean b2 = false;

// 转 boolean
boolean value = b1.booleanValue();

// String 转 boolean
boolean parsed = Boolean.parseBoolean("true");   // true
boolean parsed2 = Boolean.parseBoolean("false"); // false
boolean parsed3 = Boolean.parseBoolean("abc");   // false（只有"true"返回 true）

// 逻辑运算
Boolean.and(true, false);  // false
Boolean.or(true, false);   // true
Boolean.xor(true, true);   // false

3.4 Character

Character c = 'A';

// 转 char
char value = c.charValue();

// 判断
boolean isLetter = Character.isLetter('A');      // true
boolean isDigit = Character.isDigit('1');        // true
boolean isSpace = Character.isWhitespace(' ');   // true
boolean isUpperCase = Character.isUpperCase('A');// true
boolean isLowerCase = Character.isLowerCase('a');// true

// 转换
char upper = Character.toUpperCase('a');  // 'A'
char lower = Character.toLowerCase('A');  // 'a'

// Unicode
int codePoint = Character.codePointAt("Hello", 0);  // 72

4. 数值运算

4.1 Math 类

// 基本运算
int abs = Math.abs(-10);        // 10
double ceil = Math.ceil(3.14);  // 4.0
double floor = Math.floor(3.14);// 3.0
long round = Math.round(3.6);   // 4

// 最大最小
int max = Math.max(10, 20);     // 20
int min = Math.min(10, 20);     // 10

// 幂运算
double pow = Math.pow(2, 3);    // 8.0
double sqrt = Math.sqrt(16);    // 4.0

// 三角函数
double sin = Math.sin(Math.PI / 2);  // 1.0
double cos = Math.cos(0);            // 1.0

// 随机数
double random = Math.random();  // 0.0 ~ 1.0

4.2 BigDecimal（精确计算）

import java.math.BigDecimal;

// 创建（推荐用 String）
BigDecimal bd1 = new BigDecimal("10.50");
BigDecimal bd2 = BigDecimal.valueOf(10.50);

// ❌ 不推荐：double 精度问题
BigDecimal bd3 = new BigDecimal(10.50);  // 10.5

// 加法
BigDecimal sum = bd1.add(bd2);

// 减法
BigDecimal diff = bd1.subtract(bd2);

// 乘法
BigDecimal product = bd1.multiply(bd2);

// 除法（指定精度和舍入模式）
BigDecimal quotient = bd1.divide(bd2, 2, RoundingMode.HALF_UP);

// 比较
int cmp = bd1.compareTo(bd2);  // -1, 0, 1

// 注意：equals 比较值和精度
BigDecimal a = new BigDecimal("1.0");
BigDecimal b = new BigDecimal("1.00");
System.out.println(a.equals(b));    // false（精度不同）
System.out.println(a.compareTo(b)); // 0（值相同）

4.3 实战：金额计算

public class Money {
    private BigDecimal amount;

    public Money(String amount) {
        this.amount = new BigDecimal(amount);
    }

    public Money add(Money other) {
        return new Money(this.amount.add(other.amount).toString());
    }

    public Money multiply(int factor) {
        return new Money(this.amount.multiply(
            new BigDecimal(factor)
        ).toString());
    }

    public Money divide(int divisor) {
        return new Money(this.amount.divide(
            new BigDecimal(divisor), 
            2, 
            RoundingMode.HALF_UP
        ).toString());
    }

    @Override
    public String toString() {
        return "¥" + amount.setScale(2, RoundingMode.HALF_UP);
    }
}

// 使用
Money price = new Money("19.99");
Money total = price.multiply(3);  // ¥59.97

5. 类型转换

5.1 包装类之间转换

// int → long
Integer intVal = 100;
Long longVal = intVal.longValue();

// int → double
Integer intVal2 = 100;
Double doubleVal = intVal2.doubleValue();

// String → 各种类型
String str = "123";
int i = Integer.parseInt(str);
long l = Long.parseLong(str);
double d = Double.parseDouble(str);
boolean b = Boolean.parseBoolean(str);

5.2 数值格式化

import java.text.DecimalFormat;

DecimalFormat df = new DecimalFormat("#,###.##");

// 格式化
String formatted = df.format(1234567.89);  // "1,234,567.89"

// 百分比
DecimalFormat percent = new DecimalFormat("#.##%");
String percentStr = percent.format(0.8765);  // "87.65%"

// 货币
DecimalFormat currency = new DecimalFormat("¥#,###.##");
String currencyStr = currency.format(1234.5);  // "¥1,234.5"

6. 常见问题

6.1 空指针异常

Integer num = null;

// ❌ 空指针
int value = num;  // NullPointerException

// ✅ 先检查
int value2 = (num != null) ? num : 0;

// ✅ 使用 Optional
int value3 = Optional.ofNullable(num).orElse(0);

6.2 比较错误

Integer a = 1000;
Integer b = 1000;

// ❌ 错误：比较引用
System.out.println(a == b);  // false

// ✅ 正确：比较值
System.out.println(a.equals(b));  // true

// ✅ 或拆箱比较
System.out.println(a.intValue() == b.intValue());  // true

6.3 BigDecimal 精度

// ❌ 错误：double 精度问题
BigDecimal bd1 = new BigDecimal(0.1);  // 0.10000000000000000555...

// ✅ 正确：使用 String
BigDecimal bd2 = new BigDecimal("0.1");  // 0.1

// ✅ 或 valueOf
BigDecimal bd3 = BigDecimal.valueOf(0.1);  // 0.1

7. 本章小结

核心知识点

✅ 包装类：8 种基本类型对应包装类
✅ 自动装箱/拆箱：基本类型与包装类自动转换
✅ 常用方法：parseInt、toString、valueOf
✅ BigDecimal：精确小数计算

最佳实践

  1. 集合用包装类：List
  2. 金额用 BigDecimal：不用 double
  3. 比较用 equals：不用 ==
  4. 注意空指针：包装类可能为 null
  5. BigDecimal 用 String：避免精度问题
