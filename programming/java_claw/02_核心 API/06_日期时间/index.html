17_日期时间

1. 日期时间 API 概述

1.1 Java 中的日期时间

Java 中有三代日期时间 API：

代际 类 问题
第一代 Date、Calendar 设计差、线程不安全
第二代 SimpleDateFormat 线程不安全
第三代 Java 8+ Date/Time API 设计好、线程安全 ⭐

推荐使用 Java 8+ 的 Date/Time API

2. Java 8+ 日期时间 API

2.1 LocalDate（日期）

import java.time.*;

// 当前日期
LocalDate today = LocalDate.now();

// 指定日期
LocalDate date = LocalDate.of(2026, 3, 26);
LocalDate date2 = LocalDate.parse("2026-03-26");

// 获取年月日
int year = date.getYear();
Month month = date.getMonth();
int day = date.getDayOfMonth();

// 计算
LocalDate tomorrow = today.plusDays(1);
LocalDate nextWeek = today.plusWeeks(1);
LocalDate nextMonth = today.plusMonths(1);
LocalDate lastYear = today.minusYears(1);

// 比较
boolean isBefore = date.isBefore(date2);
boolean isAfter = date.isAfter(date2);
boolean isEqual = date.isEqual(date2);

// 格式化
DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd");
String formatted = date.format(formatter);

2.2 LocalTime（时间）

// 当前时间
LocalTime now = LocalTime.now();

// 指定时间
LocalTime time = LocalTime.of(14, 30, 0);
LocalTime time2 = LocalTime.parse("14:30:00");

// 获取时分秒
int hour = time.getHour();
int minute = time.getMinute();
int second = time.getSecond();

// 计算
LocalTime later = time.plusHours(2);
LocalTime earlier = time.minusMinutes(30);

2.3 LocalDateTime（日期时间）

// 当前日期时间
LocalDateTime now = LocalDateTime.now();

// 指定日期时间
LocalDateTime dt = LocalDateTime.of(2026, 3, 26, 14, 30, 0);
LocalDateTime dt2 = LocalDateTime.parse("2026-03-26T14:30:00");

// 日期和时间转换
LocalDate date = dt.toLocalDate();
LocalTime time = dt.toLocalTime();

// 组合
LocalDateTime dt3 = LocalDateTime.of(date, time);

// 计算
LocalDateTime tomorrow = now.plusDays(1);
LocalDateTime nextHour = now.plusHours(1);

// 格式化
DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");
String formatted = now.format(formatter);

2.4 Instant（时间戳）

// 当前时间戳
Instant now = Instant.now();

// 毫秒数
long epochMilli = now.toEpochMilli();
long epochSecond = now.getEpochSecond();

// 从毫秒创建
Instant instant = Instant.ofEpochMilli(1679836800000L);

// 计算
Instant later = now.plusSeconds(60);
Instant earlier = now.minusMinutes(10);

// 与 LocalDateTime 转换
LocalDateTime ldt = LocalDateTime.ofInstant(now, ZoneId.systemDefault());

3. 日期格式化

3.1 DateTimeFormatter

// 内置格式
DateTimeFormatter f1 = DateTimeFormatter.ISO_LOCAL_DATE;
DateTimeFormatter f2 = DateTimeFormatter.ISO_LOCAL_DATE_TIME;

// 自定义格式
DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");

// 格式化
LocalDateTime now = LocalDateTime.now();
String str = now.format(formatter);

// 解析
LocalDateTime parsed = LocalDateTime.parse("2026-03-26 14:30:00", formatter);

3.2 常用格式

// 常见格式
"yyyy-MM-dd"           // 2026-03-26
"yyyy/MM/dd"           // 2026/03/26
"yyyy 年 MM 月 dd 日"    // 2026 年 03 月 26 日
"HH:mm:ss"             // 14:30:00
"yyyy-MM-dd HH:mm:ss"  // 2026-03-26 14:30:00
"EEEE, MMMM dd, yyyy"  // Thursday, March 26, 2026

4. 时区处理

4.1 ZoneId（时区）

// 系统默认时区
ZoneId systemZone = ZoneId.systemDefault();

// 指定时区
ZoneId shanghai = ZoneId.of("Asia/Shanghai");
ZoneId tokyo = ZoneId.of("Asia/Tokyo");
ZoneId newyork = ZoneId.of("America/New_York");

// 所有可用时区
Set<String> availableIds = ZoneId.getAvailableZoneIds();

4.2 ZonedDateTime（带时区的日期时间）

// 当前时区时间
ZonedDateTime now = ZonedDateTime.now();

// 指定时区
ZonedDateTime shanghaiTime = ZonedDateTime.now(ZoneId.of("Asia/Shanghai"));
ZonedDateTime tokyoTime = ZonedDateTime.now(ZoneId.of("Asia/Tokyo"));

// 转换时区
ZonedDateTime nyTime = shanghaiTime.withZoneSameInstant(ZoneId.of("America/New_York"));

// 格式化
DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss Z");
String formatted = now.format(formatter);

5. 日期计算

5.1 Period（日期间隔）

// 两个日期的间隔
LocalDate start = LocalDate.of(2020, 1, 1);
LocalDate end = LocalDate.of(2026, 3, 26);
Period period = Period.between(start, end);

int years = period.getYears();
int months = period.getMonths();
int days = period.getDays();

// 创建间隔
Period oneYear = Period.ofYears(1);
Period oneMonth = Period.ofMonths(1);
Period oneDay = Period.ofDays(1);

// 计算
LocalDate future = start.plus(oneYear);

5.2 Duration（时间间隔）

// 两个时间的间隔
LocalTime start = LocalTime.of(10, 0);
LocalTime end = LocalTime.of(12, 30);
Duration duration = Duration.between(start, end);

long hours = duration.toHours();
long minutes = duration.toMinutes();
long seconds = duration.getSeconds();

// 创建间隔
Duration oneHour = Duration.ofHours(1);
Duration oneMinute = Duration.ofMinutes(1);

// 计算
LocalTime later = start.plus(oneHour);

6. 旧版 API（了解）

6.1 Date

// 创建
Date now = new Date();

// 时间戳
long time = now.getTime();

// 格式化
SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss");
String str = sdf.format(now);

// 解析
Date parsed = sdf.parse("2026-03-26 14:30:00");

// 注意：SimpleDateFormat 不是线程安全的

6.2 Calendar

// 创建
Calendar cal = Calendar.getInstance();

// 设置
cal.set(2026, 2, 26, 14, 30, 0);  // 月份从 0 开始

// 获取
int year = cal.get(Calendar.YEAR);
int month = cal.get(Calendar.MONTH);  // 0-11
int day = cal.get(Calendar.DAY_OF_MONTH);

// 计算
cal.add(Calendar.DAY_OF_MONTH, 1);
cal.add(Calendar.MONTH, -1);

// 转 Date
Date date = cal.getTime();

7. 实战示例

7.1 计算年龄

public static int calculateAge(LocalDate birthday) {
    return Period.between(birthday, LocalDate.now()).getYears();
}

// 使用
LocalDate birthday = LocalDate.of(1990, 5, 15);
int age = calculateAge(birthday);

7.2 判断闰年

public static boolean isLeapYear(int year) {
    return Year.isLeap(year);
}

// 或使用 LocalDate
public static boolean isLeapYear(LocalDate date) {
    return date.isLeapYear();
}

7.3 获取本月第一天和最后一天

LocalDate today = LocalDate.now();

// 本月第一天
LocalDate firstDay = today.with(TemporalAdjusters.firstDayOfMonth());

// 本月最后一天
LocalDate lastDay = today.with(TemporalAdjusters.lastDayOfMonth());

7.4 计算工作日

public static long countWorkdays(LocalDate start, LocalDate end) {
    long count = 0;
    LocalDate current = start;

    while (!current.isAfter(end)) {
        if (current.getDayOfWeek() != DayOfWeek.SATURDAY && 
            current.getDayOfWeek() != DayOfWeek.SUNDAY) {
            count++;
        }
        current = current.plusDays(1);
    }

    return count;
}

8. 本章小结

核心知识点

✅ LocalDate：日期操作
✅ LocalTime：时间操作
✅ LocalDateTime：日期时间操作
✅ Instant：时间戳
✅ DateTimeFormatter：格式化
✅ ZoneId/ZonedDateTime：时区处理
✅ Period/Duration：日期时间间隔

最佳实践

  1. 使用 Java 8+ API：线程安全、设计更好
  2. 使用 LocalDate/LocalTime：而不是 Date/Calendar
  3. 使用 DateTimeFormatter：而不是 SimpleDateFormat
  4. 注意时区：使用 ZonedDateTime 处理时区
  5. 使用 TemporalAdjusters：简化日期调整
