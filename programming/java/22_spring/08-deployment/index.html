08. 部署与生产

打包应用

Maven 打包

# 清理并打包
./mvnw clean package

# 跳过测试打包
./mvnw clean package -DskipTests

# 打包并指定 Profile
./mvnw clean package -Pprod

生成的文件

target/
├── demo-0.0.1-SNAPSHOT.jar          # 可执行 JAR
├── demo-0.0.1-SNAPSHOT.jar.original  # 原始 JAR
└── classes/                          # 编译后的类文件

运行 JAR

# 直接运行
java -jar target/demo-0.0.1-SNAPSHOT.jar

# 指定 Profile
java -jar target/demo-0.0.1-SNAPSHOT.jar --spring.profiles.active=prod

# 指定端口
java -jar target/demo-0.0.1-SNAPSHOT.jar --server.port=8081

# 查看启动参数
java -jar target/demo-0.0.1-SNAPSHOT.jar --help

Docker 部署

Dockerfile

# 多阶段构建（推荐）

# 阶段 1：构建
FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /app
COPY pom.xml .
COPY src ./src
RUN mvn clean package -DskipTests

# 阶段 2：运行
FROM eclipse-temurin:17-jre-alpine
WORKDIR /app
COPY --from=builder /app/target/*.jar app.jar

# 非 root 用户运行（安全）
RUN addgroup -S appgroup && adduser -S appuser -G appgroup
USER appuser

EXPOSE 8080

ENTRYPOINT ["java", "-jar", "app.jar"]

自定义健康检查

@Component
public class DatabaseHealthIndicator implements HealthIndicator {

    private final DataSource dataSource;

    public DatabaseHealthIndicator(DataSource dataSource) {
        this.dataSource = dataSource;
    }

    @Override
    public Health health() {
        try {
            dataSource.getConnection().close();
            return Health.up().withDetail("database", "connected").build();
        } catch (SQLException e) {
            return Health.down(e).withDetail("database", "disconnected").build();
        }
    }
}

// 使用
@Component
public class CustomHealthIndicator implements HealthIndicator {
    @Override
    public Health health() {
        // 自定义检查逻辑
        if (isServiceAvailable()) {
            return Health.up().withDetail("service", "available").build();
        }
        return Health.down().withDetail("service", "unavailable").build();
    }
}

Micrometer 指标

// 自定义指标
@Component
public class CustomMetrics {

    private final MeterRegistry meterRegistry;
    private final Counter requestCounter;
    private final Timer requestTimer;

    public CustomMetrics(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;

        this.requestCounter = Counter.builder("app.requests.total")
            .description("Total number of requests")
            .register(meterRegistry);

        this.requestTimer = Timer.builder("app.requests.duration")
            .description("Request duration")
            .register(meterRegistry);
    }

    public void processRequest() {
        requestCounter.increment();
        requestTimer.record(() -> {
            // 业务逻辑
        });
    }
}

缓存配置

<dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-cache</artifactId>
</dependency>

<dependency>
    <groupId>com.github.ben-manes.caffeine</groupId>
    <artifactId>caffeine</artifactId>
</dependency>

@Configuration
@EnableCaching
public class CacheConfig {

    @Bean
    public CacheManager cacheManager() {
        CaffeineCacheManager cacheManager = new CaffeineCacheManager("users", "products");
        cacheManager.setCaffeine(Caffeine.newBuilder()
            .expireAfterWrite(10, TimeUnit.MINUTES)
            .maximumSize(10000));
        return cacheManager;
    }
}

// 使用缓存
@Service
public class UserService {

    @Cacheable("users")  // 读取缓存
    public User findById(Long id) {
        return userRepository.findById(id).get();
    }

    @CachePut(value = "users", key = "#user.id")  // 更新缓存
    public User update(User user) {
        return userRepository.save(user);
    }

    @CacheEvict("users")  // 清除缓存
    public void delete(Long id) {
        userRepository.deleteById(id);
    }
}

最佳实践

✅ 推荐

# 1. 使用环境变量配置敏感信息
spring:
  datasource:
    password: ${DB_PASSWORD}

# 2. 启用优雅关闭
server:
  shutdown: graceful

# 3. 配置合理的超时
spring:
  datasource:
    hikari:
      connection-timeout: 30000

# 4. 使用健康检查
management:
  endpoint:
    health:
      probes:
        enabled: true

# 5. 集中日志管理
# 使用 ELK、Loki 等

❌ 避免

# 1. 硬编码敏感信息
spring:
  datasource:
    password: mypassword  # ❌

# 2. 生产环境启用详细日志
logging:
  level:
    root: DEBUG  # ❌ 性能影响

# 3. 自动创建表
spring:
  jpa:
    hibernate:
      ddl-auto: create  # ❌ 会丢失数据

# 4. 禁用安全
management:
  endpoints:
    web:
      exposure:
        include: "*"  # ❌ 暴露所有端点
