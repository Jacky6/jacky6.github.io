    09_性能优化¶

    查询优化、索引优化、配置调优、表结构优化

    1. 性能优化概述¶

    1.1 优化层次¶

    性能优化层次（从上到下）：
┌─────────────────────────────────────┐
│  应用层：SQL 编写、缓存设计          │
├─────────────────────────────────────┤
│  数据库层：索引、查询优化            │
├─────────────────────────────────────┤
│  配置层：参数调优                    │
├─────────────────────────────────────┤
│  硬件层：CPU、内存、磁盘、网络       │
└─────────────────────────────────────┘

    1.2 优化步骤¶

    1. 发现问题（监控、慢查询日志）
2. 分析原因（EXPLAIN、SHOW PROFILE）
3. 制定方案（索引、SQL 改写、配置）
4. 实施优化
5. 验证效果
6. 持续监控

    2. 慢查询分析¶

    2.1 开启慢查询日志¶

    [mysqld]
# 开启慢查询日志
slow_query_log = 1
slow_query_log_file = /var/log/mysql/slow.log

# 慢查询阈值（秒）
long_query_time = 2

# 记录未使用索引的查询
log_queries_not_using_indexes = 1

# 记录管理语句
log_slow_admin_statements = 1

    2.2 分析慢查询¶

    # 查看慢查询数量
SELECT COUNT(*) FROM mysql.slow_log;

# 使用 mysqldumpslow 分析
mysqldumpslow /var/log/mysql/slow.log

# 按时间排序
mysqldumpslow -s t /var/log/mysql/slow.log

# 按返回记录数排序
mysqldumpslow -s r /var/log/mysql/slow.log

# 查看前 10 条
mysqldumpslow -t 10 /var/log/mysql/slow.log

    2.3 pt-query-digest¶

    # 安装 Percona Toolkit
# yum install percona-toolkit
# 或
# apt install percona-toolkit

# 分析慢查询
pt-query-digest /var/log/mysql/slow.log

# 分析输出：
# 按查询时间排序
# 显示执行次数、平均时间、锁时间等

    3. 查询优化¶

    3.1 EXPLAIN 分析¶

    -- 基本用法
EXPLAIN SELECT * FROM users WHERE username = 'zhangsan';

-- 详细分析
EXPLAIN FORMAT=JSON SELECT * FROM users WHERE username = 'zhangsan';

-- 实时分析（8.0+）
EXPLAIN ANALYZE SELECT * FROM users WHERE username = 'zhangsan';

    3.2 优化示例¶

    示例 1：避免全表扫描¶

    -- ❌ 全表扫描
SELECT * FROM users WHERE YEAR(created_at) = 2024;

-- ✅ 使用索引
SELECT * FROM users 
WHERE created_at >= '2024-01-01' AND created_at < '2025-01-01';

    示例 2：优化分页¶

    -- ❌ 深分页慢
SELECT * FROM products ORDER BY id LIMIT 100000, 10;

-- ✅ 覆盖索引 + 回表
SELECT p.* FROM products p
INNER JOIN (
    SELECT id FROM products ORDER BY id LIMIT 100000, 10
) AS tmp ON p.id = tmp.id;

-- ✅ 记录上次 ID
SELECT * FROM products WHERE id > 100000 ORDER BY id LIMIT 10;

    示例 3：优化 JOIN¶

    -- ❌ 小表驱动大表
SELECT * FROM large_table l
JOIN small_table s ON l.id = s.large_id;

-- ✅ 大表驱动小表（使用 STRAIGHT_JOIN）
SELECT STRAIGHT_JOIN * FROM small_table s
JOIN large_table l ON s.large_id = l.id;

-- ✅ 确保连接列有索引
CREATE INDEX idx_large_id ON large_table(id);
CREATE INDEX idx_large_id ON small_table(large_id);

    示例 4：优化子查询¶

    -- ❌ 相关子查询慢
SELECT * FROM products p
WHERE price > (
    SELECT AVG(price) FROM products WHERE category_id = p.category_id
);

-- ✅ 改为 JOIN
SELECT p.* FROM products p
JOIN (
    SELECT category_id, AVG(price) AS avg_price
    FROM products
    GROUP BY category_id
) AS stats ON p.category_id = stats.category_id
WHERE p.price > stats.avg_price;

    示例 5：优化 OR 条件¶

    -- ❌ OR 可能导致索引失效
SELECT * FROM users WHERE username = 'zhangsan' OR email = 'zhang@example.com';

-- ✅ 改为 UNION
SELECT * FROM users WHERE username = 'zhangsan'
UNION
SELECT * FROM users WHERE email = 'zhang@example.com';

    4. 索引优化¶

    4.1 创建合适索引¶

    -- 高频查询列
CREATE INDEX idx_username ON users(username);

-- 复合索引（高选择性列在前）
CREATE INDEX idx_status_created ON orders(status, created_at);

-- 覆盖索引
CREATE INDEX idx_status_user_amount ON orders(status, user_id, total_amount);

    4.2 避免索引失效¶

    -- ❌ 函数导致索引失效
SELECT * FROM users WHERE DATE(created_at) = '2024-03-25';

-- ✅ 改为范围查询
SELECT * FROM users 
WHERE created_at >= '2024-03-25' AND created_at < '2024-03-26';

-- ❌ 隐式转换
SELECT * FROM users WHERE phone = 13800138000;

-- ✅ 类型匹配
SELECT * FROM users WHERE phone = '13800138000';

-- ❌ LIKE 通配符开头
SELECT * FROM users WHERE username LIKE '%san';

-- ✅ LIKE 不以通配符开头
SELECT * FROM users WHERE username LIKE 'zhang%';

    4.3 索引维护¶

    -- 分析表（更新统计信息）
ANALYZE TABLE users;

-- 检查表
CHECK TABLE users;

-- 优化表（重建索引）
OPTIMIZE TABLE users;

-- 删除未使用索引
ALTER TABLE users DROP INDEX idx_unused;

    5. 表结构优化¶

    5.1 选择合适数据类型¶

    -- ❌ 浪费空间
CREATE TABLE users (
    id BIGINT,           -- 用不到那么大
    status INT,          -- 只有几个状态
    created_at DATETIME  -- 不需要时间部分
);

-- ✅ 节省空间
CREATE TABLE users (
    id INT AUTO_INCREMENT,
    status TINYINT,
    created_at DATE
);

    5.2 范式与反范式¶

    第三范式（3NF）：
- 每列与主键直接相关
- 消除传递依赖

反范式（空间换时间）：
- 适当冗余，减少 JOIN
- 提高查询性能

示例：
-- 范式化
orders(user_id, product_id, quantity)
users(id, username)
products(id, name, price)

-- 反范式化（订单表冗余）
orders(user_id, username, product_id, product_name, price, quantity)

    5.3 表分区¶

    -- 按范围分区
CREATE TABLE logs (
    id INT,
    created_at DATE,
    content TEXT
) PARTITION BY RANGE (YEAR(created_at)) (
    PARTITION p2022 VALUES LESS THAN (2023),
    PARTITION p2023 VALUES LESS THAN (2024),
    PARTITION p2024 VALUES LESS THAN (2025),
    PARTITION pmax VALUES LESS THAN MAXVALUE
);

-- 按列表分区
CREATE TABLE users (
    id INT,
    city VARCHAR(50)
) PARTITION BY LIST COLUMNS(city) (
    PARTITION p_beijing VALUES IN ('北京'),
    PARTITION p_shanghai VALUES IN ('上海'),
    PARTITION p_other VALUES IN ('其他')
);

-- 按哈希分区
CREATE TABLE sessions (
    session_id VARCHAR(64),
    data TEXT
) PARTITION BY HASH(CRC32(session_id)) PARTITIONS 8;

    6. 配置优化¶

    6.1 内存配置¶

    [mysqld]
# 缓冲池（最重要）
innodb_buffer_pool_size = 4G  # 物理内存的 50-80%

# 缓冲池实例
innodb_buffer_pool_instances = 8

# 日志缓冲
innodb_log_buffer_size = 64M

# 查询缓存（8.0 已移除）
# query_cache_size = 64M

# 键缓冲（MyISAM）
key_buffer_size = 256M

# 排序缓冲
sort_buffer_size = 4M
read_buffer_size = 2M
read_rnd_buffer_size = 4M

# 连接缓冲
join_buffer_size = 4M

    6.2 连接配置¶

    [mysqld]
# 最大连接数
max_connections = 500

# 连接超时
connect_timeout = 10

# 等待超时
wait_timeout = 28800
interactive_timeout = 28800

# 线程缓存
thread_cache_size = 64

    6.3 日志配置¶

    [mysqld]
# 慢查询日志
slow_query_log = 1
long_query_time = 2

# Binlog
log_bin = mysql-bin
binlog_format = ROW
sync_binlog = 1

# 错误日志
log_error = /var/log/mysql/error.log

    6.4 InnoDB 配置¶

    [mysqld]
# 刷新方式
innodb_flush_log_at_trx_commit = 1

# 每表表空间
innodb_file_per_table = 1

# IO 能力
innodb_io_capacity = 200
innodb_io_capacity_max = 400

# 页大小
innodb_page_size = 16K

# 线程并发
innodb_thread_concurrency = 0

    7. 硬件优化¶

    7.1 CPU¶

    • 多核 CPU 有利于并发
    • 高主频有利于单查询性能
    • NUMA 架构需要特殊配置

    7.2 内存¶

    • 越大越好（缓冲池）
    • ECC 内存提高可靠性
    • 内存频率影响性能

    7.3 磁盘¶

    类型 IOPS 适用场景
    HDD 100-200 归档、备份
    SATA SSD 10K-50K 一般应用
    NVMe SSD 100K+ 高性能应用

    7.4 网络¶

    • 万兆网络减少瓶颈
    • 低延迟网络有利复制
    • 绑定网卡提高带宽

    8. 架构优化¶

    8.1 读写分离¶

    Master（写） → Slave1（读）
           → Slave2（读）
           → Slave3（读）

    8.2 分库分表¶

    垂直分库：
用户库 → 用户服务
订单库 → 订单服务
商品库 → 商品服务

水平分表：
orders_0000, orders_0001, ..., orders_9999
（按 user_id % 10000 分表）

    8.3 缓存层¶

    应用 → Redis 缓存 → MySQL
          ↓
      缓存命中直接返回
