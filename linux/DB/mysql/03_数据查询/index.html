03_数据查询

SELECT 详解：WHERE、JOIN、聚合函数、子查询、联合查询

1. SELECT 基础

1.1 基本语法

SELECT [DISTINCT] column1, column2, ...
FROM table_name
[WHERE condition]
[GROUP BY column]
[HAVING condition]
[ORDER BY column [ASC|DESC]]
[LIMIT offset, count];

1.2 查询所有列

-- 查询所有列（不推荐在生产环境使用）
SELECT * FROM users;

-- 推荐：明确指定列
SELECT id, username, email FROM users;

1.3 使用别名

-- 列别名
SELECT 
    username AS user_name,
    email AS user_email,
    created_at AS register_date
FROM users;

-- 表别名
SELECT 
    u.username,
    o.order_no,
    o.total_amount
FROM users u
JOIN orders o ON u.id = o.user_id;

1.4 去重查询

-- DISTINCT 去重
SELECT DISTINCT status FROM users;

-- 多列去重
SELECT DISTINCT status, gender FROM users;

2. WHERE 条件过滤

2.1 比较运算符

-- 等于
SELECT * FROM users WHERE status = 1;

-- 不等于
SELECT * FROM users WHERE status != 0;
SELECT * FROM users WHERE status <> 0;

-- 大于/小于
SELECT * FROM products WHERE price > 100;
SELECT * FROM products WHERE price < 50;

-- 大于等于/小于等于
SELECT * FROM products WHERE stock >= 100;
SELECT * FROM products WHERE price <= 1000;

-- IS NULL / IS NOT NULL
SELECT * FROM users WHERE phone IS NULL;
SELECT * FROM users WHERE phone IS NOT NULL;

2.2 逻辑运算符

-- AND
SELECT * FROM users WHERE status = 1 AND gender = 1;

-- OR
SELECT * FROM users WHERE gender = 1 OR gender = 2;

-- NOT
SELECT * FROM users WHERE NOT status = 0;

-- 组合使用
SELECT * FROM products 
WHERE price > 100 AND price < 1000 AND stock > 0;

2.3 IN 和 NOT IN

-- IN
SELECT * FROM users WHERE status IN (1, 2, 3);

-- 等价于
SELECT * FROM users WHERE status = 1 OR status = 2 OR status = 3;

-- NOT IN
SELECT * FROM users WHERE status NOT IN (0, 3);

-- 子查询中使用
SELECT * FROM products 
WHERE category_id IN (SELECT id FROM categories WHERE level = 1);

2.4 BETWEEN

-- BETWEEN（包含边界）
SELECT * FROM products WHERE price BETWEEN 100 AND 500;

-- 等价于
SELECT * FROM products WHERE price >= 100 AND price <= 500;

-- NOT BETWEEN
SELECT * FROM products WHERE price NOT BETWEEN 100 AND 500;

2.5 LIKE 模糊匹配

-- % 匹配任意字符
SELECT * FROM users WHERE username LIKE '张%';  -- 张开头
SELECT * FROM users WHERE username LIKE '%三';  -- 三结尾
SELECT * FROM users WHERE username LIKE '%三%'; -- 包含三

-- _ 匹配单个字符
SELECT * FROM users WHERE phone LIKE '138____8888';

-- 转义
SELECT * FROM users WHERE username LIKE '%\%%' ESCAPE '\';

-- 正则表达式（8.0+）
SELECT * FROM users WHERE username REGEXP '^张.*三$';

2.6 日期条件

-- 日期比较
SELECT * FROM orders 
WHERE created_at >= '2024-01-01' AND created_at < '2024-02-01';

-- 使用日期函数
SELECT * FROM orders 
WHERE DATE(created_at) = '2024-03-25';

-- 最近 7 天
SELECT * FROM orders 
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY);

-- 本月
SELECT * FROM orders 
WHERE YEAR(created_at) = YEAR(NOW()) 
  AND MONTH(created_at) = MONTH(NOW());

3. ORDER BY 排序

-- 单列排序（默认 ASC）
SELECT * FROM products ORDER BY price;

-- 降序
SELECT * FROM products ORDER BY price DESC;

-- 多列排序
SELECT * FROM products ORDER BY category_id ASC, price DESC;

-- 按表达式排序
SELECT *, price * stock AS total_value 
FROM products 
ORDER BY total_value DESC;

-- 按 FIELD 自定义排序
SELECT * FROM products 
ORDER BY FIELD(status, 1, 2, 0);  -- 1 在前，2 其次，0 最后

-- 按 NULL 排序
SELECT * FROM users ORDER BY phone IS NULL, phone;

4. LIMIT 分页

-- 限制数量
SELECT * FROM products LIMIT 10;

-- 偏移量 + 数量
SELECT * FROM products LIMIT 0, 10;  -- 第 1-10 条
SELECT * FROM products LIMIT 10, 10; -- 第 11-20 条
SELECT * FROM products LIMIT 20, 10; -- 第 21-30 条

-- 分页公式
-- 第 N 页，每页 M 条：LIMIT (N-1)*M, M
SELECT * FROM products LIMIT 90, 10;  -- 第 10 页，每页 10 条

-- 带排序的分页
SELECT * FROM products ORDER BY created_at DESC LIMIT 0, 10;

5. 聚合函数

5.1 常用聚合函数

-- COUNT 计数
SELECT COUNT(*) FROM users;           -- 所有行数
SELECT COUNT(id) FROM users;          -- id 非 NULL 的行数
SELECT COUNT(DISTINCT status) FROM users;  -- 不同状态数

-- SUM 求和
SELECT SUM(price * stock) AS total_value FROM products;

-- AVG 平均值
SELECT AVG(price) AS avg_price FROM products;

-- MAX/MAX 最大/最小值
SELECT MAX(price) AS max_price FROM products;
SELECT MIN(price) AS min_price FROM products;

-- GROUP_CONCAT 字符串拼接
SELECT status, GROUP_CONCAT(username) AS users 
FROM users GROUP BY status;

5.2 GROUP BY 分组

-- 单列分组
SELECT status, COUNT(*) AS count 
FROM users 
GROUP BY status;

-- 多列分组
SELECT category_id, status, COUNT(*) AS count, SUM(price) AS total
FROM products 
GROUP BY category_id, status;

-- 分组后排序
SELECT status, COUNT(*) AS count 
FROM users 
GROUP BY status 
ORDER BY count DESC;

-- 分组后过滤（HAVING）
SELECT status, COUNT(*) AS count 
FROM users 
GROUP BY status 
HAVING count > 100;

-- WHERE vs HAVING
-- WHERE: 分组前过滤，不能使用聚合函数
-- HAVING: 分组后过滤，可以使用聚合函数
SELECT status, COUNT(*) AS count 
FROM users 
WHERE created_at >= '2024-01-01'
GROUP BY status 
HAVING count > 10;

6. JOIN 连接查询

6.1 内连接（INNER JOIN）

-- 标准语法
SELECT u.username, o.order_no, o.total_amount
FROM users u
INNER JOIN orders o ON u.id = o.user_id;

-- 省略 INNER
SELECT u.username, o.order_no
FROM users u
JOIN orders o ON u.id = o.user_id;

-- 多表连接
SELECT u.username, o.order_no, p.product_name, oi.quantity
FROM users u
JOIN orders o ON u.id = o.user_id
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id;

6.2 左连接（LEFT JOIN）

-- 左表所有记录 + 右表匹配记录
SELECT u.username, o.order_no
FROM users u
LEFT JOIN orders o ON u.id = o.user_id;

-- 查询没有订单的用户
SELECT u.username
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
WHERE o.id IS NULL;

6.3 右连接（RIGHT JOIN）

-- 右表所有记录 + 左表匹配记录
SELECT u.username, o.order_no
FROM users u
RIGHT JOIN orders o ON u.id = o.user_id;

-- 查询没有用户的订单（理论上不应该存在）
SELECT o.order_no
FROM users u
RIGHT JOIN orders o ON u.id = o.user_id
WHERE u.id IS NULL;

6.4 交叉连接（CROSS JOIN）

-- 笛卡尔积
SELECT * FROM colors CROSS JOIN sizes;

-- 等价于
SELECT * FROM colors, sizes;

6.5 自连接

-- 查询员工及其上级
SELECT 
    e.name AS employee,
    m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- 查询同一分类下的商品
SELECT p1.name, p2.name
FROM products p1
JOIN products p2 ON p1.category_id = p2.category_id
WHERE p1.id < p2.id;

7. 子查询

7.1 WHERE 子句中的子查询

-- 返回单值
SELECT * FROM products 
WHERE price > (SELECT AVG(price) FROM products);

-- 返回列表
SELECT * FROM products 
WHERE category_id IN (SELECT id FROM categories WHERE level = 1);

-- 使用 EXISTS
SELECT * FROM users u
WHERE EXISTS (SELECT 1 FROM orders o WHERE o.user_id = u.id);

-- 使用 NOT EXISTS
SELECT * FROM users u
WHERE NOT EXISTS (SELECT 1 FROM orders o WHERE o.user_id = u.id);

7.2 FROM 子句中的子查询

-- 派生表
SELECT category_id, avg_price
FROM (
    SELECT category_id, AVG(price) AS avg_price
    FROM products
    GROUP BY category_id
) AS subquery
WHERE avg_price > 100;

7.3 SELECT 子句中的子查询

-- 标量子查询
SELECT 
    p.name,
    p.price,
    (SELECT AVG(price) FROM products) AS avg_price,
    (SELECT name FROM categories WHERE id = p.category_id) AS category_name
FROM products p;

7.4 相关子查询

-- 查询每个分类下价格最高的商品
SELECT p1.*
FROM products p1
WHERE p1.price = (
    SELECT MAX(p2.price)
    FROM products p2
    WHERE p2.category_id = p1.category_id
);

8. 联合查询（UNION）

-- UNION（去重）
SELECT username, email FROM users WHERE status = 1
UNION
SELECT username, email FROM users WHERE created_at >= '2024-01-01';

-- UNION ALL（不去重，性能更好）
SELECT username, email FROM users WHERE status = 1
UNION ALL
SELECT username, email FROM users WHERE created_at >= '2024-01-01';

-- 多表联合
SELECT 'user' AS type, id, username FROM users
UNION ALL
SELECT 'admin' AS type, id, username FROM admins;

-- ORDER BY 和 LIMIT
(SELECT * FROM products WHERE price < 100 ORDER BY price LIMIT 10)
UNION ALL
(SELECT * FROM products WHERE price > 1000 ORDER BY price DESC LIMIT 10)
ORDER BY price;

9. 常用函数

9.1 字符串函数

SELECT 
    CONCAT(username, '@example.com') AS email,  -- 拼接
    LENGTH(username) AS len,                     -- 长度（字节）
    CHAR_LENGTH(username) AS char_len,           -- 长度（字符）
    UPPER(username) AS upper_name,               -- 大写
    LOWER(username) AS lower_name,               -- 小写
    TRIM('  hello  ') AS trimmed,                -- 去空格
    SUBSTRING(username, 1, 3) AS prefix,         -- 子串
    REPLACE(username, 'old', 'new') AS replaced, -- 替换
    INSTR(username, 'test') AS position,         -- 位置
    LPAD(id, 6, '0') AS padded_id                -- 左填充
FROM users;

9.2 数值函数

SELECT 
    ROUND(3.14159, 2) AS rounded,      -- 四舍五入
    CEIL(3.14) AS ceiled,              -- 向上取整
    FLOOR(3.14) AS floored,            -- 向下取整
    ABS(-10) AS absolute,              -- 绝对值
    MOD(10, 3) AS remainder,           -- 取模
    POWER(2, 3) AS powered,            -- 幂运算
    RAND() AS random,                  -- 随机数
    TRUNCATE(3.14159, 2) AS truncated  -- 截断
FROM dual;

9.3 日期函数

SELECT 
    NOW() AS now,                      -- 当前日期时间
    CURDATE() AS today,                -- 当前日期
    CURTIME() AS current_time,         -- 当前时间
    DATE(NOW()) AS date_part,          -- 日期部分
    TIME(NOW()) AS time_part,          -- 时间部分
    YEAR(NOW()) AS year,               -- 年
    MONTH(NOW()) AS month,             -- 月
    DAY(NOW()) AS day,                 -- 日
    HOUR(NOW()) AS hour,               -- 时
    MINUTE(NOW()) AS minute,           -- 分
    SECOND(NOW()) AS second,           -- 秒
    WEEKDAY(NOW()) AS weekday,         -- 星期几（0-6）
    DATE_FORMAT(NOW(), '%Y-%m-%d') AS formatted,  -- 格式化
    DATE_ADD(NOW(), INTERVAL 7 DAY) AS next_week, -- 加 7 天
    DATE_SUB(NOW(), INTERVAL 1 MONTH) AS last_month, -- 减 1 月
    DATEDIFF('2024-12-31', NOW()) AS days_left,  -- 日期差
    TIMESTAMPDIFF(YEAR, '1990-01-01', NOW()) AS age  -- 时间差
FROM dual;

9.4 条件函数

SELECT 
    IF(status = 1, '正常', '禁用') AS status_text,
    IFNULL(phone, '未填写') AS phone,
    NULLIF(username, 'admin') AS safe_username,
    CASE 
        WHEN price < 100 THEN '便宜'
        WHEN price < 500 THEN '中等'
        ELSE '昂贵'
    END AS price_level,
    CASE status
        WHEN 0 THEN '禁用'
        WHEN 1 THEN '正常'
        ELSE '未知'
    END AS status_desc
FROM products;

10. 实战示例

10.1 电商查询示例

-- 1. 查询用户订单统计
SELECT 
    u.id,
    u.username,
    COUNT(o.id) AS order_count,
    SUM(o.total_amount) AS total_spent,
    AVG(o.total_amount) AS avg_order_value,
    MAX(o.created_at) AS last_order_date
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
GROUP BY u.id, u.username
HAVING order_count > 0
ORDER BY total_spent DESC
LIMIT 10;

-- 2. 商品销售排行
SELECT 
    p.id,
    p.name,
    p.category_id,
    SUM(oi.quantity) AS total_sold,
    SUM(oi.subtotal) AS total_revenue
FROM products p
JOIN order_items oi ON p.id = oi.product_id
JOIN orders o ON oi.order_id = o.id
WHERE o.status >= 1  -- 已付款
GROUP BY p.id, p.name, p.category_id
ORDER BY total_sold DESC
LIMIT 20;

-- 3. 每月销售统计
SELECT 
    DATE_FORMAT(created_at, '%Y-%m') AS month,
    COUNT(*) AS order_count,
    SUM(total_amount) AS total_revenue,
    AVG(total_amount) AS avg_order_value
FROM orders
WHERE status >= 1
GROUP BY DATE_FORMAT(created_at, '%Y-%m')
ORDER BY month DESC;

-- 4. 库存预警
SELECT 
    id,
    name,
    stock,
    CASE 
        WHEN stock = 0 THEN '缺货'
        WHEN stock < 10 THEN '紧张'
        WHEN stock < 50 THEN '充足'
        ELSE '过剩'
    END AS stock_status
FROM products
WHERE stock < 50
ORDER BY stock ASC;

-- 5. 用户复购分析
SELECT 
    user_id,
    COUNT(*) AS order_count,
    MIN(created_at) AS first_order,
    MAX(created_at) AS last_order,
    DATEDIFF(MAX(created_at), MIN(created_at)) AS days_span
FROM orders
WHERE status >= 1
GROUP BY user_id
HAVING order_count >= 2
ORDER BY order_count DESC;

11. 本章小结

核心知识点

✅ WHERE 条件过滤
✅ ORDER BY 排序
✅ LIMIT 分页
✅ 聚合函数与 GROUP BY
✅ JOIN 连接查询
✅ 子查询
✅ UNION 联合查询
✅ 常用函数

最佳实践

  1. 避免 SELECT *：明确指定需要的列
  2. 使用表别名：提高可读性
  3. 合理使用索引：WHERE、JOIN、ORDER BY 列
  4. 分页优化：大数据量时使用覆盖索引
  5. 避免 N+1 查询：使用 JOIN 替代多次查询
