11_命令速查

MySQL 常用命令分类速查表

1. 数据库操作

命令 说明 示例
CREATE DATABASE 创建数据库 CREATE DATABASE db_name;
SHOW DATABASES 查看所有数据库 SHOW DATABASES;
USE 选择数据库 USE db_name;
DROP DATABASE 删除数据库 DROP DATABASE db_name;
ALTER DATABASE 修改数据库 ALTER DATABASE db_name CHARACTER SET utf8mb4;
SELECT DATABASE() 查看当前数据库 SELECT DATABASE();

2. 表操作

2.1 创建表

命令 说明 示例
CREATE TABLE 创建表 CREATE TABLE t (id INT PRIMARY KEY);
CREATE TABLE ... LIKE 复制表结构 CREATE TABLE t2 LIKE t1;
CREATE TABLE ... AS 复制表结构和数据 CREATE TABLE t2 AS SELECT * FROM t1;
SHOW TABLES 查看所有表 SHOW TABLES;
DESCRIBE 查看表结构 DESCRIBE table_name;
SHOW CREATE TABLE 查看建表语句 SHOW CREATE TABLE table_name;
SHOW COLUMNS 查看列信息 SHOW COLUMNS FROM table_name;
SHOW INDEX 查看索引 SHOW INDEX FROM table_name;
DROP TABLE 删除表 DROP TABLE table_name;
TRUNCATE TABLE 清空表 TRUNCATE TABLE table_name;
RENAME TABLE 重命名表 RENAME TABLE old TO new;

2.2 修改表

命令 说明 示例
ALTER TABLE ... ADD 添加列 ALTER TABLE t ADD col VARCHAR(50);
ALTER TABLE ... DROP 删除列 ALTER TABLE t DROP col;
ALTER TABLE ... MODIFY 修改列类型 ALTER TABLE t MODIFY col INT;
ALTER TABLE ... CHANGE 修改列名 ALTER TABLE t CHANGE old new INT;
ALTER TABLE ... ADD INDEX 添加索引 ALTER TABLE t ADD INDEX idx_col (col);
ALTER TABLE ... DROP INDEX 删除索引 ALTER TABLE t DROP INDEX idx_col;
ALTER TABLE ... ADD PRIMARY KEY 添加主键 ALTER TABLE t ADD PRIMARY KEY (id);
ALTER TABLE ... ADD FOREIGN KEY 添加外键 ALTER TABLE t ADD FOREIGN KEY (uid) REFERENCES u(id);
ALTER TABLE ... ENGINE 修改引擎 ALTER TABLE t ENGINE=InnoDB;
ALTER TABLE ... AUTO_INCREMENT 修改自增值 ALTER TABLE t AUTO_INCREMENT=1000;

3. 数据操作（DML）

3.1 插入

命令 说明 示例
INSERT INTO ... VALUES 插入单行 INSERT INTO t (id, name) VALUES (1, 'a');
INSERT INTO ... VALUES (...), (...) 插入多行 INSERT INTO t VALUES (1,'a'), (2,'b');
INSERT INTO ... SELECT 从查询插入 INSERT INTO t2 SELECT * FROM t1;
INSERT IGNORE 忽略错误 INSERT IGNORE INTO t (id) VALUES (1);
REPLACE INTO 替换（删除 + 插入） REPLACE INTO t (id, name) VALUES (1, 'a');
INSERT ... ON DUPLICATE KEY UPDATE 冲突时更新 INSERT INTO t (id, v) VALUES (1, 1) ON DUPLICATE KEY UPDATE v=v+1;

3.2 更新

命令 说明 示例
UPDATE ... SET 更新数据 UPDATE t SET name='b' WHERE id=1;
UPDATE ... JOIN 多表更新 UPDATE t1 JOIN t2 ON t1.id=t2.id SET t1.v=t2.v;
UPDATE ... ORDER BY 排序更新 UPDATE t SET status=0 ORDER BY created_at LIMIT 10;

3.3 删除

命令 说明 示例
DELETE FROM 删除数据 DELETE FROM t WHERE id=1;
DELETE ... ORDER BY LIMIT 限制删除 DELETE FROM t ORDER BY created_at LIMIT 100;
DELETE ... JOIN 多表删除 DELETE t1, t2 FROM t1 JOIN t2 ON t1.id=t2.id WHERE t1.status=0;
TRUNCATE TABLE 清空表 TRUNCATE TABLE t;

3.4 查询

命令 说明 示例
SELECT * FROM 查询所有列 SELECT * FROM t;
SELECT col1, col2 FROM 查询指定列 SELECT id, name FROM t;
SELECT DISTINCT 去重查询 SELECT DISTINCT status FROM t;
SELECT ... WHERE 条件查询 SELECT * FROM t WHERE status=1;
SELECT ... ORDER BY 排序查询 SELECT * FROM t ORDER BY created_at DESC;
SELECT ... LIMIT 限制结果 SELECT * FROM t LIMIT 10;
SELECT ... LIMIT offset, count 分页查询 SELECT * FROM t LIMIT 0, 10;
SELECT ... GROUP BY 分组查询 SELECT status, COUNT(*) FROM t GROUP BY status;
SELECT ... HAVING 分组过滤 SELECT status, COUNT(*) FROM t GROUP BY status HAVING COUNT(*)>10;
SELECT ... JOIN 连接查询 SELECT * FROM t1 JOIN t2 ON t1.id=t2.id;
SELECT ... UNION 联合查询 SELECT id FROM t1 UNION SELECT id FROM t2;
SELECT ... INTO OUTFILE 导出到文件 SELECT * FROM t INTO OUTFILE '/tmp/t.csv';

4. 索引操作

命令 说明 示例
CREATE INDEX 创建索引 CREATE INDEX idx_name ON t(name);
CREATE UNIQUE INDEX 创建唯一索引 CREATE UNIQUE INDEX uk_email ON t(email);
DROP INDEX 删除索引 DROP INDEX idx_name ON t;
ALTER TABLE ... ADD INDEX 添加索引 ALTER TABLE t ADD INDEX idx_name (name);
ALTER TABLE ... DROP INDEX 删除索引 ALTER TABLE t DROP INDEX idx_name;
SHOW INDEX FROM 查看索引 SHOW INDEX FROM t;
ANALYZE TABLE 分析索引 ANALYZE TABLE t;
OPTIMIZE TABLE 优化表 OPTIMIZE TABLE t;

5. 事务控制

命令 说明 示例
START TRANSACTION 开始事务 START TRANSACTION;
BEGIN 开始事务 BEGIN;
COMMIT 提交事务 COMMIT;
ROLLBACK 回滚事务 ROLLBACK;
SAVEPOINT 设置保存点 SAVEPOINT sp1;
ROLLBACK TO 回滚到保存点 ROLLBACK TO sp1;
SET autocommit 设置自动提交 SET autocommit=0;
SET TRANSACTION ISOLATION LEVEL 设置隔离级别 SET TRANSACTION ISOLATION LEVEL READ COMMITTED;

6. 用户与权限

命令 说明 示例
CREATE USER 创建用户 CREATE USER 'u'@'localhost' IDENTIFIED BY 'pwd';
DROP USER 删除用户 DROP USER 'u'@'localhost';
GRANT 授权 GRANT SELECT, INSERT ON db.* TO 'u'@'localhost';
REVOKE 撤销权限 REVOKE INSERT ON db.* FROM 'u'@'localhost';
SHOW GRANTS 查看权限 SHOW GRANTS FOR 'u'@'localhost';
FLUSH PRIVILEGES 刷新权限 FLUSH PRIVILEGES;
SET PASSWORD 修改密码 SET PASSWORD FOR 'u'@'localhost' = PASSWORD('newpwd');
ALTER USER 修改用户 ALTER USER 'u'@'localhost' IDENTIFIED BY 'newpwd';

7. 系统管理

7.1 服务器信息

命令 说明 示例
SELECT VERSION() 查看版本 SELECT VERSION();
SHOW STATUS 查看状态 SHOW STATUS;
SHOW VARIABLES 查看变量 SHOW VARIABLES;
SHOW VARIABLES LIKE 查看指定变量 SHOW VARIABLES LIKE 'max_connections';
SHOW PROCESSLIST 查看进程 SHOW PROCESSLIST;
SHOW ENGINE INNODB STATUS InnoDB 状态 SHOW ENGINE INNODB STATUS;
SELECT CONNECTION_ID() 当前连接 ID SELECT CONNECTION_ID();

7.2 配置管理

命令 说明 示例
SET GLOBAL 设置全局变量 SET GLOBAL max_connections=500;
SET SESSION 设置会话变量 SET SESSION wait_timeout=300;
SET 设置变量 SET @var=1;
SHOW GLOBAL VARIABLES 查看全局变量 SHOW GLOBAL VARIABLES;
CONFIG REWRITE 重写配置（8.0+） 不适用 MySQL

7.3 日志管理

命令 说明 示例
SHOW BINARY LOGS 查看 Binlog SHOW BINARY LOGS;
SHOW MASTER STATUS 主库状态 SHOW MASTER STATUS;
SHOW SLAVE STATUS 从库状态 SHOW SLAVE STATUS\G
PURGE BINARY LOGS 清理 Binlog PURGE BINARY LOGS TO 'mysql-bin.000010';
RESET MASTER 重置主库 RESET MASTER;
RESET SLAVE 重置从库 RESET SLAVE;
FLUSH LOGS 刷新日志 FLUSH LOGS;
FLUSH SLOW LOGS 刷新慢查询日志 FLUSH SLOW LOGS;

7.4 连接管理

命令 说明 示例
KILL 杀死连接 KILL 123;
KILL QUERY 杀死查询 KILL QUERY 123;
SHOW PROCESSLIST 查看连接 SHOW PROCESSLIST;

8. 备份与恢复

命令 说明 示例
mysqldump 逻辑备份 mysqldump -u root -p db > backup.sql
mysql 恢复数据 mysql -u root -p db < backup.sql
SOURCE 执行 SQL 文件 SOURCE /path/to/backup.sql;
SELECT ... INTO OUTFILE 导出数据 SELECT * FROM t INTO OUTFILE '/tmp/t.csv';
LOAD DATA INFILE 导入数据 LOAD DATA INFILE '/tmp/t.csv' INTO TABLE t;

9. 常用函数

9.1 字符串函数

函数 说明 示例
CONCAT 拼接字符串 CONCAT('Hello', ' ', 'World')
LENGTH 字节长度 LENGTH('中文') → 6
CHAR_LENGTH 字符长度 CHAR_LENGTH('中文') → 2
UPPER/LOWER 大小写转换 UPPER('abc') → 'ABC'
TRIM 去空格 TRIM(' abc ') → 'abc'
SUBSTRING 子串 SUBSTRING('abcdef', 1, 3) → 'abc'
REPLACE 替换 REPLACE('abc', 'b', 'x') → 'axc'
INSTR 查找位置 INSTR('abcdef', 'cd') → 3
LPAD/RPAD 填充 LPAD('1', 3, '0') → '001'
GROUP_CONCAT 分组拼接 GROUP_CONCAT(name)

9.2 数值函数

函数 说明 示例
ROUND 四舍五入 ROUND(3.14159, 2) → 3.14
CEIL/FLOOR 向上/向下取整 CEIL(3.14) → 4
ABS 绝对值 ABS(-10) → 10
MOD 取模 MOD(10, 3) → 1
POWER 幂运算 POWER(2, 3) → 8
RAND 随机数 RAND()
TRUNCATE 截断 TRUNCATE(3.14159, 2) → 3.14

9.3 日期函数

函数 说明 示例
NOW() 当前日期时间 NOW()
CURDATE() 当前日期 CURDATE()
CURTIME() 当前时间 CURTIME()
DATE() 日期部分 DATE(NOW())
TIME() 时间部分 TIME(NOW())
YEAR/MONTH/DAY 年/月/日 YEAR(NOW())
HOUR/MINUTE/SECOND 时/分/秒 HOUR(NOW())
DATE_FORMAT 格式化日期 DATE_FORMAT(NOW(), '%Y-%m-%d')
DATE_ADD 日期加法 DATE_ADD(NOW(), INTERVAL 7 DAY)
DATE_SUB 日期减法 DATE_SUB(NOW(), INTERVAL 1 MONTH)
DATEDIFF 日期差 DATEDIFF('2024-12-31', NOW())
TIMESTAMPDIFF 时间差 TIMESTAMPDIFF(YEAR, '1990-01-01', NOW())
UNIX_TIMESTAMP Unix 时间戳 UNIX_TIMESTAMP()
FROM_UNIXTIME 时间戳转日期 FROM_UNIXTIME(1711353600)

9.4 条件函数

函数 说明 示例
IF 条件判断 IF(status=1, '正常', '禁用')
IFNULL NULL 判断 IFNULL(phone, '未填写')
NULLIF 相等返回 NULL NULLIF(a, b)
CASE 条件表达式 CASE WHEN x>0 THEN '正' ELSE '负' END

9.5 聚合函数

函数 说明 示例
COUNT 计数 COUNT(*), COUNT(id)
SUM 求和 SUM(amount)
AVG 平均值 AVG(price)
MAX 最大值 MAX(price)
MIN 最小值 MIN(price)
GROUP_CONCAT 字符串拼接 GROUP_CONCAT(name)

10. 性能分析

命令 说明 示例
EXPLAIN 执行计划 EXPLAIN SELECT * FROM t WHERE id=1;
EXPLAIN FORMAT=JSON JSON 格式执行计划 EXPLAIN FORMAT=JSON SELECT ...;
EXPLAIN ANALYZE 实时分析（8.0+） EXPLAIN ANALYZE SELECT ...;
SHOW PROFILE 查询性能分析 SHOW PROFILE FOR QUERY 1;
SHOW SLOW LOG 慢查询日志 查看慢查询日志文件
SLOW_QUERY_LOG 开启慢查询 SET GLOBAL slow_query_log=ON;

11. 复制相关

命令 说明 示例
CHANGE MASTER TO 配置主从 CHANGE MASTER TO MASTER_HOST='x.x.x.x';

STOP SLAVE 停止复制 STOP SLAVE;

12. 常用组合

12.1 分页查询

-- 第 N 页，每页 M 条
SELECT * FROM table LIMIT (N-1)*M, M;

12.2 批量插入

INSERT INTO t (id, name) VALUES 
(1, 'a'), (2, 'b'), (3, 'c'), ...;

12.3 条件更新

UPDATE t SET status = CASE id
    WHEN 1 THEN 1
    WHEN 2 THEN 0
    ELSE status
END WHERE id IN (1, 2);

12.4 去重删除

DELETE t1 FROM t t1
JOIN t t2 ON t1.id > t2.id AND t1.name = t2.name;

12.5 随机查询

SELECT * FROM t ORDER BY RAND() LIMIT 10;
