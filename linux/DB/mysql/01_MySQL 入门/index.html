01_MySQL 入门

MySQL 是什么？为什么它如此流行？如何开始使用 MySQL？

1. MySQL 简介

1.1 什么是 MySQL？

MySQL 是一个开源的关系型数据库管理系统（RDBMS），由 Oracle 公司开发和维护。

核心特点：

特性 说明
开源 GPL 协议，可自由使用
关系型 支持表、行、列、外键等关系模型
高性能 优化的查询引擎和索引机制
可扩展 支持主从复制、集群、分库分表
跨平台 支持 Linux、Windows、macOS
多语言 支持 Python、Java、PHP、Go 等

1.2 MySQL 历史

1995 年  →  MySQL AB 成立
2008 年  →  Sun Microsystems 收购 MySQL AB
2010 年  →  Oracle 收购 Sun，MySQL 归 Oracle 所有
2013 年  →  MySQL 5.6 发布
2015 年  →  MySQL 5.7 发布
2018 年  →  MySQL 8.0 发布
2024 年  →  MySQL 8.4 LTS 发布

1.3 应用场景

场景 说明
Web 应用 WordPress、Drupal 等 CMS
电商系统 商品、订单、用户管理
金融系统 账户、交易、账务管理
日志系统 操作日志、访问日志存储
数据分析 BI 报表、数据仓库

4.1 配置文件位置

4.2 核心配置项

# ==================== 客户端配置 ====================
[client]
# 默认字符集
default-character-set = utf8mb4

# 默认端口
port = 3306

# ==================== 服务端配置 ====================
[mysqld]
# 端口
port = 3306

# 绑定地址（0.0.0.0 允许远程连接）
bind-address = 0.0.0.0

# 字符集
character-set-server = utf8mb4
collation-server = utf8mb4_unicode_ci

# 数据目录
datadir = /var/lib/mysql

# socket 文件
socket = /var/run/mysqld/mysqld.sock

# ==================== 连接配置 ====================
# 最大连接数
max_connections = 200

# 连接超时时间（秒）
connect_timeout = 10

# 等待超时（秒）
wait_timeout = 28800

# 交互超时（秒）
interactive_timeout = 28800

# ==================== 内存配置 ====================
# 缓冲池大小（重要！）
innodb_buffer_pool_size = 1G

# 缓冲池实例数
innodb_buffer_pool_instances = 8

# 日志缓冲区大小
innodb_log_buffer_size = 16M

# 查询缓存（8.0 已移除）
# query_cache_size = 64M

# ==================== 日志配置 ====================
# 错误日志
log_error = /var/log/mysql/error.log

# 慢查询日志
slow_query_log = 1
slow_query_log_file = /var/log/mysql/slow.log
long_query_time = 2

# 通用查询日志（生产环境慎用）
# general_log = 1
# general_log_file = /var/log/mysql/general.log

# Binlog 配置
log_bin = /var/log/mysql/mysql-bin
binlog_format = ROW
expire_logs_days = 7
max_binlog_size = 100M

# ==================== InnoDB 配置 ====================
# 存储引擎
default-storage-engine = InnoDB

# 日志文件大小
innodb_log_file_size = 256M

# 日志文件数量
innodb_log_files_in_group = 2

# 刷新方式
innodb_flush_log_at_trx_commit = 1

# 文件每表
innodb_file_per_table = 1

# ==================== 安全配置 ====================
# 跳过名称解析（提高性能）
skip-name-resolve

# 本地文件加载
local_infile = 0

# ==================== 其他配置 ====================
# PID 文件
pid-file = /var/run/mysqld/mysqld.pid

# 用户
user = mysql

# 符号链接
symbolic-links = 0

6. 第一个 MySQL 程序

6.1 创建数据库和表

-- 创建数据库
CREATE DATABASE IF NOT EXISTS shop_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

-- 选择数据库
USE shop_db;

-- 创建用户表
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建商品表
CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    price DECIMAL(10, 2) NOT NULL,
    stock INT DEFAULT 0,
    category_id INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建订单表
CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    total_amount DECIMAL(10, 2) NOT NULL,
    status ENUM('pending', 'paid', 'shipped', 'completed', 'cancelled') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建订单详情表
CREATE TABLE order_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id),
    FOREIGN KEY (product_id) REFERENCES products(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

6.3 查询数据

-- 查询所有用户
SELECT * FROM users;

-- 查询商品列表
SELECT id, name, price, stock FROM products;

-- 条件查询
SELECT * FROM products WHERE price < 100;

-- 排序查询
SELECT * FROM products ORDER BY price DESC;

-- 限制结果
SELECT * FROM products LIMIT 5;

-- 连接查询
SELECT 
    o.id AS order_id,
    u.username,
    o.total_amount,
    o.status
FROM orders o
JOIN users u ON o.user_id = u.id;

-- 聚合查询
SELECT 
    status,
    COUNT(*) AS order_count,
    SUM(total_amount) AS total_amount
FROM orders
GROUP BY status;

6.4 Python 连接 MySQL

# 安装依赖
# pip install mysql-connector-python

import mysql.connector
from mysql.connector import Error

# 连接数据库
try:
    connection = mysql.connector.connect(
        host='localhost',
        port=3306,
        user='root',
        password='your-password',
        database='shop_db'
    )

    if connection.is_connected():
        print("✅ 连接成功")

        # 创建游标
        cursor = connection.cursor()

        # 查询用户
        cursor.execute("SELECT id, username, email FROM users")
        users = cursor.fetchall()
        for user in users:
            print(f"用户：{user[1]}, 邮箱：{user[2]}")

        # 插入数据
        insert_sql = """
        INSERT INTO users (username, email, password) 
        VALUES (%s, %s, %s)
        """
        user_data = ('zhaoliu', 'zhaoliu@example.com', 'hashed_password_4')
        cursor.execute(insert_sql, user_data)
        connection.commit()
        print(f"✅ 插入成功，ID: {cursor.lastrowid}")

        # 更新数据
        update_sql = "UPDATE users SET email = %s WHERE username = %s"
        cursor.execute(update_sql, ('newemail@example.com', 'zhaoliu'))
        connection.commit()

        # 删除数据
        delete_sql = "DELETE FROM users WHERE username = %s"
        cursor.execute(delete_sql, ('zhaoliu',))
        connection.commit()

except Error as e:
    print(f"❌ 错误：{e}")
finally:
    if connection.is_connected():
        cursor.close()
        connection.close()
        print("✅ 连接已关闭")

6.5 Java 连接 MySQL（JDBC）

// Maven 依赖

import java.sql.*;

        String password = "your-password";

        String sql = "SELECT id, username, email FROM users";

        try (Connection conn = DriverManager.getConnection(url, user, password);
             Statement stmt = conn.createStatement();

            System.out.println("✅ 连接成功");

        } catch (SQLException e) {
            System.out.println("❌ 错误：" + e.getMessage());
        }
    }
}

6.6 Node.js 连接 MySQL

// npm install mysql2

const mysql = require('mysql2/promise');

async function main() {
    // 创建连接
    const connection = await mysql.createConnection({
        host: 'localhost',
        port: 3306,
        user: 'root',
        password: 'your-password',
        database: 'shop_db'
    });

    console.log('✅ 连接成功');

    try {
        // 查询用户
        const [rows] = await connection.execute(
            'SELECT id, username, email FROM users'
        );

        for (const row of rows) {
            console.log(`用户：${row.username}, 邮箱：${row.email}`);
        }

        // 插入数据
        const [result] = await connection.execute(
            'INSERT INTO users (username, email, password) VALUES (?, ?, ?)',
            ['zhaoliu', 'zhaoliu@example.com', 'hashed_password']
        );
        console.log(`✅ 插入成功，ID: ${result.insertId}`);

        // 事务示例
        await connection.beginTransaction();
        try {
            await connection.execute(
                'UPDATE products SET stock = stock - 1 WHERE id = ?',
                [1]
            );
            await connection.execute(
                'INSERT INTO orders (user_id, total_amount) VALUES (?, ?)',
                [1, 7999.00]
            );
            await connection.commit();
            console.log('✅ 事务提交成功');
        } catch (err) {
            await connection.rollback();
            console.log('❌ 事务回滚');
        }

    } catch (error) {
        console.error('❌ 错误:', error.message);
    } finally {
        await connection.end();
        console.log('✅ 连接已关闭');
    }
}

main();

-- 撤销特定权限

8. 备份与恢复

8.1 使用 mysqldump 备份

# 备份单个数据库
mysqldump -u root -p shop_db > shop_db_backup.sql

# 备份多个数据库
mysqldump -u root -p --databases db1 db2 > backup.sql

# 备份所有数据库
mysqldump -u root -p --all-databases > all_backup.sql

# 只备份结构
mysqldump -u root -p --no-data shop_db > structure.sql

# 只备份数据
mysqldump -u root -p --no-create-info shop_db > data.sql

# 压缩备份
mysqldump -u root -p shop_db | gzip > shop_db_backup.sql.gz

8.2 恢复数据

# 恢复数据库
mysql -u root -p shop_db < shop_db_backup.sql

# 恢复压缩备份
gunzip < shop_db_backup.sql.gz | mysql -u root -p shop_db

# 在 MySQL 中恢复
mysql> SOURCE /path/to/backup.sql;

8.3 自动备份脚本

#!/bin/bash
# backup-mysql.sh

BACKUP_DIR="/backup/mysql"
DATE=$(date +%Y%m%d_%H%M%S)
MYSQL_USER="root"
MYSQL_PASS="your-password"
DATABASES="shop_db testdb"

mkdir -p $BACKUP_DIR

for db in $DATABASES; do
    mysqldump -u $MYSQL_USER -p$MYSQL_PASS $db | gzip > $BACKUP_DIR/${db}_${DATE}.sql.gz
done

# 删除 7 天前的备份
find $BACKUP_DIR -name "*.sql.gz" -mtime +7 -delete

echo "Backup completed: $DATE"

9. 常见问题

Q1: 忘记 root 密码怎么办？

A:

# 1. 停止 MySQL
sudo systemctl stop mysql

# 2. 跳过授权表启动
sudo mysqld --skip-grant-tables &

# 3. 无密码登录
mysql -u root

# 4. 修改密码
USE mysql;
UPDATE user SET authentication_string=PASSWORD('newpassword') WHERE User='root';
FLUSH PRIVILEGES;
EXIT;

# 5. 重启 MySQL
sudo systemctl restart mysql

Q2: 如何允许远程连接？

A:

-- 修改用户 host
UPDATE mysql.user SET host = '%' WHERE user = 'root';
FLUSH PRIVILEGES;

-- 或创建新用户
CREATE USER 'root'@'%' IDENTIFIED BY 'password';
GRANT ALL PRIVILEGES ON *.* TO 'root'@'%' WITH GRANT OPTION;
FLUSH PRIVILEGES;

Q3: 字符集乱码怎么办？

A:

# my.cnf 配置
[client]
default-character-set = utf8mb4

[mysqld]
character-set-server = utf8mb4
collation-server = utf8mb4_unicode_ci

Q4: 如何查看慢查询？

A:

-- 开启慢查询日志
SET GLOBAL slow_query_log = 'ON';
SET GLOBAL long_query_time = 2;

-- 查看慢查询
SELECT * FROM mysql.slow_log;

-- 或查看日志文件
# /var/log/mysql/slow.log
