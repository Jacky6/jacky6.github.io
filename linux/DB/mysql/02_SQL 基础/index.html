02_SQL 基础

MySQL 数据类型、表操作、约束详解

1. 数据类型

1.1 数值类型

类型 大小 范围 说明
TINYINT 1 字节 -128 ~ 127 小整数，常用于布尔值
SMALLINT 2 字节 -32768 ~ 32767 较小整数
MEDIUMINT 3 字节 -8388608 ~ 8388607 中等整数
INT/INTEGER 4 字节 -21 亿 ~ 21 亿 标准整数
BIGINT 8 字节 -922 亿亿 ~ 922 亿亿 大整数
FLOAT 4 字节 单精度浮点数 近似值
DOUBLE 8 字节 双精度浮点数 近似值
DECIMAL(M,D) 可变 精确小数 精确值，用于金额 
-- 示例
CREATE TABLE numeric_types (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tiny_int TINYINT,           -- 年龄、状态
    small_int SMALLINT,         -- 端口号
    medium_int MEDIUMINT,
    int_col INT,                -- 常用
    big_int BIGINT,             -- 用户 ID、订单号
    float_col FLOAT(10,2),      -- 单精度
    double_col DOUBLE(10,2),    -- 双精度
    decimal_col DECIMAL(10,2)   -- 金额（精确）
);

-- 插入示例
INSERT INTO numeric_types VALUES
(NULL, 100, 30000, 5000000, 2000000000, 9223372036854775807, 
 3.14, 3.1415926, 99999999.99);

1.2 字符串类型

类型 说明 最大长度 适用场景
CHAR(N) 定长字符串 255 字符 固定长度（如手机号）
VARCHAR(N) 变长字符串 65535 字节 常用（如用户名、邮箱）
TINYTEXT 小文本 255 字节 短文本
TEXT 文本 64KB 文章内容
MEDIUMTEXT 中文本 16MB 长文章
LONGTEXT 大文本 4GB 超大文本
BLOB 二进制数据 64KB 图片、文件
ENUM 枚举 65535 值 状态、类型
SET 集合 64 成员 多选标签 
-- 示例
CREATE TABLE string_types (
    id INT AUTO_INCREMENT PRIMARY KEY,
    char_col CHAR(11),          -- 手机号（固定 11 位）
    varchar_col VARCHAR(100),   -- 用户名
    text_col TEXT,              -- 文章内容
    blob_col BLOB,              -- 图片
    enum_col ENUM('active', 'inactive', 'deleted'),  -- 状态
    set_col SET('read', 'write', 'delete')  -- 权限
);

-- 插入示例
INSERT INTO string_types VALUES
(NULL, '13800138000', 'zhangsan', '这是一篇文章...', 
 0x89504E47, 'active', 'read,write');

1.3 日期时间类型

类型 大小 范围 格式 说明
DATE 3 字节 1000-01-01 ~ 9999-12-31 YYYY-MM-DD 日期
TIME 3 字节 -838:59:59 ~ 838:59:59 HH:MM:SS 时间
DATETIME 8 字节 1000-01-01 00:00:00 ~ 9999-12-31 23:59:59 YYYY-MM-DD HH:MM:SS 日期时间
TIMESTAMP 4 字节 1970-01-01 00:00:01 ~ 2038-01-19 03:14:07 YYYY-MM-DD HH:MM:SS 时间戳
YEAR 1 字节 1901 ~ 2155 YYYY 年份 
-- 示例
CREATE TABLE datetime_types (
    id INT AUTO_INCREMENT PRIMARY KEY,
    date_col DATE,              -- 生日
    time_col TIME,              -- 时长
    datetime_col DATETIME,      -- 创建时间
    timestamp_col TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    year_col YEAR               -- 年份
);

-- 插入示例
INSERT INTO datetime_types VALUES
(NULL, '2024-03-25', '12:30:00', '2024-03-25 12:30:00', NOW(), 2024);

-- 日期函数
SELECT 
    CURDATE(),      -- 当前日期
    CURTIME(),      -- 当前时间
    NOW(),          -- 当前日期时间
    UNIX_TIMESTAMP(),  -- Unix 时间戳
    FROM_UNIXTIME(1711353600);  -- 时间戳转日期

1.4 JSON 类型（5.7+）

-- 创建 JSON 列
CREATE TABLE json_example (
    id INT AUTO_INCREMENT PRIMARY KEY,
    data JSON
);

-- 插入 JSON
INSERT INTO json_example (data) VALUES
('{"name": "张三", "age": 25, "skills": ["Python", "Java"]}');

-- 查询 JSON
SELECT data->>'$.name' AS name FROM json_example;
SELECT JSON_EXTRACT(data, '$.age') FROM json_example;

-- 更新 JSON
UPDATE json_example 
SET data = JSON_SET(data, '$.email', 'zhangsan@example.com');

2. 创建数据库

-- 创建数据库
CREATE DATABASE shop_db;

-- 指定字符集
CREATE DATABASE shop_db 
  CHARACTER SET utf8mb4 
  COLLATE utf8mb4_unicode_ci;

-- 如果不存在则创建
CREATE DATABASE IF NOT EXISTS shop_db;

-- 查看所有数据库
SHOW DATABASES;

-- 查看创建语句
SHOW CREATE DATABASE shop_db;

-- 选择数据库
USE shop_db;

-- 查看当前数据库
SELECT DATABASE();

-- 删除数据库
DROP DATABASE IF EXISTS shop_db;

3. 创建表

3.1 基本语法

CREATE TABLE [IF NOT EXISTS] table_name (
    column1 datatype [constraints],
    column2 datatype [constraints],
    ...
    PRIMARY KEY (column),
    FOREIGN KEY (column) REFERENCES other_table(column),
    INDEX (column)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

3.2 完整示例

CREATE TABLE users (
    -- 主键
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    -- 基本信息
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名',
    email VARCHAR(100) NOT NULL UNIQUE COMMENT '邮箱',
    password VARCHAR(255) NOT NULL COMMENT '密码',
    avatar VARCHAR(255) DEFAULT NULL COMMENT '头像',

    -- 个人信息
    nickname VARCHAR(50) DEFAULT NULL COMMENT '昵称',
    gender TINYINT DEFAULT 0 COMMENT '性别：0 未知 1 男 2 女',
    birthday DATE DEFAULT NULL COMMENT '生日',
    phone VARCHAR(20) DEFAULT NULL COMMENT '手机号',

    -- 状态
    status TINYINT DEFAULT 1 COMMENT '状态：0 禁用 1 正常',
    is_deleted TINYINT DEFAULT 0 COMMENT '删除标记：0 否 1 是',

    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    last_login_at TIMESTAMP NULL DEFAULT NULL COMMENT '最后登录时间',

    -- 索引
    INDEX idx_username (username),
    INDEX idx_email (email),
    INDEX idx_status (status),
    INDEX idx_created_at (created_at)

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户表';

4. 约束

4.1 主键约束（PRIMARY KEY）

-- 创建时指定
CREATE TABLE users (
    id INT PRIMARY KEY,
    username VARCHAR(50)
);

-- 复合主键
CREATE TABLE user_roles (
    user_id INT,
    role_id INT,
    PRIMARY KEY (user_id, role_id)
);

-- 自增主键
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50)
);

-- 添加主键
ALTER TABLE users ADD PRIMARY KEY (id);

-- 删除主键
ALTER TABLE users DROP PRIMARY KEY;

4.2 外键约束（FOREIGN KEY）

-- 创建时指定
CREATE TABLE orders (
    id INT PRIMARY KEY,
    user_id INT NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 带级联操作
CREATE TABLE orders (
    id INT PRIMARY KEY,
    user_id INT NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id)
      ON DELETE CASCADE    -- 删除用户时删除订单
      ON UPDATE CASCADE    -- 更新用户 ID 时更新订单
);

-- 添加外键
ALTER TABLE orders 
ADD CONSTRAINT fk_orders_user 
FOREIGN KEY (user_id) REFERENCES users(id);

-- 删除外键
ALTER TABLE orders DROP FOREIGN KEY fk_orders_user;

4.3 唯一约束（UNIQUE）

-- 创建时指定
CREATE TABLE users (
    id INT PRIMARY KEY,
    email VARCHAR(100) UNIQUE
);

-- 复合唯一
CREATE TABLE user_roles (
    user_id INT,
    role_id INT,
    UNIQUE KEY uk_user_role (user_id, role_id)
);

-- 添加唯一约束
ALTER TABLE users ADD UNIQUE (email);

-- 删除唯一约束
ALTER TABLE users DROP INDEX email;

4.4 非空约束（NOT NULL）

-- 创建时指定
CREATE TABLE users (
    id INT PRIMARY KEY,
    username VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL
);

-- 修改为 NOT NULL
ALTER TABLE users MODIFY username VARCHAR(50) NOT NULL;

-- 修改为 NULL
ALTER TABLE users MODIFY username VARCHAR(50) NULL;

4.5 默认约束（DEFAULT）

-- 创建时指定
CREATE TABLE users (
    id INT PRIMARY KEY,
    status TINYINT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 修改默认值
ALTER TABLE users ALTER status SET DEFAULT 0;

-- 删除默认值
ALTER TABLE users ALTER status DROP DEFAULT;

4.6 检查约束（CHECK，8.0.16+）

-- 创建时指定
CREATE TABLE products (
    id INT PRIMARY KEY,
    price DECIMAL(10,2) CHECK (price > 0),
    stock INT CHECK (stock >= 0),
    status ENUM('active', 'inactive') CHECK (status IN ('active', 'inactive'))
);

-- 添加检查约束
ALTER TABLE users ADD CONSTRAINT chk_age CHECK (age >= 0 AND age <= 150);

5. ALTER TABLE 操作

5.1 添加列

-- 添加单列
ALTER TABLE users ADD COLUMN phone VARCHAR(20);

-- 添加列到指定位置
ALTER TABLE users ADD COLUMN nickname VARCHAR(50) AFTER username;

-- 添加列到第一列
ALTER TABLE users ADD COLUMN id BIGINT FIRST;

-- 添加多列
ALTER TABLE users 
  ADD COLUMN phone VARCHAR(20),
  ADD COLUMN address VARCHAR(255);

5.2 修改列

-- 修改数据类型
ALTER TABLE users MODIFY COLUMN phone VARCHAR(30);

-- 修改列名和数据类型
ALTER TABLE users CHANGE COLUMN phone mobile VARCHAR(20);

-- 添加/删除 NOT NULL
ALTER TABLE users MODIFY COLUMN email VARCHAR(100) NOT NULL;
ALTER TABLE users MODIFY COLUMN email VARCHAR(100) NULL;

-- 修改默认值
ALTER TABLE users ALTER COLUMN status SET DEFAULT 1;

5.3 删除列

-- 删除单列
ALTER TABLE users DROP COLUMN phone;

-- 删除多列
ALTER TABLE users 
  DROP COLUMN phone,
  DROP COLUMN address;

5.4 添加/删除索引

-- 添加普通索引
ALTER TABLE users ADD INDEX idx_username (username);

-- 添加唯一索引
ALTER TABLE users ADD UNIQUE INDEX uk_email (email);

-- 添加复合索引
ALTER TABLE users ADD INDEX idx_name_age (username, age);

-- 删除索引
ALTER TABLE users DROP INDEX idx_username;

5.5 添加/删除约束

-- 添加主键
ALTER TABLE users ADD PRIMARY KEY (id);

-- 删除主键
ALTER TABLE users DROP PRIMARY KEY;

-- 添加外键
ALTER TABLE orders ADD CONSTRAINT fk_user 
  FOREIGN KEY (user_id) REFERENCES users(id);

-- 删除外键
ALTER TABLE orders DROP FOREIGN KEY fk_user;

5.6 修改表名

-- 修改表名
ALTER TABLE users RENAME TO user_accounts;

-- 或使用 RENAME TABLE
RENAME TABLE old_table TO new_table;

5.7 修改表选项

-- 修改字符集
ALTER TABLE users CONVERT TO CHARACTER SET utf8mb4;

-- 修改注释
ALTER TABLE users COMMENT = '用户信息表';

-- 修改自增起始值
ALTER TABLE users AUTO_INCREMENT = 1000;

6. 删除表

-- 删除表
DROP TABLE users;

-- 如果存在则删除
DROP TABLE IF EXISTS users;

-- 删除多个表
DROP TABLE IF EXISTS users, orders, products;

-- 截断表（删除所有数据，保留结构）
TRUNCATE TABLE users;

-- TRUNCATE vs DELETE
-- TRUNCATE: 删除所有数据，重置自增，不能回滚，速度快
-- DELETE: 可以带 WHERE，保留自增，可以回滚，速度慢

7. 临时表

-- 创建临时表
CREATE TEMPORARY TABLE temp_users (
    id INT,
    username VARCHAR(50)
);

-- 临时表只在当前会话可见
-- 会话结束后自动删除

-- 从查询结果创建临时表
CREATE TEMPORARY TABLE temp_active_users AS
SELECT * FROM users WHERE status = 1;

    table_name,
    ROUND(data_length / 1024 / 1024, 2) AS data_size_mb,
    ROUND(index_length / 1024 / 1024, 2) AS index_size_mb,
    table_rows
FROM information_schema.tables 
WHERE table_schema = 'shop_db' AND table_name = 'users';

9. 实战示例

9.1 电商系统表结构

-- 用户表
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    nickname VARCHAR(50),
    avatar_url VARCHAR(255),
    phone VARCHAR(20),
    status TINYINT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_username (username),
    INDEX idx_email (email),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 商品分类表
CREATE TABLE categories (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    parent_id INT UNSIGNED DEFAULT NULL,
    level TINYINT DEFAULT 1,
    sort_order INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (parent_id) REFERENCES categories(id),
    INDEX idx_parent (parent_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='商品分类';

-- 商品表
CREATE TABLE products (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(200) NOT NULL,
    description TEXT,
    category_id INT UNSIGNED,
    price DECIMAL(10,2) NOT NULL CHECK (price >= 0),
    stock INT DEFAULT 0 CHECK (stock >= 0),
    status TINYINT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (category_id) REFERENCES categories(id),
    INDEX idx_category (category_id),
    INDEX idx_status (status),
    INDEX idx_price (price)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='商品表';

-- 订单表
CREATE TABLE orders (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_no VARCHAR(50) NOT NULL UNIQUE,
    user_id BIGINT UNSIGNED NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    status TINYINT DEFAULT 0 COMMENT '0 待付款 1 已付款 2 已发货 3 已完成',
    shipping_address JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    paid_at TIMESTAMP NULL,
    shipped_at TIMESTAMP NULL,
    completed_at TIMESTAMP NULL,
    FOREIGN KEY (user_id) REFERENCES users(id),
    INDEX idx_user (user_id),
    INDEX idx_status (status),
    INDEX idx_order_no (order_no)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单表';

-- 订单详情表
CREATE TABLE order_items (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_id BIGINT UNSIGNED NOT NULL,
    product_id BIGINT UNSIGNED NOT NULL,
    product_name VARCHAR(200) NOT NULL,
    quantity INT NOT NULL CHECK (quantity > 0),
    unit_price DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id),
    INDEX idx_order (order_id),
    INDEX idx_product (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单详情';

10. 本章小结

核心知识点

✅ 数值、字符串、日期时间类型
✅ 主键、外键、唯一、非空、默认、检查约束
✅ CREATE/ALTER/DROP TABLE 操作
✅ 索引创建与管理
✅ 查看表信息

最佳实践

  1. 选择合适的数据类型：节省空间，提高性能
  2. 使用 NOT NULL：避免 NULL 值带来的复杂性
  3. 添加注释：便于理解和维护
  4. 合理设计索引：平衡查询性能和写入性能
  5. 使用外键约束：保证数据一致性
