03. 状态管理

Channel - 通道

在 LangGraph 中，Channel 是状态的基本组成单元。每个状态键对应一个 Channel。

默认 Channel 类型

当你使用 TypedDict 定义状态时，LangGraph 会自动为每个字段分配合适的 Channel：

from typing import TypedDict, List

class State(TypedDict):
    value: int              # → LastValue[int]
    messages: List[str]     # → LastValue[List[str]]

显式定义 Channel

from langgraph.graph import StateGraph
from langgraph.channels import LastValue, EphemeralValue, Topic

class State(TypedDict):
    # 普通值：记住最后一次写入
    value: int

    # 临时值：只在当前 step 有效
    temp: str

# 显式指定 Channel 类型
graph = StateGraph(State)
graph.add_channel("value", LastValue(int))
graph.add_channel("temp", EphemeralValue(str))

Channel 类型详解

LastValue - 最后值

最常用的 Channel 类型。记住最后一次写入的值。

from langgraph.channels import LastValue

class State(TypedDict):
    count: int  # 默认就是 LastValue

# 行为：
# 写入 1 → 读取 1
# 写入 2 → 读取 2（覆盖了 1）

EphemeralValue - 临时值

只在当前执行 step 有效，读取后自动清空。

from langgraph.channels import EphemeralValue

class State(TypedDict):
    temp_result: str

graph = StateGraph(State)
graph.add_channel("temp_result", EphemeralValue(str))

# 使用场景：中间计算结果，不需要持久化
def compute(state: State) -> dict:
    return {"temp_result": "computed"}

def use(state: State) -> dict:
    print(state["temp_result"])  # 可以读取
    return {}

Topic - 主题队列

类似消息队列，支持多个值的累积和消费。

from langgraph.channels import Topic
from typing import Annotated

class State(TypedDict):
    # Topic 会自动累积所有写入的值
    events: Annotated[list, Topic(str)]

# 节点 A 写入
def producer(state: State) -> dict:
    return {"events": ["event1", "event2"]}

# 节点 B 读取所有累积的事件
def consumer(state: State) -> dict:
    print(state["events"])  # ["event1", "event2"]
    return {}

BinaryOperatorAggregate - 聚合值

使用自定义函数聚合多个写入。

from langgraph.channels import BinaryOperatorAggregate
from typing import Annotated

def sum_reducer(a: int, b: int) -> int:
    return a + b

class State(TypedDict):
    total: Annotated[int, BinaryOperatorAggregate(sum_reducer)]

# 多次写入会累加
def add_5(state) -> dict:
    return {"total": 5}

def add_10(state) -> dict:
    return {"total": 10}

# 执行后 total = 5 + 10 = 15

状态归约器（Reducer）

归约器定义如何合并多个更新。

使用 Annotated 定义归约器

from typing import Annotated, List
from langgraph.graph import add_messages

class State(TypedDict):
    # 默认：新值覆盖旧值
    value: int

    # 使用 add_messages 归约器：追加消息
    messages: Annotated[List, add_messages]

自定义归约器

from typing import Annotated, List

def append_reducer(existing: List, new: List) -> List:
    """自定义列表追加归约器"""
    return existing + new

class State(TypedDict):
    items: Annotated[List[str], append_reducer]

# 使用
def add_item(state: State) -> dict:
    return {"items": ["new_item"]}

# 多次调用会累积：["item1", "item2", "new_item"]

内置归约器

from langgraph.graph import add_messages

# add_messages：智能合并消息列表
# - 根据消息 ID 去重
# - 保持消息顺序
# - 处理消息替换

class State(TypedDict):
    messages: Annotated[List, add_messages]

MessagesState - 预定义消息状态

LangGraph 提供了预定义的 MessagesState，专门用于对话场景。

from langgraph.graph import MessagesState, MessageGraph

# 方式 1：使用 MessagesState
class State(MessagesState):
    # 自动包含 messages 字段
    extra_field: str

# 方式 2：直接使用 MessageGraph
graph = MessageGraph()
graph.add_node("agent", agent_fn)
# MessageGraph 自动处理消息状态

状态快照与时间旅行

获取状态快照

# 运行图并获取检查点
config = {"configurable": {"thread_id": "1"}}
result = app.invoke({"value": 0}, config)

# 获取当前状态快照
snapshot = app.get_state(config)
print(snapshot.values)  # 当前状态
print(snapshot.next)    # 下一个要执行的节点

时间旅行（回到过去的状态）

# 获取历史
history = list(app.get_state_history(config))

# 回到之前的状态
old_state = history[1]  # 倒数第二个状态
app.update_state(config, old_state.values)

# 从那个状态继续执行
result = app.invoke(None, config)

检查点（Checkpointing）

检查点用于持久化状态，支持断点续跑。

使用内存检查点

from langgraph.checkpoint.memory import MemorySaver

memory = MemorySaver()
app = graph.compile(checkpointer=memory)

# 使用 thread_id 区分不同会话
config = {"configurable": {"thread_id": "conversation-1"}}
result = app.invoke({"messages": [...]}, config)

使用 SQLite 检查点

pip install langgraph-checkpoint-sqlite

from langgraph.checkpoint.sqlite import SqliteSaver

with SqliteSaver.from_conn_string("checkpoints.sqlite") as saver:
    app = graph.compile(checkpointer=saver)

    config = {"configurable": {"thread_id": "1"}}
    result = app.invoke({"value": 0}, config)

检查点配置

config = {
    "configurable": {
        "thread_id": "user-123",      # 会话 ID
        "checkpoint_ns": "main",      # 命名空间（用于子图）
        "checkpoint_id": "abc123",    # 指定检查点 ID
    }
}

状态模式实践

模式 1：简单状态

class State(TypedDict):
    value: int
    # 适合：简单计算、计数器

模式 2：对话状态

from langchain_core.messages import BaseMessage
from typing import Annotated, List
from langgraph.graph import add_messages

class State(TypedDict):
    messages: Annotated[List[BaseMessage], add_messages]
    # 适合：聊天机器人、多轮对话

模式 3：工作流状态

class State(TypedDict):
    input: str
    steps: List[str]
    current_step: int
    result: dict
    # 适合：多步骤处理流程

模式 4：Agent 状态

class State(TypedDict):
    messages: Annotated[List, add_messages]
    tool_calls: List[dict]
    tool_results: List[dict]
    iteration: int
    # 适合：Agent、工具调用

调试状态问题

打印状态变化

def debug_node(state: State, config: dict) -> dict:
    print(f"📍 {config.get('configurable', {}).get('thread_id', 'unknown')}")
    print(f"   状态：{state}")
    return {}

使用 stream 查看状态流

# 查看每一步的状态更新
for node, values in app.stream({"value": 0}):
    print(f"{node}: {values}")

检查状态历史

# 查看所有检查点
for state in app.get_state_history(config):
    print(f"检查点 {state.config}: {state.values}")

最佳实践

✅ 推荐

# 1. 使用 TypedDict 定义清晰的状态结构
class State(TypedDict):
    messages: Annotated[List, add_messages]
    iteration: int

# 2. 只返回需要的更新
def node(state: State) -> dict:
    return {"iteration": state["iteration"] + 1}

# 3. 使用 checkpointer 持久化重要状态
app = graph.compile(checkpointer=MemorySaver())

❌ 避免

# 1. 避免在状态中存储大对象
class BadState(TypedDict):
    huge_data: bytes  # ❌ 会占用大量内存

# 2. 避免状态键过多
class BloatedState(TypedDict):
    field1: str
    field2: str
    # ... 50 个字段 ❌

# 3. 避免忘记返回状态更新
def bad_node(state: State) -> None:
    # ❌ 没有返回值
    pass
