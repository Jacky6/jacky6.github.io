02. 核心概念

StateGraph - 状态图

StateGraph 是 LangGraph 的核心类，用于定义有状态的应用流程。

from langgraph.graph import StateGraph
from typing import TypedDict

class State(TypedDict):
    messages: list
    count: int

# 创建图时必须指定状态类型
graph = StateGraph(State)

StateGraph 的主要方法

方法 说明 示例
add_node(name, fn) 添加节点 graph.add_node("agent", agent_fn)
add_edge(from, to) 添加普通边 graph.add_edge("A", "B")
add_conditional_edges(source, condition) 添加条件边 graph.add_conditional_edges("agent", route)
compile() 编译图 app = graph.compile()
set_entry_point(node) 设置入口（旧 API） graph.set_entry_point("start")

Node - 节点

节点是执行具体任务的函数。

节点函数的签名

from typing import TypedDict

class State(TypedDict):
    value: int
    messages: list

# 方式 1：接收完整 State
def my_node(state: State) -> State:
    return {"value": state["value"] + 1}

# 方式 2：只接收需要的字段（推荐）
def my_node(value: int) -> dict:
    return {"value": value + 1}

# 方式 3：返回部分更新
def my_node(state: State) -> dict:
    # 只返回需要更新的键
    return {"messages": state["messages"] + ["hello"]}

节点的最佳实践

# ✅ 好的做法：清晰的输入输出
def call_llm(messages: list) -> dict:
    response = model.invoke(messages)
    return {"messages": messages + [response]}

# ❌ 避免：副作用不明确
def call_llm(state):
    # 不要修改外部变量
    global_counter += 1  # ❌
    return {...}

Edge - 边

边定义节点之间的执行顺序。

普通边（Normal Edge）

# A 执行完后总是执行 B
graph.add_edge("A", "B")

条件边（Conditional Edge）

from langgraph.graph import END

def route(state: State) -> str:
    if state["count"] > 10:
        return "high_node"
    else:
        return "low_node"

# 根据 route 函数的返回值决定下一步
graph.add_conditional_edges(
    source="counter",      # 从哪个节点出来
    condition=route,       # 路由函数
    mapping={              # 可选：返回值到节点名的映射
        "high_node": "high_node",
        "low_node": "low_node"
    }
)

特殊节点名

from langgraph.graph import START, END

# START：图的入口
graph.add_edge(START, "first_node")

# END：图的出口
graph.add_edge("last_node", END)

State - 状态

状态是节点之间共享的数据。

定义状态

from typing import TypedDict, List
from langchain_core.messages import BaseMessage

# 方式 1：TypedDict（推荐）
class State(TypedDict):
    messages: List[BaseMessage]
    iteration: int

# 方式 2：dataclass
from dataclasses import dataclass, field

@dataclass
class State:
    messages: List[BaseMessage] = field(default_factory=list)
    iteration: int = 0

状态更新机制

class State(TypedDict):
    a: int
    b: int
    c: int

# 初始状态：{"a": 1, "b": 2, "c": 3}

# 节点 1 返回 {"a": 10}
# 新状态：{"a": 10, "b": 2, "c": 3}  ← 只更新 a

# 节点 2 返回 {"b": 20, "c": 30}
# 新状态：{"a": 10, "b": 20, "c": 30}

关键理解：节点返回的是增量更新，不是完整状态替换。

完整示例：条件分支

from typing import TypedDict, Literal
from langgraph.graph import StateGraph, START, END, ConditionalEdges

class State(TypedDict):
    value: int

def start(state: State) -> State:
    print(f"Starting with value: {state['value']}")
    return state

def add(state: State) -> State:
    return {"value": state["value"] + 5}

def subtract(state: State) -> State:
    return {"value": state["value"] - 3}

def route(state: State) -> Literal["add", "subtract"]:
    if state["value"] > 10:
        return "subtract"
    else:
        return "add"

# 构建图
graph = StateGraph(State)
graph.add_node("start", start)
graph.add_node("add", add)
graph.add_node("subtract", subtract)

graph.add_edge(START, "start")
graph.add_conditional_edges("start", route)
graph.add_edge("add", END)
graph.add_edge("subtract", END)

app = graph.compile()

# 测试
print(app.invoke({"value": 5}))    # value: 5 → add → 10
print(app.invoke({"value": 15}))   # value: 15 → subtract → 12

图的执行流程

1. invoke(input) 被调用
       ↓
2. 状态初始化为 input
       ↓
3. 从 START 开始，找到下一个节点
       ↓
4. 执行节点函数，获取状态更新
       ↓
5. 合并状态更新到当前状态
       ↓
6. 根据边找到下一个节点
       ↓
7. 重复 4-6，直到到达 END 或没有更多节点
       ↓
8. 返回最终状态

调试技巧

打印状态变化

def debug_node(state: State) -> State:
    print(f"📍 debug_node: {state}")
    return state

graph.add_node("debug", debug_node)

获取执行轨迹

# 使用 stream 模式查看每一步
for step in app.stream({"value": 0}):
    print(step)
# 输出：{'node_name': {'key': 'updated_value'}}

可视化图结构

# 打印 Mermaid 格式
print(app.get_graph().draw_mermaid())

# 保存为图片
app.get_graph().draw_mermaid_png(output_file_path="graph.png")

常见错误

❌ 错误 1：节点返回 None

def bad_node(state: State) -> None:
    print("Doing something...")
    # 忘记返回状态更新！

# ✅ 修正
def good_node(state: State) -> dict:
    print("Doing something...")
    return {}  # 空更新也是有效的

❌ 错误 2：状态键不匹配

class State(TypedDict):
    value: int

def wrong_node(state: State) -> dict:
    return {"wrong_key": 123}  # 这个键不在 State 中！

# ✅ 修正
def right_node(state: State) -> dict:
    return {"value": 123}

❌ 错误 3：循环没有退出条件

# 这会导致无限循环！
graph.add_edge("A", "B")
graph.add_edge("B", "A")
# 没有条件边或最大迭代限制

# ✅ 修正：添加计数器
class State(TypedDict):
    count: int

def check_limit(state: State) -> str:
    if state["count"] >= 5:
        return "end"
    return "continue"
