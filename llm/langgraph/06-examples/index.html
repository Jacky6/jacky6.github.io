示例 1：简单的对话机器人

from typing import TypedDict, Annotated, List
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage
from langgraph.graph import StateGraph, START, END, add_messages
from langgraph.checkpoint.memory import MemorySaver

# 1. 定义状态
class State(TypedDict):
    messages: Annotated[List, add_messages]

# 2. 初始化模型
model = ChatOpenAI(model="gpt-4")

# 3. 定义节点
def chatbot(state: State) -> dict:
    response = model.invoke(state["messages"])
    return {"messages": [response]}

# 4. 构建图
graph = StateGraph(State)
graph.add_node("chatbot", chatbot)
graph.add_edge(START, "chatbot")
graph.add_edge("chatbot", END)

# 5. 编译（带记忆）
memory = MemorySaver()
app = graph.compile(checkpointer=memory)

# 6. 使用
config = {"configurable": {"thread_id": "conversation-1"}}

# 第一轮
result = app.invoke({"messages": [HumanMessage(content="你好！")]}, config)
print(result["messages"][-1].content)

# 第二轮（记住上下文）
result = app.invoke({"messages": [HumanMessage(content="我叫小明，你呢？")]}, config)
print(result["messages"][-1].content)

示例 2：带工具调用的 Agent

from typing import TypedDict, Annotated, List, Literal
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langgraph.graph import StateGraph, START, END, add_messages
from langgraph.checkpoint.memory import MemorySaver

# 1. 定义工具
@tool
def search_web(query: str) -> str:
    """搜索网络获取信息"""
    return f"搜索结果：{query}"

@tool
def calculate(expression: str) -> str:
    """计算数学表达式"""
    return f"计算结果：{expression}"

tools = [search_web, calculate]

# 2. 绑定工具到模型
model = ChatOpenAI(model="gpt-4")
model_with_tools = model.bind_tools(tools)

# 3. 定义状态
class State(TypedDict):
    messages: Annotated[List, add_messages]

# 4. 定义节点
def agent(state: State) -> dict:
    response = model_with_tools.invoke(state["messages"])
    return {"messages": [response]}

def tools_node(state: State) -> dict:
    tool_calls = state["messages"][-1].tool_calls
    results = []

    for call in tool_calls:
        tool_name = call["name"]
        tool_input = call["args"]

        for t in tools:
            if t.name == tool_name:
                result = t.invoke(tool_input)
                results.append(ToolMessage(
                    content=str(result),
                    tool_call_id=call["id"]
                ))

    return {"messages": results}

# 5. 定义路由
def should_continue(state: State) -> Literal["tools", END]:
    messages = state["messages"]
    last_message = messages[-1]

    if last_message.tool_calls:
        return "tools"
    return END

# 6. 构建图
graph = StateGraph(State)
graph.add_node("agent", agent)
graph.add_node("tools", tools_node)

graph.add_edge(START, "agent")
graph.add_conditional_edges("agent", should_continue)
graph.add_edge("tools", "agent")  # 工具执行后回到 agent

# 7. 编译
memory = MemorySaver()
app = graph.compile(checkpointer=memory)

# 8. 使用
config = {"configurable": {"thread_id": "agent-1"}}

result = app.invoke({
    "messages": [HumanMessage(content="帮我搜索一下 Python 的最新版本")]
}, config)

print(result["messages"][-1].content)

示例 3：多轮对话 + 人工审核

from typing import TypedDict, Annotated, List, Literal
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage
from langgraph.graph import StateGraph, START, END, add_messages
from langgraph.checkpoint.memory import MemorySaver

class State(TypedDict):
    messages: Annotated[List, add_messages]
    draft: str
    approved: bool

model = ChatOpenAI(model="gpt-4")

def generate_draft(state: State) -> dict:
    prompt = f"请为以下内容生成回复草稿：{state['messages'][-1].content}"
    response = model.invoke([HumanMessage(content=prompt)])
    return {"draft": response.content}

def human_review(state: State) -> dict:
    # 实际应用中，这里会等待用户输入
    print(f"\n=== 草稿 ===")
    print(state["draft"])
    print(f"==============\n")

    # 模拟人工审核（实际应该等待用户输入）
    approved = input("是否批准？(y/n): ").lower() == 'y'
    return {"approved": approved}

def send_response(state: State) -> dict:
    return {"messages": [AIMessage(content=state["draft"])]}

def reject(state: State) -> dict:
    return {"messages": [AIMessage(content="抱歉，我需要重新考虑这个回答。")]}

def route_after_review(state: State) -> Literal["send", "reject"]:
    return "send" if state["approved"] else "reject"

# 构建图
graph = StateGraph(State)
graph.add_node("generate", generate_draft)
graph.add_node("review", human_review)
graph.add_node("send", send_response)
graph.add_node("reject", reject)

graph.add_edge(START, "generate")
graph.add_edge("generate", "review")
graph.add_conditional_edges("review", route_after_review)
graph.add_edge("send", END)
graph.add_edge("reject", END)

# 在 review 前中断，等待人工输入
memory = MemorySaver()
app = graph.compile(checkpointer=memory, interrupt_before=["review"])

# 使用
config = {"configurable": {"thread_id": "review-1"}}

# 第一步：生成草稿
app.invoke({
    "messages": [HumanMessage(content="写一封感谢邮件")]
}, config)

# 查看草稿
snapshot = app.get_state(config)
print(f"草稿：{snapshot.values['draft']}")

# 人工审核后继续
app.update_state(config, {"approved": True})
result = app.invoke(None, config)

示例 4：多 Agent 协作

from typing import TypedDict, Annotated, List, Literal
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, AIMessage
from langgraph.graph import StateGraph, START, END, add_messages

class State(TypedDict):
    messages: Annotated[List, add_messages]
    current_agent: str

# 不同角色的 Agent
researcher = ChatOpenAI(model="gpt-4", system_message="你是研究员，负责查找和分析信息")
writer = ChatOpenAI(model="gpt-4", system_message="你是作家，负责撰写内容")
reviewer = ChatOpenAI(model="gpt-4", system_message="你是审核员，负责检查和改进内容")

def researcher_node(state: State) -> dict:
    response = researcher.invoke(state["messages"])
    return {"messages": [response], "current_agent": "writer"}

def writer_node(state: State) -> dict:
    response = writer.invoke(state["messages"])
    return {"messages": [response], "current_agent": "reviewer"}

def reviewer_node(state: State) -> dict:
    response = reviewer.invoke(state["messages"])
    content = response.content

    # 如果需要修改
    if "需要修改" in content:
        return {"messages": [response], "current_agent": "writer"}

    return {"messages": [response], "current_agent": END}

def route(state: State) -> Literal["researcher", "writer", "reviewer", END]:
    if state["current_agent"] == "researcher":
        return "researcher"
    elif state["current_agent"] == "writer":
        return "writer"
    elif state["current_agent"] == "reviewer":
        return "reviewer"
    return END

# 构建图
graph = StateGraph(State)
graph.add_node("researcher", researcher_node)
graph.add_node("writer", writer_node)
graph.add_node("reviewer", reviewer_node)

graph.add_edge(START, "researcher")
graph.add_conditional_edges("researcher", lambda s: "writer")
graph.add_conditional_edges("writer", lambda s: "reviewer")
graph.add_conditional_edges("reviewer", route)

app = graph.compile()

# 使用
result = app.invoke({
    "messages": [HumanMessage(content="写一篇关于 AI 的短文")],
    "current_agent": "researcher"
})

for msg in result["messages"]:
    print(f"{msg.type}: {msg.content[:100]}...")

示例 5：数据处理工作流

from typing import TypedDict, List
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    input_data: str
    cleaned_data: str
    analyzed_data: dict
    report: str

def clean_data(state: State) -> dict:
    # 数据清洗
    cleaned = state["input_data"].strip().lower()
    return {"cleaned_data": cleaned}

def analyze_data(state: State) -> dict:
    # 数据分析
    data = state["cleaned_data"]
    analysis = {
        "length": len(data),
        "word_count": len(data.split()),
        "has_error": "error" in data
    }
    return {"analyzed_data": analysis}

def generate_report(state: State) -> dict:
    # 生成报告
    analysis = state["analyzed_data"]
    report = f"""
    数据分析报告
    ==========
    字符数：{analysis['length']}
    单词数：{analysis['word_count']}
    包含错误：{analysis['has_error']}
    """
    return {"report": report}

# 构建线性工作流
graph = StateGraph(State)
graph.add_node("clean", clean_data)
graph.add_node("analyze", analyze_data)
graph.add_node("report", generate_report)

graph.add_edge(START, "clean")
graph.add_edge("clean", "analyze")
graph.add_edge("analyze", "report")
graph.add_edge("report", END)

app = graph.compile()

# 使用
result = app.invoke({
    "input_data": "  Hello World, this is a TEST with ERROR handling  "
})

print(result["report"])

示例 6：并行处理

from typing import TypedDict, List
from langgraph.graph import StateGraph, START, END
import time

class State(TypedDict):
    items: List[str]
    results_a: List[str]
    results_b: List[str]
    results_c: List[str]
    final_result: str

def process_a(state: State) -> dict:
    time.sleep(1)  # 模拟耗时操作
    results = [f"A:{item}" for item in state["items"]]
    return {"results_a": results}

def process_b(state: State) -> dict:
    time.sleep(1)
    results = [f"B:{item}" for item in state["items"]]
    return {"results_b": results}

def process_c(state: State) -> dict:
    time.sleep(1)
    results = [f"C:{item}" for item in state["items"]]
    return {"results_c": results}

def merge_results(state: State) -> dict:
    merged = list(zip(state["results_a"], state["results_b"], state["results_c"]))
    return {"final_result": str(merged)}

# 构建图
graph = StateGraph(State)
graph.add_node("process_a", process_a)
graph.add_node("process_b", process_b)
graph.add_node("process_c", process_c)
graph.add_node("merge", merge_results)

# 扇出：start 同时触发三个处理节点
graph.add_edge(START, "process_a")
graph.add_edge(START, "process_b")
graph.add_edge(START, "process_c")

# 扇入：所有处理完成后合并
graph.add_edge("process_a", "merge")
graph.add_edge("process_b", "merge")
graph.add_edge("process_c", "merge")
graph.add_edge("merge", END)

app = graph.compile()

# 使用
result = app.invoke({"items": ["item1", "item2", "item3"]})
print(result["final_result"])

示例 7：RAG（检索增强生成）

from typing import TypedDict, Annotated, List, Literal
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_core.messages import HumanMessage, AIMessage
from langgraph.graph import StateGraph, START, END, add_messages

class State(TypedDict):
    messages: Annotated[List, add_messages]
    question: str
    context: str
    answer: str

model = ChatOpenAI(model="gpt-4")
embeddings = OpenAIEmbeddings()

# 模拟向量数据库
def retrieve_documents(question: str) -> List[str]:
    # 实际应用中这里会查询向量数据库
    return [
        "文档 1: LangGraph 是用于构建有状态应用的库",
        "文档 2: LangGraph 支持循环、分支和并行执行",
        "文档 3: LangGraph 是 LangChain 团队开发的"
    ]

def retrieve(state: State) -> dict:
    question = state["messages"][-1].content
    docs = retrieve_documents(question)
    context = "\n\n".join(docs)
    return {"question": question, "context": context}

def generate(state: State) -> dict:
    prompt = f"""基于以下上下文回答问题：

上下文：
{state['context']}

问题：
{state['question']}

回答："""

    response = model.invoke([HumanMessage(content=prompt)])
    return {"answer": response.content, "messages": [response]}

def should_use_rag(state: State) -> Literal["retrieve", "direct"]:
    # 简单判断：如果问题包含"什么"、"如何"等词，使用 RAG
    question = state["messages"][-1].content
    rag_keywords = ["什么", "如何", "为什么", "怎么", "which", "what", "how"]

    if any(kw in question for kw in rag_keywords):
        return "retrieve"
    return "direct"

def direct_answer(state: State) -> dict:
    response = model.invoke(state["messages"])
    return {"messages": [response]}

# 构建图
graph = StateGraph(State)
graph.add_node("retrieve", retrieve)
graph.add_node("generate", generate)
graph.add_node("direct", direct_answer)

graph.add_edge(START, "retrieve")
graph.add_conditional_edges("retrieve", lambda s: "generate")
graph.add_edge("generate", END)

app = graph.compile()

# 使用
result = app.invoke({
    "messages": [HumanMessage(content="LangGraph 是什么？")]
})

print(result["answer"])

调试技巧

1. 打印每一步的状态

for node, values in app.stream(input):
    print(f"\n{'='*40}")
    print(f"节点：{node}")
    print(f"更新：{values}")
    print(f"{'='*40}")

2. 可视化图

# 保存为图片
app.get_graph().draw_mermaid_png(output_file_path="workflow.png")

# 打印 Mermaid 代码

3. 检查执行历史

config = {"configurable": {"thread_id": "test-1"}}

# 获取所有检查点
for state in app.get_state_history(config):
    print(f"检查点：{state.config}")
    print(f"状态：{state.values}")
    print(f"下一个：{state.next}")
    print("---")

性能优化建议

  1. 使用流式输出：提升用户体验 python for chunk in app.stream(input, stream_mode="messages"): yield chunk

  2. 合理使用检查点：只在需要断点续跑时启用

  3. 并行化独立任务： python graph.add_edge("start", "task_a") graph.add_edge("start", "task_b") # 并行执行

  4. 缓存昂贵操作： ```python from langgraph.types import CachePolicy

graph.add_node( "expensive", expensive_fn, cache_policy=CachePolicy(ttl_seconds=3600) ) ```
