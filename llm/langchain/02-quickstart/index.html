02. 快速开始

10 行代码创建第一个 Agent

最简示例

from langchain.agents import create_agent

def get_weather(city: str) -> str:
    """Get weather for a given city."""
    return f"It's always sunny in {city}!"

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_weather],
    system_prompt="You are a helpful assistant",
)

result = agent.invoke({"messages": [{"role": "user", "content": "what is the weather in sf"}]})
print(result["messages"][-1].content)

运行结果：

The weather in San Francisco is always sunny! (开玩笑的，实际上...)

代码解析

# 第 1 行：导入 create_agent 函数
from langchain.agents import create_agent

# 第 3-5 行：定义一个工具函数
def get_weather(city: str) -> str:
    """Get weather for a given city."""
    return f"It's always sunny in {city}!"

# 关键点：
# - 函数名会成为工具名：get_weather
# - 参数类型注解帮助 LLM 理解参数
# - 文档字符串告诉 LLM 工具的用途

# 第 7-11 行：创建 Agent
agent = create_agent(
    model="claude-sonnet-4-6",      # 使用 Claude 模型
    tools=[get_weather],            # 可用的工具列表
    system_prompt="You are a helpful assistant",  # 系统提示
)

# 第 13-14 行：运行 Agent
result = agent.invoke({
    "messages": [{"role": "user", "content": "what is the weather in sf"}]
})

# 输入格式：
# {"messages": [...]} 包含对话历史
# 每条消息：{"role": "user/assistant", "content": "..."}

# 输出格式：
# {"messages": [...]} 包含完整的对话历史
# result["messages"][-1] 是最后一条消息（AI 的回复）

完整工作流程

用户请求 → Agent 思考 → 工具调用 → 整合结果 → 回复

详细流程：

1. 用户输入
   ↓
   "what is the weather in sf"

2. Agent 思考（LLM）
   ↓
   "用户想知道天气...我需要调用 get_weather 工具..."
   "参数 city 应该是 'sf' 或 'San Francisco'"

3. 工具调用
   ↓
   get_weather(city="San Francisco")
   → 返回："It's always sunny in San Francisco!"

4. Agent 整合
   ↓
   "工具告诉我旧金山总是晴天..."
   "我来组织一个友好的回复"

5. 最终回复
   ↓
   "The weather in San Francisco is always sunny! 
    However, in reality, San Francisco has a Mediterranean 
    climate with foggy mornings..."

环境准备

1. 安装依赖

方式 1：环境变量

export ANTHROPIC_API_KEY="sk-ant-..."

set ANTHROPIC_API_KEY=sk-ant-...

cat > .env << EOF

示例 1：计算器 Agent

from langchain.agents import create_agent

def calculator(expression: str) -> str:
    """Calculate a mathematical expression.

    Args:
        expression: The mathematical expression to calculate.

    Returns:
        The result of the calculation.
    """
    try:
        # 注意：实际应用中不要直接用 eval，这里仅示例
        result = eval(expression)
        return str(result)
    except Exception as e:
        return f"Error: {e}"

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[calculator],
    system_prompt="You are a helpful assistant. Use the calculator tool for math problems.",
)

# 测试
result = agent.invoke({
    "messages": [{"role": "user", "content": "计算 (123 + 456) * 789"}]
})

print(result["messages"][-1].content)
# 输出：457659

示例 2：多工具 Agent

from langchain.agents import create_agent
import datetime
import random

def get_weather(city: str) -> str:
    """Get current weather for a city.

    Args:
        city: The name of the city.

    Returns:
        Current weather information.
    """
    # 模拟天气数据（实际应用中调用天气 API）
    conditions = ["sunny", "cloudy", "rainy", "foggy"]
    temp = random.randint(15, 35)
    return f"{city}: {random.choice(conditions)}, {temp}°C"

def get_time() -> str:
    """Get current date and time.

    Returns:
        Current date and time as a string.
    """
    return datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")

def get_random_number(min: int = 0, max: int = 100) -> str:
    """Generate a random number.

    Args:
        min: Minimum value (inclusive).
        max: Maximum value (inclusive).

    Returns:
        A random number between min and max.
    """
    return str(random.randint(min, max))

# 创建 Agent
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_weather, get_time, get_random_number],
    system_prompt="""You are a helpful assistant with access to tools.
Use the appropriate tool based on the user's request.
- For weather questions, use get_weather
- For time questions, use get_time
- For random numbers, use get_random_number""",
)

# 测试 1：询问天气
result = agent.invoke({
    "messages": [{"role": "user", "content": "北京天气怎么样？"}]
})
print(result["messages"][-1].content)

# 测试 2：询问时间
result = agent.invoke({
    "messages": [{"role": "user", "content": "现在几点了？"}]
})
print(result["messages"][-1].content)

# 测试 3：生成随机数
result = agent.invoke({
    "messages": [{"role": "user", "content": "生成一个 1 到 10 之间的随机数"}]
})
print(result["messages"][-1].content)

示例 3：对话式 Agent

from langchain.agents import create_agent

def search_knowledge_base(query: str) -> str:
    """Search the internal knowledge base.

    Args:
        query: The search query.

    Returns:
        Search results from the knowledge base.
    """
    # 模拟知识库（实际应用中连接数据库或向量存储）
    knowledge = {
        "工作时间": "我们的工作时间是周一至周五 9:00-18:00",
        "退货政策": "支持 7 天无理由退货",
        "联系方式": "客服电话：400-123-4567",
        "地址": "北京市朝阳区 xxx 路 xxx 号",
    }

    for key, value in knowledge.items():
        if key in query:
            return value

    return "抱歉，我没有找到相关信息。"

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[search_knowledge_base],
    system_prompt="""You are a customer service assistant.
Use the search_knowledge_base tool to find information.
Be polite and helpful in your responses.""",
)

# 多轮对话
messages = []

while True:
    user_input = input("你：")
    if user_input.lower() in ["exit", "quit", "拜拜"]:
        break

    messages.append({"role": "user", "content": user_input})

    result = agent.invoke({"messages": messages})

    assistant_message = result["messages"][-1]
    messages.append(assistant_message)

    print(f"助手：{assistant_message.content}\n")

运行示例：

你：你们的工作时间是什么？
助手：我们的工作时间是周一至周五 9:00-18:00。如果您有其他问题，请随时询问。

你：如何退货？
助手：我们支持 7 天无理由退货。请确保商品完好无损，并保留原始包装和发票。

你：谢谢！
助手：不客气！如果您还有其他问题，随时联系我。祝您有愉快的一天！

你：拜拜
助手：再见！期待您的再次光临！

理解 Agent 的输入输出

输入格式

# 基本输入
input_data = {
    "messages": [
        {"role": "user", "content": "你好"}
    ]
}

# 带对话历史的输入
input_data = {
    "messages": [
        {"role": "system", "content": "你是助手"},
        {"role": "user", "content": "你好"},
        {"role": "assistant", "content": "你好！有什么可以帮助你的？"},
        {"role": "user", "content": "今天天气如何？"}
    ]
}

# 调用
result = agent.invoke(input_data)

输出格式

result = agent.invoke({"messages": [...]})

# result 的结构
{
    "messages": [
        # 输入的完整消息历史
        {"role": "user", "content": "你好"},

        # AI 的回复
        {"role": "assistant", "content": "你好！..."},

        # 可能的工具调用
        {"role": "tool", "name": "get_weather", "content": "..."},

        # 最终回复
        {"role": "assistant", "content": "今天天气..."}
    ]
}

# 获取最后一条消息（AI 的最终回复）
final_response = result["messages"][-1]
print(final_response.content)

流式输出

from langchain.agents import create_agent

def get_weather(city: str) -> str:
    """Get weather for a city."""
    return f"{city} is sunny, 25°C"

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_weather],
    system_prompt="You are a helpful assistant",
)

# 流式输出
for chunk in agent.stream({"messages": [{"role": "user", "content": "北京天气"}]}):
    if "messages" in chunk:
        for message in chunk["messages"]:
            if hasattr(message, 'content') and message.content:
                print(message.content, end="", flush=True)

错误处理

from langchain.agents import create_agent

def risky_tool(data: str) -> str:
    """A tool that might fail."""
    if not data:
        raise ValueError("Data cannot be empty")
    return f"Processed: {data}"

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[risky_tool],
    system_prompt="You are a helpful assistant",
)

try:
    result = agent.invoke({
        "messages": [{"role": "user", "content": "Process this data"}]
    })
    print(result["messages"][-1].content)
except Exception as e:
    print(f"Error: {e}")

调试技巧

1. 打印中间过程

from langchain.agents import create_agent

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[...],
)

result = agent.invoke({"messages": [...]})

# 打印所有消息
for i, message in enumerate(result["messages"]):
    print(f"[{i}] {message.type}: {message.content[:100]}...")

2. 使用 LangSmith

import os

# 启用 LangSmith 追踪
os.environ["LANGSMITH_TRACING"] = "true"
os.environ["LANGSMITH_API_KEY"] = "lsv2_..."
os.environ["LANGSMITH_PROJECT"] = "my-first-agent"

# 现在所有调用都会自动追踪
agent.invoke({"messages": [...]})

# 访问 https://smith.langchain.com/ 查看追踪

3. 查看工具调用

result = agent.invoke({"messages": [...]})

# 查找工具调用消息
for message in result["messages"]:
    if message.type == "tool":
        print(f"工具调用：{message.name}")
        print(f"参数：{message.additional_kwargs}")
        print(f"结果：{message.content}")

常见问题

Q1: 如何更换模型？

# 使用 OpenAI
from langchain_openai import ChatOpenAI
agent = create_agent(
    model=ChatOpenAI(model="gpt-4o"),
    tools=[...],
)

# 使用 Google
from langchain_google_genai import ChatGoogleGenerativeAI
agent = create_agent(
    model=ChatGoogleGenerativeAI(model="gemini-2.0"),
    tools=[...],
)

Q2: 如何添加更多工具？

# 只需在 tools 列表中添加函数
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[tool1, tool2, tool3, tool4, ...],  # 添加更多工具
    system_prompt="...",
)

Q3: Agent 不调用工具怎么办？

# 1. 确保工具有清晰的文档字符串
def my_tool(param: str) -> str:
    """清晰的描述工具的用途和参数。

    Args:
        param: 参数的详细说明
    """

# 2. 在系统提示中明确说明
system_prompt = """You have access to these tools:
- tool1: 用于...
- tool2: 用于...

Always use the appropriate tool when needed."""

# 3. 提供更具体的示例
system_prompt = """Example:
User: "What's the weather in Beijing?"
You: [call get_weather with city="Beijing"]"""
