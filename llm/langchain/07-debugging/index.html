07. 调试与监控

LangSmith 简介

LangSmith 是 LangChain 团队开发的调试、监控和评估平台，帮助你： - 🔍 追踪 LLM 调用和 Agent 执行 - 🐛 调试复杂的工作流 - 📊 监控性能和成本 - ✅ 评估输出质量

启用 LangSmith

基础配置

import os

# 设置环境变量
os.environ["LANGSMITH_TRACING"] = "true"
os.environ["LANGSMITH_API_KEY"] = "lsv2_..."  # 从 https://smith.langchain.com/ 获取
os.environ["LANGSMITH_PROJECT"] = "my-project"  # 项目名称
os.environ["LANGSMITH_ENDPOINT"] = "https://api.smith.langchain.com"

使用 .env 文件

# .env

批量性能测试

import time
import statistics

def benchmark(agent, queries, iterations=5):
    """性能基准测试"""
    results = []

    for query in queries:
        times = []
        for _ in range(iterations):
            start = time.time()
            agent.invoke({"messages": [{"role": "user", "content": query}]})
            end = time.time()
            times.append(end - start)

        results.append({
            "query": query,
            "avg": statistics.mean(times),
            "min": min(times),
            "max": max(times),
            "std": statistics.stdev(times) if len(times) > 1 else 0,
        })

    return results

# 使用
queries = ["简单问题", "中等问题", "复杂问题"]
results = benchmark(agent, queries)

for r in results:
    print(f"{r['query']}:")
    print(f"  平均：{r['avg']:.2f}s")
    print(f"  最小：{r['min']:.2f}s")
    print(f"  最大：{r['max']:.2f}s")
    print()

错误追踪

捕获和记录错误

from langsmith import traceable

@traceable  # LangSmith 追踪装饰器
def run_agent(agent, messages):
    try:
        result = agent.invoke({"messages": messages})
        return {"success": True, "result": result}
    except Exception as e:
        # LangSmith 会自动记录异常
        return {"success": False, "error": str(e)}

# 使用
response = run_agent(agent, [{"role": "user", "content": "..."}])

评估输出质量

基础评估

from langsmith import Client, evaluate
from langsmith.schemas import Run, Example

def accuracy(run: Run, example: Example) -> dict:
    """准确率评估器"""
    predicted = run.outputs.get("result", "")
    expected = example.outputs.get("expected", "")
    return {"score": 1.0 if predicted == expected else 0.0, "key": "accuracy"}

def relevance(run: Run, example: Example) -> dict:
    """相关性评估器"""
    # 可以使用 LLM 评估相关性
    predicted = run.outputs.get("result", "")
    expected = example.outputs.get("expected", "")

    # 简单示例：检查是否包含关键词
    keywords = ["关键", "重要"]
    score = sum(1 for kw in keywords if kw in predicted) / len(keywords)
    return {"score": score, "key": "relevance"}

# 运行评估
client = Client()
results = evaluate(
    lambda inputs: agent.invoke(inputs),
    data="my-dataset",  # 数据集名称
    evaluators=[accuracy, relevance],
)

使用 LLM 评估

from langchain_openai import ChatOpenAI
from langsmith import evaluate

# 创建评估器 LLM
evaluator_llm = ChatOpenAI(model="gpt-4o")

def llm_evaluator(run: Run, example: Example) -> dict:
    """使用 LLM 评估输出质量"""
    predicted = run.outputs.get("result", "")
    expected = example.outputs.get("expected", "")

    # 让 LLM 评分
    prompt = f"""
    评估以下回答的质量（0-1 分）：

    预期：{expected}
    实际：{predicted}

    评分（只返回数字）：
    """

    response = evaluator_llm.invoke([{"role": "user", "content": prompt}])
    score = float(response.content.strip())

    return {"score": score, "key": "quality"}

# 运行评估
results = evaluate(
    lambda inputs: agent.invoke(inputs),
    data="my-dataset",
    evaluators=[llm_evaluator],
)

成本监控

估算成本

# 价格（每 1K tokens）
PRICES = {
    "claude-sonnet-4-6": {"input": 0.003, "output": 0.015},
    "gpt-4o": {"input": 0.0025, "output": 0.01},
}

def estimate_cost(model: str, input_tokens: int, output_tokens: int) -> float:
    """估算成本"""
    price = PRICES.get(model, {"input": 0, "output": 0})
    return (input_tokens * price["input"] + output_tokens * price["output"]) / 1000

# 使用
cost = estimate_cost("claude-sonnet-4-6", 1000, 500)
print(f"估算成本：${cost:.4f}")

生产环境监控

设置告警

# 监控错误率
def check_error_rate(runs, threshold=0.05):
    """检查错误率是否超过阈值"""
    error_count = sum(1 for run in runs if run.error)
    error_rate = error_count / len(runs)

    if error_rate > threshold:
        send_alert(f"错误率过高：{error_rate:.2%}")
        return False
    return True

# 监控响应时间
def check_response_time(runs, threshold=5.0):
    """检查平均响应时间"""
    avg_time = sum(run.duration for run in runs) / len(runs)

    if avg_time > threshold:
        send_alert(f"响应时间过长：{avg_time:.2f}s")
        return False
    return True

日志聚合

import logging
from logging.handlers import RotatingFileHandler

# 配置日志
logger = logging.getLogger("agent")
logger.setLevel(logging.INFO)

# 文件日志（带轮转）
handler = RotatingFileHandler(
    "agent.log",
    maxBytes=10*1024*1024,  # 10MB
    backupCount=5
)
handler.setFormatter(logging.Formatter(
    '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
))
logger.addHandler(handler)

# 使用
def run_agent_with_logging(agent, messages):
    logger.info(f"开始处理：{messages[0]['content'][:50]}...")

    try:
        result = agent.invoke({"messages": messages})
        logger.info("处理成功")
        return result
    except Exception as e:
        logger.error(f"处理失败：{e}", exc_info=True)
        raise

最佳实践

✅ 推荐

# 1. 始终启用 LangSmith 追踪
os.environ["LANGSMITH_TRACING"] = "true"

# 2. 添加有意义的元数据
config = {
    "metadata": {
        "user_id": user_id,
        "session_id": session_id,
        "version": "1.0.0",
    }
}

# 3. 监控关键指标
# - 响应时间
# - 错误率
# - Token 使用
# - 成本

# 4. 设置告警
# - 错误率超过阈值
# - 响应时间过长
# - 成本异常

# 5. 定期评估输出质量
# - 使用自动化评估
# - 人工抽样检查

❌ 避免

# 1. 不启用追踪
# ❌ 无法调试和监控

# 2. 忽略错误日志
# ❌ 问题难以定位

# 3. 不监控成本
# ❌ 可能产生意外费用

# 4. 在生产环境关闭追踪
# ❌ 失去可观测性

# 5. 不保护敏感信息
# ❌ 确保不在日志中泄露隐私
