04. 工具与函数

什么是工具（Tool）？

在 LangChain 中，工具是 Agent 可以调用的函数，让 Agent 能够与外部世界交互。

Agent 架构：

┌─────────────────────────────────────────┐
│              Agent                       │
├─────────────────────────────────────────┤
│                                         │
│  用户输入："北京天气如何？"              │
│       ↓                                 │
│  LLM 思考：                             │
│  "需要调用天气工具..."                  │
│       ↓                                 │
│  ┌─────────────────────────────────┐   │
│  │         工具列表                 │   │
│  │  - get_weather(city: str)       │   │
│  │  - search_web(query: str)       │   │
│  │  - calculator(expr: str)        │   │
│  │  - ...                          │   │
│  └─────────────────────────────────┘   │
│       ↓                                 │
│  调用：get_weather("北京")              │
│  返回："北京：晴，25°C"                 │
│       ↓                                 │
│  整合回复："北京今天晴朗，气温 25°C..." │
│                                         │
└─────────────────────────────────────────┘

创建简单工具

基础示例

from langchain.agents import create_agent

# 定义工具函数
def get_weather(city: str) -> str:
    """Get current weather for a city.

    Args:
        city: The name of the city.

    Returns:
        Current weather information.
    """
    # 实际应用中调用天气 API
    return f"{city} is sunny, 25°C"

# 创建 Agent
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_weather],
    system_prompt="You are a helpful assistant",
)

# 使用工具
result = agent.invoke({
    "messages": [{"role": "user", "content": "北京天气如何？"}]
})

print(result["messages"][-1].content)

工具的关键要素

def tool_name(param1: type, param2: type) -> return_type:
    """工具的文档字符串。

    详细描述工具的用途和功能。

    Args:
        param1: 参数 1 的名称和说明
        param2: 参数 2 的名称和说明

    Returns:
        返回值的说明

    Example:
        tool_name("value1", 42)
    """
    # 工具实现
    return "结果"

关键要素： 1. ✅ 清晰的函数名（会成为工具名） 2. ✅ 类型注解（帮助 LLM 理解参数） 3. ✅ 详细的文档字符串（告诉 LLM 工具用途） 4. ✅ Args 说明（参数描述） 5. ✅ Returns 说明（返回值描述）

工具参数详解

必需参数

def search_web(query: str) -> str:
    """Search the web for information.

    Args:
        query: The search query (required).

    Returns:
        Search results.
    """
    return f"Results for: {query}"

# LLM 知道 query 是必需的
# 调用时一定会提供

可选参数（带默认值）

def search_web(query: str, limit: int = 10) -> str:
    """Search the web for information.

    Args:
        query: The search query.
        limit: Maximum number of results (default: 10).

    Returns:
        Search results.
    """
    return f"Top {limit} results for: {query}"

# LLM 可以选择性提供 limit
# search_web("Python")        # limit=10 (默认)
# search_web("Python", 20)    # limit=20

多个参数

def book_flight(
    origin: str,
    destination: str,
    date: str,
    passengers: int = 1
) -> str:
    """Book a flight.

    Args:
        origin: Departure city (e.g., "Beijing").
        destination: Arrival city (e.g., "Shanghai").
        date: Travel date (YYYY-MM-DD format).
        passengers: Number of passengers (default: 1).

    Returns:
        Booking confirmation.
    """
    return f"Booked {passengers} ticket(s) from {origin} to {destination} on {date}"

实用工具示例

1. 计算器工具

def calculator(expression: str) -> str:
    """Calculate a mathematical expression.

    Args:
        expression: The mathematical expression (e.g., "2 + 2 * 3").

    Returns:
        The result of the calculation.
    """
    try:
        # 注意：生产环境不要用 eval，使用安全的解析器
        result = eval(expression)
        return str(result)
    except Exception as e:
        return f"Error: {e}"

# 使用
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[calculator],
)

result = agent.invoke({
    "messages": [{"role": "user", "content": "计算 (123 + 456) * 789"}]
})
# 输出：457659

2. 日期时间工具

import datetime
from typing import Optional

def get_current_time(timezone: Optional[str] = "UTC") -> str:
    """Get current date and time.

    Args:
        timezone: Timezone name (default: "UTC").

    Returns:
        Current date and time as string.
    """
    now = datetime.datetime.now()
    return now.strftime("%Y-%m-%d %H:%M:%S %Z")

def get_weekday(date: str) -> str:
    """Get weekday for a given date.

    Args:
        date: Date in YYYY-MM-DD format.

    Returns:
        Weekday name (e.g., "Monday").
    """
    dt = datetime.datetime.strptime(date, "%Y-%m-%d")
    return dt.strftime("%A")

# 使用
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_current_time, get_weekday],
)

4. 文件操作工具

import os
from typing import List

def read_file(file_path: str) -> str:
    """Read contents of a file.

    Args:
        file_path: Path to the file.

    Returns:
        File contents or error message.
    """
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
    except Exception as e:
        return f"Error reading file: {e}"

def write_file(file_path: str, content: str) -> str:
    """Write content to a file.

    Args:
        file_path: Path to the file.
        content: Content to write.

    Returns:
        Success or error message.
    """
    try:
        with open(file_path, 'w', encoding='utf-8') as f:
            f.write(content)
        return f"Successfully wrote to {file_path}"
    except Exception as e:
        return f"Error writing file: {e}"

def list_files(directory: str = ".") -> str:
    """List files in a directory.

    Args:
        directory: Directory path (default: current directory).

    Returns:
        List of files.
    """
    try:
        files = os.listdir(directory)
        return "\n".join(files)
    except Exception as e:
        return f"Error: {e}"

5. API 调用工具

import requests

def get_stock_price(symbol: str) -> str:
    """Get current stock price.

    Args:
        symbol: Stock symbol (e.g., "AAPL", "GOOGL").

    Returns:
        Current stock price.
    """
    try:
        # 使用 Alpha Vantage API（示例）
        url = f"https://www.alphavantage.co/query?function=GLOBAL_QUOTE&symbol={symbol}&apikey=YOUR_API_KEY"
        response = requests.get(url)
        data = response.json()

        if "Global Quote" in data:
            price = data["Global Quote"]["05. price"]
            return f"{symbol}: ${price}"
        else:
            return f"Could not find price for {symbol}"
    except Exception as e:
        return f"Error: {e}"

def send_email(to: str, subject: str, body: str) -> str:
    """Send an email.

    Args:
        to: Recipient email address.
        subject: Email subject.
        body: Email body text.

    Returns:
        Success or error message.
    """
    try:
        # 使用 SMTP 或邮件 API
        # 这里仅示例
        return f"Email sent to {to} with subject: {subject}"
    except Exception as e:
        return f"Error: {e}"

工具装饰器（高级）

使用 @tool 装饰器

from langchain.tools import tool

@tool
def search_web(query: str) -> str:
    """Search the web for information.

    Args:
        query: The search query.

    Returns:
        Search results.
    """
    return f"Results for: {query}"

# 装饰器自动处理：
# - 提取函数签名
# - 解析文档字符串
# - 创建工具配置

# 使用
agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[search_web],
)

自定义工具名称

from langchain.tools import tool

@tool("weather_lookup")
def get_weather(city: str) -> str:
    """Get weather for a city.

    Args:
        city: City name.

    Returns:
        Weather information.
    """
    return f"{city}: Sunny"

# 工具名会是 "weather_lookup" 而不是 "get_weather"

带配置的工具

from langchain.tools import Tool

def custom_search(query: str, engine: str = "google") -> str:
    """Search with configurable engine."""
    return f"Searching {engine} for: {query}"

# 创建带配置的工具
search_tool = Tool(
    name="search",
    func=custom_search,
    description="Search the web. Use engine parameter to choose search engine.",
)

agent = create_agent(
    model="claude-sonnet-4-6",
    tools=[search_tool],
)

工具最佳实践

✅ 推荐

# 1. 编写清晰的文档字符串
def search(query: str) -> str:
    """Search for information.

    Args:
        query: What to search for.

    Returns:
        Search results.
    """

# 2. 使用类型注解
def calculate(x: int, y: int) -> int:
    """Add two numbers."""
    return x + y

# 3. 处理错误
def risky_operation(data: str) -> str:
    """Do something that might fail."""
    try:
        result = do_something(data)
        return result
    except Exception as e:
        return f"Error: {e}"

# 4. 提供有用的错误信息
def divide(a: float, b: float) -> str:
    """Divide two numbers."""
    if b == 0:
        return "Error: Cannot divide by zero"
    return str(a / b)

# 5. 保持工具单一职责
def get_weather(city: str) -> str:  # ✅ 好
    """Get weather."""

def weather_and_news(city: str) -> str:  # ❌ 不好
    """Get weather and news."""

❌ 避免

# 1. 缺少文档字符串
def tool(x):  # ❌ LLM 不知道用途
    return x * 2

# 2. 缺少类型注解
def calculate(x, y):  # ❌ LLM 不知道类型
    return x + y

# 3. 不处理错误
def api_call():  # ❌ 可能崩溃
    response = requests.get(url)
    return response.json()

# 4. 工具功能过多
def do_everything(action: str, **kwargs):  # ❌ 太模糊
    """Do various things."""

# 5. 返回复杂对象
def get_data() -> dict:  # ❌ LLM 难以理解
    return {"a": 1, "b": [1,2,3], "c": {...}}

# ✅ 应该返回字符串
def get_data() -> str:  # ✅ 好
    return "Formatted data as string"

Q3: 如何处理耗时工具？

import time

def long_running_task(task_id: str) -> str:
    """A task that takes time.

    Args:
        task_id: The task identifier.

    Returns:
        Task status or result.
    """
    # 显示进度
    print(f"Starting task {task_id}...")

    # 耗时操作
    time.sleep(5)

    return f"Task {task_id} completed"
