03. 模型集成

支持的模型提供商

LangChain 支持众多 LLM 提供商，让你轻松切换和测试不同模型。

主要提供商

提供商 模型 包名 特点
Anthropic Claude 3/4 langchain-anthropic 长上下文、安全
OpenAI GPT-4/3.5 langchain-openai 功能全面、生态好
Google Gemini langchain-google-genai 多模态、免费额度
Meta Llama langchain-aws 开源、可自部署
Microsoft Phi langchain-huggingface 轻量级
AWS Bedrock langchain-aws 企业级、多模型

安装模型集成

按需安装（推荐）

# Anthropic (Claude)
pip install langchain-anthropic

# OpenAI (GPT)
pip install langchain-openai

# Google (Gemini)
pip install langchain-google-genai

# AWS Bedrock
pip install langchain-aws

# Hugging Face
pip install langchain-huggingface

一键安装所有

pip install "langchain[all]"

Anthropic (Claude)

安装和配置

pip install langchain-anthropic

import os
os.environ["ANTHROPIC_API_KEY"] = "sk-ant-..."  # 从 https://console.anthropic.com/ 获取

基础使用

from langchain_anthropic import ChatAnthropic

# 创建模型实例
model = ChatAnthropic(
    model="claude-sonnet-4-6",  # 模型名称
    temperature=0.7,             # 温度（创造性）
    max_tokens=1024,            # 最大输出 token
    timeout=60,                 # 超时时间（秒）
)

# 调用模型
response = model.invoke([{"role": "user", "content": "你好，请介绍一下自己"}])
print(response.content)

流式输出

from langchain_anthropic import ChatAnthropic

model = ChatAnthropic(model="claude-sonnet-4-6")

# 流式输出
for chunk in model.stream([{"role": "user", "content": "写一首关于春天的诗"}]):
    print(chunk.content, end="", flush=True)

异步调用

from langchain_anthropic import ChatAnthropic

model = ChatAnthropic(model="claude-sonnet-4-6")

# 异步调用
response = await model.ainvoke([{"role": "user", "content": "你好"}])
print(response.content)

结构化输出

from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

class Person(BaseModel):
    name: str = Field(description="Person's name")
    age: int = Field(description="Person's age")
    email: str = Field(description="Person's email")

model = ChatOpenAI(model="gpt-4o")

# 使用 with_structured_output
structured_model = model.with_structured_output(Person)

response = structured_model.invoke([
    {"role": "user", "content": "John Doe, 30 years old, email john@example.com"}
])

print(response.name)   # John Doe
print(response.age)    # 30
print(response.email)  # john@example.com

from langchain_google_genai import ChatGoogleGenerativeAI

image_data = httpx.get(image_url).content

        "role": "user",

            {"type": "text", "text": "这是什么图片？"},

])

print(response.content)

模型比较

性能对比

# 测试不同模型
from langchain_anthropic import ChatAnthropic
from langchain_openai import ChatOpenAI
from langchain_google_genai import ChatGoogleGenerativeAI
import time

models = {
    "Claude": ChatAnthropic(model="claude-sonnet-4-6"),
    "GPT-4o": ChatOpenAI(model="gpt-4o"),
    "Gemini": ChatGoogleGenerativeAI(model="gemini-2.0-flash"),
}

prompt = [{"role": "user", "content": "解释量子力学"}]

for name, model in models.items():
    start = time.time()
    response = model.invoke(prompt)
    end = time.time()

    print(f"\n{name}:")
    print(f"时间：{end - start:.2f}秒")
    print(f"内容：{response.content[:100]}...")

在 Agent 中使用不同模型

使用字符串（简化）

from langchain.agents import create_agent

# 使用字符串指定模型
agent = create_agent(
    model="claude-sonnet-4-6",  # 或 "gpt-4o", "gemini-2.0-flash"
    tools=[...],
)

使用模型实例（更多控制）

from langchain.agents import create_agent
from langchain_anthropic import ChatAnthropic
from langchain_openai import ChatOpenAI

# 创建模型实例
model = ChatAnthropic(
    model="claude-sonnet-4-6",
    temperature=0.7,
    max_tokens=2048,
)

# 传递给 Agent
agent = create_agent(
    model=model,
    tools=[...],
)

切换模型

from langchain.agents import create_agent

def get_weather(city: str) -> str:
    return f"{city} is sunny"

# 使用 Claude
agent_claude = create_agent(
    model="claude-sonnet-4-6",
    tools=[get_weather],
)

# 使用 GPT-4
agent_gpt = create_agent(
    model="gpt-4o",
    tools=[get_weather],
)

# 使用 Gemini
agent_gemini = create_agent(
    model="gemini-2.0-flash",
    tools=[get_weather],
)

# 测试相同的问题
question = {"messages": [{"role": "user", "content": "北京天气"}]}

print("Claude:", agent_claude.invoke(question)["messages"][-1].content)
print("GPT-4:", agent_gpt.invoke(question)["messages"][-1].content)
print("Gemini:", agent_gemini.invoke(question)["messages"][-1].content)

高级配置

超时和重试

from langchain_anthropic import ChatAnthropic

model = ChatAnthropic(
    model="claude-sonnet-4-6",
    timeout=60,              # 超时 60 秒
    max_retries=3,           # 最大重试次数
    stop_sequences=["\n\n"], # 停止序列
)

温度控制

from langchain_anthropic import ChatAnthropic

# 低温度（更确定、保守）
model_conservative = ChatAnthropic(
    model="claude-sonnet-4-6",
    temperature=0.1,
)

# 中温度（平衡）
model_balanced = ChatAnthropic(
    model="claude-sonnet-4-6",
    temperature=0.7,

# 高温度（更创造性、随机）
model_creative = ChatAnthropic(
    model="claude-sonnet-4-6",
    temperature=1.0,
)

模型选择建议

按场景选择

场景：日常对话、客服
推荐：GPT-4o-mini, Claude Haiku, Gemini Flash
原因：成本低、速度快

场景：复杂推理、代码生成
推荐：Claude Opus, GPT-4o, Gemini Pro
原因：性能强、准确性高

场景：长文档分析
推荐：Claude Sonnet/Opus (200K 上下文)
原因：上下文窗口大

场景：多模态任务
推荐：GPT-4o, Gemini 2.0
原因：原生支持图像

场景：预算有限
推荐：GPT-4o-mini, Gemini Flash
原因：价格低廉

成本优化

# 根据任务复杂度选择模型
def choose_model(task_complexity):
    if task_complexity == "simple":
        return "gpt-4o-mini"  # $0.15/1M tokens
    elif task_complexity == "medium":
        return "claude-sonnet-4-6"  # $3/1M tokens
    else:
        return "claude-opus-4-1"  # $15/1M tokens

# 使用
agent = create_agent(
    model=choose_model("medium"),
    tools=[...],
)

错误处理

from langchain_anthropic import ChatAnthropic
from anthropic import APIError, RateLimitError

model = ChatAnthropic(model="claude-sonnet-4-6")

try:
    response = model.invoke([{"role": "user", "content": "你好"}])
    print(response.content)
except RateLimitError:
    print("请求频率限制，请稍后重试")
except APIError as e:
    print(f"API 错误：{e}")
except Exception as e:
    print(f"未知错误：{e}")

最佳实践

✅ 推荐

# 1. 使用环境变量管理 API 密钥
import os
os.environ["ANTHROPIC_API_KEY"] = os.getenv("ANTHROPIC_API_KEY")

# 2. 设置合理的超时
model = ChatAnthropic(model="claude-sonnet-4-6", timeout=60)

# 3. 启用重试
model = ChatAnthropic(model="claude-sonnet-4-6", max_retries=3)

# 4. 监控成本
# 定期检查 API 使用量和费用

# 5. 测试多个模型
# 在部署前测试不同模型的效果和成本

❌ 避免

# 1. 硬编码 API 密钥
os.environ["API_KEY"] = "sk-xxx"  # ❌

# 2. 不设置超时
model = ChatAnthropic(model="claude-sonnet-4-6")  # ❌ 可能无限等待

# 3. 忽略错误处理
response = model.invoke(...)  # ❌ 没有 try-except

# 4. 在生产环境使用默认配置
# 总是根据需求调整温度、token 限制等
